/**
 * Parse ONE argument, keeping its name on the issue.
 *
 * `schema.parse(args.name)` produces an issue with an empty path, so a refusal
 * could only ever name the failing CHECK and not the argument — the half of
 * this that a model cannot work around.
 */
function field<T>(schema: z.ZodType<T>, argName: string, value: unknown): T {
  return z.object({ [argName]: schema }).parse({ [argName]: value })[
    argName
  ] as T
}

/**
 * Name the arguments a Zod check rejected, and nothing else.
 *
 * The opaque fallback below exists because a filesystem or validator error may
 * carry file contents or a secret value. A Zod issue cannot: this reports only
 * the argument's path and the kind of check that failed, never the value that
 * failed it and never the issue's own message (`invalid_value` carries the
 * received literal). Without this an argument the model can fix reads as
 * "check arguments and workspace files", so it retries the same call.
 */
function rejectedArguments(error: z.ZodError): string {
  const named = [
    ...new Set(
      error.issues.map((issue) => {
        const path = issue.path.map(String).join('.')
        return path ? `${path} (${issue.code})` : issue.code
      }),
    ),
  ]
  return named.length > 0
    ? `Sites rejected these arguments: ${named.join(', ')}. Read the tool's input schema and send values that match it.`
    : 'Could not prepare the Sites request. Check arguments and workspace files.'
}

          .parse(raw)

          ? await prepare(cwd, args.projectDirectory ?? '.', signal, {
              realtime: true,

      }

        const args = z
          .object({ siteId: uuid, name: z.string().min(1).max(80) })
          .strict()
          .parse(raw)

          '',
          'POST',

          undefined,
          signal,
        )

          .extend({ published: z.boolean(), requestId: key })
          .strict()
          .parse(raw)

          input.siteId,
          input.environment,
          input.published,
          input.requestId,
          signal,
        )
      }

          this.request(path, 'GET', undefined, undefined, signal),
          this.request(
            path + '/operations',
            'GET',
            undefined,
            undefined,
            signal,
          ),
        ])

      }

          .strict()
          .parse(raw)

          undefined,
          signal,

          args.manifestPath,

          manifestPath,

          'POST',
          bundle,
          requestId,
          signal,
        )

          path + '/' + kind,
          'POST',
          { name: field(name, 'name', args.name) },
          requestId,
          signal,
        )

          path + '/secrets/' + binding,
          'PUT',

          requestId,
          signal,
        )

      // People read the connection message; an agent also needs to know an
      // unanswered mutation may have landed and is only safe to repeat by key.
      if (error instanceof SitesClientError)
        return {
          error: error.code,
          message:

              : error.message,

      // Filesystem and validator errors must never echo file contents / secret values.
      return {
        error: 'invalid_site_request',
        message:
          error instanceof z.ZodError
            ? rejectedArguments(error)
            : 'Could not prepare the Sites request. Check arguments and workspace files.',
      }
    }
  }

async function readBounded(path: string, limit: number): Promise<Buffer> {
  const file = await open(path, 'r')
  try {
    const stat = await file.stat()
    if (!stat.isFile() || stat.size > limit)
      throw invalid(
        'Site file exceeds the upload limit or is not a regular file.',
      )
    const buffer = Buffer.alloc(Math.min(stat.size + 1, limit + 1))
    let used = 0
    while (used < buffer.length) {
      const { bytesRead } = await file.read(
        buffer,
        used,
        buffer.length - used,
        null,
      )
      if (!bytesRead) break
      used += bytesRead
    }
    if (used > limit || used > stat.size)
      throw invalid('Site file changed during upload. Rebuild and retry.')
    return buffer.subarray(0, used)
  } finally {
    await file.close()
  }
}

/** Only fixed package-manager words enter cmd.exe; project paths never do. */
export function packageInvocation(
  manager: Manager,
  action: 'install' | 'build' | 'wrangler',
  platform: string,
): Invocation {
  if (!['npm', 'pnpm', 'yarn', 'bun'].includes(manager))
    throw new Error('Unsupported package manager')
  const args =
    action === 'install'
      ? ['install']
      : action === 'build'
        ? ['run', 'build']
        : manager === 'npm'
          ? ['install', '--save-dev', '--save-exact', `wrangler@${NEXT_WRANGLER_VERSION}`]
          : manager === 'pnpm'
            ? ['add', '--save-dev', '--save-exact', `wrangler@${NEXT_WRANGLER_VERSION}`]
            : ['add', '--dev', '--exact', `wrangler@${NEXT_WRANGLER_VERSION}`]
  if (platform !== 'win32' || manager === 'bun')
    return { command: manager, args, windowsVerbatimArguments: false }
  return {
    command: 'cmd.exe',
    args: ['/d', '/s', '/c', `${manager}.cmd ${args.join(' ')}`],

export function parseSitePreparationOutput(
  stdout: string,
): SitePreparationResult {
  const line = stdout
    .split('\n')
    .findLast((line) => line.startsWith(SITE_PREPARE_RESULT_PREFIX))
  if (!line) throw new Error('Site preparation did not return a result.')
  const value = JSON.parse(
    line.slice(SITE_PREPARE_RESULT_PREFIX.length),
  ) as SitePreparationResult
  const names = (list: unknown) =>
    Array.isArray(list) &&
    list.every(
      (name) => typeof name === 'string' && /^[A-Z][A-Z0-9_]{0,63}$/.test(name),
    )
  if (
    !['manifest', 'worker', 'static', 'next'].includes(value.kind) ||
    (value.kind !== 'next' && typeof value.manifestPath !== 'string') ||
    typeof value.usesAuth !== 'boolean' ||
    !names(value.storage?.d1) ||
    !names(value.storage?.r2)

export async function prepareSiteLocally(
  cwd: string,
  signal?: AbortSignal,
): Promise<SitePreparationResult> {
  const stdout = await new Promise<string>((resolve, reject) => {
    const buildSignal = AbortSignal.any([
      AbortSignal.timeout(15 * 60 * 1000),
      ...(signal ? [signal] : []),
    ])
    const child = spawn(
      'node',
      ['--input-type=module', '-e', sitePreparationScript()],
      {
        cwd,
        signal: buildSignal,
        detached: process.platform !== 'win32',
        env: nextPreparationEnvironment(process.env),
        stdio: ['ignore', 'pipe', 'pipe'],
      },
    )
    const stop = () => {
      if (process.platform !== 'win32' && child.pid) {
        try {
          process.kill(-child.pid, 'SIGTERM')
        } catch {
          /* Already exited. */
        }
      }
    }
    buildSignal.addEventListener('abort', stop, { once: true })
    let output = ''
    child.stdout.on('data', (data: Buffer) => {
      output = (output + data.toString()).slice(-64 * 1024)
    })
    child.stderr.on('data', () => {})
    child.once('error', () =>
      reject(
        preparationFailed(
          buildSignal.aborted
            ? 'Publishing was cancelled or the build exceeded its 15-minute limit.'
            : 'Publishing needs Node.js 20.19 or newer in the workspace PATH.',
        ),
      ),
    )
    child.once('close', (code) => {
      buildSignal.removeEventListener('abort', stop)
      if (code === 0) resolve(output)
      else reject(preparationFailed(parseSitePreparationError(output)))
    })
  })
  try {
    return parseSitePreparationOutput(stdout)
  } catch {
    throw preparationFailed()
  }
}

/** Receipts keep only codes; say what each means without vendor prose. */
export function operationFailureMessage(operation: Operation): string {
  const result = operation.result
  if (operation.state === 'unknown' || result?.error === 'cloudflare_outcome_unknown')
    return 'Cloudflare did not confirm the deployment. Check deployment activity in Manage site before publishing again.'
  const provider = result?.provider
  if (provider?.stage === 'worker_upload' && provider.codes?.includes(10021))
    return 'Cloudflare could not start the built Worker: it failed startup validation. This usually means code that cannot run on Workers, such as a Node.js-only module.'
  if (provider?.status === 429)
    return 'Cloudflare is rate limiting deployments. Try again in a minute.'
  if (result?.error === 'opennext_manifest_required')
    return 'This environment hosts a Next.js app. Publish the Next.js project to it.'
  return 'Cloudflare rejected the deployment.'
