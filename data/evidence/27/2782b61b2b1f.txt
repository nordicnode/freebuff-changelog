 * (Gmail, Linear, PostHog… through Freebuff integrations), pulled again on Refresh by the server.
 *
 * This file holds the stored shapes and the trust rules for them, with nothing but zod imported
 * (data.ts builds its schema from it). Mapping a result into rows and filling blanks live in
 * data-connect-run.ts; the server route in web/src/server/desktop-cloud/artifact-data-refresh.ts.
 *
 * A connected recipe (`refresh.how` of kind "integration") is:
 *  - `app`: the integration's slug ("linear"), for display and to check the call belongs to it;
 *  - `calls`: 1-4 READ-ONLY integration tool calls ({tool, arguments, account?}). The server runs
 *    them as the owner with read-only tools only; a tool that can change anything is refused.
 *    `{{id}}` blanks (a filter or parameter id) may appear in string argument values: plain text
 *    only, as in a command recipe;
 *  - `map`: how a call's result becomes rows, declaratively: `records`, the path to the list of
 *    records in each call's data, and `columns`, a path per dataset column (types come from the
 *    dataset's columns and values are coerced to them). No code.
 *
 * Who may run it is decided by these owner-only options on `refresh` (the existing rule: only the
 * owner changes `refresh`, data-trust.ts):
 *  - `approved`: the owner's approval of exactly this recipe and these options (the schedule
 *    included). Stamped only by `normaliseRecipeApproval` on an owner's own (non-agent) SAVE, and
 *    marked `via: APPROVAL_VIA` there: a stamp without the marker (written before the store
 *    enforced these rules, or forged) never counts. An agent's write of a connected recipe leaves
 *    it unapproved ("pending"), so it cannot run until the owner approves it in the app; a new
 *    artifact (a copy, a template's use) starts pending; a restore keeps only the stamp already
 *    stored. Any later change to the recipe or options leaves the stamp stale (untrusted).
 *  - `collaboratorsCanRefresh`: editors may press Refresh (it still runs as the owner). Off by default.
 *  - `onOpen`: refresh when the dashboard is opened, at most once per staleness window.
 *  - `schedule`: a scheduled refresh ({every, at?}). Recorded only: nothing runs it yet.
 */

export type SettingsSearchPage = { id: string; label: PageLabel }

function normalize(value: string) {
  return value
    .normalize('NFKD')
    .replace(/\p{M}/gu, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim()
}

// A single missing, extra, or mistyped letter; short words must match literally.
function nearWord(query: string, word: string) {
  if (query.length < 4 || Math.abs(query.length - word.length) > 1) return false
  let edits = 0
  for (let i = 0, j = 0; i < query.length || j < word.length; ) {
    if (query[i] === word[j]) {
      i++
      j++
      continue
    }
    if (++edits > 1) return false
    if (query.length >= word.length) i++
    if (word.length >= query.length) j++
  }
  return true
}

function score(query: string, label: string, keywords = '') {
  const name = normalize(label)
  if (name === query) return 100
  if (name.startsWith(query)) return 90
  if (name.includes(query)) return 80
  const words = normalize(`${label} ${keywords}`).split(' ')
  let fuzzy = false
  for (const token of query.split(' ')) {
    if (words.some((word) => word.startsWith(token))) continue
    if (!words.some((word) => nearWord(token, word))) return 0
    fuzzy = true
  }
  return fuzzy ? 10 : 50
}

export function searchSettings<T extends SettingsSearchPage>(
  query: string,
  pages: readonly T[],
  surface: Surface,
  hidden: readonly string[] = [],
): { page: T; detail?: string }[] {
  const normalized = normalize(query)
  if (!normalized) return query.trim() ? [] : pages.map((page) => ({ page }))
  return pages
    .map((page) => {
      let best = score(normalized, page.label)
      let detail: string | undefined
      for (const topic of topics[page.label]) {
        if (topic.surface && topic.surface !== surface) continue
        if (hidden.includes(topic.label)) continue
        const value = score(
          normalized,
          topic.label,
          `${page.label} ${topic.keywords}`,
        )
        if (value > best) {
          best = value
          detail = topic.label
        }
      }
      return { page, detail, score: best }
    })
    .filter((result) => result.score > 0)
    .sort((a, b) => b.score - a.score)
}

  op: ArtifactDiffOp
  /** Human: 'Slide 3 “Pricing”', 'Cell Sheet1!B3', 'Paragraph 4'. */
  label: string
  /** Which fields changed (slides, design, whiteboard, data, sheet formats). */
  fields?: string[]
  /** A short textual rendering of the old unit (capped). */
  before?: string
  after?: string
  /** Word diff of `before` → `after`, when both are text. */
  words?: DiffSegment[]
  /** Sheet: value, formula and rich text equal; only format or style keys differ. */
  formatOnly?: boolean
  /** Whether `restoreArtifactChange` can apply it. */
  restorable: boolean
  /** For grouping in the UI: the tab name, the slide label, the board title. */
  group?: string
  /** Extra lines (a dataset's row count delta and sample changed rows). */
  detail?: string[]
}
export type ArtifactDiff = {
  kind: ArtifactKind
  ok: boolean
  changes: ArtifactDiffChange[]
  omitted: number
  counts: { unit: string; added: number; removed: number; changed: number; moved: number; renamed: number }[]
  formatOnlyCount: number
  summary: string
}

type RestoreResult = { ok: true; body: string } | { ok: false; reason: string }

const DEFAULT_CAP = 200
const DEFAULT_ROW_SAMPLES = 3
const DEFAULT_MAX_LINES = 40

export function diffArtifactBodies(
  kind: ArtifactKind,
  before: string,
  after: string,
  options: { cap?: number; rowSamples?: number } = {},
): ArtifactDiff {
  const cap = Math.max(0, Math.floor(options.cap ?? DEFAULT_CAP))
  const computed = compute(kind, before, after, { rowSamples: options.rowSamples ?? DEFAULT_ROW_SAMPLES })
  return finish(kind, computed, cap)
}

export function describeArtifactDiff(
  kind: ArtifactKind,
  before: string,
  after: string,
  options: { maxLines?: number } = {},
): string {
  const maxLines = Math.max(0, Math.floor(options.maxLines ?? DEFAULT_MAX_LINES))
  const diff = diffArtifactBodies(kind, before, after, { cap: Number.MAX_SAFE_INTEGER })
  const lines = [diff.summary]
  const shown = diff.changes.slice(0, maxLines)
  for (const change of shown) {
    lines.push(bulletOf(change))
    for (const extra of change.detail ?? []) lines.push(`  ${oneLine(extra, 160)}`)
  }
  const more = diff.changes.length - shown.length
  if (more > 0) lines.push(`…and ${more} more`)
  return lines.join('\n')
}

export function restoreArtifactChange(
  kind: ArtifactKind,
  before: string,
  after: string,
  current: string,
  changeId: string,
): RestoreResult {
  try {
    const computed = compute(kind, before, after, { rowSamples: 0 })
    const entry = computed.entries.find((e) => e.change.id === changeId)
    if (!entry) return fail(`No change "${changeId}" between these versions`)
    if (!entry.restore) return fail(entry.reason ?? `${entry.change.label} cannot be restored on its own`)
    return entry.restore(current)
  } catch {
    return fail('The change could not be restored')
  }
}

/** A sheet cell's value: `v` or `f`, and the rich-text `runs` whose texts must join to `v`. */

const clone = <T>(value: T): T => structuredClone(value)

function clip(text: string, max = TEXT_CAP): string {
  return text.length > max ? `${text.slice(0, Math.max(0, max - 1))}…` : text
}

function oneLine(text: string, max: number): string {
  return clip(text.replace(/\s+/g, ' ').trim(), max)
}

/** A value as short text: strings as they are, everything else as compact JSON. */
function show(value: unknown): string {
  if (value === undefined) return ''
  if (typeof value === 'string') return value
  try {
    return JSON.stringify(value) ?? ''
  } catch {
    return ''
  }
}

const quoted = (text: string | undefined, max = 40) => (text ? ` “${oneLine(text, max)}”` : '')

/** The changed fields of `o`, one per line (`field: value`), or the bare value for one field. */
function renderFields(o: Obj | undefined, fields: readonly string[], render: (key: string, value: unknown) => string = (_k, v) => show(v)): string | undefined {
  if (!o) return undefined
  const present = fields.filter((f) => has(o, f))
  if (present.length === 0) return fields.length === 1 ? '' : undefined
  if (fields.length === 1) return clip(render(fields[0], o[fields[0]]))
  return clip(present.map((f) => `${f}: ${render(f, o[f])}`).join('\n'))
}

const TEXT_FIELDS = ['title', 'name', 'subtitle', 'label', 'text', 'body', 'markdown', 'notes', 'alt', 'url'] as const

/** An added or removed item as text: its text-ish fields, else its JSON. */
function renderItem(o: Obj | undefined): string | undefined {
  if (!o) return undefined
  const lines = TEXT_FIELDS.filter((f) => typeof o[f] === 'string' && o[f] !== '').map((f) => `${f}: ${String(o[f])}`)
  return clip(lines.length > 0 ? lines.join('\n') : show(o))
}

/** Restore through a parsed copy of the current body; `apply` mutates it or names a refusal. */
function jsonRestorer(apply: (root: Obj) => string | null): Restorer {
  return (current) => {
    const root = parseObject(current)
    if (!root) return fail('The current version is not valid JSON')
    const copy = clone(root)
    const refusal = apply(copy)
    if (refusal !== null) return fail(refusal)
    if (deepEqual(copy, root)) return fail('Nothing to restore: the current version already has the old value')
    return { ok: true, body: JSON.stringify(copy) }
  }
}

class Collector {
  readonly entries: Entry[] = []
  private readonly ids = new Set<string>()
  add(input: ChangeInput, restore?: Restorer, reason?: string) {
    let id = input.id
    for (let n = 2; this.ids.has(id); n += 1) id = `${input.id}#${n}`
    this.ids.add(id)
    const change: ArtifactDiffChange = { ...stripUndefined(input), id, restorable: restore !== undefined }
    const entry: Entry = { change }
    if (restore) entry.restore = restore
    if (reason) entry.reason = reason
    this.entries.push(entry)
  }
}

function stripUndefined<T extends object>(o: T): T {
  const out: Obj = {}
  for (const [k, v] of Object.entries(o)) if (v !== undefined) out[k] = v
  return out as T
}

/** before/after and, when both are text, the word diff. */
function textual(before: string | undefined, after: string | undefined): Pick<ChangeInput, 'before' | 'after' | 'words'> {
  const out: Pick<ChangeInput, 'before' | 'after' | 'words'> = {}
  if (before !== undefined) out.before = clip(before)
  if (after !== undefined) out.after = clip(after)
  if (before !== undefined && after !== undefined && (before !== '' || after !== '')) out.words = wordDiff(clip(before), clip(after))
  return out
}

function compute(kind: ArtifactKind, before: string, after: string, ctx: Ctx): Computed {
  try {
    const c = new Collector()
    if (kind === 'doc') {
      diffDoc(before, after, c)
      return { ok: true, entries: c.entries }
    }
    const b = parseObject(before)
    const a = parseObject(after)
    if (!b || !a) return wholeBody(before, after, 'one version is not valid JSON')
    if (before === after || deepEqual(b, a)) return { ok: true, entries: [] }
    JSON_DIFFS[kind](b, a, c, ctx)
    return { ok: true, entries: c.entries }
  } catch {
    return wholeBody(before, after, 'the versions could not be compared')
  }
}

function wholeBody(before: string, after: string, why: string): Computed {
  if (before === after) return { ok: false, entries: [] }
  return {
    ok: false,
    entries: [
      {
        change: { id: 'body', unit: 'body', op: 'changed', label: 'Whole body', before: clip(before), after: clip(after), restorable: false },
        reason: `Not comparable: ${why}`,
      },
    ],
  }
}

const OPS: ArtifactDiffOp[] = ['changed', 'added', 'removed', 'moved', 'renamed']

function plural(unit: string, n: number): string {
  if (n === 1) return unit
  if (unit === 'body') return 'bodies'
  if (/[^aeiou]y$/.test(unit)) return `${unit.slice(0, -1)}ies`
  if (/(s|x|ch|sh)$/.test(unit)) return `${unit}es`
  return `${unit}s`
}

function finish(kind: ArtifactKind, computed: Computed, cap: number): ArtifactDiff {
  // Value changes first; format-only cell changes after them (stable otherwise).
  const all = computed.entries.map((e) => e.change)
  const ordered = [...all.filter((c) => !c.formatOnly), ...all.filter((c) => c.formatOnly)]
  const counts: ArtifactDiff['counts'] = []
  const byUnit = new Map<string, ArtifactDiff['counts'][number]>()
  for (const change of ordered) {
    let row = byUnit.get(change.unit)
    if (!row) {
      row = { unit: change.unit, added: 0, removed: 0, changed: 0, moved: 0, renamed: 0 }
      byUnit.set(change.unit, row)
      counts.push(row)
    }
    row[change.op] += 1
  }
  const formatOnlyCount = ordered.filter((c) => c.formatOnly).length
  let summary: string
  if (!computed.ok && ordered.length > 0) summary = 'The whole body changed (the versions could not be compared structurally)'
  else if (ordered.length === 0) summary = 'No changes'
  else {
    summary = counts
      .map((row) => {
        const parts = OPS.filter((op) => row[op] > 0).map((op, i) => (i === 0 ? `${row[op]} ${plural(row.unit, row[op])} ${op}` : `${row[op]} ${op}`))
        const fmt = row.unit === 'cell' && formatOnlyCount > 0 ? ` (${formatOnlyCount} format only)` : ''
        return parts.join(', ') + fmt
      })
      .join('; ')
  }
  return {
    kind,
    ok: computed.ok,
    changes: ordered.slice(0, cap),
    omitted: Math.max(0, ordered.length - cap),
    counts,
    formatOnlyCount,
    summary,
  }
}

function bulletOf(change: ArtifactDiffChange): string {
  let line = `- ${change.op} ${change.label}`
  if (change.group && !change.label.includes(change.group)) line += ` (${oneLine(change.group, 60)})`
  if (change.fields && change.fields.length > 0) line += `: ${change.fields.join(', ')}`
  const b = change.before !== undefined ? oneLine(change.before, 80) : undefined
  const a = change.after !== undefined ? oneLine(change.after, 80) : undefined
  if (change.op === 'changed' || change.op === 'renamed') {
    if (b !== undefined && a !== undefined && (b || a)) line += ` — “${b}” → “${a}”`
  } else if (change.op === 'added' && a) line += ` — “${a}”`
  else if (change.op === 'removed' && b) line += ` — was “${b}”`
  return line
}

// ---------------------------------------------------------------------------------------------
// Alignment: LCS, hunks, longest increasing subsequence
// ---------------------------------------------------------------------------------------------

/** Index pairs of a longest common subsequence of `a` and `b`, or null above `maxCells`. */
function lcsPairs(a: readonly string[], b: readonly string[], maxCells: number): Array<[number, number]> | null {
  let prefix = 0
  while (prefix < a.length && prefix < b.length && a[prefix] === b[prefix]) prefix += 1
  let suffix = 0
  while (suffix < a.length - prefix && suffix < b.length - prefix && a[a.length - 1 - suffix] === b[b.length - 1 - suffix]) suffix += 1
  const x = a.slice(prefix, a.length - suffix)
  const y = b.slice(prefix, b.length - suffix)
  const pairs: Array<[number, number]> = []
  for (let i = 0; i < prefix; i += 1) pairs.push([i, i])
  if (x.length > 0 && y.length > 0) {
    if ((x.length + 1) * (y.length + 1) > maxCells) return null
    const width = y.length + 1
    const table = new Uint32Array((x.length + 1) * width)
    for (let i = x.length - 1; i >= 0; i -= 1) {
      for (let j = y.length - 1; j >= 0; j -= 1) {
        table[i * width + j] = x[i] === y[j] ? table[(i + 1) * width + j + 1] + 1 : Math.max(table[(i + 1) * width + j], table[i * width + j + 1])
      }
    }
    let i = 0
    let j = 0
    while (i < x.length && j < y.length) {
      if (x[i] === y[j]) {
        pairs.push([prefix + i, prefix + j])
        i += 1
        j += 1
      } else if (table[(i + 1) * width + j] >= table[i * width + j + 1]) i += 1
      else j += 1
    }
  }
  for (let k = 0; k < suffix; k += 1) pairs.push([a.length - suffix + k, b.length - suffix + k])
  return pairs
}

/** The unmatched stretches between matched pairs: a[aStart..aEnd) became b[bStart..bEnd). */
type Hunk = { aStart: number; aEnd: number; bStart: number; bEnd: number }

function hunksOf(pairs: Array<[number, number]>, aLength: number, bLength: number): Hunk[] {
  const out: Hunk[] = []
  let pa = 0
  let pb = 0
  for (const [ia, ib] of [...pairs, [aLength, bLength] as [number, number]]) {
    if (ia > pa || ib > pb) out.push({ aStart: pa, aEnd: ia, bStart: pb, bEnd: ib })
    pa = ia + 1
    pb = ib + 1
  }
  return out
}

/** Indices (into `seq`) of one longest strictly increasing subsequence. O(n log n). */
function lisIndices(seq: readonly number[]): Set<number> {
  const tails: number[] = []
  const prev: number[] = new Array<number>(seq.length).fill(-1)
  for (let i = 0; i < seq.length; i += 1) {
    let lo = 0
    let hi = tails.length
    while (lo < hi) {
      const mid = (lo + hi) >> 1
      if (seq[tails[mid]] < seq[i]) lo = mid + 1
      else hi = mid
    }
    if (lo > 0) prev[i] = tails[lo - 1]
    tails[lo] = i
  }
  const out = new Set<number>()
  let k = tails.length > 0 ? tails[tails.length - 1] : -1
  while (k >= 0) {
    out.add(k)
    k = prev[k]
  }
  return out
}

/** Ids present on both sides whose relative order changed (not in a longest kept order). */
function reorderedIds(beforeIds: readonly string[], afterIds: readonly string[]): Set<string> {
  const beforeIndex = new Map(beforeIds.map((id, i) => [id, i]))
  const common = afterIds.filter((id) => beforeIndex.has(id))
  const keep = lisIndices(common.map((id) => beforeIndex.get(id) ?? 0))
  return new Set(common.filter((_id, i) => !keep.has(i)))
}

// ---------------------------------------------------------------------------------------------
// Word diff
// ---------------------------------------------------------------------------------------------

export function wordDiff(before: string, after: string): DiffSegment[] {
  if (before === after) return before ? [{ op: 'same', text: before }] : []
  const a = before.match(/\s+|[^\s]+/g) ?? []
  const b = after.match(/\s+|[^\s]+/g) ?? []
  const pairs = lcsPairs(a, b, WORD_LCS_CELLS)
  const out: DiffSegment[] = []
  const push = (op: DiffSegment['op'], text: string) => {
    if (!text) return
    const last = out[out.length - 1]
    if (last && last.op === op) last.text += text
    else out.push({ op, text })
  }
  if (!pairs) {
    // Too big to align word by word: the common prefix and suffix, the middle replaced whole.
    let prefix = 0
    while (prefix < a.length && prefix < b.length && a[prefix] === b[prefix]) prefix += 1
    let suffix = 0
    while (suffix < a.length - prefix && suffix < b.length - prefix && a[a.length - 1 - suffix] === b[b.length - 1 - suffix]) suffix += 1
    push('same', a.slice(0, prefix).join(''))
    push('del', a.slice(prefix, a.length - suffix).join(''))
    push('add', b.slice(prefix, b.length - suffix).join(''))
    push('same', a.slice(a.length - suffix).join(''))
    return out
  }
  let i = 0
  let j = 0
  for (const [pi, pj] of [...pairs, [a.length, b.length] as [number, number]]) {
    push('del', a.slice(i, pi).join(''))
    push('add', b.slice(j, pj).join(''))
    if (pi < a.length) push('same', a[pi])
    i = pi + 1
    j = pj + 1
  }
  return out
}

// Keyed lists (slides, elements, data lists, boards, styles…)

type ItemOp = 'added' | 'removed' | 'changed' | 'moved'
type ItemDiff = {
  id: string
  op: ItemOp
  before?: Obj
  after?: Obj
  beforeIndex: number
  afterIndex: number
  /** Changed keys; 'order' when its position among the kept items changed. */
  fields: string[]
}

/** Items in `before`/`after` (lists of objects keyed by `id`), in display order. Null when not keyable. */
function diffKeyed(before: unknown, after: unknown, options: { skip?: readonly string[]; geometry?: readonly string[] } = {}): ItemDiff[] | null {
  const bl = before === undefined ? [] : before
  const al = after === undefined ? [] : after
  if (!Array.isArray(bl) || !Array.isArray(al)) return null
  const bm = keyed(bl)
  const am = keyed(al)
  if (!bm || !am) return null
  const skip = new Set(options.skip ?? [])
  const geometry = new Set(options.geometry ?? [])
  const beforeIds = [...bm.keys()]
  const afterIds = [...am.keys()]
  const beforeAt = new Map(beforeIds.map((id, i) => [id, i]))
  const afterAt = new Map(afterIds.map((id, i) => [id, i]))
  const moved = reorderedIds(beforeIds, afterIds)
  const positioned: Array<{ pos: number; diff: ItemDiff }> = []
  afterIds.forEach((id, afterIndex) => {
    const a = am.get(id) as Obj
    const beforeIndex = beforeAt.get(id) ?? -1
    if (beforeIndex < 0) {
      positioned.push({ pos: afterIndex, diff: { id, op: 'added', after: a, beforeIndex: -1, afterIndex, fields: [] } })
      return
    }
    const b = bm.get(id) as Obj
    const fields = [...new Set([...Object.keys(b), ...Object.keys(a)])].filter((k) => !skip.has(k) && !deepEqual(b[k], a[k]))
    const reordered = moved.has(id)
    if (fields.length === 0 && !reordered) return
    const op: ItemOp = fields.every((f) => geometry.has(f)) ? 'moved' : 'changed'
    positioned.push({ pos: afterIndex, diff: { id, op, before: b, after: a, beforeIndex, afterIndex, fields: reordered ? [...fields, 'order'] : fields } })
  })
  beforeIds.forEach((id, beforeIndex) => {
    if (am.has(id)) return
    let pos = -0.5
    for (let k = beforeIndex - 1; k >= 0; k -= 1) {
      const at = afterAt.get(beforeIds[k])
      if (at !== undefined) {
        pos = at + 0.5
        break
      }
    }
    positioned.push({ pos, diff: { id, op: 'removed', before: bm.get(id) as Obj, beforeIndex, afterIndex: -1, fields: [] } })
  })
  return positioned.sort((x, y) => x.pos - y.pos).map((p) => p.diff)
}

/** Where an item goes back: after the nearest item that preceded it in `beforeIds` and `list` still holds. */
function placeIndex(list: readonly unknown[], beforeIds: readonly string[], id: string): number {
  const at = beforeIds.indexOf(id)
  for (let k = at - 1; k >= 0; k -= 1) {
    const found = list.findIndex((item) => idOf(item) === beforeIds[k])
    if (found >= 0) return found + 1
  }
  return 0
}

type ListAt = {
  /** The list in the current root, or why it cannot be reached. */
  get: (root: Obj, create: boolean) => unknown[] | string
  noun: string
}

type KeyedRestoreOptions = {
  /** The fewest items the list may keep (a deck's one slide). */
  min?: number
  /** A refusal for putting `item` (the old version) back into `root`. */
  checkPut?: (root: Obj, item: Obj) => string | null
  /** A refusal for taking item `id` out of `root`. */
  checkRemove?: (root: Obj, id: string) => string | null
  /** After taking item `id` out: the list's follow-up (a whiteboard drops arrows bound to it). */
  afterRemove?: (list: unknown[], id: string) => unknown[]
}

/** Restore one keyed item's old state: fields back, position back, re-inserted or taken out. */
function restoreKeyed(root: Obj, at: ListAt, d: ItemDiff, beforeIds: readonly string[], options: KeyedRestoreOptions = {}): string | null {
  const got = at.get(root, d.op === 'removed')
  if (typeof got === 'string') return got
  const list = got
  const index = list.findIndex((item) => idOf(item) === d.id)
  if (d.op === 'removed') {
    if (index >= 0) return `${at.noun} "${d.id}" is already in the current version`
    if (!d.before) return 'Nothing to put back'
    const refusal = options.checkPut?.(root, d.before) ?? null
    if (refusal) return refusal
    list.splice(placeIndex(list, beforeIds, d.id), 0, clone(d.before))
    return null
  }
  if (index < 0) return `${at.noun} "${d.id}" is no longer in the current version`
  if (d.op === 'added') {
    if (options.min !== undefined && list.length <= options.min) return `The last ${at.noun.toLowerCase()} cannot be removed`
    const refusal = options.checkRemove?.(root, d.id) ?? null
    if (refusal) return refusal
    list.splice(index, 1)
    if (options.afterRemove) {
      const next = options.afterRemove(list, d.id)
      list.splice(0, list.length, ...next)
    }
    return null
  }
  // changed / moved: the changed fields back, then the position.
  const item = list[index]
  if (!isObj(item) || !d.before) return `${at.noun} "${d.id}" is not an object`
  for (const field of d.fields) {
    if (field === 'order') continue
    if (has(d.before, field)) item[field] = clone(d.before[field])
    else delete item[field]
  }
  const refusal = options.checkPut?.(root, item) ?? null
  if (refusal) return refusal
  if (d.fields.includes('order')) {
    list.splice(index, 1)
    list.splice(placeIndex(list, beforeIds, d.id), 0, item)
  }
  return null
}

const idsOf = (list: unknown): string[] => (Array.isArray(list) ? list.map(idOf).filter((id): id is string => id !== null) : [])

/** A list field on the root (`root[key]`). */
const rootList = (key: string, noun: string): ListAt => ({
  noun,
  get: (root, create) => {
    if (root[key] === undefined && create) root[key] = []
    const list = root[key]
    return Array.isArray(list) ? list : `The current version has no ${key}`
  },
})

const capitalise = (text: string) => text.charAt(0).toUpperCase() + text.slice(1)

/** Top-level fields outside the kind's units: the title, then any other setting. */
function topLevel(b: Obj, a: Obj, skip: readonly string[], c: Collector) {
  const skipped = new Set(skip)
  for (const key of [...new Set([...Object.keys(b), ...Object.keys(a)])]) {
    if (skipped.has(key) || deepEqual(b[key], a[key])) continue
    const op: ArtifactDiffOp = !has(b, key) ? 'added' : !has(a, key) ? 'removed' : 'changed'
    const restore = jsonRestorer((root) => {
      if (has(b, key)) root[key] = clone(b[key])
      else delete root[key]
      return null
    })
    const isTitle = key === 'title'
    c.add(
      {
        id: isTitle ? 'title' : `setting:${key}`,
        unit: isTitle ? 'title' : 'setting',
        op,
        label: isTitle ? 'Title' : capitalise(key.replace(/([a-z])([A-Z])/g, '$1 $2').toLowerCase()),
        ...textual(has(b, key) ? show(b[key]) : op === 'added' ? undefined : '', has(a, key) ? show(a[key]) : op === 'removed' ? undefined : ''),
      },
      restore,
    )
  }
}

/** The change texts of one keyed item. */
function itemText(d: ItemDiff, render?: (key: string, value: unknown) => string) {
  if (d.op === 'added') return textual(undefined, renderItem(d.after))
  if (d.op === 'removed') return textual(renderItem(d.before), undefined)
  const fields = d.fields.filter((f) => f !== 'order')
  if (fields.length === 0) return {}
  return textual(renderFields(d.before, fields, render) ?? '', renderFields(d.after, fields, render) ?? '')
}

const itemFields = (d: ItemDiff) => (d.op === 'changed' || d.op === 'moved' ? { fields: d.fields } : {})

function blockLabel(source: string, index: number): string {
  const first = source.split('\n')[0] ?? ''
  const heading = /^ {0,3}#{1,6}\s+(.*)$/.exec(first)
  if (heading) return `Heading${quoted(heading[1].replace(/#+\s*$/, ''))}`
  const n = index + 1
  if (/^\s*(```|~~~)/.test(first)) return `Code block ${n}`
  if (/^\s*\|/.test(first)) return `Table ${n}`
  if (/^\s*([-*+]|\d+[.)])\s/.test(first)) return `List ${n}`
  if (/^\s*>/.test(first)) return `Quote ${n}`
  if (/^\s*!\[/.test(first)) return `Image ${n}`
  return `Paragraph ${n}`
}

function docRestorer(plan: (blocks: readonly string[]) => DocBlockEdit | string): Restorer {
  return (current) => {
    const { content, suggestions } = splitDocSuggestions(current)
    const edit = plan(splitBlocks(content))
    if (typeof edit === 'string') return fail(edit)
    const next = applyDocBodyEdit(content, edit)
    if (next === null) return fail('The block changed again and cannot be restored')
    if (next === content) return fail('Nothing to restore: the current version already has the old text')
    const records = carryDocBlockIds(content, suggestions, edit, next)
    if (records === null) return fail('Restoring this block would change the surrounding blocks')
    return { ok: true, body: joinDocSuggestions(next, records) }
  }
}

const occurrences = (blocks: readonly string[], source: string) =>
  blocks.reduce<number[]>((out, block, i) => (block === source ? [...out, i] : out), [])

/** The one index of `source` in `blocks`, or a refusal. */
function uniqueBlock(blocks: readonly string[], source: string, what: string): number | string {
  const found = occurrences(blocks, source)
  if (found.length === 0) return `${what} is no longer in the current version`
  if (found.length > 1) return `${what} appears more than once in the current version`
  return found[0]
}

function diffDoc(before: string, after: string, c: Collector) {
  const bb = splitBlocks(splitDocSuggestions(before).content)
  const ab = splitBlocks(splitDocSuggestions(after).content)
  const pairs = lcsPairs(bb, ab, ALIGN_LCS_CELLS)
  if (!pairs) {
    c.add(
      { id: 'body', unit: 'body', op: 'changed', label: 'Whole document', ...textual(clip(before), clip(after)) },
      (current) => (current === after ? { ok: true, body: before } : fail('The document changed again since; it is too long to restore block by block')),
    )
    return
  }
  const bTabs = docTabs(bb)
  const aTabs = docTabs(ab)
  const groupOf = (tabs: DocTab[], index: number) => {
    const tab = tabs.length > 1 ? tabOfBlock(tabs, index) : undefined
    return tab ? { group: tab.name } : {}
  }
  const tabIdAt = (tabs: DocTab[], marker: number) => tabs.find((t) => t.marker === marker)?.id ?? String(marker + 1)

  const removedRestore = (bi: number) =>
    docRestorer((blocks) => {
      const source = bb[bi]
      if (bi === 0) {
        if (blocks[0] === source) return 'That block is already back'
        return { index: 0, span: 0, before: '', after: source }
      }
      const anchor = occurrences(blocks, bb[bi - 1])
      if (anchor.length === 1) {
        if (blocks[anchor[0] + 1] === source) return 'That block is already back'
        return { index: anchor[0] + 1, span: 0, before: '', after: source }
      }
      const following = bi + 1 < bb.length ? occurrences(blocks, bb[bi + 1]) : []
      if (following.length === 1) {
        if (blocks[following[0] - 1] === source) return 'That block is already back'
        return { index: following[0], span: 0, before: '', after: source }
      }
      return 'The blocks around it changed since; it cannot be placed back'
    })
  const addedRestore = (ai: number, what: string) =>
    docRestorer((blocks) => {
      const at = uniqueBlock(blocks, ab[ai], what)
      if (typeof at === 'string') return at
      return { index: at, before: blocks[at], after: '' }
    })
  const changedRestore = (bi: number, ai: number, what: string) =>
    docRestorer((blocks) => {
      const at = uniqueBlock(blocks, ab[ai], what)
      if (typeof at === 'string') return at
      return { index: at, before: blocks[at], after: bb[bi] }
    })

  for (const h of hunksOf(pairs, bb.length, ab.length)) {
    const removed: number[] = []
    const added: number[] = []
    const removedTabs: number[] = []
    const addedTabs: number[] = []
    for (let i = h.aStart; i < h.aEnd; i += 1) (parseTabMarker(bb[i]) !== null ? removedTabs : removed).push(i)
    for (let j = h.bStart; j < h.bEnd; j += 1) (parseTabMarker(ab[j]) !== null ? addedTabs : added).push(j)
    // Tab markers: paired by position within the hunk (a renamed tab's marker changed text).
    const tabPairs = Math.min(removedTabs.length, addedTabs.length)
    for (let k = 0; k < Math.max(removedTabs.length, addedTabs.length); k += 1) {
      const bi = removedTabs[k]
      const ai = addedTabs[k]
      const oldName = bi !== undefined ? parseTabMarker(bb[bi]) ?? '' : ''
      const newName = ai !== undefined ? parseTabMarker(ab[ai]) ?? '' : ''
      if (k < tabPairs) {
        c.add(
          { id: `tab:${tabIdAt(aTabs, ai)}`, unit: 'tab', op: 'renamed', label: `Tab${quoted(newName)}`, ...textual(oldName, newName) },
          changedRestore(bi, ai, `Tab “${newName}”`),
        )
      } else if (ai !== undefined) {
        c.add({ id: `tab:${tabIdAt(aTabs, ai)}`, unit: 'tab', op: 'added', label: `Tab${quoted(newName)}`, ...textual(undefined, newName) }, addedRestore(ai, `Tab “${newName}”`))
      } else if (bi !== undefined) {
        c.add({ id: `tab:${tabIdAt(bTabs, bi)}-removed`, unit: 'tab', op: 'removed', label: `Tab${quoted(oldName)}`, ...textual(oldName, undefined) }, removedRestore(bi))
      }
    }
    const blockPairs = Math.min(removed.length, added.length)
    for (let k = 0; k < Math.max(removed.length, added.length); k += 1) {
      const bi = removed[k]
      const ai = added[k]
      if (k < blockPairs) {
        const label = blockLabel(ab[ai], ai)
        c.add(
          { id: `block:${ai + 1}`, unit: 'block', op: 'changed', label, ...textual(bb[bi], ab[ai]), ...groupOf(aTabs, ai) },
          changedRestore(bi, ai, label),
        )
      } else if (ai !== undefined) {
        const label = blockLabel(ab[ai], ai)
        c.add({ id: `block:${ai + 1}`, unit: 'block', op: 'added', label, ...textual(undefined, ab[ai]), ...groupOf(aTabs, ai) }, addedRestore(ai, label))
      } else if (bi !== undefined) {
        c.add(
          { id: `block:${bi + 1}-removed`, unit: 'block', op: 'removed', label: blockLabel(bb[bi], bi), ...textual(bb[bi], undefined), ...groupOf(bTabs, bi) },
          removedRestore(bi),
        )
      }
    }
  }
}

type JsonDiff = (b: Obj, a: Obj, c: Collector, ctx: Ctx) => void

function slideLabel(slide: Obj | undefined, index: number): string {
  const title = slide && typeof slide.title === 'string' ? slide.title : undefined
  return `Slide ${index + 1}${quoted(title)}`
}

const SLIDE_ELEMENT_NAMES: Record<string, string> = { text: 'Text box', rect: 'Rectangle', ellipse: 'Ellipse', line: 'Line', arrow: 'Arrow', image: 'Image' }

function slideElementLabel(el: Obj | undefined): string {
  const type = el && typeof el.type === 'string' ? el.type : 'element'
  const text = el && typeof el.text === 'string' ? el.text : el && typeof el.alt === 'string' ? el.alt : undefined
  return `${SLIDE_ELEMENT_NAMES[type] ?? capitalise(type)}${quoted(text)}`
}

const slidesAt = rootList('slides', 'Slide')

const elementsOf = (slideId: string): ListAt => ({
  noun: 'Element',
  get: (root, create) => {
    const slides = root.slides
    if (!Array.isArray(slides)) return 'The current version has no slides'
    const slide = slides.find((s) => idOf(s) === slideId)
    if (!isObj(slide)) return `Slide "${slideId}" is no longer in the current version`
    if (slide.elements === undefined && create) slide.elements = []
    return Array.isArray(slide.elements) ? slide.elements : `Slide "${slideId}" has no elements`
  },
})

const diffSlides: JsonDiff = (b, a, c) => {
  topLevel(b, a, ['slides'], c)
  const items = diffKeyed(b.slides, a.slides, { skip: ['elements'] })
  if (!items) throw new NotComparable()
  const beforeIds = idsOf(b.slides)
  const labelOf = (d: ItemDiff) => slideLabel(d.after ?? d.before, d.op === 'removed' ? d.beforeIndex : d.afterIndex)
  for (const d of items) {
    c.add(
      { id: `slide:${d.id}`, unit: 'slide', op: d.op, label: labelOf(d), ...itemFields(d), ...itemText(d) },
      jsonRestorer((root) => restoreKeyed(root, slidesAt, d, beforeIds, { min: 1 })),
    )
  }
  // Elements of the slides both versions hold.
  const bSlides = keyed(Array.isArray(b.slides) ? b.slides : [])
  const aSlides = Array.isArray(a.slides) ? a.slides : []
  aSlides.forEach((slide, index) => {
    const id = idOf(slide)
    if (!id || !isObj(slide)) return
    const old = bSlides?.get(id)
    if (!isObj(old)) return
    const elements = diffKeyed(old.elements, slide.elements, { geometry: ['x', 'y'] })
    if (!elements) throw new NotComparable()
    const group = slideLabel(slide, index)
    const elementIds = idsOf(old.elements)
    for (const d of elements) {
      c.add(
        { id: `element:${id}/${d.id}`, unit: 'element', op: d.op, label: slideElementLabel(d.after ?? d.before), group, ...itemFields(d), ...itemText(d) },
        jsonRestorer((root) => restoreKeyed(root, elementsOf(id), d, elementIds)),
      )
    }
  })
}

/** A formula with its A1 references blanked, so a row insert's reference shifts still match. */
const normalizeFormula = (f: string) => f.replace(/\$?\b[A-Z]{1,3}\$?\d+\b/g, '#')

const cellValueKey = (cell: unknown): string => {
  if (!isObj(cell)) return ''
  if (typeof cell.f === 'string') return `f${normalizeFormula(cell.f)}`
  if (cell.v === undefined || cell.v === '') return ''
  return `v${typeof cell.v}:${String(cell.v)}`
}

const cellText = (cell: unknown): string => {
  if (!isObj(cell)) return ''
  if (typeof cell.f === 'string') return cell.f
  return cell.v === undefined ? '' : String(cell.v)
}

/** v/f/runs equal; in a shifted tab, formulas equal up to their references. */
function sameCellValue(b: Obj | undefined, a: Obj | undefined, shifted: boolean): boolean {
  const pick = (cell: Obj | undefined) => {
    const out: Obj = {}
    if (cell) for (const k of CELL_VALUE) if (has(cell, k)) out[k] = cell[k]
    return out
  }
  const [pb, pa] = [pick(b), pick(a)]
  if (deepEqual(pb, pa)) return true
  if (!shifted || typeof pb.f !== 'string' || typeof pa.f !== 'string') return false
  return normalizeFormula(pb.f) === normalizeFormula(pa.f) && deepEqual({ ...pb, f: '' }, { ...pa, f: '' })
}

type Grid = { cells: Map<string, Obj>; at: Map<number, Map<number, Obj>>; rows: number; cols: number }

function gridOf(cells: unknown): Grid {
  const out: Grid = { cells: new Map(), at: new Map(), rows: 0, cols: 0 }
  if (!isObj(cells)) return out
  for (const [address, cell] of Object.entries(cells)) {
    const pos = parseAddress(address)
    if (!pos || !isObj(cell)) continue
    out.cells.set(address, cell)
    let row = out.at.get(pos.row)
    if (!row) out.at.set(pos.row, (row = new Map()))
    row.set(pos.col, cell)
    out.rows = Math.max(out.rows, pos.row + 1)
    out.cols = Math.max(out.cols, pos.col + 1)
  }
  return out
}

type LineAlign = { map: Map<number, number>; inserted: number[]; deleted: number[] }

/** Align two signature lists (rows or columns), padded to one length; blank extras are dropped. */
function alignLines(bSigs: string[], aSigs: string[]): LineAlign {
  const n = Math.max(bSigs.length, aSigs.length)
  const pad = (sigs: string[]) => [...sigs, ...new Array<string>(n - sigs.length).fill('')]
  const [bs, as] = [pad(bSigs), pad(aSigs)]
  const map = new Map<number, number>()
  const inserted: number[] = []
  const deleted: number[] = []
  const pairs = lcsPairs(bs, as, ALIGN_LCS_CELLS)
  if (!pairs) {
    for (let i = 0; i < n; i += 1) map.set(i, i)
    return { map, inserted, deleted }
  }
  for (const [i, j] of pairs) map.set(i, j)
  for (const h of hunksOf(pairs, n, n)) {
    const matched = pairWithin(bs, as, h)
    const left = new Set(matched.map(([i]) => i))
    const right = new Set(matched.map(([, j]) => j))
    for (const [i, j] of matched) map.set(i, j)
    for (let i = h.aStart; i < h.aEnd; i += 1) if (!left.has(i) && bs[i] !== '') deleted.push(i)
    for (let j = h.bStart; j < h.bEnd; j += 1) if (!right.has(j) && as[j] !== '') inserted.push(j)
  }
  return { map, inserted, deleted }
}

/**
 * Which lines of a hunk are the same line edited. As many on each side: by position (edits in
 * place). Otherwise the order-keeping pairing sharing the most cells, so deleting row 2 and
 * editing row 3 pairs the old row 3 with the new row 2, not row 2 with it.
 */
function pairWithin(bs: readonly string[], as: readonly string[], h: Hunk): Array<[number, number]> {
  const dels = h.aEnd - h.aStart
  const adds = h.bEnd - h.bStart
  const out: Array<[number, number]> = []
  if (dels === adds || dels * adds > WORD_LCS_CELLS) {
    for (let k = 0; k < Math.min(dels, adds); k += 1) out.push([h.aStart + k, h.bStart + k])
    return out
  }
  const parts = (sig: string) => new Set(sig ? sig.split('\u0001') : [])
  const bp = Array.from({ length: dels }, (_v, i) => parts(bs[h.aStart + i]))
  const ap = Array.from({ length: adds }, (_v, j) => parts(as[h.bStart + j]))
  const sim = (i: number, j: number) => {
    let n = 0
    for (const p of bp[i]) if (ap[j].has(p)) n += 1
    return n
  }
  const width = adds + 1
  const best = new Float64Array((dels + 1) * width)
  for (let i = dels - 1; i >= 0; i -= 1) {
    for (let j = adds - 1; j >= 0; j -= 1) {
      const s = sim(i, j)
      best[i * width + j] = Math.max(best[(i + 1) * width + j], best[i * width + j + 1], s > 0 ? best[(i + 1) * width + j + 1] + s : 0)
    }
  }
  let i = 0
  let j = 0
  while (i < dels && j < adds) {
    const s = sim(i, j)
    if (s > 0 && best[i * width + j] === best[(i + 1) * width + j + 1] + s) {
      out.push([h.aStart + i, h.bStart + j])
      i += 1
      j += 1
    } else if (best[(i + 1) * width + j] >= best[i * width + j + 1]) i += 1
    else j += 1
  }
  return out
}

type TabAlign = { rows: LineAlign; cols: LineAlign; shifted: boolean }

function alignTab(b: Grid, a: Grid): TabAlign {
  const rowSig = (g: Grid, r: number) => {
    const row = g.at.get(r)
    if (!row) return ''
    return [...row.entries()]
      .map(([col, cell]) => [col, cellValueKey(cell)] as const)
      .filter(([, key]) => key !== '')
      .sort((x, y) => x[0] - y[0])
      .map(([col, key]) => `${col}=${key}`)
      .join('\u0001')
  }
  const rows = alignLines(
    Array.from({ length: b.rows }, (_v, r) => rowSig(b, r)),
    Array.from({ length: a.rows }, (_v, r) => rowSig(a, r)),
  )
  // Columns by their content on the rows both sides hold (keyed by the after row).
  const colSig = (g: Grid, c: number, rowKey: (r: number) => number | undefined) => {
    const parts: string[] = []
    for (const [r, row] of g.at) {
      const key = rowKey(r)
      const cell = row.get(c)
      if (key === undefined || !cell) continue
      const value = cellValueKey(cell)
      if (value) parts.push(`${key}=${value}`)
    }
    return parts.sort().join('\u0001')
  }
  const mappedAfter = new Set(rows.map.values())
  const cols = alignLines(
    Array.from({ length: b.cols }, (_v, c) => colSig(b, c, (r) => rows.map.get(r))),
    Array.from({ length: a.cols }, (_v, c) => colSig(a, c, (r) => (mappedAfter.has(r) ? r : undefined))),
  )
  const moved = (line: LineAlign) => line.inserted.length > 0 || line.deleted.length > 0 || [...line.map].some(([i, j]) => i !== j)
  return { rows, cols, shifted: moved(rows) || moved(cols) }
}

const tabsAt = rootList('tabs', 'Tab')

function tabNameTaken(root: Obj, name: unknown, exceptId: string): boolean {
  if (typeof name !== 'string' || !Array.isArray(root.tabs)) return false
  return root.tabs.some((t) => isObj(t) && t.id !== exceptId && typeof t.name === 'string' && t.name.toLowerCase() === name.toLowerCase())
}

function rowText(g: Grid, r: number, colMap?: (c: number) => number): string {
  const row = g.at.get(r)
  if (!row) return ''
  return [...row.entries()]
    .sort((x, y) => x[0] - y[0])
    .map(([col, cell]) => `${toAddress({ row: r, col: colMap ? colMap(col) : col })}: ${cellText(cell)}`)
    .join(' | ')
}

function colText(g: Grid, col: number): string {
  const parts: string[] = []
  for (const r of [...g.at.keys()].sort((x, y) => x - y)) {
    const cell = g.at.get(r)?.get(col)
    if (cell) parts.push(`${toAddress({ row: r, col })}: ${cellText(cell)}`)
  }
  return parts.join(' | ')
}

const diffSheet: JsonDiff = (b, a, c) => {
  topLevel(b, a, ['tabs', 'revision', 'deletedTabs'], c)
  const items = diffKeyed(b.tabs, a.tabs, { skip: ['cells'] })
  if (!items) throw new NotComparable()
  const beforeIds = idsOf(b.tabs)
  const nameOf = (tab: Obj | undefined, id: string) => (tab && typeof tab.name === 'string' ? tab.name : id)
  const checkName = (root: Obj, item: Obj) =>
    tabNameTaken(root, item.name, String(item.id)) ? `Another tab is already named “${String(item.name)}”` : null

  for (const d of items) {
    const name = nameOf(d.after ?? d.before, d.id)
    if (d.op === 'added' || d.op === 'removed') {
      const count = (tab: Obj | undefined) => (tab && isObj(tab.cells) ? Object.keys(tab.cells).length : 0)
      const text = (tab: Obj | undefined) => `${nameOf(tab, d.id)} (${count(tab)} cells)`
      c.add(
        {
          id: `tab:${d.id}`,
          unit: 'tab',
          op: d.op,
          label: `Tab${quoted(name)}`,
          ...(d.op === 'added' ? textual(undefined, text(d.after)) : textual(text(d.before), undefined)),
        },
        jsonRestorer((root) =>
          restoreKeyed(root, tabsAt, d, beforeIds, {
            min: 1,
            checkPut: checkName,
            checkRemove: (r, id) =>
              Array.isArray(r.names) && r.names.some((n) => isObj(n) && n.tab === id) ? 'A defined name still points at that tab' : null,
          }),
        ),
      )
      continue
    }
    // A tab both hold: its name and place, then its other settings, then its cells.
    const renamed = d.fields.includes('name')
    const ordered = d.fields.includes('order')
    if (renamed || ordered) {
      const sub: ItemDiff = { ...d, fields: d.fields.filter((f) => f === 'name' || f === 'order') }
      c.add(
        {
          id: `tab:${d.id}`,
          unit: 'tab',
          op: renamed ? 'renamed' : 'moved',
          label: `Tab${quoted(name)}`,
          fields: sub.fields,
          ...(renamed ? textual(nameOf(d.before, d.id), name) : {}),
        },
        jsonRestorer((root) => restoreKeyed(root, tabsAt, sub, beforeIds, { checkPut: checkName })),
      )
    }
    const settings = d.fields.filter((f) => f !== 'name' && f !== 'order')
    if (settings.length > 0) {
      const sub: ItemDiff = { ...d, op: 'changed', fields: settings }
      c.add(
        { id: `format:${d.id}`, unit: 'format', op: 'changed', label: `Layout of tab${quoted(name)}`, group: name, fields: settings, ...itemText(sub) },
        jsonRestorer((root) => restoreKeyed(root, tabsAt, sub, beforeIds)),
      )
    }
  }

  const bTabs = keyed(Array.isArray(b.tabs) ? b.tabs : [])
  for (const tab of Array.isArray(a.tabs) ? a.tabs : []) {
    const id = idOf(tab)
    const old = id ? bTabs?.get(id) : undefined
    if (!id || !isObj(tab) || !isObj(old)) continue
    diffTabCells(id, nameOf(tab, id), old.cells, tab.cells, c)
  }
}

function diffTabCells(tabId: string, tabName: string, beforeCells: unknown, afterCells: unknown, c: Collector) {
  if (deepEqual(beforeCells, afterCells)) return
  const bg = gridOf(beforeCells)
  const ag = gridOf(afterCells)
  const align = alignTab(bg, ag)
  const group = tabName
  const notAlone = 'Rows or columns of this tab were inserted or deleted between the versions'

  for (const r of align.rows.inserted) {
    c.add({ id: `row:${tabId}:${r + 1}`, unit: 'row', op: 'added', label: `Row ${r + 1}`, group, ...textual(undefined, rowText(ag, r)) }, undefined, notAlone)
  }
  for (const r of align.rows.deleted) {
    c.add({ id: `row:${tabId}:${r + 1}-removed`, unit: 'row', op: 'removed', label: `Row ${r + 1}`, group, ...textual(rowText(bg, r), undefined) }, undefined, notAlone)
  }
  for (const col of align.cols.inserted) {
    c.add({ id: `column:${tabId}:${colName(col)}`, unit: 'column', op: 'added', label: `Column ${colName(col)}`, group, ...textual(undefined, colText(ag, col)) }, undefined, notAlone)
  }
  for (const col of align.cols.deleted) {
    c.add(
      { id: `column:${tabId}:${colName(col)}-removed`, unit: 'column', op: 'removed', label: `Column ${colName(col)}`, group, ...textual(colText(bg, col), undefined) },
      undefined,
      notAlone,
    )
  }

  // Cells on rows and columns both sides hold, before-address → after-address.
  const pairs: Array<{ from: string; to: string; b?: Obj; a?: Obj }> = []
  const seen = new Set<string>()
  for (const [address, cell] of bg.cells) {
    const pos = parseAddress(address)
    if (!pos) continue
    const row = align.rows.map.get(pos.row)
    const col = align.cols.map.get(pos.col)
    if (row === undefined || col === undefined) continue
    const to = toAddress({ row, col })
    seen.add(to)
    pairs.push({ from: address, to, b: cell, a: ag.cells.get(to) })
  }
  const rowsBack = new Map([...align.rows.map].map(([i, j]) => [j, i]))
  const colsBack = new Map([...align.cols.map].map(([i, j]) => [j, i]))
  for (const [address, cell] of ag.cells) {
    if (seen.has(address)) continue
    const pos = parseAddress(address)
    if (!pos) continue
    const row = rowsBack.get(pos.row)
    const col = colsBack.get(pos.col)
    if (row === undefined || col === undefined) continue
    pairs.push({ from: toAddress({ row, col }), to: address, a: cell })
  }
  pairs.sort((x, y) => {
    const [px, py] = [parseAddress(x.to), parseAddress(y.to)]
    return (px?.row ?? 0) - (py?.row ?? 0) || (px?.col ?? 0) - (py?.col ?? 0)
  })

  for (const p of pairs) {
    if (deepEqual(p.b, p.a)) continue
    const valueSame = sameCellValue(p.b, p.a, align.shifted)
    const formatKeys = [...new Set([...Object.keys(p.b ?? {}), ...Object.keys(p.a ?? {})])].filter(
      (k) => !(CELL_VALUE as readonly string[]).includes(k) && !deepEqual(p.b?.[k], p.a?.[k]),
    )
    if (valueSame && formatKeys.length === 0) continue
    // Formatting a blank cell (or clearing a blank cell's format) is format-only too.
    const formatOnly = valueSame
    const op: ArtifactDiffOp = formatOnly ? 'changed' : p.b === undefined ? 'added' : p.a === undefined ? 'removed' : 'changed'
    const label = `Cell ${tabName}!${p.to}`
    const texts = formatOnly
      ? textual(renderFields(p.b, formatKeys) ?? '', renderFields(p.a, formatKeys) ?? '')
      : {
          ...(p.b !== undefined ? { before: clip(cellText(p.b)) } : {}),
          ...(p.a !== undefined ? { after: clip(cellText(p.a)) } : {}),
          ...(typeof p.b?.v === 'string' && typeof p.a?.v === 'string' ? { words: wordDiff(clip(p.b.v), clip(p.a.v)) } : {}),
        }
    const input: ChangeInput = {
      id: `cell:${tabId}!${p.to}`,
      unit: 'cell',
      op,
      label,
      group,
      ...texts,
      ...(formatOnly ? { formatOnly: true, fields: formatKeys } : {}),
    }
    if (align.shifted) {
      c.add(input, undefined, notAlone)
      continue
    }
    const address = p.to
    const old = p.b
    const afterTab = afterCells
    c.add(
      input,
      jsonRestorer((root) => {
        const tabs = root.tabs
        const tab = Array.isArray(tabs) ? tabs.find((t) => idOf(t) === tabId) : undefined
        if (!isObj(tab)) return `Tab “${tabName}” is no longer in the current version`
        if (!isObj(tab.cells)) tab.cells = {}
        const cells = tab.cells as Obj
        if (alignTab(gridOf(afterTab), gridOf(cells)).shifted) return `Rows or columns of tab “${tabName}” moved since; restore by hand`
        if (old === undefined) delete cells[address]
        else cells[address] = clone(old)
        return null
      }),
    )
  }
}

const DATA_LISTS = [
  { key: 'sources', unit: 'source', noun: 'Source', min: 1 },
  { key: 'datasets', unit: 'dataset', noun: 'Dataset', min: 0 },
  { key: 'blocks', unit: 'block', noun: 'Block', min: 1 },
  { key: 'filters', unit: 'filter', noun: 'Filter', min: 0 },
  { key: 'params', unit: 'param', noun: 'Parameter', min: 0 },
] as const

const BLOCK_KIND_NAMES: Record<string, string> = { kpi: 'KPI', chart: 'Chart', table: 'Table', text: 'Text', section: 'Section' }

function dataLabel(unit: string, noun: string, item: Obj | undefined, id: string): string {
  const text = (['label', 'title', 'markdown'] as const).map((k) => item?.[k]).find((v): v is string => typeof v === 'string' && v !== '')
  if (unit === 'block') {
    const kind = item && typeof item.kind === 'string' ? item.kind : 'block'
    return `${BLOCK_KIND_NAMES[kind] ?? capitalise(kind)} block${quoted(text ?? id)}`
  }
  return `${noun}${quoted(text ?? id)}`
}

/** Ids a dashboard item points at: its source, its dataset(s). */
function dataRefs(item: Obj): Array<{ list: 'sources' | 'datasets'; id: string }> {
  const out: Array<{ list: 'sources' | 'datasets'; id: string }> = []
  if (typeof item.sourceId === 'string') out.push({ list: 'sources', id: item.sourceId })
  if (typeof item.datasetId === 'string') out.push({ list: 'datasets', id: item.datasetId })
  if (Array.isArray(item.datasetIds)) for (const id of item.datasetIds) if (typeof id === 'string') out.push({ list: 'datasets', id })
  return out
}

function dataCheckPut(root: Obj, item: Obj): string | null {
  for (const ref of dataRefs(item)) {
    if (!idsOf(root[ref.list]).includes(ref.id)) return `It needs ${ref.list === 'sources' ? 'source' : 'dataset'} "${ref.id}", which the current version does not have`
  }
  return null
}

function dataCheckRemove(listKey: string) {
  return (root: Obj, id: string): string | null => {
    if (listKey !== 'sources' && listKey !== 'datasets') return null
    for (const key of ['datasets', 'blocks', 'filters']) {
      const list = root[key]
      if (!Array.isArray(list)) continue
      for (const item of list) {
        if (isObj(item) && idOf(item) !== id && dataRefs(item).some((ref) => ref.list === listKey && ref.id === id)) {
          return `${capitalise(key.slice(0, -1))} "${String(item.id)}" still uses it`
        }
      }
    }
    return null
  }
}

/** A dataset's row-count delta and a few changed rows, matched by its key columns or `id`. */
function datasetDetail(b: Obj, a: Obj, samples: number): string[] {
  const bRows = Array.isArray(b.rows) ? b.rows : []
  const aRows = Array.isArray(a.rows) ? a.rows : []
  if (deepEqual(bRows, aRows)) return []
  const out = [`rows ${bRows.length} → ${aRows.length}`]
  if (samples <= 0) return out
  const keyCols = (Array.isArray(a.key) ? a.key : Array.isArray(b.key) ? b.key : null)?.filter((k): k is string => typeof k === 'string')
  const columns = Array.isArray(a.columns) ? a.columns : []
  const hasId = columns.some((col) => isObj(col) && col.key === 'id')
  const cols = keyCols && keyCols.length > 0 ? keyCols : hasId ? ['id'] : null
  const keyOf = (row: unknown, index: number) =>
    cols ? cols.map((k) => `${k}=${isObj(row) ? show(row[k]) : ''}`).join(', ') : `${index + 1}`
  const bm = new Map(bRows.map((row, i) => [keyOf(row, i), row]))
  const am = new Map(aRows.map((row, i) => [keyOf(row, i), row]))
  const lines: string[] = []
  for (const [key, row] of am) {
    if (lines.length >= samples) break
    const old = bm.get(key)
    if (old === undefined) {
      lines.push(`row ${key} added`)
      continue
    }
    if (deepEqual(old, row) || !isObj(old) || !isObj(row)) continue
    const diffs = [...new Set([...Object.keys(old), ...Object.keys(row)])]
      .filter((k) => !deepEqual(old[k], row[k]))
      .map((k) => `${k} ${show(old[k] ?? null)} → ${show(row[k] ?? null)}`)
    lines.push(`row ${key}: ${diffs.join(', ')}`)
  }
  for (const key of bm.keys()) {
    if (lines.length >= samples) break
    if (!am.has(key)) lines.push(`row ${key} removed`)
  }
  return [...out, ...lines.map((line) => clip(line, 300))]
}

const diffData: JsonDiff = (b, a, c, ctx) => {
  topLevel(b, a, DATA_LISTS.map((l) => l.key), c)
  for (const spec of DATA_LISTS) {
    const items = diffKeyed(b[spec.key], a[spec.key])
    if (!items) throw new NotComparable()
    const beforeIds = idsOf(b[spec.key])
    const at = rootList(spec.key, spec.noun)
    const render = (key: string, value: unknown) => (key === 'rows' && Array.isArray(value) ? `${value.length} rows` : show(value))
    for (const d of items) {
      const detail = spec.unit === 'dataset' && d.before && d.after ? datasetDetail(d.before, d.after, ctx.rowSamples) : []
      c.add(
        {
          id: `${spec.unit}:${d.id}`,
          unit: spec.unit,
          op: d.op,
          label: dataLabel(spec.unit, spec.noun, d.after ?? d.before, d.id),
          ...itemFields(d),
          ...(d.op === 'added' || d.op === 'removed' ? itemText(d) : itemText(d, render)),
          ...(detail.length > 0 ? { detail } : {}),
        },
        jsonRestorer((root) =>
          restoreKeyed(root, at, d, beforeIds, { min: spec.min, checkPut: dataCheckPut, checkRemove: dataCheckRemove(spec.key) }),
        ),
      )
    }
  }
}

const elementsAt = rootList('elements', 'Element')

  if (item.type !== 'arrow') return null
  const present = new Set(idsOf(root.elements))
  for (const end of [item.from, item.to]) {
    const bound = boundId(end)
    if (bound !== null && !present.has(bound)) return `The connector is bound to "${bound}", which the current version does not have`
  }
  return null
}

        }),
      ),
    )
  }
}

type FlatNode = { props: Obj; kids: string[] | null; parent: string; board: string }
type FlatDesign = { boards: Obj[]; nodes: Map<string, FlatNode>; originals: Map<string, Obj> }

const ROOT_PARENT = (boardId: string) => `board:${boardId}`

// re-derived from artifact-merge.ts (flattenDesign), not exported there; null instead of throwing
function flattenDesign(design: Obj): FlatDesign | null {
  const nodes = new Map<string, FlatNode>()
  const originals = new Map<string, Obj>()
  if (design.boards !== undefined && !Array.isArray(design.boards)) return null
  const boards: Obj[] = []
  for (const board of (design.boards as unknown[] | undefined) ?? []) {
    if (!isObj(board) || typeof board.id !== 'string' || !isObj(board.root)) return null
    const boardId = board.id
    let bad = false
    const visit = (node: unknown, parent: string): string => {
      const id = idOf(node)
      if (id === null || nodes.has(id)) {
        bad = true
        return ''
      }
      const { children, ...props } = node as Obj
      const flat: FlatNode = { props, kids: null, parent, board: boardId }
      nodes.set(id, flat)
      originals.set(id, node as Obj)
      if (children !== undefined) {
        if (!Array.isArray(children)) {
          bad = true
          return id
        }
        flat.kids = children.map((child) => visit(child, id))
      }
      return id
    }
    const { root, ...rest } = board
    const rootId = visit(root, ROOT_PARENT(boardId))
    if (bad) return null
    boards.push({ ...rest, root: rootId })
  }
  return { boards, nodes, originals }
}

const NODE_NAMES: Record<string, string> = {
  frame: 'Frame',
  text: 'Text',
  image: 'Image',
  icon: 'Icon',
  button: 'Button',
  input: 'Input',
  divider: 'Divider',
  shape: 'Shape',
  instance: 'Instance',
}

function nodeLabel(props: Obj | undefined, id: string): string {
  const type = props && typeof props.type === 'string' ? props.type : 'node'
  const text = (['name', 'text', 'label', 'placeholder', 'alt'] as const).map((k) => props?.[k]).find((v): v is string => typeof v === 'string' && v !== '')
  return `${NODE_NAMES[type] ?? capitalise(type)}${quoted(text ?? id)}`
}

type Found = { node: Obj; parent: Obj | null; board: Obj }

/** Every node of the current design's trees by id (first seen wins). */
function indexTree(root: Obj): Map<string, Found> {
  const out = new Map<string, Found>()
  const boards = Array.isArray(root.boards) ? root.boards : []
  for (const board of boards) {
    if (!isObj(board) || !isObj(board.root)) continue
    const visit = (node: unknown, parent: Obj | null) => {
      const id = idOf(node)
      if (id === null || out.has(id)) return
      out.set(id, { node: node as Obj, parent, board })
      const kids = (node as Obj).children
      if (Array.isArray(kids)) for (const kid of kids) visit(kid, node as Obj)
    }
    visit(board.root, null)
  }
  return out
}

/** Ids of a subtree (the node and everything in it). */
function subtreeIds(node: unknown, out: Set<string> = new Set()): Set<string> {
  const id = idOf(node)
  if (id !== null) out.add(id)
  if (isObj(node) && Array.isArray(node.children)) for (const kid of node.children) subtreeIds(kid, out)
  return out
}

/** `node` without any descendant whose id `taken` holds. */
function pruned(node: Obj, taken: Set<string>): Obj {
  if (!Array.isArray(node.children)) return clone(node)
  const children = node.children.filter((kid) => !taken.has(idOf(kid) ?? '')).map((kid) => (isObj(kid) ? pruned(kid, taken) : kid))
  return { ...clone(node), children }
}

/** Where a node goes back among `siblings`: after the nearest before-sibling still there. */
function placeAmong(siblings: unknown[], beforeSiblings: readonly string[], id: string): number {
  return placeIndex(siblings, beforeSiblings, id)
}

function allStrings(value: unknown, out: Set<string> = new Set()): Set<string> {
  if (typeof value === 'string') out.add(value)
  else if (Array.isArray(value)) for (const v of value) allStrings(v, out)
  else if (isObj(value)) for (const v of Object.values(value)) allStrings(v, out)
  return out
}

const diffDesign: JsonDiff = (b, a, c) => {
  const fb = flattenDesign(b)
  const fa = flattenDesign(a)
  if (!fb || !fa) throw new NotComparable()
  topLevel(b, a, ['boards', 'tokens', 'styles', 'pages', 'notes'], c)

  // Tokens: each colour by name, every other token key whole.
  const bt = isObj(b.tokens) ? b.tokens : {}
  const at = isObj(a.tokens) ? a.tokens : {}
  const bColors = isObj(bt.colors) ? bt.colors : {}
  const aColors = isObj(at.colors) ? at.colors : {}
  const tokenChange = (id: string, label: string, bv: unknown, av: unknown, set: (root: Obj, tokens: Obj) => string | null) => {
    const op: ArtifactDiffOp = bv === undefined ? 'added' : av === undefined ? 'removed' : 'changed'
    c.add(
      { id, unit: 'token', op, label, ...textual(bv === undefined ? undefined : show(bv), av === undefined ? undefined : show(av)) },
      jsonRestorer((root) => {
        if (!isObj(root.tokens)) root.tokens = { colors: {} }
        return set(root, root.tokens as Obj)
      }),
    )
  }
  for (const name of [...new Set([...Object.keys(bColors), ...Object.keys(aColors)])]) {
    if (deepEqual(bColors[name], aColors[name])) continue
    tokenChange(`token:colors.${name}`, `Colour token${quoted(name)}`, bColors[name], aColors[name], (root, tokens) => {
      if (!isObj(tokens.colors)) tokens.colors = {}
      const colors = tokens.colors as Obj
      if (bColors[name] === undefined) {
        // Taking a colour out must not leave a board, node or style naming it.
        if (allStrings([root.boards, root.styles]).has(name)) return `Colour token “${name}” is still used`
        delete colors[name]
      } else colors[name] = clone(bColors[name])
      return null
    })
  }
  for (const key of [...new Set([...Object.keys(bt), ...Object.keys(at)])]) {
    if (key === 'colors' || deepEqual(bt[key], at[key])) continue
    tokenChange(`token:${key}`, `Token${quoted(key)}`, bt[key], at[key], (_root, tokens) => {
      if (bt[key] === undefined) delete tokens[key]
      else tokens[key] = clone(bt[key])
      return null
    })
  }

  // Shared styles, pages and notes: keyed lists.
  const bStyles = isObj(b.styles) ? b.styles : {}
  const aStyles = isObj(a.styles) ? a.styles : {}
  for (const kind of ['text', 'fills'] as const) {
    const items = diffKeyed(bStyles[kind], aStyles[kind])
    if (!items) throw new NotComparable()
    const beforeIds = idsOf(bStyles[kind])
    const ref = kind === 'text' ? 'textStyle' : 'fillStyle'
    const listAt: ListAt = {
      noun: kind === 'text' ? 'Text style' : 'Fill style',
      get: (root, create) => {
        if (!isObj(root.styles)) {
          if (!create) return 'The current version has no styles'
          root.styles = {}
        }
        const styles = root.styles as Obj
        if (styles[kind] === undefined && create) styles[kind] = []
        return Array.isArray(styles[kind]) ? (styles[kind] as unknown[]) : `The current version has no ${kind} styles`
      },
    }
    for (const d of items) {
      const item = d.after ?? d.before
      const name = item && typeof item.name === 'string' ? item.name : d.id
      c.add(
        { id: `style:${kind}/${d.id}`, unit: 'style', op: d.op, label: `${listAt.noun}${quoted(name)}`, ...itemFields(d), ...itemText(d) },
        jsonRestorer((root) =>
          restoreKeyed(root, listAt, d, beforeIds, {
            checkRemove: (r, id) => ([...indexTree(r).values()].some((f) => f.node[ref] === id) ? `A node still uses ${listAt.noun.toLowerCase()} "${id}"` : null),
          }),
        ),
      )
    }
  }
  for (const spec of [
    { key: 'pages', unit: 'page', noun: 'Page' },
    { key: 'notes', unit: 'note', noun: 'Note' },
  ] as const) {
    const items = diffKeyed(b[spec.key], a[spec.key])
    if (!items) throw new NotComparable()
    const beforeIds = idsOf(b[spec.key])
    for (const d of items) {
      const item = d.after ?? d.before
      const text = item && typeof item.title === 'string' ? item.title : item && typeof item.text === 'string' ? item.text : d.id
      c.add(
        { id: `${spec.unit}:${d.id}`, unit: spec.unit, op: d.op, label: `${spec.noun}${quoted(text)}`, ...itemFields(d), ...itemText(d) },
        jsonRestorer((root) =>
          restoreKeyed(root, rootList(spec.key, spec.noun), d, beforeIds, {
            checkPut: (r, put) => (spec.key === 'notes' && typeof put.page === 'string' && !idsOf(r.pages).includes(put.page) ? `Page "${put.page}" is gone` : null),
            checkRemove: (r, id) =>
              spec.key === 'pages' && Array.isArray(r.boards) && r.boards.some((board) => isObj(board) && board.page === id) ? 'Boards still sit on that page' : null,
          }),
        ),
      )
    }
  }

  // Boards (their own settings; their nodes below).
  // A board's `root` (flattened to its id) differs only when the root was replaced by a new tree;
  // the nodes under a replaced root are then this board change, not node changes.
  const boardItems = diffKeyed(fb.boards, fa.boards, { geometry: ['x', 'y'] })
  if (!boardItems) throw new NotComparable()
  const boardIds = fb.boards.map((board) => String(board.id))
  const bOriginalBoards = keyed(Array.isArray(b.boards) ? b.boards : []) ?? new Map<string, unknown>()
  const titleOf = (board: Obj | undefined, id: string) => (board && typeof board.title === 'string' ? board.title : id)
  const boardTitle = (id: string) => titleOf(fa.boards.find((x) => x.id === id) ?? fb.boards.find((x) => x.id === id), id)
  for (const d of boardItems) {
    const original = bOriginalBoards.get(d.id)
    // Put a removed board (or a replaced root) back with its whole tree; take an added one out whole.
    const full: ItemDiff = d.op !== 'added' && isObj(original) ? { ...d, before: original } : d
    c.add(
      { id: `board:${d.id}`, unit: 'board', op: d.op, label: `Board${quoted(titleOf(d.after ?? d.before, d.id))}`, ...itemFields(d), ...itemText(d) },
      jsonRestorer((root) =>
        restoreKeyed(root, rootList('boards', 'Board'), full, boardIds, {
          checkPut: (r, put) => {
            if (d.op !== 'removed' && !d.fields.includes('root')) return null
            // Ids are unique design-wide: the old tree must not collide with nodes now elsewhere.
            const seen = new Set<string>()
            // A removed board is checked before it goes back in; a changed one is already in place.
            const roots = [...(d.op === 'removed' ? [put.root] : []), ...(Array.isArray(r.boards) ? r.boards : []).map((x) => (isObj(x) ? x.root : undefined))]
            for (const tree of roots) {
              for (const id of subtreeIds(tree)) {
                if (seen.has(id)) return `Node "${id}" of that board is now elsewhere in the design`
                seen.add(id)
              }
            }
            if (typeof put.page === 'string' && !idsOf(r.pages).includes(put.page)) return `Page "${put.page}" is gone`
            return null
          },
          checkRemove: (r, id) => (r.startBoard === id ? 'The prototype starts on that board' : null),
        }),
      ),
    )
  }

  // Nodes, design-wide. Reordered children of a frame both hold are moves.
  const reordered = new Set<string>()
  for (const [id, node] of fa.nodes) {
    const old = fb.nodes.get(id)
    if (!old || !node.kids || !old.kids) continue
    for (const kid of reorderedIds(old.kids, node.kids)) reordered.add(kid)
  }
  const restoreNode = (id: string, op: ItemOp, fields: string[]) =>
    jsonRestorer((root) => {
      const present = indexTree(root)
      const old = fb.nodes.get(id)
      const oldOriginal = fb.originals.get(id)
      if (op === 'added') {
        const found = present.get(id)
        if (!found) return `Node "${id}" is no longer in the current version`
        if (!found.parent || !Array.isArray(found.parent.children)) return 'A board’s root node cannot be removed'
        found.parent.children.splice(found.parent.children.indexOf(found.node), 1)
        return null
      }
      if (!old || !oldOriginal) return 'Nothing to put back'
      const parent = old.parent.startsWith('board:') ? null : present.get(old.parent)
      const siblings = old.parent.startsWith('board:') ? [] : fb.nodes.get(old.parent)?.kids ?? []
      if (op === 'removed') {

          op: 'added',
          label: nodeLabel(node.props, id),

        },
        restoreNode(id, 'added', []),
      )
      continue

    const order = !moved && reordered.has(id)

        op: 'removed',
        label: nodeLabel(old.props, id),
        group: boardTitle(old.board),

      },
      restoreNode(id, 'removed', []),
    )
  }

/**
 * Which image addresses an artifact may hold, shared by the kinds' validators
 * (artifact-kinds/slides.ts, whiteboard.ts, design.ts) and the renderers'
 * load decision (freebuff/web …/artifacts/remote-images.ts).
 *
 * Validation stays permissive on purpose (save and restore re-validate whole
 * stored bodies, so tightening would strand existing artifacts): an https URL,
 * or a same-origin path. Only the forms a browser would resolve OFF this
 * origin are refused: `//host`, `/\host`, a backslash anywhere, and an encoded
 * slash or backslash (`/%2F…`, `/%5C…`), plus markup characters in a path. Loading is the renderers' stricter
 * call: an uploaded asset (`/_asset/<uuid>`) always, a third-party https image
 * only after the reader opts in, anything else never.
 *
 * What an AGENT may add is stricter still (`unstoredImageSrcsAdded`, enforced by the store's write
 * pipeline for agent writes): only this artifact's stored assets, or the built-in placeholder. A
 * src the stored body already holds is kept, so existing bodies and people's own https images
 * still save; the agent imports a picture (import_artifact_image) instead of hotlinking it.
 */

import { ARTIFACT_ASSET_REF_PATTERN } from './artifacts'

const CONTROL_OR_SPACE = /[\u0000- \u007f-\u009f]/
const ENCODED_SEPARATOR = /%(?:2f|5c)/i
/** Never part of a real path (a browser percent-encodes them); refused so no export can be broken out of. */
const MARKUP = /[<>"'`]/

/** A path on this origin: one leading `/`, then a letter, digit, `_` or `.`; nothing that leaves the origin. */
export function isSameOriginImagePath(src: string): boolean {
  return (
    /^\/[A-Za-z0-9_.]/.test(src) &&
    !src.includes('\\') &&
    !CONTROL_OR_SPACE.test(src) &&
    !ENCODED_SEPARATOR.test(src) &&
    !MARKUP.test(src)
  )
}

/** An absolute `https:` URL with a host and no credentials, as the browser will read it. */
export function isRemoteImageUrl(src: string): boolean {
  if (!/^https:\/\//i.test(src) || src.includes('\\') || CONTROL_OR_SPACE.test(src)) return false
  try {
    const url = new URL(src)
    return url.protocol === 'https:' && url.hostname !== '' && url.username === '' && url.password === ''
  } catch {
    return false
  }
}

/** Image srcs that are ours to serve without a remote fetch: the placeholder new slides use. */
export const BUILT_IN_IMAGE_SRCS: ReadonlySet<string> = new Set(['/Placeholder.svg'])

/** An uploaded asset ref (`/_asset/<uuid>`) or a built-in image: what an agent write may add. */
export function isStoredImageSrc(src: string): boolean {
  return ARTIFACT_ASSET_REF_PATTERN.test(src) || BUILT_IN_IMAGE_SRCS.has(src)
}

const MARKDOWN_IMAGE = /!\[[^\]]*\]\(\s*<?([^\s)>]+)/g
const HTML_IMAGE = /<img\b[^>]*?\bsrc\s*=\s*["']([^"']+)["']/gi

/**
 * Every image src in a body, in order, distinct: a Markdown body's Markdown and HTML images, and
 * every `src` in a JSON kind (slide images and elements, design nodes and fills, whiteboard
 * images), except inside free-form records (`FREE_FORM_KEYS`: a chart's `data`, a design data
 * list's `rows`), whose columns are named by the author and may well be called "src". Empty for
 * an unreadable JSON body.
 */
export function bodyImageSrcs(format: 'markdown' | 'json', body: string): string[] {
  const found = new Set<string>()
  const note = (src: unknown) => {
    if (typeof src === 'string' && src) found.add(src)
  }
  if (format !== 'json') {
    for (const match of body.matchAll(MARKDOWN_IMAGE)) note(match[1])
    for (const match of body.matchAll(HTML_IMAGE)) note(match[1])
    return [...found]
  }
  let parsed: unknown
  try {
    parsed = JSON.parse(body)
  } catch {
    return []
  }
  const walk = (value: unknown, depth: number) => {
    if (depth > 64 || !value || typeof value !== 'object') return
    if (Array.isArray(value)) for (const item of value) walk(item, depth + 1)
    else
      for (const [key, field] of Object.entries(value)) {
        if (key === 'src') note(field)
        else if (!FREE_FORM_KEYS.has(key)) walk(field, depth + 1)
      }
  }
  walk(parsed, 0)
  return [...found]
}

/** Keys whose value is author-named data rows (chart.ts `data`, design.ts data lists' `rows`), never image holders. */
const FREE_FORM_KEYS: ReadonlySet<string> = new Set(['data', 'rows'])

/**
 * The image srcs `next` adds that are not stored images (`isStoredImageSrc`) and that `stored`
 * did not already hold: what an agent write may not introduce. `stored` is null for a new body.
 */
export function unstoredImageSrcsAdded(format: 'markdown' | 'json', stored: string | null, next: string): string[] {
  const had = new Set(stored === null ? [] : bodyImageSrcs(format, stored))
  return bodyImageSrcs(format, next).filter((src) => !isStoredImageSrc(src) && !had.has(src))
}

/**
 * Freebuff Data: calculated columns and parameters. A small, safe expression language (no
 * `eval`, no property access, no loops) that the schema checks and the app evaluates per row.
 *
 *   [revenue] / [users]                      a column, in square brackets (ids may hold "-")
 *   [price] * (1 + {growth})                 a dashboard parameter, in braces
 *   IF([latency] > {slow_ms}, "slow", "ok")  functions: IF, ROUND, ABS, MIN, MAX, COALESCE,
 *                                            AND, OR, NOT, FLOOR, CEIL
 *   [first] & " " & [last]                   & joins text
 *
 * Operators by precedence (low to high): comparison (= != <> < <= > >=), & (join), + -, * / %,
 * ^ (right-associative), unary - and +. A missing cell, a division by zero or anything
 * non-finite is null, and null spreads through arithmetic (COALESCE picks the first non-null).
 * A calculated column may read the dataset's stored columns and calculated columns listed
 * before it, never itself or a later one, so there are no cycles.
 *
 * Pure (no React, no DOM): the server validates with exactly the rules the app computes by.
 */

import type { Cell, Dashboard, DataColumn, DataRow, Dataset } from './data'

export const MAX_EXPR_LENGTH = 500
const MAX_DEPTH = 32

// ---- AST

export type CalcNode =
  | { t: 'num'; v: number }
  | { t: 'str'; v: string }
  | { t: 'col'; key: string }
  | { t: 'param'; id: string }
  | { t: 'neg'; x: CalcNode }
  | { t: 'bin'; op: string; a: CalcNode; b: CalcNode }
  | { t: 'call'; fn: CalcFunction; args: CalcNode[] }

export const CALC_FUNCTIONS = ['IF', 'ROUND', 'ABS', 'MIN', 'MAX', 'COALESCE', 'AND', 'OR', 'NOT', 'FLOOR', 'CEIL'] as const
export type CalcFunction = (typeof CALC_FUNCTIONS)[number]

const ARITY: Record<CalcFunction, [number, number]> = {
  IF: [2, 3],
  ROUND: [1, 2],
  ABS: [1, 1],
  MIN: [1, 16],
  MAX: [1, 16],
  COALESCE: [1, 16],
  AND: [1, 16],
  OR: [1, 16],
  NOT: [1, 1],
  FLOOR: [1, 1],
  CEIL: [1, 1],
}

export type CalcParse = { ok: true; node: CalcNode } | { ok: false; error: string }

type Token =
  | { k: 'num'; v: number }
  | { k: 'str'; v: string }
  | { k: 'col'; v: string }
  | { k: 'param'; v: string }
  | { k: 'name'; v: string }
  | { k: 'op'; v: string }

const ID = /^[A-Za-z0-9_-]{1,64}$/

function tokenize(source: string): Token[] | string {
  const out: Token[] = []
  let i = 0
  while (i < source.length) {
    const c = source[i]
    if (c === ' ' || c === '\t' || c === '\n' || c === '\r') {
      i++
      continue
    }
    if (c === '[' || c === '{') {
      const close = c === '[' ? ']' : '}'
      const end = source.indexOf(close, i + 1)
      if (end === -1) return `"${c}" at ${i + 1} has no closing "${close}"`
      const name = source.slice(i + 1, end).trim()
      if (!ID.test(name)) return `"${c}${name}${close}" is not a ${c === '[' ? 'column' : 'parameter'} id`
      out.push({ k: c === '[' ? 'col' : 'param', v: name })
      i = end + 1
      continue
    }
    if (c === '"') {
      let text = ''
      let j = i + 1
      for (; j < source.length; j++) {
        if (source[j] === '"') {
          if (source[j + 1] === '"') {
            text += '"'
            j++
            continue
          }
          break
        }
        text += source[j]
      }
      if (j >= source.length) return 'a text in quotes is not closed'
      out.push({ k: 'str', v: text })
      i = j + 1
      continue
    }
    const num = /^(\d+(\.\d*)?|\.\d+)([eE][+-]?\d+)?/.exec(source.slice(i))
    if (num) {
      out.push({ k: 'num', v: Number(num[0]) })
      i += num[0].length
      continue
    }
    const name = /^[A-Za-z_][A-Za-z0-9_]*/.exec(source.slice(i))
    if (name) {
      out.push({ k: 'name', v: name[0] })
      i += name[0].length
      continue
    }
    const two = source.slice(i, i + 2)
    if (two === '<=' || two === '>=' || two === '!=' || two === '<>') {
      out.push({ k: 'op', v: two === '<>' ? '!=' : two })
      i += 2
      continue
    }
    if ('+-*/%^&=<>(),'.includes(c)) {
      out.push({ k: 'op', v: c })
      i++
      continue
    }
    return `unexpected "${c}" at ${i + 1}`
  }
  return out
}

/** Parse an expression. Never throws; an error says what is wrong in plain words. */
export function parseCalc(source: string): CalcParse {
  if (source.length > MAX_EXPR_LENGTH) return { ok: false, error: `an expression is at most ${MAX_EXPR_LENGTH} characters` }
  const tokens = tokenize(source)
  if (typeof tokens === 'string') return { ok: false, error: tokens }
  if (tokens.length === 0) return { ok: false, error: 'the expression is empty' }
  let pos = 0
  let depth = 0
  const peek = () => tokens[pos]
  const isOp = (v: string) => {
    const t = tokens[pos]
    return t?.k === 'op' && t.v === v
  }
  class Fail extends Error {}
  const fail = (message: string): never => {
    throw new Fail(message)
  }
  const enter = () => {
    if (++depth > MAX_DEPTH) fail('the expression is nested too deeply')
  }

  const binary = (ops: string[], next: () => CalcNode) => (): CalcNode => {
    let left = next()
    for (let t = peek(); t?.k === 'op' && ops.includes(t.v); t = peek()) {
      pos++
      const op = t.v
      left = { t: 'bin', op, a: left, b: next() }
    }
    return left
  }

  const primary = (): CalcNode => {
    enter()
    try {
      const t = tokens[pos++]
      if (!t) return fail('the expression ends too soon')
      switch (t.k) {
        case 'num':
          return { t: 'num', v: t.v }
        case 'str':
          return { t: 'str', v: t.v }
        case 'col':
          return { t: 'col', key: t.v }
        case 'param':
          return { t: 'param', id: t.v }
        case 'name': {
          const fn = t.v.toUpperCase()
          if (fn === 'TRUE' || fn === 'FALSE') return { t: 'num', v: fn === 'TRUE' ? 1 : 0 }
          if (!(CALC_FUNCTIONS as readonly string[]).includes(fn)) {
            return fail(`unknown name "${t.v}" (columns go in [brackets], parameters in {braces})`)
          }
          if (!isOp('(')) return fail(`${fn} needs "(" after it`)
          pos++
          const args: CalcNode[] = []
          if (!isOp(')')) {
            for (;;) {
              args.push(comparison())
              if (isOp(',')) {
                pos++
                continue
              }
              break
            }
          }
          if (!isOp(')')) return fail(`${fn}( is not closed`)
          pos++
          const [min, max] = ARITY[fn as CalcFunction]
          if (args.length < min || args.length > max) {
            return fail(`${fn} takes ${min === max ? min : `${min} to ${max}`} argument${max === 1 ? '' : 's'}`)
          }
          return { t: 'call', fn: fn as CalcFunction, args }
        }
        case 'op':
          if (t.v === '(') {
            const inner = comparison()
            if (!isOp(')')) return fail('a "(" is not closed')
            pos++
            return inner
          }
          if (t.v === '-' || t.v === '+') {
            const x = power()
            return t.v === '-' ? { t: 'neg', x } : x
          }
          return fail(`unexpected "${t.v}"`)
      }
    } finally {
      depth--
    }
  }

  // ^ binds tightest and is right-associative; unary minus is read in primary.
  const power = (): CalcNode => {
    const base = primary()
    if (isOp('^')) {
      pos++
      enter()
      try {
        return { t: 'bin', op: '^', a: base, b: power() }
      } finally {
        depth--
      }
    }
    return base
  }
  const product = binary(['*', '/', '%'], power)
  const sum = binary(['+', '-'], product)
  const join = binary(['&'], sum)
  const comparison = binary(['=', '!=', '<', '<=', '>', '>='], join)

  try {
    const node = comparison()
    if (pos < tokens.length) {
      const t = tokens[pos]
      return { ok: false, error: `unexpected ${t.k === 'op' ? `"${t.v}"` : 'value'} after the expression` }
    }
    return { ok: true, node }
  } catch (error) {
    if (error instanceof Fail) return { ok: false, error: error.message }
    throw error
  }
}

/** Every column and parameter an expression reads. */
export function calcReferences(node: CalcNode): { columns: Set<string>; params: Set<string> } {
  const columns = new Set<string>()
  const params = new Set<string>()
  const walk = (n: CalcNode) => {
    if (n.t === 'col') columns.add(n.key)
    else if (n.t === 'param') params.add(n.id)
    else if (n.t === 'neg') walk(n.x)
    else if (n.t === 'bin') {
      walk(n.a)
      walk(n.b)
    } else if (n.t === 'call') n.args.forEach(walk)
  }
  walk(node)
  return { columns, params }
}

// ---- Evaluation

type Value = number | string | null

const own = (record: Record<string, unknown>, key: string) => Object.prototype.hasOwnProperty.call(record, key)

const finite = (n: number): number | null => (Number.isFinite(n) ? n : null)
const truthy = (v: Value) => (typeof v === 'number' ? v !== 0 : typeof v === 'string' ? v !== '' : false)
const asNumber = (v: Value): number | null => {
  if (typeof v === 'number') return v
  if (typeof v === 'string' && v.trim() !== '' && Number.isFinite(Number(v))) return Number(v)
  return null
}

/** One expression over one row. Never throws; anything undefined is null. */
export function evaluateCalc(node: CalcNode, row: DataRow, params: Record<string, number>): Value {
  switch (node.t) {
    case 'num':
      return node.v
    case 'str':
      return node.v
    case 'col': {
      const cell = own(row, node.key) ? row[node.key] : null
      return cell === undefined ? null : cell
    }
    case 'param':
      return own(params, node.id) && Number.isFinite(params[node.id]) ? params[node.id] : null
    case 'neg': {
      const x = asNumber(evaluateCalc(node.x, row, params))
      return x === null ? null : -x
    }
    case 'bin': {
      const a = evaluateCalc(node.a, row, params)
      const b = evaluateCalc(node.b, row, params)
      if (node.op === '&') return `${a ?? ''}${b ?? ''}`
      if (['=', '!=', '<', '<=', '>', '>='].includes(node.op)) {
        if (a === null || b === null) return node.op === '=' ? (a === b ? 1 : 0) : node.op === '!=' ? (a === b ? 0 : 1) : null
        const x = asNumber(a)
        const y = asNumber(b)
        const order =
          x !== null && y !== null ? x - y : String(a).localeCompare(String(b), 'en-US', { sensitivity: 'accent' })
        switch (node.op) {
          case '=':
            return order === 0 ? 1 : 0
          case '!=':
            return order !== 0 ? 1 : 0
          case '<':
            return order < 0 ? 1 : 0
          case '<=':
            return order <= 0 ? 1 : 0
          case '>':
            return order > 0 ? 1 : 0
          default:
            return order >= 0 ? 1 : 0
        }
      }
      const x = asNumber(a)
      const y = asNumber(b)
      if (x === null || y === null) return null
      switch (node.op) {
        case '+':
          return finite(x + y)
        case '-':
          return finite(x - y)
        case '*':
          return finite(x * y)
        case '/':
          return y === 0 ? null : finite(x / y)
        case '%':
          return y === 0 ? null : finite(x % y)
        default:
          return finite(x ** y)
      }
    }
    case 'call': {
      const arg = (i: number) => evaluateCalc(node.args[i], row, params)
      const nums = () => node.args.map((_, i) => asNumber(arg(i))).filter((n): n is number => n !== null)
      switch (node.fn) {
        case 'IF':
          return truthy(arg(0)) ? arg(1) : node.args.length > 2 ? arg(2) : null
        case 'ROUND': {
          const x = asNumber(arg(0))
          const digits = node.args.length > 1 ? asNumber(arg(1)) : 0
          if (x === null || digits === null) return null
          const f = 10 ** Math.max(-10, Math.min(10, Math.trunc(digits)))
          return finite(Math.round(x * f) / f)
        }
        case 'ABS': {
          const x = asNumber(arg(0))
          return x === null ? null : Math.abs(x)
        }
        case 'FLOOR': {
          const x = asNumber(arg(0))
          return x === null ? null : Math.floor(x)
        }
        case 'CEIL': {
          const x = asNumber(arg(0))
          return x === null ? null : Math.ceil(x)
        }
        case 'MIN': {
          const values = nums()
          return values.length ? Math.min(...values) : null
        }
        case 'MAX': {
          const values = nums()
          return values.length ? Math.max(...values) : null
        }
        case 'COALESCE':
          for (let i = 0; i < node.args.length; i++) {
            const v = arg(i)
            if (v !== null && v !== '') return v
          }
          return null
        case 'AND':
          return node.args.every((_, i) => truthy(arg(i))) ? 1 : 0
        case 'OR':
          return node.args.some((_, i) => truthy(arg(i))) ? 1 : 0
        case 'NOT':
          return truthy(arg(0)) ? 0 : 1
      }
    }
  }
}

/**
 * What is wrong with a dataset's calculated columns, each `[index, field, message]`: the
 * expression must parse, read only known columns (stored, or calculated earlier) and declared
 * parameters, and its key must not shadow a stored column.
 */
export function calculatedIssues(
  dataset: Dataset,
  paramIds: ReadonlySet<string>,
): Array<{ index: number; field: 'key' | 'expr'; message: string }> {
  const issues: Array<{ index: number; field: 'key' | 'expr'; message: string }> = []
  const known = new Set(dataset.columns.map((c) => c.key))
  ;(dataset.calculated ?? []).forEach((column, index) => {
    if (column.key === '__proto__' || column.key === 'constructor' || column.key === 'prototype') {
      issues.push({ index, field: 'key', message: `"${column.key}" is reserved; choose another key` })
      return
    }
    if (known.has(column.key)) {
      issues.push({ index, field: 'key', message: `"${column.key}" is already a column of dataset "${dataset.id}"` })
      return
    }
    const parsed = parseCalc(column.expr)
    if (!parsed.ok) issues.push({ index, field: 'expr', message: parsed.error })
    else {
      const refs = calcReferences(parsed.node)
      for (const key of refs.columns) {
        if (key === column.key) issues.push({ index, field: 'expr', message: `"${key}" cannot read itself` })
        else if (!known.has(key)) {
          issues.push({
            index,
            field: 'expr',
            message: `unknown column [${key}] (a calculation reads stored columns and calculated ones listed before it)`,
          })
        }
      }
      for (const id of refs.params) {
        if (!paramIds.has(id)) issues.push({ index, field: 'expr', message: `unknown parameter {${id}}` })
      }
    }
    known.add(column.key)
  })
  return issues
}

/** Add a calculated column, or replace the one with its key (in place). `after` places a new one. */

  column: CalculatedColumn,
  after?: string,

  if (index === -1) return noDataset(dashboard, datasetId)
  const dataset = dashboard.datasets[index]

/** An ISO day moved by `days` (negative goes back). */
export function addDays(iso: string, days: number): string {

  if (!date) return iso

/** Days from `from` to `to`, both counted. */

/**
 * A preset's range, counted back from the LATEST date in the data (a dashboard is a snapshot:
 * "last 28 days" means the 28 days the data ends with, not the 28 before today).
 */
export function presetRange(preset: DatePreset, bounds: { min: string; max: string }): { from: string; to: string } {
  const to = bounds.max
  const [y, m] = to.split('-')
  let from = to
  switch (preset) {
    case 'last-7':
      from = addDays(to, -6)
      break
    case 'last-28':
      from = addDays(to, -27)
      break
    case 'last-90':
      from = addDays(to, -89)
      break
    case 'month':
      from = `${y}-${m}-01`
      break
    case 'quarter': {
      const q = Math.floor((Number(m) - 1) / 3) * 3 + 1
      from = `${y}-${String(q).padStart(2, '0')}-01`
      break
    }
    case 'ytd':
      from = `${y}-01-01`
      break
  }
  return { from: from < bounds.min ? bounds.min : from, to }
}

// ---- Aggregation

/** Combine a column's cells. Nulls are skipped; nothing to combine is `null`, never 0. */
export function aggregate(cells: Cell[], agg: Agg): number | null {
  if (agg === 'count') return cells.filter((c) => c !== null && c !== undefined).length
  const values = cells.filter((c): c is number => typeof c === 'number' && Number.isFinite(c))
  if (values.length === 0) return null
  switch (agg) {
    case 'sum':
      return values.reduce((a, b) => a + b, 0)
    case 'avg':
      return values.reduce((a, b) => a + b, 0) / values.length
    case 'min':
      return Math.min(...values)
    case 'max':
      return Math.max(...values)
    case 'last':
      return values[values.length - 1]
  }
}

export function measure(rows: DataRow[], m: Measure): number | null {
  return aggregate(
    rows.map((row) => row[m.column] ?? null),
    m.agg,
  )
}

export function kpiValue(rows: DataRow[], value: KpiValue): number | null {
  if ('literal' in value) return value.literal
  if ('ratio' in value) {
    const numerator = measure(rows, value.ratio.numerator)
    const denominator = measure(rows, value.ratio.denominator)
    if (numerator === null || denominator === null || denominator === 0) return null
    return numerator / denominator
  }
  return measure(rows, value)
}

export type Group = { key: string | number; rows: DataRow[] }

/**
 * Rows grouped by one column. `sorted` orders groups ascending (numbers numerically, strings
 * and ISO dates lexically); otherwise groups keep the order their key first appears in.
 * Rows with no value in the column are left out.
 */
export function groupRows(rows: DataRow[], column: string, sorted: boolean): Group[] {
  const groups = new Map<string | number, DataRow[]>()
  for (const row of rows) {
    const key = row[column]
    if (key === null || key === undefined) continue
    const bucket = groups.get(key)
    if (bucket) bucket.push(row)
    else groups.set(key, [row])
  }
  const out = [...groups].map(([key, groupRowsList]) => ({ key, rows: groupRowsList }))
  if (sorted) out.sort((a, b) => compareCells(a.key, b.key))
  return out
}

function compareCells(a: Cell, b: Cell): number {
  if (typeof a === 'number' && typeof b === 'number') return a - b
  return String(a).localeCompare(String(b), LOCALE)
}

    const last = groups[groups.length - 1]
    const before = groups[groups.length - 2]
    value = last ? kpiValue(last.rows, block.value) : null
    period = last?.key ?? null
    rowCount = last?.rows.length ?? 0
    if (before) {
      previous = kpiValue(before.rows, block.value)
      previousKey = before.key
    }
  } else {
    value = kpiValue(rows, block.value)
  }

  let base: number | null = null
  let defaultLabel = ''
  if (block.delta) {
    if (block.delta.vs === 'previous') {
      base = previous
      defaultLabel =
        previousKey === null ? '' : `vs ${formatKey(previousKey, sparkType ?? 'string', true)}`
    } else {
      base = block.delta.vs
      defaultLabel = `vs ${formatValue(block.delta.vs, block.format, dashboard.currency)}`
    }
  }

  const delta =
    block.delta && value !== null && base !== null
      ? computeDelta(value, base, {
          format: block.delta.format,
          good: block.delta.good ?? 'up',
          valueFormat: block.format,
          currency: dashboard.currency,
          label: block.delta.label ?? defaultLabel,
        })
      : null

  // Compare to previous period: the same measure over the window before the range wins over the
  // block's own comparison.
  if (compareOn(state) && !('literal' in block.value) && value !== null) {
    const before = previousPeriod(dashboard, block.datasetId, state)
    const previousValue = before ? kpiResult(dashboard, block, before.state).value : null
    if (before && previousValue !== null) {
      const compared = computeDelta(value, previousValue, {
        format: block.delta?.format ?? 'percent',
        good: block.delta?.good ?? 'up',
        valueFormat: block.format,
        currency: dashboard.currency,
        label: `vs ${formatDateRange(before.from, before.to)}`,
      })
      return { value, base: previousValue, delta: compared, trend, period, rowCount }
    }
  }

  return { value, base, delta, trend, period, rowCount }
}

export function computeDelta(
  value: number,
  base: number,
  options: {
    format: 'percent' | 'absolute'
    good: 'up' | 'down'
    valueFormat: ValueFormat
    currency?: string
    label: string
  },
): DeltaResult | null {
  let change: number
  if (options.format === 'percent') {
    if (base === 0) return null
    change = (value - base) / Math.abs(base)
  } else {
    change = value - base
  }
  const magnitude = (amount: number) =>
    options.format === 'percent'
      ? formatPercent(amount)
      : options.valueFormat === 'percent'
        ? `${trimNumber(amount * 100, 1)} pp`
        : formatValue(amount, options.valueFormat, options.currency)
  // A change too small to show ("+0 pp") is no change: the arrow must agree with the digits.
  if (magnitude(Math.abs(change)) === magnitude(0)) change = 0
  const direction = change > 0 ? 'up' : change < 0 ? 'down' : 'flat'
  const sentiment = direction === 'flat' ? 'neutral' : direction === options.good ? 'good' : 'bad'
  const sign = change > 0 ? '+' : change < 0 ? '−' : ''
  return { change, direction, sentiment, text: `${sign}${magnitude(Math.abs(change))}`, label: options.label }
}

/** The format a KPI's number uses when the column alone decides. */
export function defaultFormat(type: ColumnType | undefined): ValueFormat {
  if (type === 'currency') return 'currency'
  if (type === 'percent') return 'percent'
  return 'number'
}

// ---- Charts

export type ChartSeries = {
  key: string
  label: string
  /** The same measure over the period before (compare to previous period), drawn dashed. */
  previous?: boolean
  /** A fitted trendline (not data): drawn thin and dashed. */
  trend?: boolean
  /** Drawn against the right-hand value axis. */
  axis?: 'right'
}
export type ChartData = {
  /** One object per x: `x` holds the x value, `s0…sN` each series' value. */
  rows: Array<Record<string, string | number | null>>
  series: ChartSeries[]
  xType: ColumnType
}

export const MAX_SPLIT_SERIES = 12

export function chartData(dashboard: Dashboard, block: ChartBlock, state: FilterState): ChartData {
  const dataset = findDataset(dashboard, block.datasetId)
  const rows = datasetRows(dashboard, block.datasetId, stateForBlock(state, block.id))
  return chartFromRows(dataset, block, rows)
}

/** The label of the bucket `topN` folds the smaller categories into. */
export const OTHER_LABEL = 'Other'

/** A chart's points from given rows (the chart's own filters already applied). */
export function chartFromRows(dataset: Dataset | undefined, block: ChartBlock, rows: DataRow[]): ChartData {
  const xType = findColumn(dataset, block.x)?.type ?? 'string'
  const agg = block.agg ?? 'sum'

  if (block.type === 'scatter') {
    const first = block.series[0]
    const points = rows
      .filter((r) => typeof r[block.x] === 'number' && typeof r[first.key] === 'number')
      .map((r) => ({ x: r[block.x] as number, s0: r[first.key] as number }))
      .sort((a, b) => a.x - b.x)
    const series: ChartSeries[] = [{ key: 's0', label: first.label }]
    if (block.trendline && points.length > 1) {
      const fit = linearFit(points.map((p) => [p.x, p.s0] as const))
      if (fit) {
        points.forEach((p) => Object.assign(p, { t0: fit.slope * p.x + fit.intercept }))
        series.push({ key: 't0', label: `${first.label} trend`, trend: true })
      }
    }
    return { rows: points, series, xType }
  }

  // Dates and numbers read left to right in order; categories keep the dataset's order.
  let groups = groupRows(rows, block.x, xType !== 'string')
  let other: DataRow[] | null = null
  // Top N: the N largest categories by the first series (all series summed when split).
  if (block.topN !== undefined && xType === 'string' && groups.length > block.topN) {
    const sizeKey = block.series[0].key
    const ranked = groups
      .map((g) => ({ g, size: aggregate(g.rows.map((r) => r[sizeKey] ?? null), agg) ?? -Infinity }))
      .sort((a, b) => b.size - a.size)
    const keep = new Set(ranked.slice(0, block.topN).map((r) => r.g))
    other = groups.filter((g) => !keep.has(g)).flatMap((g) => g.rows)
    groups = groups.filter((g) => keep.has(g))
  }
  const buckets: Array<{ key: string | number; rows: DataRow[]; other?: boolean }> = [...groups]
  if (other && other.length) buckets.push({ key: OTHER_LABEL, rows: other, other: true })

  let series: ChartSeries[]
  let out: Array<Record<string, string | number | null>>
  if (block.split) {
    const splitKey = block.split
    const values = groupRows(rows, splitKey, false)
      .slice(0, MAX_SPLIT_SERIES)
      .map((g) => String(g.key))
    series = values.map((v, i) => ({ key: `s${i}`, label: v }))
    const column = block.series[0].key
    out = buckets.map((g) => {
      const row: Record<string, string | number | null> = { x: g.key }
      if (g.other) row[OTHER_FLAG] = 1
      values.forEach((v, i) => {
        row[`s${i}`] = aggregate(
          g.rows.filter((r) => String(r[splitKey]) === v).map((r) => r[column] ?? null),
          agg,
        )
      })
      return row
    })
  } else {
    const single = block.type === 'pie' || block.type === 'donut' || block.type === 'funnel'
    series = (single ? block.series.slice(0, 1) : block.series).map((s, i) => ({
      key: `s${i}`,
      label: s.label,
      ...(s.axis === 'right' ? { axis: 'right' as const } : {}),
    }))
    const columns = block.series.map((s) => s.key)
    out = buckets.map((g) => {
      const row: Record<string, string | number | null> = { x: g.key }
      if (g.other) row[OTHER_FLAG] = 1
      series.forEach((s, i) => {
        row[s.key] = aggregate(
          g.rows.map((r) => r[columns[i]] ?? null),
          agg,
        )
      })
      return row
    })
  }

  // Sorting: "Other" always stays last.
  if (block.sort) {
    const tail = out.filter((r) => r[OTHER_FLAG])
    const head = out.filter((r) => !r[OTHER_FLAG])
    const size = (r: Record<string, string | number | null>) =>
      series.reduce((sum, s) => sum + (typeof r[s.key] === 'number' ? (r[s.key] as number) : 0), 0)
    if (block.sort === 'x') head.sort((a, b) => compareCells(a.x as Cell, b.x as Cell))
    else head.sort((a, b) => (block.sort === 'value-asc' ? size(a) - size(b) : size(b) - size(a)))
    out = [...head, ...tail]
  }

  // Trendline: a least-squares line through the first series, over the points in order.
  if (block.trendline && !block.split && out.length > 1) {
    const pairs = out.flatMap((r, i) => (typeof r.s0 === 'number' ? [[i, r.s0] as const] : []))
    const fit = linearFit(pairs)
    if (fit) {
      out = out.map((r, i) => ({ ...r, t0: fit.slope * i + fit.intercept }))
      series = [...series, { key: 't0', label: `${series[0].label} trend`, trend: true }]
    }
  }

  return { rows: out, series, xType }
}

/** Marks the "Other" row of a top-N chart: it is not one category, so it never filters or drills. */
export const OTHER_FLAG = '__other'
export const isOtherRow = (row: Record<string, unknown>) => Boolean(row[OTHER_FLAG])

/** Least squares over (x, y) pairs; null with fewer than two distinct x. */
export function linearFit(points: ReadonlyArray<readonly [number, number]>): { slope: number; intercept: number } | null {
  const n = points.length
  if (n < 2) return null
  let sx = 0
  let sy = 0
  for (const [x, y] of points) {
    sx += x
    sy += y
  }
  const mx = sx / n
  const my = sy / n
  let num = 0
  let den = 0
  for (const [x, y] of points) {
    num += (x - mx) * (y - my)
    den += (x - mx) * (x - mx)
  }
  if (den === 0) return null
  const slope = num / den
  return { slope, intercept: my - slope * mx }
}

/**
 * Small multiples: the chart drawn once per value of its `facet` column (at most MAX_FACETS,
 * in the order values first appear; dates and numbers ascending), each from that value's rows.
 */
export function chartFacets(
  dashboard: Dashboard,
  block: ChartBlock,
  state: FilterState,
): Array<{ key: string | number; label: string; data: ChartData }> {
  if (!block.facet) return []
  const dataset = findDataset(dashboard, block.datasetId)
  const rows = datasetRows(dashboard, block.datasetId, stateForBlock(state, block.id))
  const type = findColumn(dataset, block.facet)?.type ?? 'string'
  return groupRows(rows, block.facet, type !== 'string')
    .slice(0, MAX_FACETS)
    .map((g) => ({ key: g.key, label: formatKey(g.key, type), data: chartFromRows(dataset, block, g.rows) }))
}

/** A log axis cannot show zero or less: the chart then draws linear and says why. */
export function logScaleUsable(data: ChartData): boolean {
  return data.rows.every((row) =>
    data.series.every((s) => {
      const v = row[s.key]
      return typeof v !== 'number' || v > 0
    }),
  )
}

/** Chart types that can draw the period before beside the period shown. */
const COMPARABLE: ReadonlySet<ChartBlock['type']> = new Set(['bar', 'line', 'area', 'combo'])

export type ChartComparison = { data: ChartData; compare: { from: string; to: string } | null }

/**
 * A chart's data with "Compare to previous period" applied: when the toggle is on and a date
 * range narrows the chart's dataset, the first series measured over the window just before is
 * added as a `previous` series. An x on the range's own date column is shifted forward by the
 * range's length (so Nov 9 lines up with Oct 26 for a 14-day range); any other x (categories)
 * is matched as is. Unsplit bar, line, area and combo charts only; nothing lines up is no
 * comparison.
 */
export function chartComparison(dashboard: Dashboard, block: ChartBlock, state: FilterState): ChartComparison {
  const data = chartData(dashboard, block, state)
  if (!compareOn(state) || block.split || !COMPARABLE.has(block.type) || data.rows.length === 0) {
    return { data, compare: null }
  }
  const before = previousPeriod(dashboard, block.datasetId, stateForBlock(state, block.id))
  if (!before) return { data, compare: null }
  const previous = chartData(dashboard, block, before.state)
  const shift = block.x === before.column
  const byKey = new Map<string, number | null>()
  for (const row of previous.rows) {
    const key = shift && typeof row.x === 'string' ? addDays(row.x, before.days) : String(row.x)
    byKey.set(key, typeof row.s0 === 'number' ? row.s0 : null)
  }
  let matched = 0
  const rows = data.rows.map((row) => {
    const value = byKey.get(String(row.x)) ?? null
    if (value !== null) matched++
    return { ...row, p0: value }
  })
  if (matched === 0) return { data, compare: null }
  const label = `${data.series[0].label}, ${formatDateRange(before.from, before.to)}`
  return {
    data: { ...data, rows, series: [...data.series, { key: 'p0', label, previous: true }] },
    compare: { from: before.from, to: before.to },
  }
}

// ---- Conditional colour

export type Tone = 'good' | 'bad' | 'warn' | 'info'
type Rule = { op: '>' | '>=' | '<' | '<=' | '=' | '!='; value: number | string; tone: Tone; label?: string }

/** Does a value meet a rule? Numbers compare as numbers, text case-sensitively and only by = / !=. */
export function ruleMatches(rule: Rule, value: Cell | undefined): boolean {
  if (value === null || value === undefined) return false
  if (typeof rule.value === 'string' || typeof value === 'string') {
    const same = String(value) === String(rule.value)
    return rule.op === '=' ? same : rule.op === '!=' ? !same : false
  }
  switch (rule.op) {
    case '>':
      return value > rule.value
    case '>=':
      return value >= rule.value
    case '<':
      return value < rule.value
    case '<=':
      return value <= rule.value
    case '=':
      return value === rule.value
    case '!=':
      return value !== rule.value
  }
}

/** The first rule a value meets (rules read in order, like a spreadsheet's), or null. */
export function firstRule<R extends Rule>(rules: readonly R[] | undefined, value: Cell | undefined): R | null {
  for (const rule of rules ?? []) if (ruleMatches(rule, value)) return rule
  return null
}

export const TONE_WORD: Record<Tone, string> = { good: 'good', bad: 'needs attention', warn: 'watch', info: 'note' }

/** How a KPI's rule reads aloud: its own label, else the tone in words. */
export const ruleWord = (rule: Rule) => rule.label?.trim() || TONE_WORD[rule.tone]

/** How a combo chart draws each series: as given, else bars first and lines after. */
export function comboKinds(block: ChartBlock): Array<'bar' | 'line'> {
  return block.series.map((s, i) => s.as ?? (i === 0 ? 'bar' : 'line'))
}

export type FunnelStep = {
  key: string | number
  value: number
  /** Share of the first step (1 for the first), or null when the first is 0. */
  ofFirst: number | null
  /** Share of the step before (null for the first, or when that one is 0). */
  ofPrevious: number | null
}

/** A funnel's steps in order, from the chart's first series. Steps with no value are left out. */
export function funnelSteps(data: ChartData): FunnelStep[] {
  const points = data.rows
    .map((row) => ({ key: row.x as string | number, value: row.s0 }))
    .filter((p): p is { key: string | number; value: number } => typeof p.value === 'number')
  const first = points[0]?.value ?? 0
  return points.map((p, i) => {
    const previous = i > 0 ? points[i - 1].value : null
    return {
      key: p.key,
      value: p.value,
      ofFirst: first > 0 ? p.value / first : null,
      ofPrevious: previous !== null && previous > 0 ? p.value / previous : null,
    }
  })
}

/**
 * Where an annotation sits among a chart's x values: on its own value when shown, else (dates and
 * numbers, which read in order) on the first x after it. Null when it falls past the end or the
 * axis is categories.
 */
export function snapToX(keys: Array<string | number>, x: string | number, ordered: boolean): string | number | null {
  const exact = keys.find((k) => String(k) === String(x))
  if (exact !== undefined) return exact
  if (!ordered) return null
  const after = keys.find((k) => compareCells(k, x) > 0)
  return after ?? null
}

/**
 * A heatmap scale over a column's cells: each number maps to 0 (lowest) … 1 (highest); a column
 * of one value maps to 0.5. Anything not a number maps to null.
 */
export function heatScale(cells: Array<Cell | undefined>): (cell: Cell | undefined) => number | null {
  const values = cells.filter((c): c is number => typeof c === 'number' && Number.isFinite(c))
  if (values.length === 0) return () => null
  const min = Math.min(...values)
  const max = Math.max(...values)
  return (cell) => {
    if (typeof cell !== 'number' || !Number.isFinite(cell)) return null
    return max === min ? 0.5 : (cell - min) / (max - min)
  }
}

/** An axis tick: always short ("$80K", "45K", "12%"). */
export function formatAxis(value: number, format: ValueFormat, currency = 'USD'): string {
  if (format === 'percent') return formatPercent(value)
  if (format === 'currency' && Math.abs(value) >= 1000) {
    return numberFormat(`axis-${currency}`, {
      style: 'currency',
      currency,
      notation: 'compact',
      maximumFractionDigits: 1,
    }).format(value)
  }
  return formatValue(value, format === 'currency' ? 'currency' : 'compact', currency)
}

/** The format a chart's values take: from its first series column. */
export function chartValueFormat(dashboard: Dashboard, block: ChartBlock): ValueFormat {
  if (block.valueFormat) return block.valueFormat
  const type = findColumn(findDataset(dashboard, block.datasetId), block.series[0].key)?.type
  if (type === 'currency') return 'currency'
  if (type === 'percent') return 'percent'
  return 'compact'
}

/**
 * The format of one drawn series (`s0`…): the chart's valueFormat when set, else that series'
 * own column (a right-axis "Orders" beside "Revenue" reads as a count, not dollars).
 */
export function seriesValueFormat(dashboard: Dashboard, block: ChartBlock, seriesKey: string): ValueFormat {
  if (block.valueFormat || block.split || !/^s\d+$/.test(seriesKey)) return chartValueFormat(dashboard, block)
  const series = block.series[Number(seriesKey.slice(1))]
  const type = series ? findColumn(findDataset(dashboard, block.datasetId), series.key)?.type : undefined
  if (type === 'currency') return 'currency'
  if (type === 'percent') return 'percent'
  return 'compact'
}

// ---- Tables

export function sortRows(rows: DataRow[], column: string, dir: 'asc' | 'desc'): DataRow[] {
  const sign = dir === 'asc' ? 1 : -1
  return rows
    .map((row, index) => ({ row, index }))
    .sort((a, b) => {
      const x = a.row[column]
      const y = b.row[column]
      const xEmpty = x === null || x === undefined
      const yEmpty = y === null || y === undefined
      // Empty cells sink, whichever way the column is sorted.
      if (xEmpty || yEmpty) return xEmpty === yEmpty ? a.index - b.index : xEmpty ? 1 : -1
      const order = compareCells(x, y)
      return order === 0 ? a.index - b.index : order * sign
    })
    .map(({ row }) => row)
}

export function paginate<T>(rows: T[], page: number, pageSize: number): { rows: T[]; page: number; pages: number } {
  const pages = Math.max(1, Math.ceil(rows.length / pageSize))
  const clamped = Math.min(Math.max(0, page), pages - 1)
  return { rows: rows.slice(clamped * pageSize, (clamped + 1) * pageSize), page: clamped, pages }
}

export function tableColumns(dashboard: Dashboard, block: TableBlock): DataColumn[] {
  const dataset = findDataset(dashboard, block.datasetId)
  if (!dataset) return []
  if (!block.columns) return dataset.columns
  return block.columns
    .map((key) => findColumn(dataset, key))
    .filter((c): c is DataColumn => c !== undefined)
}

// ---- Formatting

const cache = new Map<string, Intl.NumberFormat>()
function numberFormat(key: string, options: Intl.NumberFormatOptions): Intl.NumberFormat {
  let format = cache.get(key)
  if (!format) {
    format = new Intl.NumberFormat(LOCALE, options)
    cache.set(key, format)
  }
  return format
}

function trimNumber(value: number, digits: number): string {
  return numberFormat(`n${digits}`, { maximumFractionDigits: digits }).format(value)
}

/** 0.1234 → "12.3%". */
export function formatPercent(fraction: number): string {
  return `${trimNumber(fraction * 100, 1)}%`
}

export function formatValue(value: number | null | undefined, format: ValueFormat, currency = 'USD'): string {
  if (value === null || value === undefined || !Number.isFinite(value)) return EMPTY
  switch (format) {
    case 'integer':
      return numberFormat('int', { maximumFractionDigits: 0 }).format(value)
    case 'compact':
      if (Math.abs(value) < 1000) return trimNumber(value, Math.abs(value) < 10 ? 2 : 1)
      return numberFormat('compact', { notation: 'compact', maximumFractionDigits: 1 }).format(value)
    case 'currency': {
      const abs = Math.abs(value)
      if (abs >= 100_000) {
        return numberFormat(`cc-${currency}`, {
          style: 'currency',
          currency,
          notation: 'compact',
          maximumFractionDigits: 1,
        }).format(value)
      }
      const digits = abs >= 1000 || Number.isInteger(value) ? 0 : 2
      return numberFormat(`c${digits}-${currency}`, {
        style: 'currency',
        currency,
        minimumFractionDigits: digits,
        maximumFractionDigits: digits,
      }).format(value)
    }
    case 'percent':
      return formatPercent(value)
    case 'number':
      return trimNumber(value, Math.abs(value) >= 100 ? 0 : 2)
  }
}

const dateFormats = {
  short: new Intl.DateTimeFormat(LOCALE, { timeZone: 'UTC', month: 'short', day: 'numeric' }),
  long: new Intl.DateTimeFormat(LOCALE, { timeZone: 'UTC', month: 'short', day: 'numeric', year: 'numeric' }),
  // Date and time apart: ICU versions disagree on the joiner ("," or " at") when combined.
  stampDate: new Intl.DateTimeFormat(LOCALE, {
    timeZone: 'America/Los_Angeles',
    month: 'short',
    day: 'numeric',
    year: 'numeric',
  }),
  stampTime: new Intl.DateTimeFormat(LOCALE, {
    timeZone: 'America/Los_Angeles',
    hour: 'numeric',
    minute: '2-digit',
  }),
}

function utcDate(iso: string): Date | null {
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso)
  if (!match) return null
  const date = new Date(Date.UTC(Number(match[1]), Number(match[2]) - 1, Number(match[3])))
  return Number.isNaN(date.getTime()) ? null : date
}

/** "2026-11-09" → "Nov 9, 2026" (or "Nov 9" when `short`). Anything else is returned as-is. */
export function formatDate(iso: string, short = false): string {
  const date = utcDate(iso)
  if (!date) return iso
  return (short ? dateFormats.short : dateFormats.long).format(date)
}

/** "Oct 5 – Nov 9, 2026"; the year once when both ends share it. */
export function formatDateRange(from: string, to: string): string {
  if (from === to) return formatDate(from)
  const a = utcDate(from)
  const b = utcDate(to)
  if (!a || !b) return `${from} – ${to}`
  if (a.getUTCFullYear() === b.getUTCFullYear()) return `${formatDate(from, true)} – ${formatDate(to)}`
  return `${formatDate(from)} – ${formatDate(to)}`
}

/** The as-of stamp: a date reads as a calendar day, a date-time in Pacific time. */
export function formatAsOf(asOf: string): string {
  if (/^\d{4}-\d{2}-\d{2}$/.test(asOf)) return formatDate(asOf)
  const date = new Date(asOf)
  if (Number.isNaN(date.getTime())) return asOf
  return `${dateFormats.stampDate.format(date)}, ${dateFormats.stampTime.format(date)} PT`
}

/** A group or axis key: dates read as dates, numbers formatted. */
export function formatKey(key: string | number, type: ColumnType, short = false): string {
  if (type === 'date' && typeof key === 'string') return formatDate(key, short)
  if (typeof key === 'number') return formatCell(key, type)
  return key
}

export function formatCell(value: Cell | undefined, type: ColumnType, currency?: string): string {
  if (value === null || value === undefined) return EMPTY
  if (typeof value === 'string') return type === 'date' ? formatDate(value) : value
  if (type === 'currency') {
    return numberFormat(`cell-${currency ?? 'USD'}`, {
      style: 'currency',
      currency: currency ?? 'USD',
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    }).format(value)
  }
  if (type === 'percent') return formatPercent(value)
  return trimNumber(value, Number.isInteger(value) ? 0 : 2)
}

// ---- Provenance

/**
 * The time window a block's numbers cover. A date-range filter on its dataset wins (the range
 * chosen, its open ends filled from the data); then the source's own `window`; then the span
 * of the dataset's first date column in the rows shown.
 */
export function blockWindow(dashboard: Dashboard, block: DataBlock, state: FilterState): string | null {
  const datasetId = blockDatasetId(block)
  const dataset = findDataset(dashboard, datasetId)
  for (const filter of dashboard.filters ?? []) {
    const choice = state[filter.id]
    if (filter.type !== 'date-range' || !dataset || !filter.datasetIds.includes(dataset.id)) continue
    if (!choice || choice.kind !== 'date-range' || !isFilterActive(choice)) continue
    const bounds = dateBounds(dashboard, filter)
    const from = choice.from ?? bounds?.min
    const to = choice.to ?? bounds?.max
    if (from && to) return formatDateRange(from, to)
  }
  const source = findSource(dashboard, blockSourceId(dashboard, block))
  if (source?.window) return source.window
  const dateColumn = dataset?.columns.find((c) => c.type === 'date')
  if (!dataset || !dateColumn) return null
  const dates = filterRows(dataset, dashboard.filters, state)
    .map((r) => r[dateColumn.key])
    .filter((v): v is string => typeof v === 'string')
    .sort()
  if (dates.length === 0) return null
  return formatDateRange(dates[0], dates[dates.length - 1])
}

const AGG_WORD: Record<Agg, string> = {
  sum: 'Sum',
  avg: 'Average',
  min: 'Lowest',
  max: 'Highest',
  last: 'Latest',
  count: 'Count',
}

function describeMeasure(dataset: Dataset | undefined, m: Measure): string {
  const column = findColumn(dataset, m.column)
  return `${AGG_WORD[m.agg].toLowerCase()} of ${column?.label ?? m.column}`
}

/** How a KPI's number is worked out, in one sentence: "Sum of Active users over 3 rows …". */
export function explainKpi(dashboard: Dashboard, block: KpiBlock, state: FilterState): string {
  const dataset = findDataset(dashboard, block.datasetId)
  const result = kpiResult(dashboard, block, state)
  const v = block.value
  let how: string
  if ('literal' in v) how = 'A fixed number, as reported by the source'
  else if ('ratio' in v) {
    how = `${capitalise(describeMeasure(dataset, v.ratio.numerator))} divided by ${describeMeasure(dataset, v.ratio.denominator)}`
  } else how = capitalise(describeMeasure(dataset, v))
  const parts = [how]
  if (!('literal' in v)) {
    const sparkType = block.sparkline ? findColumn(dataset, block.sparkline)?.type : undefined
    const where =
      result.period !== null && block.sparkline
        ? ` for ${findColumn(dataset, block.sparkline)?.label ?? block.sparkline} ${formatKey(result.period, sparkType ?? 'string')}`
        : ''
    parts.push(`over ${result.rowCount} ${result.rowCount === 1 ? 'row' : 'rows'}${where}`)
  }
  const filters = activeFilterSummary(dashboard, block.datasetId, state)
  if (filters) parts.push(`(${filters})`)
  return `${parts.join(' ')}.`
}

/** "Platform: Web; Weeks: Oct 5 – Nov 9, 2026", for the filters that touch a dataset. */
export function activeFilterSummary(dashboard: Dashboard, datasetId: string | undefined, state: FilterState): string {
  if (!datasetId) return ''
  const parts: string[] = []
  for (const filter of dashboard.filters ?? []) {
    const choice = state[filter.id]
    if (!filter.datasetIds.includes(datasetId) || !isFilterActive(choice)) continue
    if (choice.kind === 'select') parts.push(`${filter.label}: ${choice.value}`)
    else if (choice.kind === 'multi') parts.push(`${filter.label}: ${choice.values.join(', ')}`)
    else if (choice.kind === 'date-range') {
      const bounds = dateBounds(dashboard, filter)
      const from = choice.from ?? bounds?.min
      const to = choice.to ?? bounds?.max
      if (from && to) parts.push(`${filter.label}: ${formatDateRange(from, to)}`)
    }
  }
  const cross = crossFilter(state)
  const dataset = findDataset(dashboard, datasetId)
  if (cross && dataset?.columns.some((c) => c.key === cross.column)) {
    const type = findColumn(dataset, cross.column)?.type ?? 'string'
    parts.push(`${cross.label}: ${formatKey(cross.value, type)}`)
  }
  return parts.join('; ')
}

const capitalise = (text: string) => text.charAt(0).toUpperCase() + text.slice(1)

export const SUMMARY_LIMIT = 1600

/** A short plain-text account of a block as shown, for "Ask Buffy". Never more than ~800 chars. */
export function describeBlock(dashboard: Dashboard, block: DataBlock, state: FilterState): string {
  const source = findSource(dashboard, blockSourceId(dashboard, block))
  const window = blockWindow(dashboard, block, state)
  const filters = activeFilterSummary(dashboard, blockDatasetId(block), state)
  const provenance = [
    source ? `Source: ${source.label}` : '',
    window ? `Window: ${window}` : '',
    filters ? `Filters: ${filters}` : '',
  ]
    .filter(Boolean)
    .join('. ')

  let body: string
  switch (block.kind) {
    case 'kpi': {
      const r = kpiResult(dashboard, block, state)
      body = `${block.label}: ${formatValue(r.value, block.format, dashboard.currency)}`
      if (r.delta) body += ` (${r.delta.text} ${r.delta.label}, ${r.delta.direction})`
      const rule = firstRule(block.rules, r.value)
      if (rule) body += `. Flagged: ${ruleWord(rule)}`
      if (r.trend.length > 1) {
        body += `. Trend: ${r.trend
          .map((p) => formatValue(p.value, block.format, dashboard.currency))
          .join(', ')}`
      }
      body += `. ${explainKpi(dashboard, block, state)}`
      break
    }
    case 'chart': {
      body = describeChart(dashboard, block, state)
      break
    }
    case 'table': {
      const rows = datasetRows(dashboard, block.datasetId, state)
      const columns = tableColumns(dashboard, block)
      body = `Table "${block.title}", ${rows.length} rows; columns ${columns.map((c) => c.label).join(', ')}.`
      const sorted = block.sort ? sortRows(rows, block.sort.column, block.sort.dir) : rows
      const sample = sorted
        .slice(0, 3)
        .map((r) => columns.map((c) => formatCell(r[c.key], c.type, dashboard.currency)).join(' | '))
      if (sample.length) body += ` First rows: ${sample.join('; ')}.`
      break
    }
    case 'text':
      body = block.markdown.replace(/\s+/g, ' ').trim()
      break
    case 'section':
      body = `Section "${block.title}"${block.description ? `: ${block.description}` : ''}.`
      break
  }
  // The body gives way first: where the numbers came from always survives the cap.
  const tail = provenance ? ` ${provenance}.` : ''
  const room = SUMMARY_LIMIT - tail.length
  const clipped = body.length > room ? `${body.slice(0, Math.max(0, room - 1))}…` : body
  return `${clipped}${tail}`
}

/**
 * "Explain this chart": what it plots and how rows combine, each series' total, range and
 * change from first to last point, the reference lines and annotations, the comparison when
 * one is drawn, and the last points.
 */
export function describeChart(dashboard: Dashboard, block: ChartBlock, state: FilterState): string {
  const { data, compare } = chartComparison(dashboard, block, state)
  const format = chartValueFormat(dashboard, block)
  const dataset = findDataset(dashboard, block.datasetId)
  const xLabel = findColumn(dataset, block.x)?.label ?? block.x
  const show = (value: number | null) => formatValue(value, format, dashboard.currency)
  const parts: string[] = []
  const plotted = data.series.filter((s) => !s.previous && !s.trend)
  parts.push(
    `${capitalise(block.type.replace('-', ' '))} chart "${block.title}", ${plotted.map((s) => s.label).join(', ')} by ${xLabel}, ${data.rows.length} points.`,
  )
  if (block.split) {
    parts.push(`One series per ${findColumn(dataset, block.split)?.label ?? block.split}.`)
  }
  if (block.type !== 'scatter') {
    parts.push(`Rows sharing a ${xLabel} are combined by ${AGG_WORD[block.agg ?? 'sum'].toLowerCase()}.`)
    const ordered = data.xType !== 'string'
    for (const s of plotted.slice(0, 4)) {
      const values = data.rows.map((r) => r[s.key]).filter((v): v is number => typeof v === 'number')
      if (values.length === 0) continue
      const min = Math.min(...values)
      const max = Math.max(...values)
      const lowAt = data.rows.find((r) => r[s.key] === min)?.x as string | number
      const highAt = data.rows.find((r) => r[s.key] === max)?.x as string | number
      const total = values.reduce((a, b) => a + b, 0)
      // A total of rates or averages means nothing: those read as their average.
      const additive = (block.agg === undefined || block.agg === 'sum' || block.agg === 'count') && format !== 'percent'
      const overall = additive ? `total ${show(total)}` : `average ${show(total / values.length)}`
      let line = `${s.label}: ${overall}, high ${show(max)} (${formatKey(highAt, data.xType, true)}), low ${show(min)} (${formatKey(lowAt, data.xType, true)})`
      if (ordered && values.length > 1 && values[0] !== 0) {
        line += `, ${formatPercent((values[values.length - 1] - values[0]) / Math.abs(values[0]))} first to last`
      }
      parts.push(`${line}.`)
    }
  }
  if (block.references?.length) {
    parts.push(`Target lines: ${block.references.map((r) => `${r.label ? `${r.label} ` : ''}${show(r.value)}`).join(', ')}.`)
  }
  if (block.annotations?.length) {
    parts.push(`Marked: ${block.annotations.map((a) => `${a.label} (${formatKey(a.x, data.xType, true)})`).join(', ')}.`)
  }
  if (compare) parts.push(`Dashed: the same measure over ${formatDateRange(compare.from, compare.to)}.`)
  const points = data.rows.slice(-8).map((row) => {
    const x = formatKey(row.x as string | number, data.xType, true)
    const ys = data.series.map((s) => show(row[s.key] as number | null))
    return `${x}: ${ys.join(' / ')}`
  })
  if (points.length) parts.push(`${data.rows.length > 8 ? 'Last 8: ' : ''}${points.join('; ')}.`)
  if (block.note) parts.push(`Note: ${block.note}`)
  return parts.join(' ')
}

export function blockTitle(block: DataBlock): string {
  switch (block.kind) {
    case 'kpi':
      return block.label
    case 'text':
      return block.markdown.replace(/[#*_`>]/g, '').trim().split('\n')[0].slice(0, 80)
    default:
      return block.title
  }
}

// ---- Freshness (agent-refreshed data: nothing on the page fetches)

const HOUR_MS = 3_600_000

export type Freshness = {
  /** When the rows were pulled: the dataset's own stamp, else the page's asOf. Null when unknown. */
  fetchedAt: string | null
  /** Whether the stamp is the dataset's own (true) or the page's asOf (false). */
  own: boolean
  ageHours: number | null
  staleAfterHours: number | null
  /** Older than its recipe allows. Never stale without a threshold or a stamp. */
  stale: boolean
  /** "Nov 14, 2026, 3:00 PM PT" (or a calendar day). */
  label: string
  /** "3 days ago", "just now"; empty when unknown. */
  age: string
}

/** Milliseconds since the epoch for an ISO date (start of that UTC day) or date-time; null if unreadable. */
export function stampTime(iso: string): number | null {
  const t = Date.parse(/^\d{4}-\d{2}-\d{2}$/.test(iso) ? `${iso}T00:00:00Z` : iso)
  return Number.isNaN(t) ? null : t
}

/** "just now", "40 minutes ago", "5 hours ago", "3 days ago". */
export function formatAge(hours: number): string {
  if (hours < 1 / 60) return 'just now'
  if (hours < 1) {
    const m = Math.round(hours * 60)
    return `${m} minute${m === 1 ? '' : 's'} ago`
  }
  if (hours < 48) {
    const h = Math.round(hours)
    return `${h} hour${h === 1 ? '' : 's'} ago`
  }
  const d = Math.round(hours / 24)
  return `${d} days ago`
}

export function datasetFreshness(dashboard: Dashboard, dataset: Dataset, now: number): Freshness {
  const own = dataset.provenance?.fetchedAt
  const fetchedAt = own ?? dashboard.asOf ?? null
  const staleAfterHours = dataset.refresh?.staleAfterHours ?? null
  const at = fetchedAt === null ? null : stampTime(fetchedAt)
  // A date-only stamp means some time that day: age it from the day's end, never from its start.
  const effective = at === null || fetchedAt === null ? null : /T/.test(fetchedAt) ? at : at + 24 * HOUR_MS - 1
  const ageHours = effective === null ? null : Math.max(0, (now - effective) / HOUR_MS)
  return {
    fetchedAt,
    own: own !== undefined,
    ageHours,
    staleAfterHours,
    stale: ageHours !== null && staleAfterHours !== null && ageHours > staleAfterHours,
    label: fetchedAt === null ? '' : formatAsOf(fetchedAt),
    age: ageHours === null ? '' : formatAge(ageHours),
  }
}

/** The freshness behind a block's numbers: its dataset's, or the page's asOf for a literal KPI. */
export function blockFreshness(dashboard: Dashboard, block: DataBlock, now: number): Freshness | null {
  const dataset = findDataset(dashboard, blockDatasetId(block))
  if (dataset) return datasetFreshness(dashboard, dataset, now)
  if (block.kind !== 'kpi' || !dashboard.asOf) return null
  const at = stampTime(dashboard.asOf)
  const ageHours = at === null ? null : Math.max(0, (now - at) / HOUR_MS)
  return {
    fetchedAt: dashboard.asOf,
    own: false,
    ageHours,
    staleAfterHours: null,
    stale: false,
    label: formatAsOf(dashboard.asOf),
    age: ageHours === null ? '' : formatAge(ageHours),
  }
}

/** Datasets older than their recipe allows, in page order. */
export function staleDatasets(dashboard: Dashboard, now: number): string[] {
  return dashboard.datasets.filter((d) => datasetFreshness(dashboard, d, now).stale).map((d) => d.id)
}

/** The page's own stamp: the oldest dataset stamp when datasets carry them, else asOf. */
export function pageFreshness(dashboard: Dashboard, now: number): { label: string; stale: number; oldest: Freshness | null } {
  let oldest: Freshness | null = null
  let stale = 0
  for (const dataset of dashboard.datasets) {
    const f = datasetFreshness(dashboard, dataset, now)
    if (f.stale) stale++
    if (f.fetchedAt === null) continue
    if (!oldest || (stampTime(f.fetchedAt) ?? 0) < (stampTime(oldest.fetchedAt ?? '') ?? 0)) oldest = f
  }
  const label = oldest?.label ?? (dashboard.asOf ? formatAsOf(dashboard.asOf) : '')
  return { label, stale, oldest }
}

/**
 * A refresh recipe as typed in a form: blank fields dropped, null when there is no recipe. The
 * typed `how` replaces the legacy free-text file and command (a form never writes those again).
 */
export function cleanRefresh(input: { recipe: string; how?: RecipeHow | null; staleAfterHours?: number | null }): DatasetRefresh | null {
  const recipe = input.recipe.trim().slice(0, 2000)
  if (!recipe) return null
  const out: DatasetRefresh = { recipe }
  if (input.how) out.how = input.how
  const hours = input.staleAfterHours
  if (hours !== null && hours !== undefined && Number.isFinite(hours) && hours >= 1) out.staleAfterHours = Math.min(Math.round(hours), 24 * 366)
  return out
}

// ---- Tables: formats, totals, search, TSV

/** How a table cell reads: the block's format for that column when it sets one, else by type. */
export function formatTableCell(block: TableBlock, column: DataColumn, value: Cell | undefined, currency?: string): string {
  const format = block.formats?.[column.key]
  if (format && typeof value === 'number') return formatValue(value, format, currency)
  return formatCell(value, column.type, currency)
}

export type TableTotal = { value: number | null; agg: 'sum' | 'avg' }

/**
 * A totals row: sums for number and currency columns, averages for percent (a sum of rates means
 * nothing). Columns in `skip` (calculated per row, like a ratio) get no total: summing a per-row
 * formula is rarely the formula over the total.
 */
export function tableTotals(columns: DataColumn[], rows: DataRow[], skip: ReadonlySet<string> = new Set()): Map<string, TableTotal> {
  const out = new Map<string, TableTotal>()
  for (const column of columns) {
    if (!isNumericType(column.type) || skip.has(column.key)) continue
    const agg = column.type === 'percent' ? 'avg' : 'sum'
    out.set(column.key, { value: aggregate(rows.map((r) => r[column.key] ?? null), agg), agg })
  }
  return out
}

/** Rows whose shown text contains every word of the query (case-insensitive). */
export function searchRows(rows: DataRow[], columns: DataColumn[], query: string, currency?: string): DataRow[] {
  const words = query.trim().toLowerCase().split(/\s+/).filter(Boolean)
  if (words.length === 0) return rows
  return rows.filter((row) => {
    const text = columns
      .map((c) => {
        const v = row[c.key]
        return v === null || v === undefined ? '' : `${String(v)} ${formatCell(v, c.type, currency)}`
      })
      .join(' ')
      .toLowerCase()
    return words.every((w) => text.includes(w))
  })
}

/**
 * One TSV cell: tabs and line breaks become spaces; text a spreadsheet would run as a formula
 * gets a leading `'`. A number's shown text ("-2", "-$5.00") is never guarded: it is a value.
 */
export function tsvCell(cell: string | { text: string; numeric: true }): string {
  const text = typeof cell === 'string' ? cell : cell.text
  const flat = text.replace(/[\t\r\n]+/g, ' ')
  // Text only (export-safe.ts): a formatted number such as -5 stays a value.
  return typeof cell === 'string' ? formulaSafeCell(flat) : flat
}

/** A table as shown (formatted cells), tab-separated, for pasting into a spreadsheet. */
export function toTsv(header: string[], body: Array<Array<string | { text: string; numeric: true }>>): string {
  return [header, ...body].map((line) => line.map(tsvCell).join('\t')).join('\n')
}

// ---- Explaining a single number

/** How one chart point is worked out: "Sum of Active users over 3 rows where Week is Nov 9 …". */
export function explainChartPoint(dashboard: Dashboard, block: ChartBlock, state: FilterState, x: string | number): string {
  const dataset = findDataset(dashboard, block.datasetId)
  const own = stateForBlock(state, block.id)
  const rows = datasetRows(dashboard, block.datasetId, own)
  const xCol = findColumn(dataset, block.x)
  const matching = rows.filter((r) => r[block.x] !== null && r[block.x] !== undefined && String(r[block.x]) === String(x))
  const agg = AGG_WORD[block.agg ?? 'sum'].toLowerCase()
  const what = block.split
    ? `${findColumn(dataset, block.series[0].key)?.label ?? block.series[0].key}, one series per ${findColumn(dataset, block.split)?.label ?? block.split}`
    : block.series.map((s) => findColumn(dataset, s.key)?.label ?? s.key).join(', ')
  const calcs = block.series
    .map((s) => dataset?.calculated?.find((c) => c.key === s.key))
    .filter((c): c is NonNullable<typeof c> => c !== undefined)
    .map((c) => `${c.label} = ${c.expr}`)
  const filters = activeFilterSummary(dashboard, block.datasetId, own)
  return [
    `${capitalise(agg)} of ${what} over ${matching.length} ${matching.length === 1 ? 'row' : 'rows'} where ${xCol?.label ?? block.x} is ${formatKey(x, xCol?.type ?? 'string')}`,
    calcs.length ? `; calculated per row as ${calcs.join('; ')}` : '',
    filters ? ` (${filters})` : '',
    '.',
  ].join('')
}

/** Where one table cell comes from: a stored value, or its calculation. */
export function explainCell(dashboard: Dashboard, datasetId: string, columnKey: string): string {
  const dataset = findDataset(dashboard, datasetId)
  const calc = dataset?.calculated?.find((c) => c.key === columnKey)
  const source = findSource(dashboard, dataset?.sourceId)
  if (calc) return `${calc.label} is calculated for each row: ${calc.expr}.`
  const column = findColumn(dataset, columnKey)
  return `${column?.label ?? columnKey} is a stored value in ${dataset?.title ?? datasetId}${source ? `, from ${source.label}` : ''}.`
}

// ---- The agent's readout: computed values, so it can check its own output

/** Things worth fixing on a page as computed, in plain words (the agent reads these first). */
export function dashboardChecks(dashboard: Dashboard, now: number): string[] {
  const out: string[] = []
  const state = defaultFilterState(dashboard)
  const doubts = new Map(datasetDoubts(dashboard).map((d) => [d.datasetId, d]))
  for (const dataset of dashboard.datasets) {
    const f = datasetFreshness(dashboard, dataset, now)
    if (f.stale) out.push(`dataset ${dataset.id} is stale: fetched ${f.age}, its recipe allows ${f.staleAfterHours} hours`)
    const doubt = doubts.get(dataset.id)
    if (doubt?.edited) out.push(`dataset ${dataset.id} was edited by hand (${doubt.edited.by}, ${doubt.edited.at}) and no longer matches its source`)
    if (doubt?.failed) out.push(`dataset ${dataset.id}: the last refresh failed (${doubt.failed.slice(0, 200)})`)
    if (doubt?.voidCertificate) out.push(`dataset ${dataset.id} changed since the owner certified it`)
    for (const problem of doubt?.checks ?? []) out.push(`dataset ${dataset.id} ${problem}`)
    if (dataset.rows.length === 0 && dashboard.blocks.some((b) => blockDatasetId(b) === dataset.id)) {
      out.push(`dataset ${dataset.id} has no rows but blocks read it`)
    }
    if (dataset.rows.length > 0) {
      for (const calc of dataset.calculated ?? []) {
        const empty = dataset.rows.filter((row) => row[calc.key] === null || row[calc.key] === undefined).length
        if (empty === dataset.rows.length) {
          out.push(`calculated column ${dataset.id}.${calc.key} is empty on every row (a missing input or a division by zero?)`)
        } else if (empty > 0) {
          // A per-row null is a missing input or a division by zero there: worth knowing, not always wrong.
          out.push(`calculated column ${dataset.id}.${calc.key} is empty on ${empty} of ${dataset.rows.length} rows`)
        }
      }
    }
  }
  for (const block of dashboard.blocks) {
    if (block.kind === 'kpi') {
      if (kpiResult(dashboard, block, state).value === null) out.push(`kpi ${block.id} shows no value`)
    } else if (block.kind === 'chart') {
      const data = chartData(dashboard, block, state)
      if (data.rows.length === 0) out.push(`chart ${block.id} has no points`)
      else if (block.scale === 'log' && !logScaleUsable(data)) out.push(`chart ${block.id} asks for a log scale but has values of 0 or less; it draws linear`)
      else if (data.rows.every((r) => data.series.every((s) => r[s.key] === null))) out.push(`chart ${block.id} has points but every value is empty`)
    } else if (block.kind === 'table') {
      if (datasetRows(dashboard, block.datasetId, state).length === 0) out.push(`table ${block.id} has no rows`)
    }
  }
  return out
}

/** A compact per-series account: points, total or average, low, high, first and last. */
function seriesLine(data: ChartData, key: string, label: string, show: (v: number | null) => string, additive: boolean): string {
  const values = data.rows.map((r) => r[key]).filter((v): v is number => typeof v === 'number')
  if (values.length === 0) return `${label}: no values`
  const total = values.reduce((a, b) => a + b, 0)
  const x = (row: Record<string, unknown> | undefined) =>
    row ? formatKey(row.x as string | number, data.xType, true) : ''
  const first = data.rows.find((r) => typeof r[key] === 'number')
  const last = [...data.rows].reverse().find((r) => typeof r[key] === 'number')
  return `${label}: ${additive ? `total ${show(total)}` : `avg ${show(total / values.length)}`}, low ${show(Math.min(...values))}, high ${show(Math.max(...values))}, first ${show(values[0])} (${x(first)}), last ${show(values[values.length - 1])} (${x(last)})`
}

export const READOUT_LIMIT = 6000

/**
 * What the page computes, as the agent's `read_artifact` sees it: checks first, then each
 * dataset's freshness, each KPI's value and change, and each chart's series. Default filters
 * and parameters; calculated columns computed. Capped at `limit` characters.
 */
export function dashboardReadout(stored: Dashboard, now: number, limit = READOUT_LIMIT): string {
  // Calculated columns are computed here exactly as the page computes them.
  const dashboard = materializeDashboard(stored)
  const state = defaultFilterState(dashboard)
  const lines: string[] = []
  const checks = dashboardChecks(dashboard, now)
  lines.push(checks.length ? `Checks: ${checks.join('; ')}.` : 'Checks: nothing to flag.')
  const fresh = dashboard.datasets.map((d) => {
    const f = datasetFreshness(dashboard, d, now)

/**
 * Running a connected recipe (data-connect.ts), the pure parts: filling its `{{blanks}}`, turning
 * the integration's results into rows (`mapIntegrationRows`), and, for the owner's "Test", an
 * outline of a result's shape to map from. The server route (artifact-data-refresh.ts) does the

const BLANK = /\{\{([A-Za-z0-9_-]+)\}\}/g

/** One call's outcome, as integrations `run` returns it ({tool, account?, data} or {tool, error}). */
export type IntegrationCallOutcome = { tool: string; account?: string; data?: unknown; error?: string }

/**
 * A recipe's calls with their blanks filled from the reader's values (`options`), or the
 * defaults when there are none. A value must be plain text (`SAFE_VALUE`, not starting with
 * "-"); a string that is exactly one parameter's blank becomes that number.
 */
export function fillIntegrationCalls(
  dashboard: Dashboard,
  how: IntegrationRecipe,
  options: RecipeOptions = {},
): { calls: IntegrationCall[] } | { problem: 'blank_unsafe' } {
  let unsafe = false
  const fill = (value: unknown): unknown => {
    if (typeof value === 'string') {
      const whole = WHOLE_BLANK.exec(value)
      if (whole && dashboard.params?.some((p) => p.id === whole[1])) {
        const number = Number(blankValue(dashboard, whole[1]!, options))
        if (Number.isFinite(number)) return number
      }
      return value.replace(BLANK, (_match, id: string) => {
        const text = blankValue(dashboard, id, options) ?? ''
        if (!SAFE_VALUE.test(text) || text.startsWith('-')) unsafe = true
        return text
      })
    }
    if (Array.isArray(value)) return value.map(fill)
    if (value && typeof value === 'object')
      return Object.fromEntries(Object.entries(value as Record<string, unknown>).map(([key, item]) => [key, fill(item)]))
    return value
  }
  const calls = how.calls.map((call) => ({ ...call, arguments: fill(call.arguments) as Record<string, unknown> }))
  return unsafe ? { problem: 'blank_unsafe' } : { calls }
}

/** The value at a path, or undefined. Own properties only. */
export function readPath(value: unknown, path: string): unknown {
  const segments = pathSegments(path)
  if (!segments) return undefined
  let current = value
  for (const segment of segments) {
    if (current === null || typeof current !== 'object') return undefined
    if (typeof segment === 'number') {
      if (!Array.isArray(current)) return undefined
      current = current[segment]
    } else {
      if (Array.isArray(current) || !Object.hasOwn(current, segment)) return undefined
      current = (current as Record<string, unknown>)[segment]
    }
  }
  return current
}

const NUMERIC = new Set<ColumnType>(['number', 'currency', 'percent'])
const NUMBER_TEXT = /^[-+]?(\d+(\.\d*)?|\.\d+)([eE][-+]?\d+)?$/

/** A value coerced to a column's type: a cell, or `undefined` when it cannot be one. */
export function coerceCell(value: unknown, type: ColumnType, scale?: number): Cell | undefined {
  if (value === null || value === undefined) return null
  if (NUMERIC.has(type)) {
    let n: number
    if (typeof value === 'number') n = value
    else if (typeof value === 'string') {
      const text = value.trim().replace(/^[$€£¥]/, '').replace(/,(?=\d{3}(\D|$))/g, '').replace(/%$/, '')
      if (!NUMBER_TEXT.test(text)) return undefined
      n = Number(text)
    } else return undefined
    if (scale !== undefined) n *= scale
    return Number.isFinite(n) ? n : undefined
  }
  if (type === 'date') {
    if (typeof value === 'string') {
      const match = ISO_DATE.exec(value.trim())
      if (!match) return undefined
      const [, y, m, d] = match
      const date = new Date(Date.UTC(Number(y), Number(m) - 1, Number(d)))
      return date.getUTCMonth() === Number(m) - 1 && date.getUTCDate() === Number(d) ? `${y}-${m}-${d}` : undefined
    }
    if (typeof value === 'number' && Number.isFinite(value)) {
      // Epoch seconds or milliseconds.
      const ms = Math.abs(value) < 1e11 ? value * 1000 : value
      const date = new Date(ms)
      return Number.isNaN(date.getTime()) ? undefined : date.toISOString().slice(0, 10)
    }
    return undefined
  }
  if (typeof value === 'string') return value.slice(0, MAX_CELL_CHARS)
  if (typeof value === 'number' || typeof value === 'boolean') return String(value)
  if (Array.isArray(value) && value.every((item) => item === null || ['string', 'number', 'boolean'].includes(typeof item)))
    return value.filter((item) => item !== null).join(', ').slice(0, MAX_CELL_CHARS)
  return undefined
}

export type MappedRows =
  | { ok: true; rows: DataRow[]; dropped: number; blanked: number }
  | { ok: false; problem: ConnectProblem }

/**
 * The rows a recipe's results make: for each call in order, the list at `map.records` in its
 * data, each record read into the mapped columns and coerced to the dataset's column types.
 * Any failed call, a truncated result (integrations cut an oversized one to a text preview),
 * or a missing list fails the whole run: a partial pull is never written as if it were whole.
 * A value that cannot be its column's type becomes empty (`blanked` counts them).
 */
export function mapIntegrationRows(
  outcomes: readonly IntegrationCallOutcome[],
  map: IntegrationMap,
  columns: readonly DataColumn[],
  maxRows = INTEGRATION_MAX_ROWS,
): MappedRows {
  const types = new Map(columns.map((c) => [c.key, c.type]))
  for (const column of map.columns) if (!types.has(column.key)) return { ok: false, problem: 'unknown_column' }
  const cap = Math.min(map.limit ?? INTEGRATION_MAX_ROWS, maxRows, INTEGRATION_MAX_ROWS)
  const rows: DataRow[] = []
  let total = 0
  let blanked = 0
  for (const outcome of outcomes) {
    if (outcome.error !== undefined || !('data' in outcome)) return { ok: false, problem: 'call_failed' }
    const data = outcome.data
    if (data && typeof data === 'object' && !Array.isArray(data) && (data as { truncated?: unknown }).truncated === true)
      return { ok: false, problem: 'too_large' }
    const records = readPath(data, map.records)
    if (!Array.isArray(records)) return { ok: false, problem: 'no_list' }
    for (const record of records) {
      total++
      if (rows.length >= cap) continue
      const row: DataRow = {}
      for (const column of map.columns) {
        const raw =
          column.path === '$account'
            ? (outcome.account ?? null)
            : column.path === '$tool'
              ? outcome.tool
              : column.path === '$index'
                ? rows.length
                : readPath(record, column.path)
        const cell = coerceCell(raw, types.get(column.key)!, column.scale)
        if (cell === undefined) blanked++
        row[column.key] = cell ?? null
      }
      rows.push(row)
    }
  }
  return { ok: true, rows, dropped: total - rows.length, blanked }
}

export type ShapeOutline = {
  /** Paths to lists in the data, with their lengths (largest first). */
  lists: Array<{ path: string; length: number }>
  /** Fields of the first record at `records`: path, JSON type and a short sample. */
  fields: Array<{ path: string; type: string; sample: string }>
}

const join = (base: string, key: string | number) =>
  typeof key === 'number' ? `${base}[${key}]` : base ? `${base}.${key}` : key

/**
 * For the owner's Test only: where the lists are in a result and what a record holds, so they
 * can pick `records` and each column's path. Bounded (depth, entries, sample length).
 */
export function outlineShape(data: unknown, records?: string, max = 60): ShapeOutline {
  const lists: ShapeOutline['lists'] = []
  const visit = (value: unknown, path: string, depth: number) => {
    if (depth > 5 || lists.length >= 20) return
    if (Array.isArray(value)) {
      lists.push({ path, length: value.length })
      if (value.length && value[0] && typeof value[0] === 'object') visit(value[0], join(path, 0), depth + 1)
      return
    }
    if (value && typeof value === 'object')
      for (const [key, item] of Object.entries(value).slice(0, 40)) if (pathSegments(join(path, key))) visit(item, join(path, key), depth + 1)
  }
  visit(data, '', 0)
  lists.sort((a, b) => b.length - a.length)
  const fields: ShapeOutline['fields'] = []
  const list = readPath(data, records ?? lists[0]?.path ?? '')
  const first = Array.isArray(list) ? list.find((item) => item && typeof item === 'object') : undefined
  const walk = (value: unknown, path: string, depth: number) => {
    if (fields.length >= max) return
    if (value && typeof value === 'object' && !Array.isArray(value) && depth < 3) {
      for (const [key, item] of Object.entries(value)) if (pathSegments(join(path, key))) walk(item, join(path, key), depth + 1)
      return
    }
    if (!path) return
    const type = value === null ? 'null' : Array.isArray(value) ? 'list' : typeof value
    const text = typeof value === 'string' ? value : JSON.stringify(value) ?? ''
    fields.push({ path, type, sample: text.length > 40 ? `${text.slice(0, 39)}…` : text })
  }
  if (first) walk(first, '', 0)
  return { lists, fields }
}

/**
 * Connected datasets: a dataset whose rows come from one of the OWNER's connected apps
 * (Gmail, Linear, PostHog… through Freebuff integrations), pulled again on Refresh by the server.
 *
 * This file holds the stored shapes and the trust rules for them, with nothing but zod imported
 * (data.ts builds its schema from it). Mapping a result into rows and filling blanks live in
 * data-connect-run.ts; the server route in web/src/server/desktop-cloud/artifact-data-refresh.ts.
 *
 * A connected recipe (`refresh.how` of kind "integration") is:
 *  - `app`: the integration's slug ("linear"), for display and to check the call belongs to it;
 *  - `calls`: 1-4 READ-ONLY integration tool calls ({tool, arguments, account?}). The server runs
 *    them as the owner with read-only tools only; a tool that can change anything is refused.
 *    `{{id}}` blanks (a filter or parameter id) may appear in string argument values: plain text
 *    only, as in a command recipe;
 *  - `map`: how a call's result becomes rows, declaratively: `records`, the path to the list of
 *    records in each call's data, and `columns`, a path per dataset column (types come from the
 *    dataset's columns and values are coerced to them). No code.
 *
 * Who may run it is decided by these owner-only options on `refresh` (the existing rule: only the
 * owner changes `refresh`, data-trust.ts):
 *  - `approved`: the owner's approval of exactly this recipe and these options (the schedule
 *    included). Stamped only by `normaliseRecipeApproval` on an owner's own (non-agent) SAVE, and
 *    marked `via: APPROVAL_VIA` there: a stamp without the marker (written before the store
 *    enforced these rules, or forged) never counts. An agent's write of a connected recipe leaves
 *    it unapproved ("pending"), so it cannot run until the owner approves it in the app; a new
 *    artifact (a copy, a template's use) starts pending; a restore keeps only the stamp already
 *    stored. Any later change to the recipe or options leaves the stamp stale (untrusted).
 *  - `collaboratorsCanRefresh`: editors may press Refresh (it still runs as the owner). Off by default.
 *  - `onOpen`: refresh when the dashboard is opened, at most once per staleness window.
 *  - `schedule`: a scheduled refresh ({every, at?}). Recorded only: nothing runs it yet.

const FORBIDDEN_SEGMENTS = new Set(['__proto__', 'prototype', 'constructor'])
/** Read from the call, not its data: the account's handle, the tool, the record's position. */
export const SPECIAL_PATHS = ['$account', '$tool', '$index'] as const

/**
 * A path into a JSON value, as segments: `data.items`, `fields.Name`, `labels[0]`, `[2].id`;
 * the empty path is the value itself. Null when it is not a valid path.
 */
export function pathSegments(path: string): Array<string | number> | null {
  if (path.length > INTEGRATION_PATH_MAX_CHARS) return null
  if (path === '') return []
  const out: Array<string | number> = []
  const re = /\[(\d{1,5})\]|(?:^|\.)([A-Za-z0-9_$@-]+)/y
  let at = 0
  while (at < path.length) {
    re.lastIndex = at
    const match = re.exec(path)
    if (!match || match.index !== at) return null
    if (match[1] !== undefined) out.push(Number(match[1]))
    else {
      // A name after the start needs its dot; "a..b" and a leading "." fail the sticky match.
      if (at > 0 && path[at] !== '.') return null
      if (at === 0 && path[0] === '.') return null
      const name = match[2]!
      if (FORBIDDEN_SEGMENTS.has(name)) return null
      out.push(name)
    }
    at = re.lastIndex
    if (out.length > PATH_MAX_SEGMENTS) return null
  }
  return out
}

/** A JSON value small and plain enough to be a tool argument. */
function argumentsProblem(value: unknown, depth = 0): string | null {
  if (depth > 6) return 'arguments nest too deep'
  if (value === null || typeof value === 'string' || typeof value === 'boolean') return null
  if (typeof value === 'number') return Number.isFinite(value) ? null : 'numbers must be finite'
  if (Array.isArray(value)) {
    for (const item of value) {
      const problem = argumentsProblem(item, depth + 1)
      if (problem) return problem
    }
    return null
  }
  if (typeof value === 'object') {
    for (const [key, item] of Object.entries(value as Record<string, unknown>)) {
      if (FORBIDDEN_SEGMENTS.has(key)) return `"${key}" is not an argument name`
      const problem = argumentsProblem(item, depth + 1)
      if (problem) return problem
    }
    return null
  }
  return 'arguments are JSON values'
}

  schedule?: DatasetSchedule
  approved?: RecipeApproval
  staleAfterHours?: number
}

export const isIntegrationRecipe = (how: { kind: string } | undefined | null): how is IntegrationRecipe =>
  how?.kind === 'integration'

/** Every string in a recipe's call arguments: where `{{blank}}`s may appear (data.ts checks their ids). */
export function integrationBlankTexts(how: IntegrationRecipe): string[] {
  const out: string[] = []
  const walk = (value: unknown) => {
    if (typeof value === 'string') out.push(value)
    else if (Array.isArray(value)) value.forEach(walk)
    else if (value && typeof value === 'object') Object.values(value).forEach(walk)
  }
  for (const call of how.calls) walk(call.arguments)
  return out
}

/** The blank ids a connected recipe uses. */
export function integrationBlankIds(how: IntegrationRecipe): string[] {
  const ids = new Set<string>()
  for (const text of integrationBlankTexts(how)) for (const match of text.matchAll(BLANK)) ids.add(match[1]!)
  return [...ids]
}

/** JSON with object keys sorted, so equal values have equal text. */
export function canonicalJson(value: unknown): string {
  if (value === undefined) return 'null'
  if (value === null || typeof value !== 'object') return JSON.stringify(value)
  if (Array.isArray(value)) return `[${value.map(canonicalJson).join(',')}]`
  const record = value as Record<string, unknown>
  return `{${Object.keys(record)
    .filter((key) => record[key] !== undefined)
    .sort()
    .map((key) => `${JSON.stringify(key)}:${canonicalJson(record[key])}`)
    .join(',')}}`
}

/**
 * What an approval vouches for: the recipe and the options that decide who runs it and when, as
 * text. The schedule is in it although nothing runs schedules yet: when one does, it must run only
 * the schedule the owner approved, so changing it already invalidates the approval.
 */
export function approvalSubject(refresh: ConnectRefresh): string {
  return canonicalJson({
    how: refresh.how ?? null,
    collaboratorsCanRefresh: refresh.collaboratorsCanRefresh === true,
    onOpen: refresh.onOpen === true,
    schedule: refresh.schedule ?? null,
  })
}

/** A connected recipe's state: not connected, pending the owner's approval, or approved as it stands. */
export function connectedRecipeState(refresh: ConnectRefresh | undefined): 'none' | 'pending' | 'approved' {
  if (!refresh || !isIntegrationRecipe(refresh.how)) return 'none'
  const stamp = refresh.approved
  // The full text, never a hash: a recipe changed in any way after approval is not approved. And
  // only the enforced pipeline's stamp counts (`via`): an older or forged one is pending.
  return stamp && stamp.via === APPROVAL_VIA && stamp.of === approvalSubject(refresh) ? 'approved' : 'pending'
}

/**
 * How a write may treat approvals: `mint` (a save: the owner's own new stamp is stamped), `keep`
 * (a restore: only the stamp already stored survives, so restoring an old version never approves
 * what it carries), `strip` (a new artifact, a copy or a template's use: nothing starts approved,
 * least of all by an account that never saw the recipe).
 */
export type ApprovalMode = 'mint' | 'keep' | 'strip'

export type ApprovalWriter = { owner: boolean; agent: boolean; now: string; approvals?: ApprovalMode }

/**
 * The `refresh` to store, by the approval rules: only the owner's own write (never an agent's,
 * not even theirs) may add or change an approval, and it is stamped here (who, when, exactly
 * what) whatever the writer sent. A recipe that is not connected carries no approval.
 */
export function normaliseRecipeApproval<R extends ConnectRefresh>(
  old: R | undefined,
  next: R | undefined,
  writer: ApprovalWriter,
  person: string,
): R | undefined {
  if (!next) return next
  const { approved, ...rest } = next
  if (!isIntegrationRecipe(next.how)) return approved === undefined ? next : (rest as R)
  if (approved === undefined) return next
  const mode = writer.approvals ?? 'mint'
  if (mode === 'strip') return rest as R
  // The stored stamp, exactly as stored (never given the marker here: a legacy stamp stays legacy).
  const unchanged = old?.approved !== undefined && canonicalJson(approved) === canonicalJson(old.approved)
  if (unchanged) return next
  // A new or changed stamp: the owner approving now, or anyone else's attempt (dropped: pending).
  if (mode === 'keep' || !writer.owner || writer.agent) return rest as R
  return { ...rest, approved: { by: person, at: writer.now, of: approvalSubject(next), via: APPROVAL_VIA } } as R
}

/** The approval stamp a client sends to approve (the store replaces who and when). */
export function approvalRequest(refresh: ConnectRefresh, now: string): RecipeApproval {
  return { by: 'owner', at: now, of: approvalSubject(refresh) }
}

/** Who may press Refresh on a connected dataset: the owner; an editor when the owner allowed it. */
export function mayRefreshConnected(role: string, refresh: ConnectRefresh | undefined): boolean {
  if (role === 'owner') return true
  return role === 'editor' && refresh?.collaboratorsCanRefresh === true
}

/** Hours between refreshes-on-open: the dataset's staleness window, else one hour. */
export function onOpenWindowHours(refresh: ConnectRefresh | undefined): number {
  return refresh?.staleAfterHours ?? 1
}

/**
 * Whether opening the dashboard should refresh this dataset now: the owner asked for it, the
 * recipe is approved, and the rows are older than the window (or were never fetched).
 */
export function refreshOnOpenDue(
  dataset: { refresh?: ConnectRefresh; provenance?: { fetchedAt?: string }; lastRun?: { at: string } },
  now: number,
): boolean {
  const refresh = dataset.refresh
  if (!refresh?.onOpen || connectedRecipeState(refresh) !== 'approved') return false
  const window = onOpenWindowHours(refresh) * 3_600_000
  // A failed attempt counts too: opening a page never retries a broken source in a loop.
  const stamps = [dataset.provenance?.fetchedAt, dataset.lastRun?.at].map((at) => (at ? Date.parse(at) : NaN)).filter(Number.isFinite)
  if (!stamps.length) return true
  return now - Math.max(...stamps) >= window
}

+        i += 1

+    rows.push(row)
+  }

+/**
+ * A1 addresses: column letters, cell keys and ranges. Pure, no React.
+ *
+ * Rows and columns are 0-based internally (`{ row: 0, col: 0 }` is `A1`).
+ */
+
+export type CellPos = { row: number; col: number }
+
+/** An inclusive rectangle, normalised so `top <= bottom` and `left <= right`. */
+export type CellRect = { top: number; left: number; bottom: number; right: number }
+
+const ADDRESS = /^([A-Z]{1,3})([1-9][0-9]{0,6})$/
+
+/** `0` → `A`, `25` → `Z`, `26` → `AA`. */
+export function colName(col: number): string {
+  let n = col + 1
+  let name = ''
+  while (n > 0) {
+    const rem = (n - 1) % 26
+    name = String.fromCharCode(65 + rem) + name
+    n = Math.floor((n - 1) / 26)
+  }
+  return name
+}
+
+/** `A` → `0`, `AA` → `26`. Letters must be upper case. */
+export function colIndex(name: string): number {
+  let n = 0
+  for (let i = 0; i < name.length; i += 1) n = n * 26 + (name.charCodeAt(i) - 64)
+  return n - 1
+}
+
+export function toAddress(pos: CellPos): string {
+  return `${colName(pos.col)}${pos.row + 1}`
+}
+
+/** `B3` → `{ row: 2, col: 1 }`; anything else (lower case, `$`, `A0`) → `null`. */
+export function parseAddress(address: string): CellPos | null {
+  const match = ADDRESS.exec(address)
+  if (!match) return null
+  return { row: Number(match[2]) - 1, col: colIndex(match[1]) }
+}
+
+export function rectOf(a: CellPos, b: CellPos): CellRect {
+  return {
+    top: Math.min(a.row, b.row),
+    bottom: Math.max(a.row, b.row),
+    left: Math.min(a.col, b.col),
+    right: Math.max(a.col, b.col),
+  }
+}
+
+export function rectContains(rect: CellRect, pos: CellPos): boolean {
+  return pos.row >= rect.top && pos.row <= rect.bottom && pos.col >= rect.left && pos.col <= rect.right
+}
+
+export function rectSize(rect: CellRect): { rows: number; cols: number } {
+  return { rows: rect.bottom - rect.top + 1, cols: rect.right - rect.left + 1 }
+}

+/** `A1:C3`, or `A1` for a single cell. */
+export function rectToRange(rect: CellRect): string {
+  const start = toAddress({ row: rect.top, col: rect.left })
+  if (rect.top === rect.bottom && rect.left === rect.right) return start
+  return `${start}:${toAddress({ row: rect.bottom, col: rect.right })}`
+}
+
+/** `A1:C3` or `A1` → a rect; `null` when malformed. (No sheet prefix, no `$`.) */
+export function parseRange(range: string): CellRect | null {
+  const [first, second, extra] = range.split(':')
+  if (extra !== undefined || !first) return null
+  const a = parseAddress(first)
+  const b = second === undefined ? a : parseAddress(second)
+  if (!a || !b) return null
+  return rectOf(a, b)
+}

+/**
+ * A range with an optional tab prefix: `Summary!A1:C3`, `'Q4 plan'!B2`.
+ * Returns the tab name (unquoted) when present.
+ */
+export function parseSheetRange(source: string): { tab?: string; rect: CellRect } | null {
+  const bang = source.lastIndexOf('!')
+  if (bang === -1) {
+    const rect = parseRange(source)
+    return rect ? { rect } : null
+  }
+  let tab = source.slice(0, bang)
+  if (tab.startsWith("'")) {
+    if (!tab.endsWith("'") || tab.length < 3) return null
+    tab = tab.slice(1, -1).replace(/''/g, "'")
+  }
+  if (!tab) return null
+  const rect = parseRange(source.slice(bang + 1))
+  return rect ? { tab, rect } : null
+}

+    ? name
+    : `'${name.replace(/'/g, "''")}'`

+/**
+ * Freebuff Sheets — rich text inside one cell, as styled runs (`cell.runs`) over the plain text
+ * `cell.v`. The editor types it Markdown-ish: **bold**, *italic*, ~~strike~~, ++underline++ and
+ * [label](https://url); a cell that holds runs edits back as the same markup. Pure.
+ *
+ * `v` stays the plain text, so formulas, find, sort and CSV read the words, never the markup.
+ */
+
+import { safeLinkHref } from './safe-link'
+import type { SheetCell, TextRun } from './sheet'
+
+const LINK = /^\[([^\]\n]+)\]\(([^)\s]+)\)/
+const WORD = /[\p{L}\p{N}]/u
+const MARKS: Array<{ open: string; key: 'bold' | 'italic' | 'strike' | 'underline' }> = [
+  { open: '**', key: 'bold' },
+  { open: '~~', key: 'strike' },
+  { open: '++', key: 'underline' },
+  { open: '*', key: 'italic' },
+]
+
+/** A run with no style of its own folds into its plain neighbours. */
+function push(runs: TextRun[], run: TextRun) {
+  if (!run.text) return
+  const last = runs[runs.length - 1]
+  const plain = (r: TextRun) => Object.keys(r).length === 1
+  if (last && plain(last) && plain(run)) last.text += run.text
+  else runs.push(run)
+}
+
+/**
+ * Markup → plain text plus runs; `null` when `raw` holds no markup (so it is ordinary text).
+ * An unsafe link (`javascript:`) stays literal text.
+ */
+export function parseRichText(raw: string): { text: string; runs: TextRun[] } | null {
+  const runs: TextRun[] = []
+  let plain = ''
+  let styled = false
+  let i = 0
+  while (i < raw.length) {
+    const rest = raw.slice(i)
+    const link = LINK.exec(rest)
+    if (link) {
+      const href = safeLinkHref(link[2])
+      if (href) {
+        push(runs, { text: plain })
+        plain = ''
+        push(runs, { text: link[1], link: href })
+        styled = true
+        i += link[0].length
+        continue
+      }
+    }
+    const mark = MARKS.find((m) => rest.startsWith(m.open))
+    if (mark) {
+      const close = rest.indexOf(mark.open, mark.open.length + 1)
+      const inner = close > 0 ? rest.slice(mark.open.length, close) : ''
+      // Stricter than Markdown so arithmetic and globs stay text ("2*3*4", "*.ts,*.tsx"): the
+      // markers sit at word edges, the marked text starts and ends with a letter or digit,
+      // and never crosses a line.
+      const before = i === 0 ? '' : raw[i - 1]
+      const after = raw[i + close + mark.open.length] ?? ''
+      const edge = (ch: string) => ch === '' || (!WORD.test(ch) && ch !== '*' && ch !== '~' && ch !== '+')
+      if (
+        inner &&
+        WORD.test(inner[0]) &&
+        WORD.test(inner[inner.length - 1]) &&
+        edge(before) &&
+        edge(after) &&
+        !inner.includes('\n') &&
+        !(mark.open === '*' && inner.includes('*'))
+      ) {
+        push(runs, { text: plain })
+        plain = ''
+        const run: TextRun = { text: inner }
+        run[mark.key] = true
+        push(runs, run)
+        styled = true
+        i += close + mark.open.length
+        continue
+      }
+    }
+    plain += raw[i]
+    i += 1
+  }
+  if (!styled) return null
+  push(runs, { text: plain })
+  return { text: runs.map((run) => run.text).join(''), runs }
+}
+
+/** Runs → the markup the editor shows (the inverse of `parseRichText` for what it makes). */
+export function richTextToMarkup(runs: readonly TextRun[]): string {
+  return runs
+    .map((run) => {
+      let text = run.text
+      if (run.link) return `[${text}](${run.link})`
+      if (run.italic) text = `*${text}*`
+      if (run.strike) text = `~~${text}~~`
+      if (run.underline) text = `++${text}++`
+      if (run.bold) text = `**${text}**`
+      return text
+    })
+    .join('')
+}
+
+/** The runs a cell shows, or null when they no longer match its text (a stale edit). */
+export function cellRuns(cell: SheetCell | undefined): TextRun[] | null {
+  if (!cell?.runs || typeof cell.v !== 'string') return null
+  return cell.runs.map((run) => run.text).join('') === cell.v ? cell.runs : null
+}
