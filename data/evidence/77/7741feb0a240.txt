/**
 * The bench's static oracle (bench-v2, B2): what a skill SHOULD do on one
 * bench cell (repository snapshot × OS × environment), decided with no run:
 *
 * - `should_refuse`: the skill's own preflight refuses the repository (or
 *   the OS) before any edit. The real `runSponsoredSkillPreflight` decides
 *   it over a static context of the snapshot, and Track A's fit oracle
 *   (`sponsoredSkillFitRefusal` over `sponsoredRepoFacts`) must agree.
 * - `should_fail_gracefully`: the skill fits, but the environment takes away
 *   a toolchain the run needs (`BenchEnvironment.hides`), so the run should
 *   stop cleanly, not crash or report a delivery.
 * - `should_run`: neither; the run should deliver.
 *
 * Checks it cannot decide statically (commands, scripts, globs over source
 * files, lockfile text it does not keep) are `unverified`, never a refusal:
 * the live run decides them. Free and deterministic, so it covers the whole
 * pool on three OSes on every bench run.
 */

/**
 * The bench planner (bench-v2, B4): which cells (pool entry × OS ×
 * environment) one bench run of a skill covers, and which of them run live.
 *
 * - The STATIC oracle always covers all of the pool on three OSes: free, and
 *   it catches a route or fit mistake on the extended set.
 * - LIVE runs use `core` entries only (sweeps add `extended`), and only cells
 *   the oracle labels `should_run`, plus a few `should_refuse` and
 *   environment cells to prove the refusal and the graceful stop.
 * - Every pick is by `hash(skill.id, …)`, never by version, so successive
 *   versions of a skill hit the same cells and their receipts compare.
 *
 * Modes (Owen, 2026-10-14):
 * - `smoke` (about 1h, ≤12 Lab runs): Lab `default` on 10 stratified
 *   `should_run` core repositories plus one run per smoke environment.
 * - `full` (about 3h of Lab plus a Desktop matrix): every Lab `should_run`
 *   core cell (failures re-run, up to 3 attempts), the full environments
 *   sampled, 3 live `should_refuse` cells, and Desktop shards (Linux about 12
 *   runs, Windows ≤18, macOS ≤6; macOS only on a minor or major bump).
 * - `sweep` (weekly, and when a skill is first seeded): `full` with the
 *   extended entries live too. It never gates a PR.
 */

/**
 * A second thread editing package.json while the sponsored run works in the
 * same checkout. Every 20 seconds, for at most 40 minutes, a detached writer
 * sets `freebuffLabConcurrentEdit` to its next count and logs that count
 * outside the checkout. The probe stops it and compares.
 */
const CONCURRENT_WRITER = `set -eu
mkdir -p "${LAB}"
rm -f "${LAB}/edits"
[ -f package.json ] || { echo 'No package.json in this repository; the collision case needs one.' >&2; exit 3; }
setsid nohup python3 -c '
import json, os, time
log = os.path.expanduser("~/.cache/freebuff-lab/edits")
for n in range(1, 121):
    time.sleep(20)
    try:
        with open("package.json") as f:
            data = json.load(f)
    except Exception:
        with open(log, "a") as f:
            f.write("unreadable\\n")
        continue
    data["freebuffLabConcurrentEdit"] = n
    with open("package.json", "w") as f:
        json.dump(data, f, indent=2)
        f.write("\\n")
    with open(log, "a") as f:
        f.write("%d\\n" % n)
' >/dev/null 2>&1 < /dev/null &
echo $! > "${LAB}/writer.pid"`

/**
 * A run whose writer edited package.json and whose last edit is missing
 * afterwards was overwritten from a stale copy, or the file was left invalid.
 * The writer's edit can land after the agent's last write, so the file
 * holding the LAST edit is the only clean outcome. Null: nothing to measure.
 */
export function labProbeCollided(probe: LabBatchRun['probe']): boolean | null {
  if (!probe || probe.kind !== 'concurrent-package-json') return null
  if (!probe.editsWritten) return null
  return (
    !probe.packageJsonValid ||
    probe.unreadable > 0 ||
    probe.markerInFile !== probe.lastEdit
  )
}

export function summarizeLabBatch(batch: LabBatch): LabBatchSummary {
  const graded = batch.runs.flatMap((run) => (run.grade ? [run.grade] : []))
  const count = <K extends string>(keys: K[]) => {
    const totals = new Map<K, number>()
    for (const key of keys) totals.set(key, (totals.get(key) ?? 0) + 1)
    return [...totals].sort((a, b) => b[1] - a[1] || (a[0] < b[0] ? -1 : 1))
  }
  const byTier = { perfect: 0, handoff: 0, partial: 0, broken: 0, lost: 0 }
  for (const grade of graded) byTier[grade.tier]++
  const probes = batch.runs.flatMap((run) => {
    const collided = labProbeCollided(run.probe)
    return collided === null ? [] : [collided]
  })
  return {
    planned: batch.repositories.length * batch.repetitions,
    started: batch.runs.length,
    graded: graded.length,
    perfect: byTier.perfect,
    perfectRate: graded.length ? byTier.perfect / graded.length : null,
    byTier,
    byCause: count(
      graded.filter((g) => g.tier !== 'perfect').map((g) => g.causeKey),
    ).map(([causeKey, n]) => ({ causeKey, count: n })),

      batch.environment === 'concurrent-package-json'
        ? {
            measured: probes.length,
            collided: probes.filter(Boolean).length,
          }
        : null,
  }
}

/** The first run-less slot: each repository's slots are claimed in order. */

  batch: Pick<LabBatch, 'repositories' | 'repetitions' | 'runs'>,
  busyProjects: ReadonlySet<string> = new Set(),
): { projectId: string; slot: number } | null {
  for (let slot = 0; slot < batch.repetitions; slot++)
    for (const { projectId } of batch.repositories) {
      if (busyProjects.has(projectId)) continue
      if (
        !batch.runs.some(
          (run) => run.projectId === projectId && run.slot === slot,
        )
      )
        return { projectId, slot }

/**
 * The bench's static oracle (bench-v2, B2): what a skill SHOULD do on one
 * bench cell (repository snapshot × OS × environment), decided with no run:
 *
 * - `should_refuse`: the skill's own preflight refuses the repository (or
 *   the OS) before any edit. The real `runSponsoredSkillPreflight` decides
 *   it over a static context of the snapshot, and Track A's fit oracle
 *   (`sponsoredSkillFitRefusal` over `sponsoredRepoFacts`) must agree.
 * - `should_fail_gracefully`: the skill fits, but the environment takes away
 *   a toolchain the run needs (`BenchEnvironment.hides`), so the run should
 *   stop cleanly, not crash or report a delivery.
 * - `should_run`: neither; the run should deliver.
 *
 * Checks it cannot decide statically (commands, scripts, globs over source
 * files, lockfile text it does not keep) are `unverified`, never a refusal:
 * the live run decides them. Free and deterministic, so it covers the whole
 * pool on three OSes on every bench run.
 */

  BENCH_HEALED_TOOLCHAINS,
  type BenchEnvironment,
  type BenchOs,
  type BenchToolchain,
} from './agentic-bench-env'
import {
  benchSnapshotForOs,
  type BenchSnapshot,
} from './agentic-bench-snapshot'

  evaluateSponsoredSkillFit,
  sponsoredFitCheckDecidable,
  sponsoredRepoFacts,
  sponsoredStaticCheckContext,
  type SponsoredRepoFacts,
  type SponsoredSkillFitVerdict,
} from './sponsored-skill-fit'

  sponsoredSkillFitEcosystems,
  sponsoredSkillGlob,
  sponsoredSkillPreflightChecks,
  type SponsoredSkill,
  type SponsoredSkillCheck,
} from './sponsored-skill'

  runSponsoredSkillPreflight,
  SPONSORED_PREFLIGHT_OS_CHECK_ID,
} from './sponsored-step-driver'

export type BenchExpectation =
  | 'should_run'
  | 'should_refuse'
  | 'should_fail_gracefully'

export type BenchOracleVerdict = {
  expected: BenchExpectation
  /**
   * False when the environment cannot be set up on this repository (the
   * collision case needs a root package.json): the planner never runs it.
   */
  applicable: boolean
  /** One line: the refusing check, the hidden toolchain, or `fits`. */
  reason: string
  /** The fit oracle's verdict on the same facts. */
  fit: SponsoredSkillFitVerdict['fit']
  /** Preflight checks the snapshot cannot decide; the live run decides them. */
  unverified: string[]
  /**
   * The fit oracle and the real preflight disagree on a check both can
   * decide: a bug in one of them. The label follows the real preflight.
   */
  fidelity?: string
}

/** The toolchains a run of `skill` on this repository needs. */
export function benchRunToolchains(
  skill: SponsoredSkill,
  facts: SponsoredRepoFacts,
): BenchToolchain[] {
  const hasNode = facts.manifests.some((m) => m.ecosystem === 'npm')
  const hasPython = facts.manifests.some((m) => m.ecosystem === 'pypi')
  const routes = sponsoredSkillFitEcosystems(skill) ?? [
    ...new Set(
      skill.requirements.runtimes.map((r) =>
        r.name === 'python' ? 'python' : 'node',
      ),
    ),
  ]
  // The route this repository takes: the first the skill offers that the
  // repository has a manifest for, else the skill's first.
  const route =
    routes.find((r) => (r === 'node' ? hasNode : hasPython)) ?? routes[0]
  const needs = new Set<BenchToolchain>()
  if (route === 'node') needs.add('node')
  if (route === 'python') {
    needs.add('python')
    if (skill.packages.some((p) => p.ecosystem === 'pypi')) needs.add('pip')
  }
  // An npm tool needs Node only where a check runs it; instructions that use
  // it "when npx is available" do not make Node a requirement.
  const npmTools = new Set(
    skill.tools.filter((t) => t.source === 'npm').map((t) => t.name),
  )
  const checks = [
    ...sponsoredSkillPreflightChecks(skill),
    ...skill.steps.flatMap((step) => step.checks),
  ]
  if (
    checks.some(
      (check) =>
        check.kind === 'command_succeeds' &&
        (npmTools.has(check.argv[0] ?? '') ||
          ['node', 'npm', 'npx'].includes(check.argv[0] ?? '')),
    )
  )
    needs.add('node')
  return [...needs]
}

const contentless = (snapshot: BenchSnapshot) =>
  Object.entries(snapshot.files)
    .filter(([, text]) => text === null)
    .map(([path]) => path)

/** Whether the snapshot can decide this check at all. */
function staticallyDecidable(
  check: SponsoredSkillCheck,
  snapshot: BenchSnapshot,
): boolean {
  if (!sponsoredFitCheckDecidable(check)) return false
  if (check.kind === 'file_contains') {
    const re = sponsoredSkillGlob(check.path)
    if (contentless(snapshot).some((path) => re.test(path))) return false
  }
  return true
}

export async function benchRepoFacts(
  snapshot: BenchSnapshot,
): Promise<SponsoredRepoFacts> {
  return sponsoredRepoFacts({
    paths: Object.keys(snapshot.files),
    complete: snapshot.complete,
    read: (path) => snapshot.files[path] ?? null,
  })
}

/** THE label of one bench cell. */
export async function benchExpectation(
  skill: SponsoredSkill,
  snapshot: BenchSnapshot,
  os: BenchOs,
  env: Pick<BenchEnvironment, 'hides' | 'needsRootPackageJson'>,
): Promise<BenchOracleVerdict> {
  const onOs = benchSnapshotForOs(snapshot, os)
  const facts = await benchRepoFacts(onOs)
  const fit = evaluateSponsoredSkillFit(skill, { os, repo: facts })
  const applicable = !env.needsRootPackageJson || 'package.json' in onOs.files

  const files = Object.fromEntries(
    Object.entries(onOs.files).map(([path, text]) => [path, text ?? '']),
  )
  const preflight = await runSponsoredSkillPreflight({
    skill,
    ctx: sponsoredStaticCheckContext(files),
    platform: os,
  })
  const declared = new Map(
    sponsoredSkillPreflightChecks(skill).map((check) => [check.id, check]),
  )
  const unverified: string[] = []
  const refused: string[] = []
  for (const result of preflight.checks) {
    if (result.status !== 'fail') continue
    if (result.id === SPONSORED_PREFLIGHT_OS_CHECK_ID) {
      refused.push(result.id)
      continue
    }
    const check = declared.get(result.id)
    if (check && staticallyDecidable(check, onOs)) refused.push(result.id)
    else unverified.push(result.id)
  }
  // A check the static run did not fail can still be one it could not
  // decide (a command it never ran because an earlier check failed).
  for (const check of declared.values())
    if (
      !staticallyDecidable(check, onOs) &&
      !unverified.includes(check.id) &&
      !refused.includes(check.id) &&
      check.kind !== 'no_privileged_secrets' &&
      check.kind !== 'changes_within'
    )
      unverified.push(check.id)

  let fidelity: string | undefined
  if (fit.fit === 'misfit' && !refused.length)
    fidelity = `the fit oracle refuses on ${fit.checkId} (${fit.reason}) but the real preflight passes every check it can decide`
  if (fit.fit === 'fits' && refused.length)
    fidelity = `the fit oracle says fits but the real preflight refuses on ${refused.join(', ')}`

  const base = {
    applicable,
    fit: fit.fit,
    unverified,
    ...(fidelity ? { fidelity } : {}),
  }
  if (refused.length)
    return {
      ...base,
      expected: 'should_refuse',
      reason: `preflight refuses: ${refused.join(', ')}`,
    }
  const missing = benchRunToolchains(skill, facts).filter(
    (tool) =>
      env.hides.includes(tool) && !BENCH_HEALED_TOOLCHAINS.includes(tool),
  )
  if (missing.length)
    return {
      ...base,
      expected: 'should_fail_gracefully',
      reason: `the environment has no ${missing.join(' or ')}`,
    }
  return { ...base, expected: 'should_run', reason: 'fits' }
}

  'not_a_fit',
  'platform_unsupported',

  'environment_limited',

export type BenchCellResult = 'pass' | 'fail' | 'mismatch' | 'infra'

/**
 * THE pass predicate of one live run against its label. A run that did
 * the opposite of its label (delivered where it should refuse, refused
 * where it should run) is a `mismatch`, always red.
 */
export function benchCellPass(
  expected: BenchExpectation,
  outcome: BenchRunOutcome,
): BenchCellResult {
  if (outcome.infra || !outcome.tier) return 'infra'
  const refused =
    outcome.stoppedBecause === 'preflight_failed' ||
    BENCH_REFUSAL_CAUSES.includes(outcome.causeKey ?? '')
  switch (expected) {
    case 'should_run':
      if (outcome.tier === 'perfect') return 'pass'
      return refused ? 'mismatch' : 'fail'
    case 'should_refuse':
      if (refused) return 'pass'
      return outcome.tier === 'perfect' ? 'mismatch' : 'fail'
    case 'should_fail_gracefully':
      if (outcome.tier === 'perfect') return 'mismatch'
      return refused || BENCH_GRACEFUL_CAUSES.includes(outcome.causeKey ?? '')
        ? 'pass'
        : 'fail'
  }
}

/**
 * The bench planner (bench-v2, B4): which cells (pool entry × OS ×
 * environment) one bench run of a skill covers, and which of them run live.
 *
 * - The STATIC oracle always covers all of the pool on three OSes: free, and
 *   it catches a route or fit mistake on the extended set.
 * - LIVE runs use `core` entries only (sweeps add `extended`), and only cells
 *   the oracle labels `should_run`, plus a few `should_refuse` and
 *   environment cells to prove the refusal and the graceful stop.
 * - Every pick is by `hash(skill.id, …)`, never by version, so successive
 *   versions of a skill hit the same cells and their receipts compare.
 *
 * Modes (Owen, 2026-10-14):
 * - `smoke` (about 1h, ≤12 Lab runs): Lab `default` on 10 stratified
 *   `should_run` core repositories plus one run per smoke environment.
 * - `full` (about 3h of Lab plus a Desktop matrix): every Lab `should_run`
 *   core cell (failures re-run, up to 3 attempts), the full environments
 *   sampled, 3 live `should_refuse` cells, and Desktop shards (Linux about 12
 *   runs, Windows ≤18, macOS ≤6; macOS only on a minor or major bump).
 * - `sweep` (weekly, and when a skill is first seeded): `full` with the
 *   extended entries live too. It never gates a PR.

const cellId = (entry: string, os: BenchOs, env: string) =>
  `${entry}|${os}|${env}`

/** Deterministic order for picks: by hash of the skill id (not version) and the key. */
const byHash = <T>(
  skillId: string,
  salt: string,
  items: T[],
  key: (t: T) => string,
) =>
  [...items].sort((a, b) => {
    const ha = sha256Hex(`${skillId}|${salt}|${key(a)}`)
    const hb = sha256Hex(`${skillId}|${salt}|${key(b)}`)
    return ha < hb ? -1 : ha > hb ? 1 : 0
  })

/** The surfaces a mode and bump require (B5: patch → Lab + Windows; minor/major → all). */
export function benchSurfaces(
  mode: BenchMode,
  bump: BenchBump,
): Array<'lab' | `desktop-${BenchOs}`> {
  if (mode === 'smoke') return ['lab']
  return bump === 'patch'
    ? ['lab', 'desktop-linux', 'desktop-win32']
    : ['lab', 'desktop-linux', 'desktop-win32', 'desktop-darwin']
}

function toCell(
  entry: LabBenchV2Entry,
  os: BenchOs,
  surface: BenchCell['surface'],
  env: string,
  verdict: BenchOracleVerdict,
  attempts: number,
): BenchCell {
  return {
    id: cellId(entry.key, os, env),
    entry: entry.key,
    os,
    surface,
    env,
    expected: verdict.expected,
    reason: verdict.reason,
    unverified: verdict.unverified,
    ...(verdict.fidelity ? { fidelity: verdict.fidelity } : {}),
    attempts,
  }
}

/** Round-robin over buckets so a pick of n spans stacks, not one bucket. */
function stratified(
  skillId: string,
  entries: LabBenchV2Entry[],
  n: number,
): LabBenchV2Entry[] {
  const buckets = new Map<string, LabBenchV2Entry[]>()
  for (const entry of byHash(skillId, 'stratify', entries, (e) => e.key)) {
    const list = buckets.get(entry.bucket) ?? []
    list.push(entry)
    buckets.set(entry.bucket, list)
  }
  const order = byHash(skillId, 'buckets', [...buckets.keys()], (b) => b)
  const out: LabBenchV2Entry[] = []
  while (out.length < n && order.some((b) => buckets.get(b)!.length))
    for (const bucket of order) {
      const next = buckets.get(bucket)!.shift()
      if (next && out.length < n) out.push(next)
    }
  return out
}

export async function planBench(
  skill: SponsoredSkill,
  mode: BenchMode,
  options: { bump?: BenchBump; pool?: readonly LabBenchV2Entry[] } = {},
): Promise<BenchPlan> {
  const bump = options.bump ?? 'patch'
  const pool = options.pool ?? LAB_BENCH_V2
  const L = BENCH_PLAN_LIMITS

  const verdicts = new Map<string, BenchOracleVerdict>()
  const label = async (
    entry: LabBenchV2Entry,
    os: BenchOs,
    env: BenchEnvironment,
  ) => {
    const id = cellId(entry.key, os, env.key)
    let verdict = verdicts.get(id)
    if (!verdict) {
      verdict = await benchExpectation(
        skill,
        benchV2Snapshot(entry.key),
        os,
        env,
      )
      verdicts.set(id, verdict)
    }
    return verdict
  }

  const staticCells: BenchCell[] = []
  for (const entry of pool)
    for (const os of BENCH_OSES) {
      const env = benchDefaultEnvironment(
        os === 'linux' ? 'lab' : 'desktop',
        os,
      )
      staticCells.push(
        toCell(entry, os, 'static', 'default', await label(entry, os, env), 0),
      )
    }

  const liveTiers = mode === 'sweep' ? ['core', 'extended'] : ['core']
  const livePool = pool.filter((e) => liveTiers.includes(e.tier))
  const labPool = livePool.filter((e) => e.kind === 'github')
  const live: BenchCell[] = []
  const add = async (
    entry: LabBenchV2Entry,
    os: BenchOs,
    env: BenchEnvironment,
    attempts: number,
  ) => {
    const verdict = await label(entry, os, env)
    if (!verdict.applicable) return false
    const id = cellId(entry.key, os, env.key)
    if (live.some((c) => c.id === id)) return false
    live.push(toCell(entry, os, env.surface, env.key, verdict, attempts))
    return true
  }
  const labDefault = benchDefaultEnvironment('lab', 'linux')
  const runsOn = async (
    entry: LabBenchV2Entry,
    os: BenchOs,
    env: BenchEnvironment,
  ) => (await label(entry, os, env)).expected === 'should_run'

  // Lab, default environment.
  const labShouldRun: LabBenchV2Entry[] = []
  for (const entry of labPool)
    if (await runsOn(entry, 'linux', labDefault)) labShouldRun.push(entry)
  if (mode === 'smoke') {
    for (const entry of stratified(
      skill.id,
      labShouldRun,
      L.smokeLabRepositories,
    ))
      await add(entry, 'linux', labDefault, 1)
  } else {
    for (const entry of labShouldRun)
      await add(
        entry,
        'linux',
        labDefault,
        mode === 'full' ? L.fullAttempts : 1,
      )
  }

  // Lab environment cases, sampled where the skill runs.
  const labEnvs = BENCH_ENVIRONMENTS.filter(
    (env) =>
      env.surface === 'lab' &&
      env.key !== labDefault.key &&
      (mode === 'smoke' ? env.tier === 'smoke' : true),
  )
  const perEnv = mode === 'smoke' ? 1 : L.fullEnvironmentCells
  for (const env of labEnvs) {
    let picked = 0
    for (const entry of byHash(skill.id, env.key, labShouldRun, (e) => e.key)) {
      if (picked >= perEnv) break
      if (await add(entry, 'linux', env, 1)) picked++
    }
  }

  // Live refusals: the skill must stop itself on a repository it does not fit.
  if (mode !== 'smoke') {
    const refusing: LabBenchV2Entry[] = []
    for (const entry of labPool)
      if (
        (await label(entry, 'linux', labDefault)).expected === 'should_refuse'
      )
        refusing.push(entry)
    for (const entry of byHash(
      skill.id,
      'refuse',
      refusing,
      (e) => e.key,
    ).slice(0, L.fullRefusalCells))
      await add(entry, 'linux', labDefault, 1)
  }

  // Desktop shards.
  const surfaces = benchSurfaces(mode, bump)
  for (const os of BENCH_OSES) {
    if (!surfaces.includes(`desktop-${os}`)) continue
    const envs = BENCH_ENVIRONMENTS.filter(
      (env) => env.surface === 'desktop' && env.os === os,
    )
    const base = benchDefaultEnvironment('desktop', os)
    const shouldRun: LabBenchV2Entry[] = []
    for (const entry of livePool)
      if (await runsOn(entry, os, base)) shouldRun.push(entry)
    for (const entry of stratified(skill.id, shouldRun, L.desktop[os].default))
      await add(entry, os, base, 1)
    for (const env of envs.filter((e) => e.key !== base.key)) {
      let picked = 0
      for (const entry of byHash(skill.id, env.key, shouldRun, (e) => e.key)) {
        if (picked >= L.desktop[os].perEnvironment) break
        if (await add(entry, os, env, 1)) picked++
      }
    }
  }

  // Lab batches: one per environment, split to the Lab's caps. Each attempt
  // after the first is a re-run of that cell's failures, planned by the runner.
  const labBatches: BenchLabBatchPlan[] = []
  const byEnv = new Map<string, BenchCell[]>()
  for (const cell of live.filter((c) => c.surface === 'lab'))
    byEnv.set(cell.env, [...(byEnv.get(cell.env) ?? []), cell])
  const size = Math.min(LAB_BATCH_MAX_REPOSITORIES, LAB_BATCH_MAX_RUNS)
  for (const [envKey, cells] of byEnv) {
    const env = BENCH_ENVIRONMENTS.find((e) => e.key === envKey)!
    for (let i = 0; i < cells.length; i += size) {
      const chunk = cells.slice(i, i + size)
      labBatches.push({
        environment: env.lab!,
        repositories: chunk.map((c) => {
          const entry = pool.find((e) => e.key === c.entry)!
          if (entry.kind !== 'github')
            throw new Error(`${c.entry} is not on GitHub.`)
          return labBenchPinSpec(entry)
        }),
        cells: chunk.map((c) => c.id),
      })
    }
  }

  const desktopShards: BenchDesktopShardPlan[] = []
  for (const cell of live.filter((c) => c.surface === 'desktop')) {
    let shard = desktopShards.find(
      (s) =>
        s.os === cell.os &&
        s.env === cell.env &&
        s.entries.length < L.desktopShardEntries,
    )
    if (!shard) {
      shard = { os: cell.os, env: cell.env, entries: [], cells: [] }
      desktopShards.push(shard)
    }
    shard.entries.push(cell.entry)
    shard.cells.push(cell.id)
  }

  const labRuns = live
    .filter((c) => c.surface === 'lab')
    .reduce((n, c) => n + c.attempts, 0)
  const desktopRuns = Object.fromEntries(
    BENCH_OSES.map((os) => [
      os,
      live.filter((c) => c.surface === 'desktop' && c.os === os).length,
    ]),
  ) as Record<BenchOs, number>
  const githubMinutesLinuxEquivalent = BENCH_OSES.reduce(
    (n, os) =>
      n + desktopRuns[os] * L.desktopRunMinutes * L.githubMinuteMultiplier[os],
    0,
  )
  return {
    benchVersion: LAB_BENCH_V2_VERSION,
    mode,
    bump,
    skill: sponsoredSkillKey(skill),
    skillSha256: sponsoredSkillSha256(skill),
    static: staticCells,
    live,
    labBatches,
    desktopShards,
    estimate: {
      labRuns,
      desktopRuns,
      labWallMinutes:
        Math.ceil(labRuns / L.labConcurrentRuns) * L.labRunMinutes,
      // Shards run in parallel; the longest is one OS's shard of one env.
      desktopWallMinutes: Math.max(
        0,
        ...desktopShards.map((s) => s.entries.length * L.desktopRunMinutes),
      ),
      githubMinutesLinuxEquivalent,
    },
  }
}

/**
 * Bench receipts (bench-v2, B5): what one bench run of one skill version
 * found, committed next to the version as
 * `common/src/ads/sponsored-skills/<id>/<version>.bench.json` and indexed in
 * `sponsored-skills/receipts.ts`. CI checks the newest version of every skill
 * has a current one (`sponsored-skill-bench-receipt.test.ts`) and runs
 * nothing: the bench itself runs only when someone dispatches it.
 */
import {
  benchCellPass,
  type BenchCellResult,
  type BenchExpectation,
  type BenchRunOutcome,
} from './agentic-bench-oracle'
import { dropZ } from '../util/two-proportion-z'

import type { BenchOs, BenchSurface } from './agentic-bench-env'
import type { BenchBump, BenchMode, BenchPlan } from './agentic-bench-plan'

/** Smoke and full verdicts need at least this overall `should_run` pass rate. */
export const BENCH_MIN_PASS_RATE = 0.5
/** A drop of this many points in the `should_run` rate, at z ≥ BENCH_MIN_DROP_Z, regresses. */
export const BENCH_MAX_RATE_DROP = 0.1
export const BENCH_MIN_DROP_Z = 2
/** A cell regresses when it passed before and now passes 0 of at least this many attempts. */
export const BENCH_CELL_REGRESSION_ATTEMPTS = 3

export type BenchReceiptCell = {
  id: string
  entry: string
  os: BenchOs
  surface: BenchSurface
  env: string
  expected: BenchExpectation
  attempts: number
  passes: number
  results: BenchCellResult[]
  causeKeys: string[]
  /** One `gs://codebuff-ads-evals-text/bench/…` trace per attempt, when uploaded. */
  traces?: string[]
}

export type BenchVerdict = {
  result: 'pass' | 'regressed' | 'failed'
  /** The receipt it was compared with (`<id>@<version>`), or null for a first. */
  against: string | null
  reasons: string[]
}

export type BenchReceipt = {
  v: 1
  skill: string
  /** `sponsoredSkillSha256` of the version benched; must equal its lock line. */
  skillSha256: string
  benchVersion: string
  mode: BenchMode
  bump: BenchBump
  surfaces: string[]
  createdAt: string
  labBatchIds: string[]
  actionsRunUrl: string | null
  costUsd: number | null
  static: {
    cells: number
    byExpected: Record<BenchExpectation, number>
    labelsSha256: string
    fidelity: string[]
  }
  cells: BenchReceiptCell[]
  verdict: BenchVerdict
}

const rateOf = (cells: readonly BenchReceiptCell[]) => {
  const run = cells.filter((c) => c.expected === 'should_run')
  const hits = run.reduce((n, c) => n + Math.min(c.passes, 1), 0)
  return { hits, n: run.length, rate: run.length ? hits / run.length : null }
}

/** One cell's attempts folded: a pass stops the retries, a mismatch is never forgiven. */
export function benchReceiptCell(
  cell: Pick<
    BenchReceiptCell,
    'id' | 'entry' | 'os' | 'surface' | 'env' | 'expected'
  >,
  outcomes: readonly BenchRunOutcome[],
  traces?: string[],
): BenchReceiptCell {
  const results = outcomes.map((o) => benchCellPass(cell.expected, o))
  return {
    ...cell,
    attempts: results.filter((r) => r !== 'infra').length,
    passes: results.filter((r) => r === 'pass').length,
    results,
    causeKeys: outcomes.map((o) => o.causeKey ?? 'none'),
    ...(traces?.length ? { traces } : {}),
  }
}

/**
 * A receipt's own verdict, with no previous version: no label mismatch, at
 * least one pass in every language family the skill runs on, and an overall
 * `should_run` rate of at least BENCH_MIN_PASS_RATE.
 */
export function benchOwnVerdict(
  cells: readonly BenchReceiptCell[],
  families: ReadonlyMap<string, string>,
): string[] {
  const reasons: string[] = []
  for (const cell of cells)
    if (cell.results.includes('mismatch'))
      reasons.push(
        `${cell.id}: expected ${cell.expected}, the run did the opposite`,
      )
  const run = cells.filter((c) => c.expected === 'should_run')
  const perFamily = new Map<string, number>()
  for (const cell of run) {
    const family = families.get(cell.entry) ?? 'other'
    perFamily.set(
      family,
      (perFamily.get(family) ?? 0) + Math.min(cell.passes, 1),
    )
  }
  for (const [family, passes] of perFamily)
    if (!passes) reasons.push(`No should_run cell passed for ${family}.`)
  const { rate } = rateOf(cells)
  if (rate !== null && rate < BENCH_MIN_PASS_RATE)
    reasons.push(
      `should_run pass rate ${(rate * 100).toFixed(0)}% is below ${BENCH_MIN_PASS_RATE * 100}%.`,
    )
  for (const cell of cells)
    if (cell.attempts === 0 && cell.results.length)
      reasons.push(`${cell.id}: every attempt was infra; not measured.`)
  return reasons
}

/**
 * THE regression rule between two receipts of the same skill (B5): a cell
 * regresses if it had at least one pass before and passes 0 of
 * BENCH_CELL_REGRESSION_ATTEMPTS now; the overall `should_run` rate
 * regresses when it drops BENCH_MAX_RATE_DROP or more at z ≥ BENCH_MIN_DROP_Z.
 * Only cells both receipts ran are compared.
 */
export function compareBenchReceipts(
  previous: Pick<BenchReceipt, 'skill' | 'cells'>,
  next: Pick<BenchReceipt, 'cells'>,
): { regressed: boolean; reasons: string[]; shared: number } {
  const reasons: string[] = []
  const before = new Map(previous.cells.map((c) => [c.id, c]))
  const shared = next.cells.filter((c) => before.has(c.id))
  for (const cell of shared) {
    const old = before.get(cell.id)!
    if (
      old.passes >= 1 &&
      cell.passes === 0 &&
      cell.attempts >= BENCH_CELL_REGRESSION_ATTEMPTS
    )
      reasons.push(
        `${cell.id}: passed ${old.passes}/${old.attempts} in ${previous.skill}, 0/${cell.attempts} now`,
      )
  }
  const a = rateOf(shared.map((c) => before.get(c.id)!))
  const b = rateOf(shared)
  if (a.rate !== null && b.rate !== null) {
    const drop = a.rate - b.rate
    const z = dropZ(a, b)
    if (drop >= BENCH_MAX_RATE_DROP && z >= BENCH_MIN_DROP_Z)
      reasons.push(
        `should_run rate fell ${(drop * 100).toFixed(0)} points (${(a.rate * 100).toFixed(0)}% → ${(b.rate * 100).toFixed(0)}%, z ${z.toFixed(1)})`,
      )
  }
  return { regressed: reasons.length > 0, reasons, shared: shared.length }
}

/** A receipt from a plan and its live outcomes, judged against the previous version's receipt. */
export function buildBenchReceipt(input: {
  plan: BenchPlan
  cells: BenchReceiptCell[]
  labelsSha256: string
  families: ReadonlyMap<string, string>
  previous: BenchReceipt | null
  labBatchIds: string[]
  actionsRunUrl: string | null
  costUsd: number | null
  createdAt: string
  surfaces?: string[]
}): BenchReceipt {
  const { plan } = input
  const byExpected = {
    should_run: 0,
    should_refuse: 0,
    should_fail_gracefully: 0,
  }
  for (const cell of plan.static) byExpected[cell.expected]++
  const fidelity = plan.static.flatMap((c) =>
    c.fidelity ? [`${c.id}: ${c.fidelity}`] : [],
  )
  const own = benchOwnVerdict(input.cells, input.families)
  if (fidelity.length)
    own.push(
      `${fidelity.length} static cell(s) where the fit oracle and the preflight disagree.`,
    )
  const compared = input.previous
    ? compareBenchReceipts(input.previous, { cells: input.cells })
    : null
  return {
    v: 1,
    skill: plan.skill,
    skillSha256: plan.skillSha256,
    benchVersion: plan.benchVersion,
    mode: plan.mode,
    bump: plan.bump,
    surfaces:
      input.surfaces ??
      [
        ...new Set(
          input.cells.map((c) =>
            c.surface === 'lab' ? 'lab' : `desktop-${c.os}`,
          ),
        ),
      ].sort(),
    createdAt: input.createdAt,
    labBatchIds: input.labBatchIds,
    actionsRunUrl: input.actionsRunUrl,
    costUsd: input.costUsd,
    static: {
      cells: plan.static.length,
      byExpected,
      labelsSha256: input.labelsSha256,
      fidelity,
    },
    cells: input.cells,
    verdict: {
      result: own.length
        ? 'failed'
        : compared?.regressed
          ? 'regressed'
          : 'pass',
      against: input.previous?.skill ?? null,
      reasons: [...own, ...(compared?.reasons ?? [])],
    },
  }
}

/** Markdown for a PR body, between `<!-- bench -->` markers the workflow rewrites. */
export function benchVerdictMarkdown(receipt: BenchReceipt): string {
  const lines = [

    '',
    '| Cell | Expected | Passes | Causes | Trace |',
    '| --- | --- | ---: | --- | --- |',
    ...receipt.cells.map(
      (c) =>
        `| ${c.id} | ${c.expected} | ${c.passes}/${c.attempts}${c.results.includes('mismatch') ?     ...(receipt.actionsRunUrl ? ['', `Run: ${receipt.actionsRunUrl}`] : []),

}

+/**
+ * How many standard errors `current` is BELOW `baseline` (two-proportion z,
+ * pooled). Positive = a drop. Infinity when both are certain and differ.
+ * Shared by the skill-health alert and the bench's receipt comparison.
+ */
+export function dropZ(
+  baseline: { hits: number; n: number },
+  current: { hits: number; n: number },
+): number {
+  if (baseline.n === 0 || current.n === 0) return 0
+  const p1 = baseline.hits / baseline.n
+  const p2 = current.hits / current.n
+  const pooled = (baseline.hits + current.hits) / (baseline.n + current.n)
+  const se = Math.sqrt(pooled * (1 - pooled) * (1 / baseline.n + 1 / current.n))
+  if (se === 0) return p1 > p2 ? Infinity : 0
+  return (p1 - p2) / se
+}

+    .filter(([, text]) => text === null)
+    .map(([path]) => path)

+/** Whether the snapshot can decide this check at all. */
+function staticallyDecidable(
+  check: SponsoredSkillCheck,
+  snapshot: BenchSnapshot,
+): boolean {
+  if (!sponsoredFitCheckDecidable(check)) return false
+  if (check.kind === 'file_contains') {
+    const re = sponsoredSkillGlob(check.path)
+    if (contentless(snapshot).some((path) => re.test(path))) return false
+  }
+  return true
+}

+export async function benchRepoFacts(
+  snapshot: BenchSnapshot,
+): Promise<SponsoredRepoFacts> {
+  return sponsoredRepoFacts({
+    paths: Object.keys(snapshot.files),
+    complete: snapshot.complete,
+    read: (path) => snapshot.files[path] ?? null,
+  })

+/** THE label of one bench cell. */
+export async function benchExpectation(
+  skill: SponsoredSkill,
+  snapshot: BenchSnapshot,
+  os: BenchOs,
+  env: Pick<BenchEnvironment, 'hides' | 'needsRootPackageJson'>,
+): Promise<BenchOracleVerdict> {
+  const onOs = benchSnapshotForOs(snapshot, os)
+  const facts = await benchRepoFacts(onOs)
+  const fit = evaluateSponsoredSkillFit(skill, { os, repo: facts })
+  const applicable = !env.needsRootPackageJson || 'package.json' in onOs.files
+
+  const files = Object.fromEntries(
+    Object.entries(onOs.files).map(([path, text]) => [path, text ?? '']),
+  )
+  const preflight = await runSponsoredSkillPreflight({
+    skill,
+    ctx: sponsoredStaticCheckContext(files),
+    platform: os,
+  })
+  const declared = new Map(
+    sponsoredSkillPreflightChecks(skill).map((check) => [check.id, check]),
+  )
+  const unverified: string[] = []
+  const refused: string[] = []
+  for (const result of preflight.checks) {
+    if (result.status !== 'fail') continue
+    if (result.id === SPONSORED_PREFLIGHT_OS_CHECK_ID) {
+      refused.push(result.id)
+      continue
+    }
+    const check = declared.get(result.id)
+    if (check && staticallyDecidable(check, onOs)) refused.push(result.id)
+    else unverified.push(result.id)
+  }
+  // A check the static run did not fail can still be one it could not
+  // decide (a command it never ran because an earlier check failed).
+  for (const check of declared.values())
+    if (
+      !staticallyDecidable(check, onOs) &&
+      !unverified.includes(check.id) &&
+      !refused.includes(check.id) &&
+      check.kind !== 'no_privileged_secrets' &&
+      check.kind !== 'changes_within'
+    )
+      unverified.push(check.id)
+
+  let fidelity: string | undefined
+  if (fit.fit === 'misfit' && !refused.length)
+    fidelity = `the fit oracle refuses on ${fit.checkId} (${fit.reason}) but the real preflight passes every check it can decide`
+  if (fit.fit === 'fits' && refused.length)
+    fidelity = `the fit oracle says fits but the real preflight refuses on ${refused.join(', ')}`
+
+  const base = {
+    applicable,
+    fit: fit.fit,
+    unverified,
+    ...(fidelity ? { fidelity } : {}),
+  }
+  if (refused.length)
+    return {
+      ...base,
+      expected: 'should_refuse',
+      reason: `preflight refuses: ${refused.join(', ')}`,
+    }
+  const missing = benchRunToolchains(skill, facts).filter(
+    (tool) =>
+      env.hides.includes(tool) && !BENCH_HEALED_TOOLCHAINS.includes(tool),
+  )
+  if (missing.length)
+    return {
+      ...base,
+      expected: 'should_fail_gracefully',
+      reason: `the environment has no ${missing.join(' or ')}`,
+    }
+  return { ...base, expected: 'should_run', reason: 'fits' }
+}

+/** What one live run of a cell did, from a Lab grade or a Desktop leg's result. */
+export type BenchRunOutcome = {
+  tier: 'perfect' | 'handoff' | 'partial' | 'broken' | 'lost' | null
+  causeKey: string | null
+  /** The step driver's stop reason when known (`preflight_failed`, …). */
+  stoppedBecause?: string | null
+  /** The run never tested the skill: no grade, a failed setup, an infra cause. */
+  infra?: boolean
+}
+
+/** Grader causes that mean the run stopped at the skill's own refusal. */
+export const BENCH_REFUSAL_CAUSES: readonly string[] = [
+  'preflight_refused',
+  'not_a_fit',
+  'platform_unsupported',
+]
+
+/** Grader causes that mean a clean stop for a missing toolchain. */
+export const BENCH_GRACEFUL_CAUSES: readonly string[] = [
+  ...BENCH_REFUSAL_CAUSES,
+  'missing_node',
+  'missing_toolchain',
+  'environment_limited',
+]

+export type BenchCellResult = 'pass' | 'fail' | 'mismatch' | 'infra'
+
+/**
+ * THE pass predicate of one live run against its label. A run that did
+ * the opposite of its label (delivered where it should refuse, refused
+ * where it should run) is a `mismatch`, always red.
+ */
+export function benchCellPass(
+  expected: BenchExpectation,
+  outcome: BenchRunOutcome,
+): BenchCellResult {
+  if (outcome.infra || !outcome.tier) return 'infra'
+  const refused =
+    outcome.stoppedBecause === 'preflight_failed' ||
+    BENCH_REFUSAL_CAUSES.includes(outcome.causeKey ?? '')
+  switch (expected) {
+    case 'should_run':
+      if (outcome.tier === 'perfect') return 'pass'
+      return refused ? 'mismatch' : 'fail'
+    case 'should_refuse':
+      if (refused) return 'pass'
+      return outcome.tier === 'perfect' ? 'mismatch' : 'fail'
+    case 'should_fail_gracefully':
+      if (outcome.tier === 'perfect') return 'mismatch'
+      return refused || BENCH_GRACEFUL_CAUSES.includes(outcome.causeKey ?? '')
+        ? 'pass'
+        : 'fail'
+  }

+const rateOf = (cells: readonly BenchReceiptCell[]) => {
+  const run = cells.filter((c) => c.expected === 'should_run')
+  const hits = run.reduce((n, c) => n + Math.min(c.passes, 1), 0)
+  return { hits, n: run.length, rate: run.length ? hits / run.length : null }

+/** One cell's attempts folded: a pass stops the retries, a mismatch is never forgiven. */
+export function benchReceiptCell(
+  cell: Pick<
+    BenchReceiptCell,
+    'id' | 'entry' | 'os' | 'surface' | 'env' | 'expected'
+  >,
+  outcomes: readonly BenchRunOutcome[],
+  traces?: string[],

+    passes: results.filter((r) => r === 'pass').length,

+/**
+ * A receipt's own verdict, with no previous version: no label mismatch, at
+ * least one pass in every language family the skill runs on, and an overall
+ * `should_run` rate of at least BENCH_MIN_PASS_RATE.
+ */
+export function benchOwnVerdict(
+  cells: readonly BenchReceiptCell[],
+  families: ReadonlyMap<string, string>,
+): string[] {
+  const reasons: string[] = []
+  for (const cell of cells)
+    if (cell.results.includes('mismatch'))
+      reasons.push(
+        `${cell.id}: expected ${cell.expected}, the run did the opposite`,
+      )
+  const run = cells.filter((c) => c.expected === 'should_run')
+  const perFamily = new Map<string, number>()
+  for (const cell of run) {
+    const family = families.get(cell.entry) ?? 'other'
+    perFamily.set(
+      family,
+      (perFamily.get(family) ?? 0) + Math.min(cell.passes, 1),
+    )
+  }
+  for (const [family, passes] of perFamily)
+    if (!passes) reasons.push(`No should_run cell passed for ${family}.`)
+  const { rate } = rateOf(cells)
+  if (rate !== null && rate < BENCH_MIN_PASS_RATE)
+    reasons.push(
+      `should_run pass rate ${(rate * 100).toFixed(0)}% is below ${BENCH_MIN_PASS_RATE * 100}%.`,
+    )
+  for (const cell of cells)
+    if (cell.attempts === 0 && cell.results.length)
+      reasons.push(`${cell.id}: every attempt was infra; not measured.`)
+  return reasons
+}

+/**
+ * THE regression rule between two receipts of the same skill (B5): a cell
+ * regresses if it had at least one pass before and passes 0 of
+ * BENCH_CELL_REGRESSION_ATTEMPTS now; the overall `should_run` rate
+ * regresses when it drops BENCH_MAX_RATE_DROP or more at z ≥ BENCH_MIN_DROP_Z.
+ * Only cells both receipts ran are compared.
+ */
+export function compareBenchReceipts(
+  previous: Pick<BenchReceipt, 'skill' | 'cells'>,
+  next: Pick<BenchReceipt, 'cells'>,
+): { regressed: boolean; reasons: string[]; shared: number } {
+  const reasons: string[] = []
+  const before = new Map(previous.cells.map((c) => [c.id, c]))
+  const shared = next.cells.filter((c) => before.has(c.id))
+  for (const cell of shared) {
+    const old = before.get(cell.id)!
+    if (
+      old.passes >= 1 &&
+      cell.passes === 0 &&
+      cell.attempts >= BENCH_CELL_REGRESSION_ATTEMPTS
+    )
+      reasons.push(
+        `${cell.id}: passed ${old.passes}/${old.attempts} in ${previous.skill}, 0/${cell.attempts} now`,
+      )
+  }
+  const a = rateOf(shared.map((c) => before.get(c.id)!))
+  const b = rateOf(shared)
+  if (a.rate !== null && b.rate !== null) {
+    const drop = a.rate - b.rate
+    const z = dropZ(a, b)
+    if (drop >= BENCH_MAX_RATE_DROP && z >= BENCH_MIN_DROP_Z)
+      reasons.push(
+        `should_run rate fell ${(drop * 100).toFixed(0)} points (${(a.rate * 100).toFixed(0)}% → ${(b.rate * 100).toFixed(0)}%, z ${z.toFixed(1)})`,
+      )
+  }
+  return { regressed: reasons.length > 0, reasons, shared: shared.length }
+}

+          : 'pass',

+      reasons: [...own, ...(compared?.reasons ?? [])],

+  }

+/**
+ * The bench planner (bench-v2, B4): which cells (pool entry × OS ×
+ * environment) one bench run of a skill covers, and which of them run live.
+ *
+ * - The STATIC oracle always covers all of the pool on three OSes: free, and
+ *   it catches a route or fit mistake on the extended set.
+ * - LIVE runs use `core` entries only (sweeps add `extended`), and only cells
+ *   the oracle labels `should_run`, plus a few `should_refuse` and
+ *   environment cells to prove the refusal and the graceful stop.
+ * - Every pick is by `hash(skill.id, …)`, never by version, so successive
+ *   versions of a skill hit the same cells and their receipts compare.
+ *
+ * Modes (Owen, 2026-10-14):
+ * - `smoke` (about 1h, ≤12 Lab runs): Lab `default` on 10 stratified
+ *   `should_run` core repositories plus one run per smoke environment.
+ * - `full` (about 3h of Lab plus a Desktop matrix): every Lab `should_run`
+ *   core cell (failures re-run, up to 3 attempts), the full environments
+ *   sampled, 3 live `should_refuse` cells, and Desktop shards (Linux about 12
+ *   runs, Windows ≤18, macOS ≤6; macOS only on a minor or major bump).
+ * - `sweep` (weekly, and when a skill is first seeded): `full` with the
+ *   extended entries live too. It never gates a PR.
+ */

+const cellId = (entry: string, os: BenchOs, env: string) =>
+  `${entry}|${os}|${env}`

+/** Deterministic order for picks: by hash of the skill id (not version) and the key. */
+const byHash = <T>(
+  skillId: string,
+  salt: string,
+  items: T[],
+  key: (t: T) => string,
+) =>
+  [...items].sort((a, b) => {
+    const ha = sha256Hex(`${skillId}|${salt}|${key(a)}`)
+    const hb = sha256Hex(`${skillId}|${salt}|${key(b)}`)
+    return ha < hb ? -1 : ha > hb ? 1 : 0
+  })

+/** The surfaces a mode and bump require (B5: patch → Lab + Windows; minor/major → all). */
+export function benchSurfaces(
+  mode: BenchMode,
+  bump: BenchBump,
+): Array<'lab' | `desktop-${BenchOs}`> {
+  if (mode === 'smoke') return ['lab']
+  return bump === 'patch'
+    ? ['lab', 'desktop-linux', 'desktop-win32']
+    : ['lab', 'desktop-linux', 'desktop-win32', 'desktop-darwin']

+function toCell(
+  entry: LabBenchV2Entry,
+  os: BenchOs,
+  surface: BenchCell['surface'],
+  env: string,
+  verdict: BenchOracleVerdict,
+  attempts: number,
+): BenchCell {
+  return {
+    id: cellId(entry.key, os, env),
+    entry: entry.key,
+    os,
+    surface,
+    env,
+    expected: verdict.expected,
+    reason: verdict.reason,
+    unverified: verdict.unverified,
+    ...(verdict.fidelity ? { fidelity: verdict.fidelity } : {}),
+    attempts,
+  }

+/** Round-robin over buckets so a pick of n spans stacks, not one bucket. */
+function stratified(
+  skillId: string,
+  entries: LabBenchV2Entry[],
+  n: number,
+): LabBenchV2Entry[] {
+  const buckets = new Map<string, LabBenchV2Entry[]>()
+  for (const entry of byHash(skillId, 'stratify', entries, (e) => e.key)) {
+    const list = buckets.get(entry.bucket) ?? []
+    list.push(entry)
+    buckets.set(entry.bucket, list)
+  }
+  const order = byHash(skillId, 'buckets', [...buckets.keys()], (b) => b)
+  const out: LabBenchV2Entry[] = []
+  while (out.length < n && order.some((b) => buckets.get(b)!.length))
+    for (const bucket of order) {
+      const next = buckets.get(bucket)!.shift()
+      if (next && out.length < n) out.push(next)
+    }
+  return out
+}

+export async function planBench(
+  skill: SponsoredSkill,
+  mode: BenchMode,
+  options: { bump?: BenchBump; pool?: readonly LabBenchV2Entry[] } = {},
+): Promise<BenchPlan> {
+  const bump = options.bump ?? 'patch'
+  const pool = options.pool ?? LAB_BENCH_V2
+  const L = BENCH_PLAN_LIMITS
+
+  const verdicts = new Map<string, BenchOracleVerdict>()
+  const label = async (
+    entry: LabBenchV2Entry,
+    os: BenchOs,
+    env: BenchEnvironment,
+  ) => {
+    const id = cellId(entry.key, os, env.key)
+    let verdict = verdicts.get(id)
+    if (!verdict) {
+      verdict = await benchExpectation(
+        skill,
+        benchV2Snapshot(entry.key),
+        os,
+        env,
+      )
+      verdicts.set(id, verdict)
+    }
+    return verdict
+  }
+
+  const staticCells: BenchCell[] = []
+  for (const entry of pool)
+    for (const os of BENCH_OSES) {
+      const env = benchDefaultEnvironment(
+        os === 'linux' ? 'lab' : 'desktop',
+        os,
+      )
+      staticCells.push(
+        toCell(entry, os, 'static', 'default', await label(entry, os, env), 0),
+      )
+    }
+
+  const liveTiers = mode === 'sweep' ? ['core', 'extended'] : ['core']
+  const livePool = pool.filter((e) => liveTiers.includes(e.tier))
+  const labPool = livePool.filter((e) => e.kind === 'github')
+  const live: BenchCell[] = []
+  const add = async (
+    entry: LabBenchV2Entry,
+    os: BenchOs,
+    env: BenchEnvironment,
+    attempts: number,
+  ) => {
+    const verdict = await label(entry, os, env)
+    if (!verdict.applicable) return false
+    const id = cellId(entry.key, os, env.key)
+    if (live.some((c) => c.id === id)) return false
+    live.push(toCell(entry, os, env.surface, env.key, verdict, attempts))
+    return true
+  }
+  const labDefault = benchDefaultEnvironment('lab', 'linux')
+  const runsOn = async (
+    entry: LabBenchV2Entry,
+    os: BenchOs,
+    env: BenchEnvironment,
+  ) => (await label(entry, os, env)).expected === 'should_run'
+
+  // Lab, default environment.
+  const labShouldRun: LabBenchV2Entry[] = []
+  for (const entry of labPool)
+    if (await runsOn(entry, 'linux', labDefault)) labShouldRun.push(entry)
+  if (mode === 'smoke') {
+    for (const entry of stratified(
+      skill.id,
+      labShouldRun,
+      L.smokeLabRepositories,
+    ))
+      await add(entry, 'linux', labDefault, 1)
+  } else {
+    for (const entry of labShouldRun)
+      await add(
+        entry,
+        'linux',
+        labDefault,
+        mode === 'full' ? L.fullAttempts : 1,
+      )
+  }
+
+  // Lab environment cases, sampled where the skill runs.
+  const labEnvs = BENCH_ENVIRONMENTS.filter(
+    (env) =>
+      env.surface === 'lab' &&
+      env.key !== labDefault.key &&
+      (mode === 'smoke' ? env.tier === 'smoke' : true),
+  )
+  const perEnv = mode === 'smoke' ? 1 : L.fullEnvironmentCells
+  for (const env of labEnvs) {
+    let picked = 0
+    for (const entry of byHash(skill.id, env.key, labShouldRun, (e) => e.key)) {
+      if (picked >= perEnv) break
+      if (await add(entry, 'linux', env, 1)) picked++
+    }
+  }
+
+  // Live refusals: the skill must stop itself on a repository it does not fit.
+  if (mode !== 'smoke') {
+    const refusing: LabBenchV2Entry[] = []
+    for (const entry of labPool)
+      if (
+        (await label(entry, 'linux', labDefault)).expected === 'should_refuse'
+      )
+        refusing.push(entry)
+    for (const entry of byHash(
+      skill.id,
+      'refuse',
+      refusing,
+      (e) => e.key,
+    ).slice(0, L.fullRefusalCells))
+      await add(entry, 'linux', labDefault, 1)
+  }
+
+  // Desktop shards.
+  const surfaces = benchSurfaces(mode, bump)
+  for (const os of BENCH_OSES) {
+    if (!surfaces.includes(`desktop-${os}`)) continue
+    const envs = BENCH_ENVIRONMENTS.filter(
+      (env) => env.surface === 'desktop' && env.os === os,
+    )
+    const base = benchDefaultEnvironment('desktop', os)
+    const shouldRun: LabBenchV2Entry[] = []
+    for (const entry of livePool)
+      if (await runsOn(entry, os, base)) shouldRun.push(entry)
+    for (const entry of stratified(skill.id, shouldRun, L.desktop[os].default))
+      await add(entry, os, base, 1)
+    for (const env of envs.filter((e) => e.key !== base.key)) {
+      let picked = 0
+      for (const entry of byHash(skill.id, env.key, shouldRun, (e) => e.key)) {
+        if (picked >= L.desktop[os].perEnvironment) break
+        if (await add(entry, os, env, 1)) picked++
+      }
+    }
+  }
+
+  // Lab batches: one per environment, split to the Lab's caps. Each attempt
+  // after the first is a re-run of that cell's failures, planned by the runner.
+  const labBatches: BenchLabBatchPlan[] = []
+  const byEnv = new Map<string, BenchCell[]>()
+  for (const cell of live.filter((c) => c.surface === 'lab'))
+    byEnv.set(cell.env, [...(byEnv.get(cell.env) ?? []), cell])
+  const size = Math.min(LAB_BATCH_MAX_REPOSITORIES, LAB_BATCH_MAX_RUNS)
+  for (const [envKey, cells] of byEnv) {
+    const env = BENCH_ENVIRONMENTS.find((e) => e.key === envKey)!
+    for (let i = 0; i < cells.length; i += size) {
+      const chunk = cells.slice(i, i + size)
+      labBatches.push({
+        environment: env.lab!,
+        repositories: chunk.map((c) => {
+          const entry = pool.find((e) => e.key === c.entry)!
+          if (entry.kind !== 'github')
+            throw new Error(`${c.entry} is not on GitHub.`)
+          return labBenchPinSpec(entry)
+        }),
+        cells: chunk.map((c) => c.id),
+      })
+    }
+  }
+
+  const desktopShards: BenchDesktopShardPlan[] = []
+  for (const cell of live.filter((c) => c.surface === 'desktop')) {
+    let shard = desktopShards.find(
+      (s) =>
+        s.os === cell.os &&
+        s.env === cell.env &&
+        s.entries.length < L.desktopShardEntries,
+    )
+    if (!shard) {
+      shard = { os: cell.os, env: cell.env, entries: [], cells: [] }
+      desktopShards.push(shard)
+    }
+    shard.entries.push(cell.entry)
+    shard.cells.push(cell.id)
+  }
+
+  const labRuns = live
+    .filter((c) => c.surface === 'lab')
+    .reduce((n, c) => n + c.attempts, 0)
+  const desktopRuns = Object.fromEntries(
+    BENCH_OSES.map((os) => [
+      os,
+      live.filter((c) => c.surface === 'desktop' && c.os === os).length,
+    ]),
+  ) as Record<BenchOs, number>
+  const githubMinutesLinuxEquivalent = BENCH_OSES.reduce(
+    (n, os) =>
+      n + desktopRuns[os] * L.desktopRunMinutes * L.githubMinuteMultiplier[os],
+    0,
+  )
+  return {
+    benchVersion: LAB_BENCH_V2_VERSION,
+    mode,
+    bump,
+    skill: sponsoredSkillKey(skill),
+    skillSha256: sponsoredSkillSha256(skill),
+    static: staticCells,
+    live,
+    labBatches,
+    desktopShards,
+    estimate: {
+      labRuns,
+      desktopRuns,
+      labWallMinutes:
+        Math.ceil(labRuns / L.labConcurrentRuns) * L.labRunMinutes,
+      // Shards run in parallel; the longest is one OS's shard of one env.
+      desktopWallMinutes: Math.max(
+        0,
+        ...desktopShards.map((s) => s.entries.length * L.desktopRunMinutes),
+      ),
+      githubMinutesLinuxEquivalent,
+    },
+  }
+}
