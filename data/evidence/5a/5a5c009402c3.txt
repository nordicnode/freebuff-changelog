```
/**
 * Where each turn of a local Claude Code chat ended in the CLI's own transcript, so Roll back can
 * cut the model's memory at a message (web history-store.ts `rollbackChat`, claude-rollback.ts).
 *
 * Desktop's Claude harness (claude-code.ts) appends one mark to the chat's saved state at the end
 * of every turn that produced a reply. A rollback keeps the marks of the runs it keeps and resumes
 * the last one's session AT its last entry, branched into a new session (`resumeSessionAt` with
 * `forkSession`), which is how Claude Code's own rewind drops the turns after a message. A mark
 * names its own session: a fork or rollback moves the chat to a new session id, and the kept turns
 * still live in the transcript of the session they ran in, which stays on this computer.
 */
```

```
/** Strip SDK URL/person enrichment before either capture or the log mirror. */
export function sanitizeProductFunnelCapture<
  T extends {
    event: string
    properties: Record<string, unknown>
  },
>(event: T, projectToken: string): T | null {
  if (event.event !== PRODUCT_FUNNEL_EVENT) return event
  const fields: Record<string, unknown> = {}
  for (const key of productFunnelEventSchema.keyof().options) {
    if (event.properties[key] !== undefined) fields[key] = event.properties[key]
  }
  const parsed = productFunnelEventSchema.safeParse(fields)
  if (!parsed.success) return null
  // posthog-js adds this public ingest key before before_send; transport does
  // not restore it after sanitization. Never copy an untrusted event token.
  const properties: Record<string, unknown> = {
    ...parsed.data,
    token: projectToken,
  }
  for (const key of [
    'distinct_id',
    '$insert_id',
    '$session_id',
    '$is_identified',
  ]) {
    const value = event.properties[key]
    if (typeof value === 'string' || typeof value === 'boolean')
      properties[key] = value
  }
  const sanitized = { ...event, properties }
  Reflect.deleteProperty(sanitized, '$set')
  Reflect.deleteProperty(sanitized, '$set_once')
  return sanitized
}
```

```
/**
 * The user-facing message for a native turn's journal `error`, or null when it
 * is not one we can name. Desktop builds before these messages existed wrote the
 * SDK's own text for every abort ("Claude Code process aborted by user").
 *
 * The CLI runs on the user's computer with their own sign-in, so most failures
 * are theirs to fix there: signed out, plan limit used up, a model their plan
 * cannot use. Only fixed classes are named; the CLI's raw text never is (it can
 * quote the request).
 */
export function deviceHarnessFailureMessage(
  message: string,
  harness?: DeviceHarness,
): string | null {
  if (
    message === DEVICE_HARNESS_TIME_LIMIT_MESSAGE ||
    message === LEGACY_DEVICE_HARNESS_TIME_LIMIT_MESSAGE ||
    message === DEVICE_HARNESS_STOPPED_MESSAGE
  )
    return message
  if (/process aborted by user/i.test(message))
    return DEVICE_HARNESS_STOPPED_MESSAGE
  if (message.length <= 1000 && DESKTOP_WORDED.some((pattern) => pattern.test(message)))
    return message
  if (!harness) return null
  const label = HARNESS_LABEL[harness]
  const account = HARNESS_ACCOUNT[harness]
  if (
    /\b(hit your (usage )?limit|usage limit reached|(5-hour|weekly|session) limit reached)\b/i.test(
      message,
    )
  ) {
    const reset = limitReset(message)
    return `Your ${account} plan’s ${label} usage limit is used up${reset ? ` until ${reset}` : ''}. Wait for it to reset, or ${PICK_FREEBUFF}.`
  }
  const unsupported =
    /The '([\w.:/-]{1,80})' model is not supported when using Codex with a ChatGPT account/.exec(
      message,
    )
  if (unsupported)
    return `Your ChatGPT plan can’t use ${unsupported[1]} in Codex. Choose another model from the model picker.`
  if (/spend cap set by the owner of your workspace/i.test(message))
    return `Your ${account} workspace’s spend cap for ${label} is used up. Ask a workspace owner to raise it, or ${PICK_FREEBUFF}.`
  if (/\bworkspace is out of credits\b/i.test(message))
    return `Your ${account} workspace is out of ${label} credits. Add credits there, or ${PICK_FREEBUFF}.`
  if (/^Quota exceeded\. Check your plan and billing details|credit balance is too low/i.test(message))
    return `The account ${label} signs in with on your computer is out of quota or credit. Check its plan and billing, or ${PICK_FREEBUFF}.`
  if (new RegExp(`^Could not install ${label}\\b`).test(message))
    return `Freebuff couldn’t install ${label} on your computer. Check that computer’s internet connection, then send your message again.`
  return null
}
```

```
/** Capability detection is presentation only; native IPC checks its sender/account. */
export function hostedWorkspaceHost(
  value: unknown,
): HostedWorkspaceHost | null {
  if (!value || typeof value !== 'object') return null
  const host = value as Partial<HostedWorkspaceHost>
  return host.protocolVersion === HOSTED_WORKSPACE_PROTOCOL_VERSION &&
    typeof host.platform === 'string' &&
    typeof host.appVersion === 'string' &&
    ['status', 'openLocalProject', 'disconnect', 'returnToDesktop'].every(
      (name) => typeof host[name as keyof HostedWorkspaceHost] === 'function',
    )
    ? (host as HostedWorkspaceHost)
    : null
}
```

```
export type SettingsSearchPage = { id: string; label: PageLabel }

function normalize(value: string) {
  return value
    .normalize('NFKD')
    .replace(/\p{M}/gu, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim()
}

// A single missing, extra, or mistyped letter; short words must match literally.
function nearWord(query: string, word: string) {
  if (query.length < 4 || Math.abs(query.length - word.length) > 1) return false
  let edits = 0
  for (let i = 0, j = 0; i < query.length || j < word.length; ) {
    if (query[i] === word[j]) {
      i++
      j++
      continue
    }
    if (++edits > 1) return false
    if (query.length >= word.length) i++
    if (word.length >= query.length) j++
  }
  return true
}

function score(query: string, label: string, keywords = '') {
  const name = normalize(label)
  if (name === query) return 100
  if (name.startsWith(query)) return 90
  if (name.includes(query)) return 80
  const words = normalize(`${label} ${keywords}`).split(' ')
  let fuzzy = false
  for (const token of query.split(' ')) {
    if (words.some((word) => word.startsWith(token))) continue
    if (!words.some((word) => nearWord(token, word))) return 0
    fuzzy = true
  }
  return fuzzy ? 10 : 50
}

export function searchSettings<T extends SettingsSearchPage>(
  query: string,
  pages: readonly T[],
  surface: Surface,
  hidden: readonly string[] = [],
): { page: T; detail?: string }[] {
  const normalized = normalize(query)
  if (!normalized) return query.trim() ? [] : pages.map((page) => ({ page }))
  return pages
    .map((page) => {
      let best = score(normalized, page.label)
      let detail: string | undefined
      for (const topic of topics[page.label]) {
        if (topic.surface && topic.surface !== surface) continue
        if (hidden.includes(topic.label)) continue
        const value = score(
          normalized,
          topic.label,
          `${page.label} ${topic.keywords}`,
        )
        if (value > best) {
          best = value
          detail = topic.label
        }
      }
      return { page, detail, score: best }
    })
    .filter((result) => result.score > 0)
    .sort((a, b) => b.score - a.score)
}
```

```
/**
 * Where each turn of a local Claude Code chat ended in the CLI's own transcript, so Roll back can
 * cut the model's memory at a message (web history-store.ts `rollbackChat`, claude-rollback.ts).
 *
 * Desktop's Claude harness (claude-code.ts) appends one mark to the chat's saved state at the end
 * of every turn that produced a reply. A rollback keeps the marks of the runs it keeps and resumes
 * the last one's session AT its last entry, branched into a new session (`resumeSessionAt` with
 * `forkSession`), which is how Claude Code's own rewind drops the turns after a message. A mark
 * names its own session: a fork or rollback moves the chat to a new session id, and the kept turns
 * still live in the transcript of the session they ran in, which stays on this computer.
 */
export type ClaudeTurnMark = {
  /** The Freebuff run (`freebuff_cloud_run.id`) this turn answered. */
  run: string
  /** The CLI session the turn ran in. */
  session: string
  /** The uuid of the turn's last top-level transcript entry: its final assistant message. */
  end: string
}

/** Only the most recent turns can be rolled back to; older marks are dropped. */
export const CLAUDE_TURN_MARKS_KEPT = 200

/** CLI session ids and transcript entry uuids; anything else is refused before it reaches a CLI. */
export const CLAUDE_TRANSCRIPT_ID = /^[a-zA-Z0-9_-]{1,128}$/

export function isClaudeTurnMark(value: unknown): value is ClaudeTurnMark {
  if (!value || typeof value !== 'object') return false
  const { run, session, end } = value as Record<string, unknown>
  return (
    typeof run === 'string' &&
    run.length > 0 &&
    typeof session === 'string' &&
    CLAUDE_TRANSCRIPT_ID.test(session) &&
    typeof end === 'string' &&
    CLAUDE_TRANSCRIPT_ID.test(end)
  )
}

/** The well-formed marks a saved state carries, oldest first. */
export function claudeTurnMarks(state: unknown): ClaudeTurnMark[] {
  if (!state || typeof state !== 'object') return []
  const turns = (state as { turns?: unknown }).turns
  return Array.isArray(turns) ? turns.filter(isClaudeTurnMark) : []
}
```
