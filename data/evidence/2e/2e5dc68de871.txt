/**
 * Hosting addresses users may not claim, shared by freebuff.app hosting
 * (`<slug>.freebuff.app`) and Sites (`<label>.freebuff.site`). Either is a
 * public address on our own domain, so a claimed `login` or `account` is a
 * ready-made phishing page carrying our name. Also covers the hostnames
 * infrastructure tends to expect (`api`, `cdn`, `mail`) and ones that ARE
 * infrastructure — `auth.freebuff.app` is the sign-in email relay generated
 * projects call — whose meaning must never be user-controlled.
 *
 * freebuff.app checks it in two layers: at claim time (createDeployment / slug
 * rename in freebuff/web/convex/deployment.ts, where the user gets a friendly
 * error) and again in the deploy pipeline itself (getOrCreateVercelProject),
 * which every path — web, cloud, agent redeploys, browser-publish finalize —
 * funnels through, so a slug that slipped in through an older row or a path
 * without the claim check still cannot reach a `<reserved>.freebuff.app`
 * domain. Sites checks it, with `impersonatesFreebuff` below, in
 * siteSubdomainError (sites/subdomain.ts) whenever an address is chosen.
 *
 * Offensive names are refused at the same places by
 * freebuff/web/lib/offensiveHostingSlugs.ts, which stays out of common/
 * (published) so its terms never leave the private repo.
 */

/**
 * Parse ONE argument, keeping its name on the issue.
 *
 * `schema.parse(args.name)` produces an issue with an empty path, so a refusal
 * could only ever name the failing CHECK and not the argument — the half of
 * this that a model cannot work around.
 */
function field<T>(schema: z.ZodType<T>, argName: string, value: unknown): T {
  return z.object({ [argName]: schema }).parse({ [argName]: value })[
    argName
  ] as T
}

/**
 * Name the arguments a Zod check rejected, and nothing else.
 *
 * The opaque fallback below exists because a filesystem or validator error may
 * carry file contents or a secret value. A Zod issue cannot: this reports only
 * the argument's path and the kind of check that failed, never the value that
 * failed it and never the issue's own message (`invalid_value` carries the
 * received literal). Without this an argument the model can fix reads as
 * "check arguments and workspace files", so it retries the same call.
 */
function rejectedArguments(error: z.ZodError): string {
  const named = [
    ...new Set(
      error.issues.flatMap((issue) => {
        const path = issue.path.map(String).join('.')
        // An extra argument's issue sits on the object itself; its NAME (a key
        // the model chose, never a value) is what it has to drop.
        if (issue.code === 'unrecognized_keys')
          return issue.keys.map(
            (key) =>
              `${path ? `${path}.` : ''}${key.slice(0, 64)} (unrecognized_keys)`,
          )
        return [`${path || 'the arguments object'} (${issue.code})`]
      }),
    ),
  ]
  return named.length > 0
    ? `Sites rejected these arguments: ${named.join(', ')}. Read the tool's input schema and send values that match it.`
    : 'Could not prepare the Sites request. Check arguments and workspace files.'
}

import {
  impersonatesFreebuff,
  isReservedHostingSlug,
  withoutFreebuffBrand,
} from './reserved-names'

/**
 * DNS labels only. Reserved: the generated address namespace, the DNS target,
 * the hosting reserved words (`login`, `admin`…), any Freebuff or Codebuff
 * look-alike (reserved-names.ts), and punycode `xn--` labels, which a browser
 * can render as Unicode look-alikes no ASCII rule sees.
 */
export function siteSubdomainError(input: string): string | null {
  const name = input.trim().toLowerCase()
  if (!/^[a-z0-9](?:[a-z0-9-]{0,61}[a-z0-9])?$/.test(name))
    return 'Use 1–63 letters, numbers or hyphens. Start and end with a letter or number.'
  if (
    name === 'customers' ||
    /^site-[a-f0-9]{32}$/.test(name) ||
    name.startsWith('xn--') ||
    isReservedHostingSlug(name) ||
    impersonatesFreebuff(name)
  )
    return 'This subdomain is reserved. Choose another name.'
  return null
}

const SHORT_ID_LENGTH = 6
const PREVIEW_SUFFIX = '-preview'
// Leaves room for `-<id>-preview` inside a 63-character DNS label.
const NAME_SLUG_MAX = 63 - 1 - SHORT_ID_LENGTH - PREVIEW_SUFFIX.length

/**
 * `My Cool App!` → `my-cool-app`; falls back to `site` when nothing survives.
 * Words that read as Freebuff or Codebuff are dropped rather than refused, so
 * `Freebuff Airdrop` → `airdrop`: a generated address never reads as ours.
 */
export function siteNameSlug(name: string) {
  const slug = withoutFreebuffBrand(
    name
      .normalize('NFKD')
      .replace(/[\u0300-\u036f]/g, '')
      .toLowerCase()
      .replace(/[^a-z0-9]+/g, '-'),
  )
    .slice(0, NAME_SLUG_MAX)
    .replace(/-+$/, '')
  return slug || 'site'
}

/** Lowercase base36 suffix that keeps generated addresses distinct. */
export function siteShortId(random: Uint8Array) {
  if (random.length < SHORT_ID_LENGTH) throw new Error('Not enough randomness.')
  const alphabet = 'abcdefghijklmnopqrstuvwxyz0123456789'
  return Array.from(
    random.subarray(0, SHORT_ID_LENGTH),
    (byte) => alphabet[byte % alphabet.length],
  ).join('')
}

/** The addresses a new site starts with, before any rename. */
export function generatedSiteSubdomains(slug: string, shortId: string) {
  const production = `${slug}-${shortId}`
  return { production, preview: production + PREVIEW_SUFFIX }
}

export function siteHostname(env: {
  subdomain?: string | null
  worker_name: string
  domain: string
}) {
  return `${env.subdomain ?? env.worker_name}.${env.domain}`
}

export interface SiteSubdomainAvailability {
  subdomain: string
  available: boolean
}

/**
 * Hosting addresses users may not claim, shared by freebuff.app hosting
 * (`<slug>.freebuff.app`) and Sites (`<label>.freebuff.site`). Either is a
 * public address on our own domain, so a claimed `login` or `account` is a
 * ready-made phishing page carrying our name. Also covers the hostnames
 * infrastructure tends to expect (`api`, `cdn`, `mail`) and ones that ARE
 * infrastructure — `auth.freebuff.app` is the sign-in email relay generated
 * projects call — whose meaning must never be user-controlled.
 *
 * freebuff.app checks it in two layers: at claim time (createDeployment / slug
 * rename in freebuff/web/convex/deployment.ts, where the user gets a friendly

 /** Lowercase base36 suffix that keeps generated addresses distinct. */
 export function siteShortId(random: Uint8Array) {
   if (random.length < SHORT_ID_LENGTH) throw new Error('Not enough randomness.')
   const alphabet = 'abcdefghijklmnopqrstuvwxyz0123456789'
   return Array.from(
     random.subarray(0, SHORT_ID_LENGTH),
     (byte) => alphabet[byte % alphabet.length],
   ).join('')
 }
