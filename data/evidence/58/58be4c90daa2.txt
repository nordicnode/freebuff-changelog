import { z } from 'zod/v4'
import { FILE_UPLOAD_BASE64_MAX, validFileUploadData } from './file-upload'
import { BACKGROUND_TASK_ID } from './hosted-tasks'

const id = z.string().regex(/^[a-zA-Z0-9_-]{1,120}$/)
const scope = { projectId: id, threadId: id }
const path = z.string().max(4096)
const dimensions = { cols: z.number().int().min(2).max(500), rows: z.number().int().min(1).max(200) }
const terminal = { ...scope, sessionId: z.string().regex(/^[a-zA-Z0-9_-]{1,128}$/) }
const viewer = { ...terminal, viewerId: z.string().uuid() }
const generation = { ...viewer, generation: z.string().min(1).max(128) }

/** Selectors only. The server derives ownership and binding; the device resolves paths. */
export const devicePanelRequest = z.discriminatedUnion('panel', [
  z.object({ panel: z.literal('files'), input: z.discriminatedUnion('operation', [
    z.object({ ...scope, operation: z.literal('list'), directory: path.optional(), query: z.string().max(512).optional() }).strict(),
    z.object({ ...scope, operation: z.literal('read'), path, expanded: z.boolean() }).strict(),
    z.object({ ...scope, operation: z.literal('save'), path, content: z.string().max(512 * 1024), revision: z.string().regex(/^[a-f0-9]{40,64}$/) }).strict(),
    z.object({ ...scope, operation: z.literal('create'), path, kind: z.enum(['file', 'directory']) }).strict(),
    z.object({ ...scope, operation: z.literal('upload'), path, data: z.string().max(FILE_UPLOAD_BASE64_MAX).refine(value => validFileUploadData(value)) }).strict(),
    z.object({ ...scope, operation: z.literal('download'), path, offset: z.number().int().min(0).max(Number.MAX_SAFE_INTEGER) }).strict(),
  ]) }).strict(),
  z.object({ panel: z.literal('git'), input: z.discriminatedUnion('operation', [
    z.object({ ...scope, operation: z.literal('changes'), scope: z.enum(['all', 'uncommitted']) }).strict(),
    z.object({ ...scope, operation: z.literal('diff'), scope: z.enum(['all', 'uncommitted']), path, untracked: z.boolean() }).strict(),
    z.object({ ...scope, operation: z.literal('summary'), live: z.boolean() }).strict(),
    z.object({ ...scope, operation: z.literal('pullRequest'),
      repository: z.string().regex(/^[A-Za-z0-9-]{1,39}\/[A-Za-z0-9_.-]{1,100}$/), number: z.number().int().min(1).max(Number.MAX_SAFE_INTEGER) }).strict(),
  ]) }).strict(),
  z.object({ panel: z.literal('tasks'), input: z.discriminatedUnion('operation', [
    z.object({ ...scope, operation: z.literal('list') }).strict(),
    z.object({ ...scope, operation: z.literal('stop'), taskId: z.string().regex(BACKGROUND_TASK_ID) }).strict(),
  ]) }).strict(),
  z.object({ panel: z.literal('terminal'), input: z.discriminatedUnion('operation', [
    z.object({ ...terminal, operation: z.literal('attach'), cols: dimensions.cols.optional(), rows: dimensions.rows.optional(),
      resumeGeneration: z.string().regex(/^[a-zA-Z0-9_-]{1,128}$/).optional() }).strict(),
    z.object({ ...viewer, operation: z.literal('read') }).strict(),
    z.object({ ...viewer, operation: z.literal('detach') }).strict(),
    z.object({ ...generation, operation: z.literal('input'), data: z.string().max(16 * 1024) }).strict(),
    z.object({ ...generation, ...dimensions, operation: z.literal('resize') }).strict(),
    z.object({ ...terminal, operation: z.literal('close') }).strict(),
  ]) }).strict(),
])
export type DevicePanelRequest = z.infer<typeof devicePanelRequest>
export const DEVICE_PANEL_MS = 25_000
export const DEVICE_PANEL_LIMIT = 32
export const DEVICE_PANEL_BYTES = 4 * 1024 * 1024
export const devicePanelOperation = z.object({
  id: z.string().uuid(), sessionId: z.string().uuid(), bindingId: z.string().uuid(),
  expiresAt: z.number().int(), request: devicePanelRequest,
}).strict()
export type DevicePanelOperation = z.infer<typeof devicePanelOperation>
export const devicePanelResult = z.discriminatedUnion('ok', [
  z.object({ ok: z.literal(true), value: z.unknown() }).strict(),
  z.object({ ok: z.literal(false), error: z.string().max(4000) }).strict(),
])
export type DevicePanelResult = z.infer<typeof devicePanelResult>

- common/src/desktop-cloud/hosted-git.ts (45 lines):
```
import type { ChangesScope, CloudChanges, FileDiffResult } from './changes'
import type { HostedBrowserScope } from './hosted-workspace'
import type { PullRequestSnapshot, PullRequestStatus } from './pull-request'

/**
 * Where a chat's changes live and where they go: its branch, what "branch changes" compare
 * against, and its pull request. `live` asks Desktop to refresh the PR (one cached `gh` read);
 * without it the answer is what Desktop already saved, which the sidebar reads for every row.
 * Desktop builds before 2026-11 reject the operation; callers treat that as "nothing known".
 */
export interface GitSummary {
  branch: string | null
  /** The branch "Branch changes" diff against; null when the chat has none (a shared checkout). */
  base: string | null
  worktree: boolean
  pullRequest: PullRequestStatus | null
  /** The open PR no longer merges cleanly. */
  conflict: boolean
  /**
   * Where this chat's PR would be on github.com: its `origin` and the branch it owns. Lets the
   * browser read the PR with the account's GitHub connection when `gh` cannot answer (not installed,
   * signed out, another account). Null for a shared checkout; absent from Desktop before 2026-11.
   */
  github?: { repository: string; branch: string } | null
}

/** Git reads are scoped to the native chat, with no renderer-supplied repository path. */
export type HostedGitRequest = HostedBrowserScope & (
  | { operation: 'changes'; scope: ChangesScope }
  | { operation: 'diff'; scope: ChangesScope; path: string; untracked: boolean }
  | { operation: 'summary'; live: boolean }
  /** A PR the chat touched, read with the computer's `gh`; `pullRequest` is null when `gh` cannot see it.
   *  An object, never a bare null, which the device relay refuses. Desktop builds before 2026-11-16 reject it. */
  | { operation: 'pullRequest'; repository: string; number: number }
)
export type HostedGitResponses = {
  changes: CloudChanges
  diff: FileDiffResult
  summary: GitSummary
  pullRequest: PullRequestSnapshot
}
export type HostedGitHost = {
  request<T extends HostedGitRequest>(input: T): Promise<HostedGitResponses[T['operation']]>
}
```
