/**
 * Which placement a stored `via` names, and whether it was a copied link or an
 * emailed invite, for `freebuff.referral.activated` (referral-v2.ts). A null
 * or unknown `via` is the unmarked bucket, with no method.
 */
export function referralPlacementFromVia(via: string | null | undefined): {
  placement: string
  method: 'link' | 'email' | null
} {
  return (
    (via ? PLACEMENT_BY_VIA.get(via) : undefined) ?? {
      placement: UNMARKED_REFERRAL_PLACEMENT,
      method: null,
    }
  )
}

/** True for a `via` one of our builders puts on a link or an emailed invite. */
export function isKnownReferralVia(via: unknown): via is string {
  return typeof via === 'string' && PLACEMENT_BY_VIA.has(via)
}

/** Strip SDK URL/person enrichment before either capture or the log mirror. */
export function sanitizeProductFunnelCapture<
  T extends {
    event: string
    properties: Record<string, unknown>
  },
>(event: T, projectToken: string): T | null {
  if (event.event !== PRODUCT_FUNNEL_EVENT) return event
  const fields: Record<string, unknown> = {}
  for (const key of productFunnelEventSchema.keyof().options) {
    if (event.properties[key] !== undefined) fields[key] = event.properties[key]
  }
  const parsed = productFunnelEventSchema.safeParse(fields)
  if (!parsed.success) return null
  // posthog-js adds this public ingest key before before_send; transport does
  // not restore it after sanitization. Never copy an untrusted event token.
  const properties: Record<string, unknown> = {
    ...parsed.data,
    token: projectToken,
  }
  for (const key of [
    'distinct_id',
    '$insert_id',
    '$session_id',
    '$is_identified',
  ]) {
    const value = event.properties[key]
    if (typeof value === 'string' || typeof value === 'boolean')
      properties[key] = value
  }
  const sanitized = { ...event, properties }
  Reflect.deleteProperty(sanitized, '$set')
  Reflect.deleteProperty(sanitized, '$set_once')
  return sanitized
}

export type SettingsSearchPage = { id: string; label: PageLabel }

function normalize(value: string) {
  return value
    .normalize('NFKD')
    .replace(/\p{M}/gu, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim()
}

// A single missing, extra, or mistyped letter; short words must match literally.
function nearWord(query: string, word: string) {
  if (query.length < 4 || Math.abs(query.length - word.length) > 1) return false
  let edits = 0
  for (let i = 0, j = 0; i < query.length || j < word.length; ) {
    if (query[i] === word[j]) {
      i++
      j++
      continue
    }
    if (++edits > 1) return false
    if (query.length >= word.length) i++
    if (word.length >= query.length) j++
  }
  return true
}

function score(query: string, label: string, keywords = '') {
  const name = normalize(label)
  if (name === query) return 100
  if (name.startsWith(query)) return 90
  if (name.includes(query)) return 80
  const words = normalize(`${label} ${keywords}`).split(' ')
  let fuzzy = false
  for (const token of query.split(' ')) {
    if (words.some((word) => word.startsWith(token))) continue
    if (!words.some((word) => nearWord(token, word))) return 0
    fuzzy = true
  }
  return fuzzy ? 10 : 50
}

export function searchSettings<T extends SettingsSearchPage>(
  query: string,
  pages: readonly T[],
  surface: Surface,
  hidden: readonly string[] = [],
): { page: T; detail?: string }[] {
  const normalized = normalize(query)
  if (!normalized) return query.trim() ? [] : pages.map((page) => ({ page }))
  return pages
    .map((page) => {
      let best = score(normalized, page.label)
      let detail: string | undefined
      for (const topic of topics[page.label]) {
        if (topic.surface && topic.surface !== surface) continue
        if (hidden.includes(topic.label)) continue
        const value = score(
          normalized,
          topic.label,
          `${page.label} ${topic.keywords}`,
        )
        if (value > best) {
          best = value
          detail = topic.label
        }
      }
      return { page, detail, score: best }
    })
    .filter((result) => result.score > 0)
    .sort((a, b) => b.score - a.score)
}
