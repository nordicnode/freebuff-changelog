File common/src/ads/sponsored-step-driver.ts:
/**
 * The step driver: the harness, not the model, decides when a sponsored
 * skill's steps are done (piece 3 of COD-822; the format is
 * `docs/ads/agentic/sponsored-skills.md`).
 *
 * After the run's turn has done its work, the driver walks the skill's steps
 * in order and runs each step's checks through THE skill evaluator
 * (`evaluateSponsoredSkillChecks`), the same one the sponsored eval scores
 * with. A step with no failing check is done, whatever the model said. A
 * failing `agent` or `check` step goes back to the agent with the failed
 * checks (`sponsoredSkillCheckFeedback`), to fix and stop; the checks run
 * again after each fix. The step's `maxAttempts` counts the first try (the
 * turn's own work) plus fixes, and the run as a whole gets at most
 * {@link SPONSORED_STEP_MAX_FIXES_PER_RUN}. Then the step is BLOCKED, and the
 * walk ends there: later steps usually build on it, and prompting them would
 * spend the run's compute on work that cannot pass. They are still checked
 * once, so the result says what is true on disk.
 *
 * - A `user` step is never prompted or retried. It is done only once its
 *   checks pass (the user handed over the credential), else it needs the user.
 * - A step that passed before a later fix is checked again at the end: a fix
 *   for step 4 can break step 2.
 *
 * DELIVERED means every `agent` and `check` step passed. That is the run's
 * completion, not the model writing "Done:".
 *
 * Pure: the caller hands in the check context (its worktree, its sandbox) and
 * how to prompt the agent, so the whole loop runs in tests without a model,
 * a sandbox or a disk.
 */

/**
 * THE fit oracle: whether a sponsored skill's preflight would refuse a
 * repository, decided from the repository's FACTS before anything is
 * offered. Pure, no I/O: serving (freebuff-web), Desktop, the Web worker, the
 * CLI and the bench all call it, over facts built by the one builder here
 * ({@link sponsoredRepoFacts}).
 *
 * Each preflight check gets a three-valued verdict, ANDed across checks:
 * - `requirements.os` against the device OS;
 * - `file_exists` when its glob can only match fit files
 *   (`SPONSORED_FIT_FILE_NAMES`), so the facts' listing decides it;
 * - `file_contains changedOnly:false` against the manifests' dependency names
 *   (projected back into manifest-shaped text and matched with the check's

 * - an `absent` check on a marker file (`.coderabbit.yaml`);
 * - a format-2 `fit` block directly.
 * Command and script checks are never fit checks: always `unknown`.
 *
 * Only a KNOWN misfit refuses ({@link sponsoredSkillFitRefusal}): an unknown
 * verdict serves, as today, and the skill's own preflight decides in the run.
 * Doc: `docs/ads/agentic/sponsored-skills.md`, "Fit".
 */

/**
 * How a skill's next version is numbered (docs/ads/agentic/sponsored-skills.md,
 * "Versions"). Shared by the skill improvement loop, which numbers its
 * candidates, and the Agentic Ads Lab, which numbers an admin's edits.
 */

import type { SponsoredSkill } from './sponsored-skill'

export type SkillBump = 'patch' | 'minor' | 'major'

/** Key order is not meaning: a skill read back from jsonb has its keys reordered. */
const sorted = (value: unknown): unknown =>
  Array.isArray(value)
    ? value.map(sorted)
    : value && typeof value === 'object'
      ? Object.fromEntries(
          Object.keys(value)
            .sort()
            .map((key) => [
              key,
              sorted((value as Record<string, unknown>)[key]),
            ]),
        )
      : value
const same = (a: unknown, b: unknown) =>
  JSON.stringify(sorted(a)) === JSON.stringify(sorted(b))

/** A check without its label: what decides pass or fail. */
const checkLogic = ({
  label: _label,
  ...check
}: SponsoredSkill['preflight'][number]) => check

/**
 * The bump a change needs. Major: steps added, removed, reordered or
 * re-kinded, a user step or its credential changed, or what the user
 * consents to or needs changed. Minor: a check's logic, a check added or
 * removed, attempts, a package or tool. Patch: everything else (wording,
 * knowledge, labels).
 */
export function requiredSkillBump(
  current: SponsoredSkill,
  next: SponsoredSkill,
): SkillBump {
  const shape = (s: SponsoredSkill) =>
    s.steps.map((step) => [step.id, step.kind])
  const userSteps = (s: SponsoredSkill) =>
    s.steps.filter((step) => step.kind === 'user')
  if (
    !same(shape(current), shape(next)) ||
    !same(userSteps(current), userSteps(next)) ||
    !same(current.consent, next.consent) ||
    !same(current.requirements, next.requirements) ||
    !same(current.desktop, next.desktop)
  )
    return 'major'
  const logic = (s: SponsoredSkill) => ({
    preflight: s.preflight.map(checkLogic),
    fit: s.fit,
    steps: s.steps.map((step) => [
      step.checks.map(checkLogic),
      step.maxAttempts,
    ]),
    packages: s.packages,
    tools: s.tools,
  })
  return same(logic(current), logic(next)) ? 'patch' : 'minor'
}

export function bumpVersion(version: string, bump: SkillBump): string {
  const [major = 0, minor = 0, patch = 0] = version
    .split('-')[0]!
    .split('.')
    .map(Number)
  if (bump === 'major') return `${major + 1}.0.0`
  if (bump === 'minor') return `${major}.${minor + 1}.0`
  return `${major}.${minor}.${patch + 1}`
}

/**

 * `{a,b}` is alternation. The step driver matches paths with this too, so
 * review and execution never disagree on what a glob covers.
 */
export function sponsoredSkillGlob(glob: string): RegExp {
  let source = ''
  let depth = 0
  for (let i = 0; i < glob.length; i++) {
    const char = glob[i]!
    if (char === '*') {
      if (glob[i + 1] === '*') {
        i++
        if (glob[i + 1] === '/') {
          i++
          source += '(?:.*/)?'
        } else {
          source += '.*'
        }
      } else {
        source += '[^/]*'
      }
    } else if (char === '?') source += '[^/]'
    else if (char === '{') {
      depth++
      source += '(?:'
    } else if (char === '}' && depth > 0) {
      depth--
      source += ')'
    } else if (char === ',' && depth > 0) source += '|'
    else source += char.replace(/[.+^$()|[\]\\]/g, '\\$&')
  }
  return new RegExp(`^${source}$`)
}

function validGlob(glob: unknown): glob is string {
  if (typeof glob !== 'string' || !glob || glob.length > 200) return false
  if (glob.startsWith('/') || glob.split('/').includes('..')) return false
  let depth = 0
  for (const char of glob) {
    if (char === '{') depth++
    if (char === '}') depth--
    if (depth < 0) return false
  }
  return depth === 0
}

function validRegex(source: unknown, flags: unknown): boolean {
  if (typeof source !== 'string' || !source || source.length > 300) return false
  if (
    flags !== undefined &&
    (typeof flags !== 'string' || !REGEX_FLAGS.test(flags))
  )
    return false
  try {
    new RegExp(source, flags as string | undefined)
    return true
  } catch {
    return false
  }
}

function text(
  errors: SponsoredSkillParseError[],
  path: string,
  value: unknown,
  max: number,
  { allowEmpty = false } = {},
): string {
  if (typeof value !== 'string' || (!allowEmpty && !value.trim())) {
    errors.push({ path, message: 'must be a non-empty string' })
    return ''
  }
  if (value.length > max) {
    errors.push({ path, message: `must be at most ${max} characters` })
  }
  return value
}

function timeout(
  errors: SponsoredSkillParseError[],
  path: string,
  value: unknown,
): void {
  if (value === undefined) return
  if (
    typeof value !== 'number' ||
    !Number.isInteger(value) ||
    value < 1 ||
    value > MAX_SKILL_TIMEOUT_SECONDS
  ) {
    errors.push({
      path,
      message: `must be an integer from 1 to ${MAX_SKILL_TIMEOUT_SECONDS}`,
    })
  }
}

function parseCheck(
  errors: SponsoredSkillParseError[],
  path: string,
  value: unknown,
  ids: Set<string>,
  tools: ReadonlySet<string>,
  stepKind: SponsoredSkillStepKind | 'preflight',
): void {
  if (!isRecord(value)) {
    errors.push({ path, message: 'must be an object' })
    return
  }
  if (typeof value.id !== 'string' || !SLUG.test(value.id)) {
    errors.push({ path: `${path}.id`, message: 'must be a lowercase slug' })
  } else if (ids.has(value.id)) {
    errors.push({
      path: `${path}.id`,
      message: `duplicate check id "${value.id}"`,
    })
  } else ids.add(value.id)
  text(errors, `${path}.label`, value.label, MAX_SKILL_TITLE_CHARS)

  const at = (field: string) => `${path}.${field}`
  switch (value.kind) {
    case 'file_exists':
      if (!validGlob(value.path))
        errors.push({
          path: at('path'),
          message: 'must be a repo-relative glob',
        })
      return
    case 'file_contains':
      if (!validGlob(value.path))
        errors.push({
          path: at('path'),
          message: 'must be a repo-relative glob',
        })
      if (!validRegex(value.pattern, value.flags)) {
        errors.push({
          path: at('pattern'),
          message: 'must be a valid regex (flags: i, m, s)',
        })
      }
      if (value.absent !== undefined && typeof value.absent !== 'boolean') {
        errors.push({ path: at('absent'), message: 'must be a boolean' })
      }
      if (
        value.changedOnly !== undefined &&
        typeof value.changedOnly !== 'boolean'
      ) {
        errors.push({ path: at('changedOnly'), message: 'must be a boolean' })
      }
      return
    case 'package_declared':
      if (typeof value.name !== 'string' || !NPM_NAME.test(value.name)) {
        errors.push({
          path: at('name'),
          message: 'must be an npm package name',
        })
      }
      return
    case 'env_key_declared':
      if (!validGlob(value.path))
        errors.push({
          path: at('path'),
          message: 'must be a repo-relative glob',
        })
      if (typeof value.key !== 'string' || !ENV_KEY.test(value.key)) {
        errors.push({
          path: at('key'),
          message: 'must be an UPPER_SNAKE env key',
        })
      }
      return
    case 'user_confirmed':
      if (stepKind !== 'user') {
        errors.push({ path, message: 'user_confirmed belongs on a user step' })
      }
      return
    case 'credential_provided':
      if (stepKind !== 'user') {
        errors.push({
          path,
          message: 'credential_provided belongs on a user step',
        })
      }
      if (typeof value.envKey !== 'string' || !ENV_KEY.test(value.envKey)) {
        errors.push({
          path: at('envKey'),
          message: 'must be an UPPER_SNAKE env key',
        })
      }
      return
    case 'script_passes':
      if (!oneOf(SPONSORED_SKILL_SCRIPTS, value.script)) {
        errors.push({
          path: at('script'),
          message: `must be one of ${SPONSORED_SKILL_SCRIPTS.join(', ')}`,
        })
      }
      if (
        value.whenMissing !== undefined &&
        !oneOf(['skip', 'fail'] as const, value.whenMissing)
      ) {
        errors.push({
          path: at('whenMissing'),
          message: 'must be skip or fail',
        })
      }
      timeout(errors, at('timeoutSeconds'), value.timeoutSeconds)
      return
    case 'command_succeeds': {
      const argv = value.argv
      if (
        !Array.isArray(argv) ||
        argv.length === 0 ||
        argv.length > 16 ||
        !argv.every(
          (token) => typeof token === 'string' && ARGV_TOKEN.test(token),
        )
      ) {
        errors.push({
          path: at('argv'),
          message: 'must be 1-16 plain argv tokens (no shell syntax)',
        })
      } else if (argv[0] !== 'node' && !tools.has(argv[0] as string)) {
        errors.push({
          path: at('argv'),
          message: `argv[0] must be node or a declared tool, not "${argv[0]}"`,
        })
      }
      timeout(errors, at('timeoutSeconds'), value.timeoutSeconds)
      return
    }
    case 'no_privileged_secrets':
    case 'changes_within':
      if (
        !Array.isArray(value.paths) ||
        value.paths.length === 0 ||
        !value.paths.every(validGlob)
      ) {
        errors.push({
          path: at('paths'),
          message: 'must be a non-empty list of repo-relative globs',
        })
      }
      return
    default:
      errors.push({
        path: at('kind'),
        message: `unknown check kind ${JSON.stringify(value.kind)}`,
      })
  }
}

/**
 * Validates an untrusted value (a stored campaign row, a file in review) as a
 * skill. Every error is reported, not just the first, so a reviewer or the
 * authoring skill can fix them in one pass.
 */
const SHA256_HEX = /^[0-9a-f]{64}$/
/** HTTPS, a dotted host and a plain path: nothing a shell or URL parser could read twice. */
const RELEASE_URL =
  /^https:\/\/[a-z0-9-]+(?:\.[a-z0-9-]+)+\/[A-Za-z0-9._~%+\/-]+$/
const RELEASE_BINARY_SEGMENT = /^[A-Za-z0-9_][A-Za-z0-9._+-]*$/

function validReleaseUrl(url: unknown): url is string {
  if (typeof url !== 'string' || url.length > 2_048 || !RELEASE_URL.test(url))
    return false
  try {
    const parsed = new URL(url)
    return (
      parsed.protocol === 'https:' &&
      !parsed.username &&
      !parsed.password &&
      !parsed.port &&
      !parsed.search &&
      !parsed.hash &&
      // An IP literal or a local name is never a vendor's release host.
      !/^[0-9.]+$/.test(parsed.hostname) &&
      !/(?:^|\.)(?:localhost|local|internal)$/.test(parsed.hostname) &&
      // The URL parser folds `..` away, so look at the text as written.
      !url
        .slice(`https://${parsed.host}`.length)
        .split('/')
        .map((part) => part.toLowerCase().replaceAll('%2e', '.'))
        .some((part) => part === '..' || part === '.')
    )
  } catch {
    return false
  }
}

function validReleaseBinary(binary: unknown): binary is string {
  if (typeof binary !== 'string' || !binary || binary.length > 200) return false
  const parts = binary.split('/')
  return parts.every(
    (part) => RELEASE_BINARY_SEGMENT.test(part) && part !== '..',
  )
}

function parseToolRelease(
  errors: SponsoredSkillParseError[],
  path: string,
  release: unknown,
): void {
  if (!isRecord(release)) {
    errors.push({ path, message: 'must be an object' })
    return
  }
  if (
    typeof release.version !== 'string' ||
    !EXACT_SEMVER.test(release.version)
  )
    errors.push({ path: `${path}.version`, message: 'must be an exact semver' })
  const assets = release.assets
  if (
    !Array.isArray(assets) ||
    assets.length === 0 ||
    assets.length > MAX_SPONSORED_TOOL_RELEASE_ASSETS
  ) {
    errors.push({
      path: `${path}.assets`,
      message: `must list 1-${MAX_SPONSORED_TOOL_RELEASE_ASSETS} assets`,
    })
    return
  }
  const targets = new Set<string>()
  assets.forEach((asset, index) => {
    const at = `${path}.assets[${index}]`
    if (!isRecord(asset)) {
      errors.push({ path: at, message: 'must be an object' })
      return
    }
    if (!(SPONSORED_TOOL_RELEASE_OS as readonly unknown[]).includes(asset.os))
      errors.push({
        path: `${at}.os`,
        message: `must be one of ${SPONSORED_TOOL_RELEASE_OS.join(', ')}`,
      })
    if (
      !(SPONSORED_TOOL_RELEASE_ARCH as readonly unknown[]).includes(asset.arch)
    )
      errors.push({
        path: `${at}.arch`,
        message: `must be one of ${SPONSORED_TOOL_RELEASE_ARCH.join(', ')}`,
      })
    if (!validReleaseUrl(asset.url))
      errors.push({
        path: `${at}.url`,
        message:
          'must be a plain https URL on a public host, with no query, fragment, port or credentials',
      })
    if (typeof asset.sha256 !== 'string' || !SHA256_HEX.test(asset.sha256))
      errors.push({
        path: `${at}.sha256`,
        message: 'must be 64 lowercase hex characters',
      })
    if (
      !(SPONSORED_TOOL_RELEASE_ARCHIVES as readonly unknown[]).includes(
        asset.archive,
      )
    )
      errors.push({
        path: `${at}.archive`,
        message: `must be one of ${SPONSORED_TOOL_RELEASE_ARCHIVES.join(', ')}`,
      })
    if (!validReleaseBinary(asset.binary))
      errors.push({
        path: `${at}.binary`,
        message: 'must be a relative path inside the archive, without ..',
      })
    const target = `${String(asset.os)}-${String(asset.arch)}`
    if (targets.has(target))
      errors.push({ path: at, message: `duplicate asset for ${target}` })
    targets.add(target)
  })
}

/** The release asset a harness on `os`/`arch` installs for `tool`, or null. */
export function sponsoredToolReleaseAsset(
  tool: SponsoredSkillTool,
  os: string,
  arch: string,
): SponsoredSkillToolReleaseAsset | null {
  if (tool.source !== 'path' || !tool.release) return null
  return (
    tool.release.assets.find(
      (asset) => asset.os === os && asset.arch === arch,
    ) ?? null
  )
}

export function parseSponsoredSkill(value: unknown): SponsoredSkillParseResult {
  const errors: SponsoredSkillParseError[] = []
  if (!isRecord(value))
    return { ok: false, errors: [{ path: '', message: 'must be an object' }] }

  if (!(SPONSORED_SKILL_FORMATS as readonly unknown[]).includes(value.format)) {
    errors.push({
      path: 'format',
      message: `must be one of ${SPONSORED_SKILL_FORMATS.join(', ')}`,
    })
  }
  const format2 = value.format === 2
  if (format2) {
    const os =
      isRecord(value.requirements) && Array.isArray(value.requirements.os)
        ? (value.requirements.os as string[])
        : null
    parseSponsoredSkillFit(errors, value.fit, os)
  } else if (value.fit !== undefined) {
    errors.push({ path: 'fit', message: 'a fit block needs format 2' })
  }
  if (typeof value.id !== 'string' || !SLUG.test(value.id)) {
    errors.push({ path: 'id', message: 'must be a lowercase slug' })
  }
  if (typeof value.version !== 'string' || !EXACT_SEMVER.test(value.version)) {
    errors.push({ path: 'version', message: 'must be an exact semver' })
  }
  if (typeof value.advertiser !== 'string' || !SLUG.test(value.advertiser)) {
    errors.push({ path: 'advertiser', message: 'must be a lowercase slug' })
  }
  text(errors, 'changelog', value.changelog, MAX_SKILL_GOAL_CHARS)
  text(errors, 'goal', value.goal, MAX_SKILL_GOAL_CHARS)
  text(errors, 'fits', value.fits, MAX_SKILL_GOAL_CHARS)
  if (!isRecord(value.consent)) {
    errors.push({ path: 'consent', message: 'must be an object' })
  } else {
    text(errors, 'consent.title', value.consent.title, MAX_SKILL_TITLE_CHARS)
    text(
      errors,
      'consent.summary',
      value.consent.summary,
      MAX_SKILL_SUMMARY_CHARS,
    )
  }

  if (value.desktop !== undefined) {
    const desktop = value.desktop
    if (
      !isRecord(desktop) ||
      Object.keys(desktop).some(
        (key) => !['interactiveTerminal', 'browserOrigins'].includes(key),
      )
    ) {
      errors.push({
        path: 'desktop',
        message: 'must declare only interactiveTerminal and browserOrigins',
      })
    } else {
      if (
        desktop.interactiveTerminal !== undefined &&
        typeof desktop.interactiveTerminal !== 'boolean'
      )
        errors.push({
          path: 'desktop.interactiveTerminal',
          message: 'must be a boolean',
        })
      if (
        desktop.browserOrigins !== undefined &&
        (!Array.isArray(desktop.browserOrigins) ||
          desktop.browserOrigins.length > 10 ||
          desktop.browserOrigins.some((origin) => {
            if (typeof origin !== 'string' || origin.length > 2048) return true
            try {
              const u = new URL(origin)
              return (
                u.protocol !== 'https:' ||
                u.origin !== origin ||
                !!u.username ||
                !!u.password
              )
            } catch {
              return true
            }
          }))
      )
        errors.push({
          path: 'desktop.browserOrigins',
          message:
            'must be at most 10 exact HTTPS origins, without paths or credentials',
        })
    }
  }

  const requirements = value.requirements
  if (!isRecord(requirements)) {
    errors.push({ path: 'requirements', message: 'must be an object' })
  } else {
    const list = <T extends string>(field: string, allowed: readonly T[]) => {
      const items = requirements[field]
      if (
        !Array.isArray(items) ||
        items.length === 0 ||
        !items.every((item) => oneOf(allowed, item))
      ) {
        errors.push({
          path: `requirements.${field}`,
          message: `must be a non-empty list of ${allowed.join(', ')}`,
        })
      }
    }
    list('os', SPONSORED_SKILL_OS)
    list('packageManagers', SPONSORED_SKILL_PACKAGE_MANAGERS)
    const fitRoutes =
      format2 && isRecord(value.fit) && isRecord(value.fit.routes)
        ? Object.keys(value.fit.routes)
        : null
    // `frameworks` is the JS-only enum: a format-2 skill with no node route
    // lists none, and one with a node route must list at least one.
    if (fitRoutes && !fitRoutes.includes('node')) {
      if (
        !Array.isArray(requirements.frameworks) ||
        requirements.frameworks.length !== 0
      )
        errors.push({
          path: 'requirements.frameworks',
          message: 'must be empty for a skill with no node fit route',
        })
    } else list('frameworks', SPONSORED_SKILL_FRAMEWORKS)
    if (fitRoutes && Array.isArray(requirements.runtimes)) {
      const names = requirements.runtimes.map((runtime) =>
        isRecord(runtime) ? runtime.name : null,
      )
      for (const [route, runtime] of [
        ['node', 'node'],
        ['python', 'python'],
      ] as const) {
        if (fitRoutes.includes(route) && !names.includes(runtime))
          errors.push({
            path: 'requirements.runtimes',
            message: `must name ${runtime}: the fit block has a ${route} route`,
          })
      }
    }
    const runtimes = requirements.runtimes
    if (
      !Array.isArray(runtimes) ||
      runtimes.length === 0 ||
      !runtimes.every(
        (runtime) =>
          isRecord(runtime) &&
          oneOf(SPONSORED_SKILL_RUNTIMES, runtime.name) &&
          (runtime.minVersion === undefined ||
            (typeof runtime.minVersion === 'string' &&
              /^\d+(?:\.\d+){0,2}$/.test(runtime.minVersion))),
      )
    ) {
      errors.push({
        path: 'requirements.runtimes',
        message: 'must be a non-empty list of { name, minVersion? }',
      })
    }
  }

  if (
    !Array.isArray(value.knowledge) ||
    value.knowledge.length > MAX_SKILL_KNOWLEDGE_ITEMS
  ) {
    errors.push({
      path: 'knowledge',
      message: `must be a list of at most ${MAX_SKILL_KNOWLEDGE_ITEMS} facts`,
    })
  } else {
    value.knowledge.forEach((fact, index) =>
      text(errors, `knowledge[${index}]`, fact, MAX_SKILL_KNOWLEDGE_CHARS),
    )
  }

  const packages = Array.isArray(value.packages) ? value.packages : null
  if (!packages) errors.push({ path: 'packages', message: 'must be a list' })
  if (packages && packages.length > MAX_SKILL_PACKAGES) {
    errors.push({
      path: 'packages',
      message: `must declare at most ${MAX_SKILL_PACKAGES} packages`,
    })
  }
  /** npm names only: `package_declared` reads package.json. */
  const packageNames = new Set<string>()
  const pypiNames = new Set<string>()
  packages?.forEach((pkg, index) => {
    const path = `packages[${index}]`
    if (isRecord(pkg) && pkg.ecosystem === 'pypi') {
      if (typeof pkg.name !== 'string' || !PYPI_PACKAGE_NAME.test(pkg.name)) {
        errors.push({
          path,
          message:
            'must be { ecosystem: "pypi", name, version } with a PyPI project name',
        })
        return
      }
      if (
        typeof pkg.version !== 'string' ||
        !PYPI_EXACT_VERSION.test(pkg.version)
      ) {
        errors.push({
          path: `${path}.version`,
          message:
            'must be an exact PEP 440 release such as 2.32.3, not a range, extra or URL',
        })
      }
      if (pkg.dev !== undefined && typeof pkg.dev !== 'boolean') {
        errors.push({ path: `${path}.dev`, message: 'must be a boolean' })
      }
      const normalized = normalizePypiName(pkg.name)
      if (pypiNames.has(normalized))
        errors.push({ path, message: `duplicate package ${pkg.name}` })
      pypiNames.add(normalized)
      return
    }
    if (
      !isRecord(pkg) ||
      pkg.ecosystem !== 'npm' ||
      typeof pkg.name !== 'string' ||
      !NPM_NAME.test(pkg.name)
    ) {
      errors.push({
        path,
        message:
          'must be { ecosystem: "npm", name, version } or { ecosystem: "pypi", name, version }',
      })
      return
    }
    if (typeof pkg.version !== 'string' || !EXACT_SEMVER.test(pkg.version)) {
      errors.push({
        path: `${path}.version`,
        message: 'must be an exact version, not a range',
      })
    }
    if (pkg.dev !== undefined && typeof pkg.dev !== 'boolean') {
      errors.push({ path: `${path}.dev`, message: 'must be a boolean' })
    }
    if (packageNames.has(pkg.name))
      errors.push({ path, message: `duplicate package ${pkg.name}` })
    packageNames.add(pkg.name)
  })

  const tools = new Set<string>()
  if (!Array.isArray(value.tools))
    errors.push({ path: 'tools', message: 'must be a list' })
  else {
    value.tools.forEach((tool, index) => {
      const path = `tools[${index}]`
      if (
        !isRecord(tool) ||
        typeof tool.name !== 'string' ||
        !SLUG.test(tool.name)
      ) {
        errors.push({ path, message: 'must have a slug name' })
        return
      }
      if (tool.source === 'npm') {
        if (
          typeof tool.package !== 'string' ||
          !NPM_NAME.test(tool.package) ||
          typeof tool.version !== 'string' ||
          !EXACT_SEMVER.test(tool.version)
        ) {
          errors.push({
            path,
            message:
              'an npm tool needs { package, version } with an exact version',
          })
          return
        }
      } else if (tool.source === 'path') {
        if (typeof tool.installHint !== 'string' || !tool.installHint.trim()) {
          errors.push({ path, message: 'a path tool needs an installHint' })
          return
        }
        if (tool.release !== undefined)
          parseToolRelease(errors, `${path}.release`, tool.release)
      } else {
        errors.push({ path, message: 'source must be npm or path' })
        return
      }
      if (tool.name === 'node')
        errors.push({ path, message: 'a tool may not be named node' })
      tools.add(tool.name)
    })
  }

  const checkIds = new Set<string>()
  const compiledFit =
    format2 && !errors.some((error) => error.path.startsWith('fit'))
      ? compileSponsoredSkillFit(value.fit as SponsoredSkillFit)
      : []
  for (const check of compiledFit) checkIds.add(check.id)
  if (
    !Array.isArray(value.preflight) ||
    value.preflight.length + compiledFit.length > MAX_SKILL_PREFLIGHT_CHECKS
  ) {
    errors.push({
      path: 'preflight',
      message: `must be a list of at most ${MAX_SKILL_PREFLIGHT_CHECKS - compiledFit.length} checks (${MAX_SKILL_PREFLIGHT_CHECKS} with the fit block's)`,
    })
  } else {
    value.preflight.forEach((check, index) =>
      parseCheck(
        errors,
        `preflight[${index}]`,
        check,
        checkIds,
        tools,
        'preflight',
      ),
    )
  }

  const steps = value.steps
  if (
    !Array.isArray(steps) ||
    steps.length === 0 ||
    steps.length > MAX_SKILL_STEPS
  ) {
    errors.push({
      path: 'steps',
      message: `must list 1-${MAX_SKILL_STEPS} steps`,
    })
  } else {
    const stepIds = new Set<string>()
    steps.forEach((step, index) => {
      const path = `steps[${index}]`
      if (!isRecord(step)) {
        errors.push({ path, message: 'must be an object' })
        return
      }
      if (typeof step.id !== 'string' || !SLUG.test(step.id)) {
        errors.push({ path: `${path}.id`, message: 'must be a lowercase slug' })
      } else if (stepIds.has(step.id)) {
        errors.push({
          path: `${path}.id`,
          message: `duplicate step id "${step.id}"`,
        })
      } else stepIds.add(step.id)
      const kind = oneOf(SPONSORED_SKILL_STEP_KINDS, step.kind)
        ? step.kind
        : null
      if (!kind) {
        errors.push({
          path: `${path}.kind`,
          message: 'must be agent, user or check',
        })
      }
      const title = text(
        errors,
        `${path}.title`,
        step.title,
        MAX_SKILL_TITLE_CHARS,
      )
      if (/\n/.test(title))
        errors.push({ path: `${path}.title`, message: 'must be one line' })
      text(
        errors,
        `${path}.instructions`,
        step.instructions,
        MAX_SKILL_INSTRUCTIONS_CHARS,
        {
          allowEmpty: kind === 'check',
        },
      )
      if (step.maxAttempts !== undefined) {
        if (
          typeof step.maxAttempts !== 'number' ||
          !Number.isInteger(step.maxAttempts) ||
          step.maxAttempts < 1 ||
          step.maxAttempts > MAX_SKILL_ATTEMPTS
        ) {
          errors.push({
            path: `${path}.maxAttempts`,
            message: `must be 1-${MAX_SKILL_ATTEMPTS}`,
          })
        }
      }
      if (kind === 'user') {
        const credential = step.credential
        if (
          credential !== undefined &&
          (!isRecord(credential) ||
            declaredRunCredentials(
              credentialDirective(credential as SponsoredRunCredential),
            ).length !== 1)
        ) {
          errors.push({
            path: `${path}.credential`,
            message:
              'a user step needs { env, label, getUrl } valid as a requires-credential line (env ends in _KEY/_TOKEN/_SECRET/_PASSWORD, label ≤ 60 chars, https getUrl)',
          })
        }
      } else if (step.credential !== undefined) {
        errors.push({
          path: `${path}.credential`,
          message: 'only a user step asks for a credential',
        })
      }
      if (
        !Array.isArray(step.checks) ||
        step.checks.length === 0 ||
        step.checks.length > MAX_SKILL_CHECKS_PER_STEP
      ) {
        // A step with no check is a step the harness cannot call done.
        errors.push({
          path: `${path}.checks`,
          message: `must list 1-${MAX_SKILL_CHECKS_PER_STEP} checks`,
        })
      } else if (kind) {
        step.checks.forEach((check, checkIndex) =>
          parseCheck(
            errors,
            `${path}.checks[${checkIndex}]`,
            check,
            checkIds,
            tools,
            kind,
          ),
        )
      }
    })
  }

  if (
    !Array.isArray(value.outcomes) ||
    !value.outcomes.every((outcome) =>
      oneOf(['api_key_issued', 'mcp_installed'] as const, outcome),
    )
  ) {
    errors.push({
      path: 'outcomes',
      message: 'must be a list of api_key_issued, mcp_installed',
    })
  }

  if (errors.length) return { ok: false, errors }
  const skill = value as unknown as SponsoredSkill

  const credentials = skill.steps.flatMap((step) =>
    step.credential ? [step.credential] : [],
  )
  if (
    declaredRunCredentials(renderSponsoredSkillProcedure(skill)).length !==
    credentials.length
  ) {
    errors.push({
      path: 'steps',
      message: 'user steps must ask for at most two distinct credentials',
    })
  }
  for (const check of allChecks(skill)) {
    if (
      check.kind === 'credential_provided' &&
      !credentials.some((credential) => credential.env === check.envKey)
    ) {
      errors.push({
        path: check.id,
        message: `checks credential ${check.envKey}, which no user step asks for`,
      })
    }
  }

  for (const check of allChecks(skill)) {
    if (check.kind === 'package_declared' && !packageNames.has(check.name)) {
      errors.push({
        path: check.id,
        message: pypiNames.has(normalizePypiName(check.name))
          ? `checks package ${check.name}, a pypi package: package_declared reads package.json, so check a Python package with file_contains or command_succeeds`
          : `checks package ${check.name}, which the skill does not declare`,
      })
    }
  }
  const rendered = renderSponsoredSkillProcedure(skill)
  if (rendered.length > MAX_SKILL_RENDERED_CHARS) {
    errors.push({
      path: 'steps',
      message: `renders to ${rendered.length} characters; the limit is ${MAX_SKILL_RENDERED_CHARS}`,
    })
  }
  // The run card and the grader count steps, and read "needs the user", from
  // the rendered text (sponsored-procedure-steps.ts). The kinds here must
  // agree, or the grader would forgive an agent step as the user's.
  const parsed = parseSponsoredProcedureSteps(rendered)
  if (parsed.length !== skill.steps.length) {
    errors.push({
      path: 'steps',
      message: `renders to ${parsed.length} parsed steps, not ${skill.steps.length}`,
    })
  } else {
    skill.steps.forEach((step, index) => {
      if (parsed[index]!.needsUser !== (step.kind === 'user')) {
        errors.push({
          path: `steps[${index}]`,
          message:
            step.kind === 'user'
              ? 'a user step must read as the user\'s action, e.g. "Ask the user to …"'
              : 'its wording reads as a user action; reword it or make it a user step',
        })
      }
    })
  }
  return errors.length ? { ok: false, errors } : { ok: true, skill }
}

function allChecks(skill: SponsoredSkill): SponsoredSkillCheck[] {
  return [
    ...sponsoredSkillPreflightChecks(skill),
    ...skill.steps.flatMap((step) => step.checks),
  ]
}

/** Semver order (prerelease below its release), for picking the latest version. */
export function compareSponsoredSkillVersions(a: string, b: string): number {
  const parse = (version: string) => {
    const [core = '', pre] = version.split('-', 2)
    return { parts: core.split('.').map(Number), pre }
  }
  const left = parse(a)
  const right = parse(b)
  for (let i = 0; i < 3; i++) {
    const diff = (left.parts[i] ?? 0) - (right.parts[i] ?? 0)
    if (diff) return diff
  }
  if (left.pre === right.pre) return 0
  if (left.pre === undefined) return 1
  if (right.pre === undefined) return -1
  return left.pre < right.pre ? -1 : 1
}

export function sponsoredSkillKey(

function canonical(value: unknown): unknown {
  if (Array.isArray(value)) return value.map(canonical)
  if (isRecord(value)) {
    return Object.fromEntries(
      Object.keys(value)
        .filter((key) => value[key] !== undefined)
        .sort()
        .map((key) => [key, canonical(value[key])]),
    )
  }
  return value
}

/** Canonical JSON: keys sorted at every depth, `undefined` dropped. Array order is meaning. */
export function canonicalizeSponsoredSkill(skill: SponsoredSkill): string {
  return JSON.stringify(canonical(skill))
}

/**
 * THE hash Accept locks. It covers every field, so a changed step, check,
 * package or consent line is a different skill, and a stored proposal whose
 * hash no longer matches the version it names is refused rather than run.
 */
export function sponsoredSkillSha256(
  skill: SponsoredSkill,
): `sha256:${string}` {
  return `sha256:${sha256Hex(canonicalizeSponsoredSkill(skill))}`
}

/**
 * The first line of a rendered skill: `skill: <id>@<version> sha256:<hex>`.
 * The existing pipeline hashes and locks the procedure TEXT at Accept
 * (`procedure_sha256`, `compute_procedure_sha256`); because this line carries
 * the skill's own hash, locking the text locks every check and package too.
 */
export const SPONSORED_SKILL_DIRECTIVE = 'skill:'
const SKILL_DIRECTIVE_LINE =
  /^skill:\s+([a-z][a-z0-9-]{0,63})@(\d+\.\d+\.\d+(?:-[0-9A-Za-z.-]+)?)\s+(sha256:[0-9a-f]{64})\s*$/

/** The skill a procedure text names, or null for a legacy free-text procedure. */
export function sponsoredSkillDirective(
  procedure: string | null | undefined,
): { id: string; version: string; sha256: `sha256:${string}` } | null {
  const first = procedure?.replace(/\r\n?/g, '\n').trimStart().split('\n', 1)[0]
  const match = first ? SKILL_DIRECTIVE_LINE.exec(first) : null
  if (!match) return null
  return {
    id: match[1]!,
    version: match[2]!,
    sha256: match[3] as `sha256:${string}`,
  }
}

/**
 * Whether `skill` is the one an accepted procedure text locked: same id,
 * version and hash. The step driver refuses to run a skill that fails this.
 */
export function sponsoredSkillMatchesProcedure(
  skill: SponsoredSkill,
  procedure: string,
): boolean {
  const directive = sponsoredSkillDirective(procedure)
  return (
    directive !== null &&
    directive.id === skill.id &&
    directive.version === skill.version &&
    directive.sha256 === sponsoredSkillSha256(skill) &&
    procedure.trim() === renderSponsoredSkillProcedure(skill)
  )
}

function credentialDirective(credential: SponsoredRunCredential): string {
  return [
    `requires-credential: env=${credential.env}`,
    `label="${credential.label}"`,
    ...(credential.getUrl ? [`get_url=${credential.getUrl}`] : []),
  ].join(' ')
}

function indent(body: string, spaces: number): string {
  const pad = ' '.repeat(spaces)
  return body
    .split('\n')
    .map((line) => (line.trim() ? pad + line : ''))
    .join('\n')
}

export function renderSponsoredSkillProcedure(skill: SponsoredSkill): string {
  const lines: string[] = [
    `${SPONSORED_SKILL_DIRECTIVE} ${sponsoredSkillKey(skill)} ${sponsoredSkillSha256(skill)}`,
    // The directives today's Desktop already reads, so a skill runs on the
    // current harness before the step driver ships.
    ...skill.steps.flatMap((step) =>
      step.credential ? [credentialDirective(step.credential)] : [],
    ),
    ...(skill.outcomes.length
      ? [`outcomes: ${skill.outcomes.join(', ')}`]
      : []),
    '',
    `Goal: ${skill.consent.summary}`,
    '',
    'Fit every step to this repository: its framework, layout and conventions decide where',
    'files go and how code is written. Each step is done when its checks pass, however you get there.',
    '',
  ]
  if (skill.desktop) {
    lines.push(SPONSORED_DESKTOP_TOOLS_DECLARATION)
    if (skill.desktop.interactiveTerminal)
      lines.push(
        '[freebuff product prompt text omitted]',
      )
    if (skill.desktop.browserOrigins?.length)
      lines.push(
        `- Use preview_open to show CLI authentication links to the user on these origins only: ${skill.desktop.browserOrigins.join(', ')}. Preserve the CLI URL exactly, including OAuth state; do not append attribution parameters. The user signs in directly in the browser; call wait_for_cli_login to wait, then verify with the CLI. For this native CLI flow, use these tools instead of the static request_sponsored_signup link. Never request passwords or tokens in chat.`,
      )
    lines.push(
      '- CLI login state lives in this run’s private HOME. Do not claim a global installation or login in the user’s regular shell.',
      '',
    )
  }
  if (skill.knowledge.length) {
    lines.push('What to know:')
    for (const fact of skill.knowledge) lines.push(`- ${fact}`)
    lines.push('')
  }
  const npmTools = skill.tools.filter((tool) => tool.source === 'npm')
  if (skill.packages.length || npmTools.length) {
    lines.push('Pinned installs (install exactly these, nothing else):')
    for (const pkg of skill.packages) {
      // npm lines are unchanged byte for byte: shipped procedure hashes
      // (`procedure_sha256`) are of this text.
      lines.push(
        pkg.ecosystem === 'pypi'
          ? `- ${pkg.name}==${pkg.version} (pip${pkg.dev ? ', dev' : ''})`
          : `- ${pkg.name}@${pkg.version}${pkg.dev ? ' (dev)' : ''}`,
      )
    }
    for (const tool of npmTools) {
      lines.push(
        `- \`${tool.name}\` = npx --yes ${tool.package}@${tool.version}`,
      )
    }
    if (skill.packages.some((pkg) => pkg.ecosystem === 'pypi')) {
      lines.push(
        'Python (pip) packages install from wheels only, one plain command: `pip install --only-binary=:all: <name>==<version>`, or `uv add --no-build <name>==<version>` in a uv project.',
      )
    }
    lines.push('')
  }
  const pathTools = skill.tools.filter((tool) => tool.source === 'path')
  if (pathTools.length) {
    lines.push(
      'Already-installed CLIs (do not install them; if one is missing, stop and report it):',
    )
    for (const tool of pathTools) {
      if (tool.source === 'path')
        lines.push(`- \`${tool.name}\` (${tool.installHint})`)
    }
    lines.push('')
  }
  skill.steps.forEach((step, index) => {
    lines.push(`${index + 1}. ${step.title}`)
    if (step.instructions.trim())
      lines.push(indent(step.instructions.trim(), 3))
    lines.push(
      indent(
        `Done when: ${step.checks.map((check) => check.label).join('; ')}.`,
        3,
      ),
    )
    lines.push('')
  })
  return lines.join('\n').trim()
}

/**
 * The link Desktop's sidebar sign-up opens and attributes
 * (`freebuff-desktop/src/shared/sponsored-signup.ts`) in a procedure TEXT:
 * the FIRST `https://` URL whose `bfcid` parameter is the `{bfcid}` slot (or
 * its URL-encoded form). Null means a run of this text converts with no
 * attribution: `{{advertiserLink}}` alone does not count, because it is text
 * for the model, not a link Desktop opens. Mirrors `sponsoredSignupStep`: a
 * first match Desktop would refuse (two bfcids, credentials, over 2,048
 * chars) is null too.
 */
export function sponsoredProcedureTrackedLink(
  procedure: string | null | undefined,
): string | null {
  if (!procedure) return null
  for (const match of procedure.matchAll(/https:\/\/[^\s<>"'`]+/gi)) {
    const raw = match[0].replace(/[.,;!?)\]]+$/, '')
    try {
      const url = new URL(raw)
      const ids = url.searchParams.getAll('bfcid')
      if (!ids.some((value) => /^\{bfcid\}$/i.test(value))) continue
      // Desktop refuses (opens nothing for) the first match on these.
      if (ids.length !== 1 || url.username || url.password || raw.length > 2048)
        return null
      return raw
    } catch {
      // Not a URL; keep looking.
    }
  }
  return null
}

/** {@link sponsoredProcedureTrackedLink} of the skill's rendered procedure. */
export function sponsoredSkillTrackedLink(
  skill: SponsoredSkill,
): string | null {
  return sponsoredProcedureTrackedLink(renderSponsoredSkillProcedure(skill))
}

/**
 * Why a procedure text may not be saved for attribution, or null. A text
 * that names a sponsored skill (`skill: <id>@<version> sha256:…`) must carry
 * a tracked link: a skill is ours to write, so a missing one is our bug and
 * loses every conversion's attribution (COD-882). A legacy free-text
 * procedure is the advertiser's own words and is only flagged at review.
 */
export function sponsoredProcedureTrackedLinkError(
  procedure: string | null | undefined,
): string | null {
  const directive = sponsoredSkillDirective(procedure)
  if (!directive) return null
  if (sponsoredProcedureTrackedLink(procedure)) return null
  return `Sponsored skill ${directive.id}@${directive.version} ${SPONSORED_SKILL_TRACKED_LINK_ERROR}`
}

/**
 * The step driver: the harness, not the model, decides when a sponsored
 * skill's steps are done (piece 3 of COD-822; the format is
 * `docs/ads/agentic/sponsored-skills.md`).
 *
 * After the run's turn has done its work, the driver walks the skill's steps
 * in order and runs each step's checks through THE skill evaluator
 * (`evaluateSponsoredSkillChecks`), the same one the sponsored eval scores
 * with. A step with no failing check is done, whatever the model said. A
 * failing `agent` or `check` step goes back to the agent with the failed
 * checks (`sponsoredSkillCheckFeedback`), to fix and stop; the checks run
 * again after each fix. The step's `maxAttempts` counts the first try (the
 * turn's own work) plus fixes, and the run as a whole gets at most
 * {@link SPONSORED_STEP_MAX_FIXES_PER_RUN}. Then the step is BLOCKED, and the
 * walk ends there: later steps usually build on it, and prompting them would
 * spend the run's compute on work that cannot pass. They are still checked
 * once, so the result says what is true on disk.
 *
 * - A `user` step is never prompted or retried. It is done only once its
 *   checks pass (the user handed over the credential), else it needs the user.
 * - A step that passed before a later fix is checked again at the end: a fix
 *   for step 4 can break step 2.
 *
 * DELIVERED means every `agent` and `check` step passed. That is the run's
 * completion, not the model writing "Done:".
 *
 * Pure: the caller hands in the check context (its worktree, its sandbox) and
 * how to prompt the agent, so the whole loop runs in tests without a model,
 * a sandbox or a disk.
 */

/**
 * Where the skill a run drives came from: this build's registry, the server
 * (verified against the consented text or the hash recorded at accept), or
 * nowhere, and then the procedure runs as free text with no step checks.
 */
export type SponsoredSkillSource = 'bundled' | 'server' | 'none'

export type SponsoredSkillSelection =
  | { skill: SponsoredSkill; source: Exclude<SponsoredSkillSource, 'none'> }
  | { skill: null; source: 'none' }

const NO_SKILL: SponsoredSkillSelection = { skill: null, source: 'none' }

/**
 * The skill an accepted procedure names, and where it came from. Only when
 * the text is exactly that skill's rendering: this build's registry first,
 * then the skill the Accept response offered (`offered`, untrusted JSON, held
 * to the procedure by `deliveredSponsoredSkillForProcedure`),
 * else none, and a legacy or unmatched procedure runs as free text.
 */
export function selectSponsoredProcedureSkill(
  procedure: string | null | undefined,
  offered?: unknown,
): SponsoredSkillSelection {
  const directive = sponsoredSkillDirective(procedure)
  if (!directive || !procedure) return NO_SKILL
  const bundled = resolveSponsoredSkill(`${directive.id}@${directive.version}`)
  if (bundled && sponsoredSkillMatchesProcedure(bundled, procedure))
    return { skill: bundled, source: 'bundled' }
  const server = deliveredSponsoredSkillForProcedure(offered, procedure)
  return server ? { skill: server, source: 'server' } : NO_SKILL
}

  procedure: string | null | undefined,
  offered?: unknown,

 * copy stored with the run (`stored`, the verified skill as accepted) first,
 * so a resume never depends on this build's registry; then the registry.
 * Null when neither matches, and then the caller may ask the server
 * (`deliveredSponsoredSkillForRef` on its answer) or run the text undriven.

  /** False once the run is aborted or its compute grant is about to lapse. */
  canContinue: () => boolean
  maxFixesPerRun?: number
  /** Progress only: the step about to be checked (a UI's "Step 2/4"). */
  onStep?: (step: SponsoredStepProgress) => void

/** Which step the driver is on, for a progress line. Never affects the walk. */
export type SponsoredStepProgress = {
  number: number
  total: number
  title: string
}

/** `<id>@<version>`, as the registry keys it. */
function skillKey(skill: SponsoredSkill): string {
  return `${skill.id}@${skill.version}`
}

async function check(
  deps: Pick<SponsoredStepDriverDeps, 'ctx' | 'skill'>,
  step: Pick<SponsoredSkillStep, 'checks'>,
): Promise<{ passed: boolean; results: SponsoredSkillCheckResult[] }> {
  try {
    return await evaluateSponsoredSkillChecks(step.checks, deps.ctx, deps.skill)
  } catch (error) {
    // The evaluator never throws for a check's own failure; this is the
    // context failing (a disk error). A failure, not a pass.
    return {
      passed: false,
      results: [
        {
          id: 'check-error',
          label: 'Run the checks',
          status: 'fail',
          detail: error instanceof Error ? error.message : String(error),
        },
      ],
    }
  }
}

/** Walk the steps, check each, and have the agent fix what fails. */
export async function driveSponsoredSkillSteps(
  deps: SponsoredStepDriverDeps,
): Promise<SponsoredStepDriveResult> {
  const maxPerRun = deps.maxFixesPerRun ?? SPONSORED_STEP_MAX_FIXES_PER_RUN
  const total = deps.skill.steps.length
  const results: SponsoredStepResult[] = []
  // The fix count at which each step last passed: a later fix may undo it.
  const passedAt = new Map<number, number>()
  let fixes = 0
  let walking = true
  let stoppedBecause: SponsoredStepDriveResult['stoppedBecause']

  for (const [index, step] of deps.skill.steps.entries()) {
    const number = index + 1
    const base = { number, id: step.id, title: step.title, kind: step.kind }
    try {
      deps.onStep?.({ number, total, title: step.title })
    } catch {
      // Progress is display only; it never stops the walk.
    }
    let outcome = await check(deps, step)

    if (step.kind === 'user') {
      results.push({
        ...base,
        status: outcome.passed ? 'passed' : 'needs_user',
        fixes: 0,
        checks: outcome.results,
      })
      continue
    }

    const walkingAtEntry = walking
    // The turn's own work was the first attempt.
    const maxFixes = (step.maxAttempts ?? DEFAULT_SKILL_MAX_ATTEMPTS) - 1
    let stepFixes = 0
    while (walking && !outcome.passed) {
      if (stepFixes >= maxFixes || fixes >= maxPerRun) break
      if (!deps.canContinue()) {
        walking = false
        stoppedBecause = 'out_of_time'
        break
      }
      stepFixes += 1
      fixes += 1
      const completed = await deps.prompt(
        sponsoredStepFixPrompt(
          step,
          number,
          total,
          outcome.results,
          stepFixes,
          maxFixes,
        ),
      )
      if (!completed) {
        walking = false
        stoppedBecause = 'turn_failed'
      }
      outcome = await check(deps, step)
    }

    if (outcome.passed) passedAt.set(number, fixes)
    // Blocked: the driver tried, or would have but the fix budget is spent.
    // Not reached: the walk had already stopped (or ran out of time) first.
    const blocked = stepFixes > 0 || (walkingAtEntry && walking)
    results.push({
      ...base,
      status: outcome.passed ? 'passed' : blocked ? 'blocked' : 'not_reached',
      fixes: stepFixes,
      checks: outcome.results,
    })
    if (!outcome.passed) walking = false
  }

  for (const result of results) {
    const at = passedAt.get(result.number)
    if (at === undefined || at === fixes) continue
    const again = await check(deps, deps.skill.steps[result.number - 1]!)
    result.checks = again.results
    if (!again.passed) result.status = 'blocked'
  }

  return {
    skill: skillKey(deps.skill),
    steps: results,
    delivered: results.every(
      (result) => result.kind === 'user' || result.status === 'passed',
    ),
    fixes,
    ...(stoppedBecause ? { stoppedBecause } : {}),
  }
}

export type SponsoredStepLoopDeps<State> = {
  skill: SponsoredSkill
  ctx: SponsoredSkillCheckContext
  /** One more segment of the run's turn; resolves null when it did not complete. */
  resume: (prompt: string, previous: State) => Promise<State | null>
  /** False once the run is aborted or out of time. */
  canContinue: () => boolean
  /** Takes the run's credentials out of command output before the agent reads it. */
  redact?: (text: string) => string
  /** A `report_friction` blocker, sent for each step left blocked. */
  onFriction?: (input: SponsoredFrictionInput) => void
  /** Progress only (see {@link SponsoredStepDriverDeps.onStep}). */
  onStep?: (step: SponsoredStepProgress) => void
}

/**
 * The driver run inside a turn, for any runtime (Desktop's harness, the Cloud
 * runner): fixes are more segments of the SAME run (same transcript), and the
 * state it resolves is the last segment that COMPLETED, so a fix that errored
 * leaves the turn where it was rather than failing work already on disk.
 */
export async function runSponsoredStepLoop<State>(
  initial: State,
  deps: SponsoredStepLoopDeps<State>,
): Promise<{ state: State; result: SponsoredStepDriveResult }> {
  const redact = deps.redact ?? identity
  let state = initial

  const result = await driveSponsoredSkillSteps({
    skill: deps.skill,
    ctx: redactedCheckContext(deps.ctx, redact),
    canContinue: deps.canContinue,
    ...(deps.onStep ? { onStep: deps.onStep } : {}),
    prompt: async (text) => {
      const next = await deps.resume(text, state)
      if (next === null) return false
      state = next
      return true
    },
  })

  // The report the user reads must match the checks, not the model's memory
  // of what it tried. Only after a fix: otherwise the first segment's report
  // already described the work the checks then confirmed.
  if (result.fixes > 0 && deps.canContinue()) {
    const next = await deps.resume(sponsoredStepReportPrompt(result), state)
    if (next !== null) state = next
  }

  for (const step of result.steps) {
    if (step.kind === 'user' || step.status !== 'blocked') continue
    deps.onFriction?.({
      blocking: true,
      category: step.checks.some(
        (check) =>
          check.status === 'fail' &&
          /\bexited\b|could not run/.test(check.detail),
      )
        ? 'command_failed'
        : 'other',
      detail: `Step check: step ${step.number} still failed its checks after ${step.fixes} fix${
        step.fixes === 1 ? '' : 'es'
      } by the agent, so Freebuff marked it blocked.`,
      step: step.number,
    })
  }
  return { state, result }
}

const identity = (text: string) => text

/** Command output reaches the agent and the report (COD-665): redacted first. */
function redactedCheckContext(
  ctx: SponsoredSkillCheckContext,
  redact: (text: string) => string,
): SponsoredSkillCheckContext {
  return {
    ...ctx,
    runScript: async (script, timeout) => {
      const out = await ctx.runScript(script, timeout)
      return out === 'missing' ? out : { ...out, output: redact(out.output) }
    },
    runCommand: async (argv, timeout) => {
      const out = await ctx.runCommand(argv, timeout)
      return { ...out, output: redact(out.output) }
    },
  }
}

const PREFLIGHT_DETAIL_CHARS = 200

/** A failed preflight check, as the run's error and friction name it. */
function preflightFailureLine(
  skill: SponsoredSkill,
  failed: SponsoredSkillCheckResult,
): string {
  // One line: the runs' diagnostics keep only an error's first line.
  const oneLine = failed.detail.replace(/\s+/g, ' ').trim()
  const detail =
    oneLine.length > PREFLIGHT_DETAIL_CHARS
      ? `${oneLine.slice(0, PREFLIGHT_DETAIL_CHARS - 1)}…`
      : oneLine
  const declared = sponsoredSkillPreflightChecks(skill).find(
    (
      check,
    ): check is Extract<SponsoredSkillCheck, { kind: 'command_succeeds' }> =>
      check.id === failed.id && check.kind === 'command_succeeds',
  )
  const tool = declared
    ? (skill.tools ?? []).find((candidate) => candidate.name === declared.argv[0])
    : undefined
  const hint =
    tool?.source === 'path'
      ? ` Install \`${tool.name}\`: ${tool.installHint}`
      : ''
  return `"${failed.label}" (${detail}).${hint}`
}

/**
 * THE PREFLIGHT, run before the agent's first turn: the skill's
 * `requirements.os` against the platform the checks run on, then its
 * `preflight` checks through the same evaluator as the steps. On a failure no
 * agent turn runs and nothing is edited: the caller ends the run with
 * `reason`, and it is reported the way a blocked step is (the
 * `report_friction` blocker is sent here; `result` is the step report, every
 * step not reached). An unsupported OS runs no command: the skill's tool
 * check says nothing more on a platform the skill does not support.
 */
export async function runSponsoredSkillPreflight(
  deps: SponsoredSkillPreflightDeps,
): Promise<SponsoredSkillPreflightOutcome> {
  const { skill } = deps
  const checks: SponsoredSkillCheckResult[] = []
  const os = sponsoredSkillOs(deps.platform)
  const supported = skill.requirements?.os ?? []
  if (os && supported.length > 0) {
    const ok = supported.includes(os)
    checks.push({
      id: SPONSORED_PREFLIGHT_OS_CHECK_ID,
      label: `Runs on ${supported.join(' or ')}`,
      status: ok ? 'pass' : 'fail',
      detail: ok
        ? `this run is on ${os}`
        : `this skill supports ${supported.join(', ')}; this run is on ${os}`,
    })
  }
  if (checks.every((result) => result.status !== 'fail')) {
    const outcome = await check(
      { skill, ctx: redactedCheckContext(deps.ctx, deps.redact ?? identity) },
      { checks: sponsoredSkillPreflightChecks(skill) },
    )
    checks.push(...outcome.results)
  }
  const failed = checks.filter((result) => result.status === 'fail')
  if (failed.length === 0) return { passed: true, checks }

  const reason = `Preflight failed before any edit: ${failed
    .map((result) => preflightFailureLine(skill, result))
    .join(' ')}`
  deps.onFriction?.({
    blocking: true,
    category: failed.some(
      (result) => result.id === SPONSORED_PREFLIGHT_OS_CHECK_ID,
    )
      ? 'project_mismatch'
      : 'missing_toolchain',
    // The server keeps the first SPONSORED_FRICTION_DETAIL_MAX characters.
    detail: reason.slice(0, SPONSORED_FRICTION_DETAIL_MAX),
  })
  return {
    passed: false,
    checks,
    reason,
    result: {
      skill: skillKey(skill),
      steps: skill.steps.map((step, index) => ({
        number: index + 1,
        id: step.id,
        title: step.title,
        kind: step.kind,
        status: 'not_reached',
        fixes: 0,
        checks: [],
      })),
      delivered: false,
      fixes: 0,
      stoppedBecause: 'preflight_failed',
      preflight: checks,
    },
  }
}

/** Expands `{a,b}` alternation (nested too) into every literal alternative. */
export function expandSponsoredGlobBraces(glob: string): string[] {
  const open = glob.indexOf('{')
  if (open < 0) return [glob]
  let depth = 0
  let close = -1
  const commas: number[] = []
  for (let i = open; i < glob.length; i++) {
    const char = glob[i]
    if (char === '{') depth++
    else if (char === '}') {
      depth--
      if (depth === 0) {
        close = i
        break
      }
    } else if (char === ',' && depth === 1) commas.push(i)
  }
  if (close < 0) return [glob]
  const head = glob.slice(0, open)
  const tail = glob.slice(close + 1)
  const bounds = [open, ...commas, close]
  const parts: string[] = []
  for (let i = 0; i + 1 < bounds.length; i++)
    parts.push(glob.slice(bounds[i]! + 1, bounds[i + 1]))
  return parts.flatMap((part) => expandSponsoredGlobBraces(head + part + tail))
}

/** A one-segment basename glob (`*`, `?`) as a regex. */
function basenameRegex(glob: string): RegExp {
  let source = ''
  for (const char of glob) {
    if (char === '*') source += '[^/]*'
    else if (char === '?') source += '[^/]'
    else source += char.replace(/[.+^$()|[\]\\{}]/g, '\\$&')
  }
  return new RegExp(`^${source}$`)
}

const FIT_NAME_REGEXES = SPONSORED_FIT_FILE_NAMES.map(basenameRegex)

/** Whether a concrete basename is a fit file (listed in repository facts). */
export function isSponsoredFitFileName(basename: string): boolean {
  return FIT_NAME_REGEXES.some((re) => re.test(basename))
}

/**
 * Whether every path `glob` can match has a fit-file basename, so a
 * repository's facts (which list every fit file) decide it. The last segment
 * of each brace alternative must be one of {@link SPONSORED_FIT_FILE_NAMES}
 * exactly, or a literal that is a fit file name.
 */
export function sponsoredFitGlobDecidable(glob: string): boolean {
  return expandSponsoredGlobBraces(glob).every((alternative) => {
    const basename = alternative.split('/').at(-1) ?? ''
    if (!basename || basename.includes('**')) return false
    if ((SPONSORED_FIT_FILE_NAMES as readonly string[]).includes(basename))
      return true
    return !/[*?]/.test(basename) && isSponsoredFitFileName(basename)
  })
}

function validRouteGlob(glob: unknown): glob is string {
  return (
    typeof glob === 'string' &&
    glob.length > 0 &&
    glob.length <= 200 &&
    !glob.startsWith('/') &&
    !glob.split('/').includes('..') &&
    sponsoredFitGlobDecidable(glob)
  )
}

/**
 * Validates a `fit` block, pushing every problem. `os` is the skill's
 * `requirements.os`, which the block's `os` must equal.
 */
export function parseSponsoredSkillFit(
  errors: SponsoredSkillParseError[],
  value: unknown,
  requirementsOs: readonly string[] | null,
): void {
  const all = ['darwin', 'linux', 'win32']
  if (!isRecord(value)) {
    errors.push({ path: 'fit', message: 'format 2 needs a fit object' })
    return
  }
  for (const key of Object.keys(value)) {
    if (!['os', 'routes', 'mustNotExist'].includes(key))
      errors.push({ path: `fit.${key}`, message: 'is not a fit field' })
  }
  if (value.os !== undefined) {
    if (
      !Array.isArray(value.os) ||
      value.os.length === 0 ||
      !value.os.every((os) => all.includes(os as string)) ||
      new Set(value.os).size !== value.os.length
    )
      errors.push({
        path: 'fit.os',
        message: 'must be a non-empty list of darwin, linux, win32',
      })
  }
  if (requirementsOs) {
    const want = [...((value.os as string[] | undefined) ?? all)].sort()
    const got = [...requirementsOs].sort()
    if (want.join(',') !== got.join(','))
      errors.push({
        path: 'requirements.os',
        message: `must equal fit.os (${want.join(', ')})`,
      })
  }
  const routes = value.routes
  if (!isRecord(routes) || Object.keys(routes).length === 0) {
    errors.push({
      path: 'fit.routes',
      message: `must declare at least one of ${SPONSORED_FIT_ROUTE_ECOSYSTEMS.join(', ')}`,
    })
  } else {
    for (const [ecosystem, route] of Object.entries(routes)) {
      const at = `fit.routes.${ecosystem}`
      if (
        !(SPONSORED_FIT_ROUTE_ECOSYSTEMS as readonly string[]).includes(
          ecosystem,
        )
      ) {
        errors.push({
          path: at,
          message: `unknown ecosystem; one of ${SPONSORED_FIT_ROUTE_ECOSYSTEMS.join(', ')}`,
        })
        continue
      }
      if (!isRecord(route)) {
        errors.push({ path: at, message: 'must be an object' })
        continue
      }
      for (const key of Object.keys(route)) {
        if (!['frameworks', 'anyDependency'].includes(key))
          errors.push({ path: `${at}.${key}`, message: 'is not a route field' })
      }
      const table = SPONSORED_FIT_ECOSYSTEMS[ecosystem as SponsoredFitRouteEcosystem]
      if (route.frameworks !== undefined) {
        const known = Object.keys(table.frameworks)
        if (
          !Array.isArray(route.frameworks) ||
          !route.frameworks.every((name) => known.includes(name as string)) ||
          new Set(route.frameworks).size !== route.frameworks.length
        )
          errors.push({
            path: `${at}.frameworks`,
            message: `must be distinct names from ${known.join(', ')}`,
          })
      }
      if (route.anyDependency !== undefined) {
        const name = ecosystem === 'node' ? NPM_NAME : PYPI_NAME
        if (
          !Array.isArray(route.anyDependency) ||
          route.anyDependency.length > MAX_FIT_ROUTE_DEPENDENCIES ||
          !route.anyDependency.every(
            (dep) => typeof dep === 'string' && name.test(dep),
          ) ||
          new Set(route.anyDependency).size !== route.anyDependency.length
        )
          errors.push({
            path: `${at}.anyDependency`,
            message: `must be at most ${MAX_FIT_ROUTE_DEPENDENCIES} distinct ${ecosystem === 'node' ? 'npm' : 'PyPI'} package names`,
          })
      }
    }
  }
  if (value.mustNotExist !== undefined) {
    if (
      !Array.isArray(value.mustNotExist) ||
      value.mustNotExist.length === 0 ||
      value.mustNotExist.length > MAX_FIT_MUST_NOT_EXIST ||
      !value.mustNotExist.every(validRouteGlob)
    )
      errors.push({
        path: 'fit.mustNotExist',
        message: `must be 1-${MAX_FIT_MUST_NOT_EXIST} repo-relative globs over fit file names (${SPONSORED_FIT_MARKER_NAMES.join(', ')} …)`,
      })
  }
}

/** The routes a fit block declares, in a fixed order. */
export function sponsoredFitRoutes(
  fit: SponsoredSkillFit,
): Array<{ ecosystem: SponsoredFitRouteEcosystem; dependencies: string[] | null }> {
  const out: Array<{
    ecosystem: SponsoredFitRouteEcosystem
    dependencies: string[] | null
  }> = []
  for (const ecosystem of SPONSORED_FIT_ROUTE_ECOSYSTEMS) {
    const route = fit.routes[ecosystem] as
      | SponsoredFitRoute<string>
      | undefined
    if (!route) continue
    const table = SPONSORED_FIT_ECOSYSTEMS[ecosystem].frameworks as Record<
      string,
      readonly string[]
    >
    const deps = [
      ...(route.frameworks ?? []).flatMap((name) => table[name] ?? []),
      ...(route.anyDependency ?? []),
    ]
    out.push({
      ecosystem,
      dependencies: deps.length ? [...new Set(deps)] : null,
    })
  }
  return out
}

function escapeRegex(text: string): string {
  return text.replace(/[.*+?^${}()|[\]\\/]/g, '\\$&')
}

/** A PyPI name as a regex: case-insensitive by flag, `-`, `_` and `.` interchangeable (PEP 503). */
function pypiNameRegex(name: string): string {
  return name
    .split(/[-_.]+/)
    .map(escapeRegex)
    .join('[-_.]')
}

/** Start of the file, whatever the `m` flag says. */
const FILE_START = '(?<![\\s\\S])'
/** At a line start, in a file that is not a JSON object (so never package.json). */
const PYTHON_LINE = `^(?<=${FILE_START}(?!\\s*\\{)[\\s\\S]*)[ \\t]*["']?`
/** After a Python requirement's name: a version, extra, marker, comment, separator or the end of the line. */
const PYTHON_NAME_END = `["']?(?=[ \\t]*(?:[<>=!~;\\[\\],#@]|$))`

/**
 * The regex alternative a route compiles to, matched per manifest file:
 * - node: the dependency is a JSON key (`"next":`).
 * - python: a line of a non-JSON manifest starts with the requirement
 *   (`fastapi>=0.1`, `"fastapi[all]",`, `fastapi = "^0.110"`).
 * - a route with no dependency list: any manifest of the ecosystem
 *   (package.json is a JSON object; a Python manifest is not).
 */
function routePattern(route: {
  ecosystem: SponsoredFitRouteEcosystem
  dependencies: string[] | null
}): string {
  if (route.ecosystem === 'node') {
    return route.dependencies
      ? `"(?:${route.dependencies.map(escapeRegex).join('|')})"\\s*:`
      : `${FILE_START}\\s*\\{`
  }
  return route.dependencies
    ? `${PYTHON_LINE}(?:${route.dependencies.map(pypiNameRegex).join('|')})${PYTHON_NAME_END}`
    : `${FILE_START}(?!\\s*\\{)`
}

/** Each declared route's manifest basenames, one brace glob for every depth. */
export function sponsoredFitManifestGlob(fit: SponsoredSkillFit): string {
  const names = sponsoredFitRoutes(fit).flatMap(
    (route) => SPONSORED_FIT_ECOSYSTEMS[route.ecosystem].manifests,
  )
  return `**/{${names.join(',')}}`
}

/**
 * THE fit block as preflight checks, run ahead of the skill's own
 * `preflight` by the step driver (`sponsoredSkillPreflightChecks`):
 * 1. `fit-manifest`: a manifest of a declared route exists;
 * 2. `fit-route`: one such manifest declares a route's dependency (dropped
 *    when every route is unconstrained);
 * 3. `fit-must-not-exist`: none of `mustNotExist` exists (even empty).
 * The OS rides `requirements.os`, which the step driver already checks.
 * Generated, never hand-edited, so the regex cap on authored checks does not
 * apply.
 */
export function compileSponsoredSkillFit(
  fit: SponsoredSkillFit,
): SponsoredSkillCheck[] {
  const routes = sponsoredFitRoutes(fit)
  const path = sponsoredFitManifestGlob(fit)
  const ecosystems = routes.map((route) =>
    route.ecosystem === 'node' ? 'a package.json' : 'a Python manifest',
  )
  const checks: SponsoredSkillCheck[] = [
    {
      id: SPONSORED_FIT_CHECK_IDS.manifest,
      label: `the project has ${ecosystems.join(' or ')}`,
      kind: 'file_exists',
      path,
    },
  ]
  if (routes.some((route) => route.dependencies)) {
    checks.push({
      id: SPONSORED_FIT_CHECK_IDS.route,
      label: `the project uses ${routes
        .map((route) =>
          route.dependencies
            ? route.dependencies.slice(0, 4).join(', ') +
              (route.dependencies.length > 4 ? ' …' : '')
            : `any ${route.ecosystem} project`,
        )
        .join(' or ')}`,
      kind: 'file_contains',
      path,
      pattern: routes.map(routePattern).join('|'),
      flags: 'im',
      changedOnly: false,
    })
  }
  if (fit.mustNotExist?.length) {
    checks.push({
      id: SPONSORED_FIT_CHECK_IDS.mustNotExist,
      label: `the project has no ${fit.mustNotExist.join(' or ')} yet`,
      kind: 'file_contains',
      path:
        fit.mustNotExist.length === 1
          ? fit.mustNotExist[0]!
          : `{${fit.mustNotExist.join(',')}}`,
      // `^` matches every file, empty ones too: any match is a failure.
      pattern: '^',
      absent: true,
      changedOnly: false,
    })
  }
  return checks
}

/**
 * THE fit oracle: whether a sponsored skill's preflight would refuse a
 * repository, decided from the repository's FACTS before anything is
 * offered. Pure, no I/O: serving (freebuff-web), Desktop, the Web worker, the
 * CLI and the bench all call it, over facts built by the one builder here
 * ({@link sponsoredRepoFacts}).
 *
 * Each preflight check gets a three-valued verdict, ANDed across checks:
 * - `requirements.os` against the device OS;
 * - `file_exists` when its glob can only match fit files
 *   (`SPONSORED_FIT_FILE_NAMES`), so the facts' listing decides it;
 * - `file_contains changedOnly:false` against the manifests' dependency names
 *   (projected back into manifest-shaped text and matched with the check's
 *   own regex);
 * - `package_declared` against the root package.json's names;
 * - an `absent` check on a marker file (`.coderabbit.yaml`);
 * - a format-2 `fit` block directly.
 * Command and script checks are never fit checks: always `unknown`.
 *
 * Only a KNOWN misfit refuses ({@link sponsoredSkillFitRefusal}): an unknown
 * verdict serves, as today, and the skill's own preflight decides in the run.
 * Doc: `docs/ads/agentic/sponsored-skills.md`, "Fit".

export type SponsoredRepoFacts = {
  v: 1
  /** Every fit-file path found (basename in SPONSORED_FIT_FILE_NAMES), repo-relative, sorted. */
  paths: string[]
  /**
   * The listing covered the whole project and nothing was cut, so a fit file

   */
  complete: boolean
  /** Exact paths known to be absent even when `complete` is false (legacy facts). */
  absent?: string[]

type MaybePromise<T> = T | Promise<T>

/**
 * THE builder of repository facts. `paths` is the project's file listing,
 * repo-relative (either separator; tracked and untracked, ignored files
 * excluded, like the step driver's `files()`); `complete` says that listing
 * is the whole project. `read` returns a file's text, or null when it is
 * missing or unreadable. Reads only manifests (≤16, ≤256 KiB each) and fit
 * marker files.
 */
export async function sponsoredRepoFacts(input: {
  paths: Iterable<string>
  complete: boolean
  read: (path: string) => MaybePromise<string | null>
}): Promise<SponsoredRepoFacts> {
  let complete = input.complete
  const found = new Set<string>()
  for (const raw of input.paths) {
    const path = raw.replace(/\\/g, '/').replace(/^\.\//, '')
    if (!path || path.split('/').includes('node_modules')) continue
    const basename = path.split('/').at(-1) ?? ''
    if (isSponsoredFitFileName(basename)) found.add(path)
  }
  let paths = [...found].sort(byDepthThenPath)
  if (paths.length > MAX_REPO_FACT_PATHS) {
    paths = paths.slice(0, MAX_REPO_FACT_PATHS)
    complete = false
  }

  const manifests: SponsoredRepoManifest[] = []
  let budget = MAX_REPO_FACT_DEPENDENCY_NAMES
  const manifestPaths = paths.filter((path) =>
    manifestEcosystem(path.split('/').at(-1) ?? ''),
  )
  for (const path of manifestPaths.slice(0, MAX_REPO_FACT_MANIFESTS)) {
    const basename = path.split('/').at(-1)!
    const ecosystem = manifestEcosystem(basename)!
    const text = await safeRead(input.read, path)
    const parsed =
      text === null || text.length > MAX_REPO_FACT_MANIFEST_BYTES
        ? { names: [], complete: false }
        : manifestDependencyNames(basename, text)
    const names = parsed.names.slice(0, Math.max(0, budget))
    budget -= names.length
    manifests.push({
      path,
      ecosystem,
      names,
      complete: parsed.complete && names.length === parsed.names.length,
    })
  }

  const empty: string[] = []
  for (const path of paths) {
    const basename = path.split('/').at(-1) ?? ''
    if (!(SPONSORED_FIT_MARKER_NAMES as readonly string[]).includes(basename))
      continue
    const text = await safeRead(input.read, path)
    if (text !== null && !/\S/.test(text)) empty.push(path)
  }

  return {
    v: 1,
    paths: paths.sort(),
    complete,
    manifests: manifests.sort((a, b) => a.path.localeCompare(b.path)),
    ...(empty.length ? { empty } : {}),
  }
}

async function safeRead(
  read: (path: string) => MaybePromise<string | null>,
  path: string,
): Promise<string | null> {
  try {
    return await read(path)
  } catch {
    return null
  }
}

  const depth = a.split('/').length - b.split('/').length

/**
 * The dependency names one manifest declares. Never throws: a manifest that
 * does not parse is `complete: false` with whatever was read.
 */
export function manifestDependencyNames(
  basename: string,
  text: string,
): { names: string[]; complete: boolean } {
  const names = new Set<string>()
  let complete = true
  const add = (name: string | null) => {
    if (name && name.length <= MAX_DEPENDENCY_NAME_CHARS) names.add(name)
  }
  if (basename === 'package.json') {
    try {
      const parsed = JSON.parse(text.replace(/^﻿/, '')) as unknown
      if (!parsed || typeof parsed !== 'object' || Array.isArray(parsed))
        return { names: [], complete: false }
      for (const field of [
        'dependencies',
        'devDependencies',
        'peerDependencies',
        'optionalDependencies',
      ]) {
        const deps = (parsed as Record<string, unknown>)[field]
        if (deps && typeof deps === 'object' && !Array.isArray(deps))
          for (const key of Object.keys(deps)) add(key)
      }
    } catch {
      return { names: [], complete: false }
    }
    return { names: [...names], complete }
  }
  const lines = text.replace(/^﻿/, '').split(/\r?\n/)
  if (/^requirements[^/]*\.txt$/.test(basename)) {
    for (const line of lines) add(requirementName(line.replace(/\s#.*$|^#.*$/, '')))
    return { names: [...names], complete }
  }
  if (basename === 'Pipfile' || basename === 'pyproject.toml') {
    let table = ''
    let arrayOpen = false
    for (const rawLine of lines) {
      const line = rawLine.replace(/\s+#.*$/, '').trim()
      if (!line || line.startsWith('#')) continue
      const header = /^\[\[?([^\]]+)\]\]?$/.exec(line)
      if (header) {
        table = header[1]!.trim()
        arrayOpen = false
        continue
      }
      if (arrayOpen) {
        for (const literal of line.matchAll(/["']([^"']+)["']/g))
          add(requirementName(literal[1]!))
        if (outsideStrings(line).includes(']')) arrayOpen = false
        continue
      }
      const assign = /^([A-Za-z0-9_.\-"']+)\s*=\s*(.*)$/.exec(line)
      if (!assign) continue
      const key = assign[1]!.replace(/["']/g, '')
      const value = assign[2]!
      const keyTable =
        table === 'packages' ||
        table === 'dev-packages' ||
        /^tool\.poetry(?:\.group\.[^.]+)?\.(?:dev-)?dependencies$/.test(table)
      if (keyTable) {
        if (key !== 'python') add(requirementName(key))
        continue
      }
      const arrayTable =
        (table === 'project' && key === 'dependencies') ||
        table === 'project.optional-dependencies' ||
        table === 'dependency-groups' ||
        (table === 'tool.uv' && key === 'dev-dependencies')
      if (arrayTable && value.startsWith('[')) {
        for (const literal of value.matchAll(/["']([^"']+)["']/g))
          add(requirementName(literal[1]!))
        if (!outsideStrings(value).slice(1).includes(']')) arrayOpen = true
      }
    }
    return { names: [...names], complete }
  }
  if (basename === 'setup.cfg') {
    let collecting = false
    let table = ''
    for (const rawLine of lines) {
      const header = /^\s*\[([^\]]+)\]\s*$/.exec(rawLine)
      if (header) {
        table = header[1]!.trim()
        collecting = false
        continue
      }
      const assign = /^([A-Za-z0-9_.-]+)\s*=\s*(.*)$/.exec(rawLine)
      if (assign) {
        collecting =
          (table === 'options' && assign[1] === 'install_requires') ||
          table === 'options.extras_require'
        if (collecting) add(requirementName(assign[2]!.replace(/#.*$/, '')))
        continue
      }
      if (collecting && /^\s+\S/.test(rawLine))
        add(requirementName(rawLine.replace(/#.*$/, '')))
    }
    return { names: [...names], complete }
  }
  if (basename === 'setup.py') {
    // A setup.py is code: read the literal install_requires / extras_require
    // lists, and say the answer may be partial.
    complete = false
    for (const block of text.matchAll(
      /(?:install_requires|extras_require)\s*=\s*([[{][\s\S]*?[\]}])\s*[,)]/g,
    )) {
      for (const literal of block[1]!.matchAll(/["']([^"']+)["']/g))
        add(requirementName(literal[1]!))
    }
    return { names: [...names], complete }
  }
  return { names: [], complete: false }
}

      /** The checks the facts could not decide (`fit` for the block, `requirements-os`). */
      unknownChecks: string[]

  | {

      reason: SponsoredFitMisfitReason
      checkId: string

export const SPONSORED_FIT_OS_CHECK_ID = 'requirements-os'
export const SPONSORED_FIT_BLOCK_CHECK_ID = 'fit'
export const SPONSORED_FIT_WEB_CHECK_ID = 'web'

type Facts = SponsoredRepoFacts

function pathsMatching(facts: Facts, glob: string): string[] {
  const re = sponsoredSkillGlob(glob)
  return facts.paths.filter((path) => re.test(path))
}

/** Whether no fit path the glob could match exists, as far as the facts prove. */
function provenAbsent(facts: Facts, glob: string): boolean {
  if (facts.complete) return sponsoredFitGlobDecidable(glob)
  // Without a full listing, only a glob of exact paths the facts name absent.
  const literals = expandSponsoredGlobBraces(glob)
  return (
    literals.every((literal) => !/[*?]/.test(literal)) &&
    literals.every((literal) => facts.absent?.includes(literal))
  )
}

function manifestAt(facts: Facts, path: string): SponsoredRepoManifest | null {
  return facts.manifests.find((manifest) => manifest.path === path) ?? null
}

/**
 * A manifest as text a check's regex can read: package.json as a JSON
 * object of its dependencies, a Python manifest as one requirement per line.
 * What the facts carry of the real file, never more.
 */
export function sponsoredManifestProjection(
  manifest: SponsoredRepoManifest,
): string {
  if (manifest.ecosystem === 'npm')
    return `{\n${manifest.names.map((name) => `  "${name}": "*"`).join(',\n')}\n}\n`
  return `${manifest.names.join('\n')}\n`
}

/** Whether a check can ever be decided from repository facts. */
export function sponsoredFitCheckDecidable(check: SponsoredSkillCheck): boolean {
  switch (check.kind) {
    case 'file_exists':
      return sponsoredFitGlobDecidable(check.path)
    case 'file_contains':
      return (check.changedOnly ?? true) === false && sponsoredFitGlobDecidable(check.path)
    case 'package_declared':
    case 'no_privileged_secrets':
    case 'changes_within':
      return true
    default:
      return false
  }
}

/** One preflight check's three-valued verdict over the facts. */
export function evaluateSponsoredFitCheck(
  check: SponsoredSkillCheck,
  facts: Facts | null | undefined,
): SponsoredFitCheckVerdict {
  // Before the first edit nothing has changed: these pass in every preflight.
  if (check.kind === 'no_privileged_secrets' || check.kind === 'changes_within')
    return 'pass'
  if (!facts) return 'unknown'
  switch (check.kind) {
    case 'file_exists':
      if (pathsMatching(facts, check.path).length) return 'pass'
      return provenAbsent(facts, check.path) ? 'fail' : 'unknown'
    case 'file_contains': {
      // A changed-only check reads the run's own changes; none exist yet.
      if ((check.changedOnly ?? true) !== false) return 'unknown'
      let re: RegExp
      try {
        re = new RegExp(check.pattern, check.flags)
      } catch {
        return 'unknown'
      }
      const files = pathsMatching(facts, check.path)
      if (check.absent) {
        let unknown = false
        for (const path of files) {
          if (facts.empty?.includes(path)) {
            if (re.test('')) return 'fail'
            continue
          }
          const manifest = manifestAt(facts, path)
          const isMarker = (SPONSORED_FIT_MARKER_NAMES as readonly string[]).includes(
            path.split('/').at(-1) ?? '',
          )
          // A present marker that is not empty has a non-blank character:
          // a pattern any such text matches fails it.
          if (isMarker && (check.pattern === '\\S' || check.pattern === '^'))
            return 'fail'
          if (manifest && re.test(sponsoredManifestProjection(manifest))) return 'fail'
          if (check.pattern === '^') return 'fail'
          unknown = true
        }
        if (unknown) return 'unknown'
        if (files.length === 0 && !provenAbsent(facts, check.path)) return 'unknown'
        return 'pass'
      }
      let unknown = false
      for (const path of files) {
        const manifest = manifestAt(facts, path)
        if (facts.empty?.includes(path)) {
          if (re.test('')) return 'pass'
          continue
        }
        if (!manifest) {
          unknown = true
          continue
        }
        if (re.test(sponsoredManifestProjection(manifest))) return 'pass'
        if (!manifest.complete) unknown = true
      }
      if (unknown) return 'unknown'
      if (files.length === 0 && !provenAbsent(facts, check.path)) return 'unknown'
      return 'fail'
    }
    case 'package_declared': {
      const root = manifestAt(facts, 'package.json')
      if (root) {
        if (root.names.includes(check.name)) return 'pass'
        return root.complete ? 'fail' : 'unknown'
      }
      if (facts.paths.includes('package.json')) return 'unknown'
      return provenAbsent(facts, 'package.json') ? 'fail' : 'unknown'
    }
    default:
      return 'unknown'
  }
}

function routeVerdict(
  facts: Facts,
  ecosystem: SponsoredFitRouteEcosystem,
  dependencies: string[] | null,
): SponsoredFitCheckVerdict {
  const glob = `**/{${SPONSORED_FIT_ECOSYSTEMS[ecosystem].manifests.join(',')}}`
  const files = pathsMatching(facts, glob)
  if (!files.length) return provenAbsent(facts, glob) ? 'fail' : 'unknown'
  if (!dependencies) return 'pass'
  const wanted = new Set(
    ecosystem === 'python' ? dependencies.map(normalizePypiName) : dependencies,
  )
  let unknown = false
  for (const path of files) {
    const manifest = manifestAt(facts, path)
    if (!manifest) {
      unknown = true
      continue
    }
    const names =
      ecosystem === 'python' ? manifest.names.map(normalizePypiName) : manifest.names
    if (names.some((name) => wanted.has(name))) return 'pass'
    if (!manifest.complete) unknown = true
  }
  return unknown ? 'unknown' : 'fail'
}

/** A format-2 fit block over the facts: the routes ORed, then `mustNotExist`. */
function fitBlockVerdict(
  skill: SponsoredSkill,
  facts: Facts | null | undefined,
): { verdict: SponsoredFitCheckVerdict; reason?: SponsoredFitMisfitReason } {
  const fit = skill.fit
  if (!fit) return { verdict: 'pass' }
  if (!facts) return { verdict: 'unknown' }
  const routes = sponsoredFitRoutes(fit)
  const verdicts = routes.map((route) =>
    routeVerdict(facts, route.ecosystem, route.dependencies),
  )
  let route: SponsoredFitCheckVerdict = verdicts.includes('pass')
    ? 'pass'
    : verdicts.includes('unknown')
      ? 'unknown'
      : 'fail'
  if (route === 'fail') {
    const anyManifest = routes.some(
      (r) =>
        pathsMatching(
          facts,
          `**/{${SPONSORED_FIT_ECOSYSTEMS[r.ecosystem].manifests.join(',')}}`,
        ).length > 0,
    )
    return { verdict: 'fail', reason: anyManifest ? 'no_route' : 'no_manifest' }
  }
  for (const glob of fit.mustNotExist ?? []) {
    if (pathsMatching(facts, glob).length)
      return { verdict: 'fail', reason: 'file_present' }
    if (!provenAbsent(facts, glob)) route = route === 'pass' ? 'unknown' : route
  }
  return { verdict: route }
}

function checkMisfitReason(check: SponsoredSkillCheck): SponsoredFitMisfitReason {
  switch (check.kind) {
    case 'file_exists':
      return 'file_missing'
    case 'package_declared':
      return 'package_missing'
    case 'file_contains':
      return check.absent ? 'file_present' : 'file_content'
    default:
      return 'file_content'
  }
}

/**
 * THE fit verdict of one skill for one repository and device. `fits` only
 * when every fit check passes; `misfit` as soon as one is proven to fail
 * (the first, in the order the step driver runs them); `unknown` otherwise.
 */
export function evaluateSponsoredSkillFit(
  skill: SponsoredSkill,
  input: SponsoredFitInput,
): SponsoredSkillFitVerdict {
  const unknownChecks: string[] = []
  const misfit = (
    reason: SponsoredFitMisfitReason,
    checkId: string,
  ): SponsoredSkillFitVerdict => ({
    fit: 'misfit',
    reason,
    checkId,
    proof: 'known',
    unknownChecks,
  })
  if (input.web) {
    const web = sponsoredSkillWebRefusal(skill)
    if (web) return misfit(web, SPONSORED_FIT_WEB_CHECK_ID)
  }
  const supported = skill.requirements?.os ?? []
  if (supported.length) {
    if (!input.os) unknownChecks.push(SPONSORED_FIT_OS_CHECK_ID)
    else if (!supported.includes(input.os))
      return misfit('os', SPONSORED_FIT_OS_CHECK_ID)
  }
  if (skill.fit) {
    const block = fitBlockVerdict(skill, input.repo)
    if (block.verdict === 'fail')
      return misfit(block.reason ?? 'no_route', SPONSORED_FIT_BLOCK_CHECK_ID)
    if (block.verdict === 'unknown') unknownChecks.push(SPONSORED_FIT_BLOCK_CHECK_ID)
  }
  for (const check of skill.preflight ?? []) {
    const verdict = evaluateSponsoredFitCheck(check, input.repo)
    if (verdict === 'fail') return misfit(checkMisfitReason(check), check.id)
    if (verdict === 'unknown') unknownChecks.push(check.id)
  }
  return unknownChecks.length ? { fit: 'unknown', unknownChecks } : { fit: 'fits' }
}

export type SponsoredSkillFitRefusal = {
  reason: SponsoredFitMisfitReason
  checkId: string
}

/**
 * Why `skill` must not be offered for this repository and device, or null.
 * Refuses ONLY a known misfit: an unknown verdict serves.
 */
export function sponsoredSkillFitRefusal(
  skill: SponsoredSkill,
  input: SponsoredFitInput,
): SponsoredSkillFitRefusal | null {
  const verdict = evaluateSponsoredSkillFit(skill, input)
  return verdict.fit === 'misfit' && verdict.proof === 'known'
    ? { reason: verdict.reason, checkId: verdict.checkId }
    : null
}

/**
 * The repository's ecosystems, for logs: which manifests exist, never what
 * they declare. `root` counts only manifests at the repository root.
 */
export function sponsoredRepoFactsSummary(facts: SponsoredRepoFacts | null | undefined): {
  ecosystems: string[]
  rootEcosystems: string[]
  complete: boolean | null
  empty: boolean | null
  monorepo: boolean | null
} {
  if (!facts)
    return { ecosystems: [], rootEcosystems: [], complete: null, empty: null, monorepo: null }
  const ecosystemOf = (basename: string): string | null => {
    if (basename === 'package.json' || basename === 'deno.json' || basename === 'deno.jsonc')
      return basename === 'package.json' ? 'node' : 'deno'
    if (manifestEcosystem(basename) === 'pypi') return 'python'
    if (basename === 'go.mod') return 'go'
    if (basename === 'Cargo.toml') return 'rust'
    if (basename === 'composer.json') return 'php'
    if (basename === 'Gemfile') return 'ruby'
    if (basename === 'pom.xml' || basename.startsWith('build.gradle')) return 'java'
    return null
  }
  const all = new Set<string>()
  const root = new Set<string>()
  let packageJsons = 0
  for (const path of facts.paths) {
    const parts = path.split('/')
    const ecosystem = ecosystemOf(parts.at(-1) ?? '')
    if (!ecosystem) continue
    all.add(ecosystem)
    if (parts.length === 1) root.add(ecosystem)
    if (parts.at(-1) === 'package.json') packageJsons++
  }
  return {
    ecosystems: [...all].sort(),
    rootEcosystems: [...root].sort(),
    complete: facts.complete,
    empty: facts.complete ? facts.paths.length === 0 : null,
    monorepo:
      packageJsons > 1 ||
      facts.paths.includes('pnpm-workspace.yaml') ||
      (all.size > 1 && root.size < all.size),
  }
}

+    errors.push({ path: 'fit', message: 'format 2 needs a fit object' })
+    return

+  for (const key of Object.keys(value)) {
+    if (!['os', 'routes', 'mustNotExist'].includes(key))
+      errors.push({ path: `fit.${key}`, message: 'is not a fit field' })

+      errors.push({
+        path: 'fit.os',
+        message: 'must be a non-empty list of darwin, linux, win32',

+      ) {
+        errors.push({
+          path: at,
+          message: `unknown ecosystem; one of ${SPONSORED_FIT_ROUTE_ECOSYSTEMS.join(', ')}`,
+        })
+        continue
+      }
+      if (!isRecord(route)) {
+        errors.push({ path: at, message: 'must be an object' })
+        continue
+      }
+      for (const key of Object.keys(route)) {
+        if (!['frameworks', 'anyDependency'].includes(key))
+          errors.push({ path: `${at}.${key}`, message: 'is not a route field' })
+      }
+      const table = SPONSORED_FIT_ECOSYSTEMS[ecosystem as SponsoredFitRouteEcosystem]
+      if (route.frameworks !== undefined) {
+        const known = Object.keys(table.frameworks)
+        if (
+          !Array.isArray(route.frameworks) ||
+          !route.frameworks.every((name) => known.includes(name as string)) ||
+          new Set(route.frameworks).size !== route.frameworks.length

+            path: `${at}.frameworks`,
+            message: `must be distinct names from ${known.join(', ')}`,
+          })

+/** The routes a fit block declares, in a fixed order. */
+export function sponsoredFitRoutes(
+  fit: SponsoredSkillFit,
+): Array<{ ecosystem: SponsoredFitRouteEcosystem; dependencies: string[] | null }> {
+  const out: Array<{
+    ecosystem: SponsoredFitRouteEcosystem
+    dependencies: string[] | null
+  }> = []
+  for (const ecosystem of SPONSORED_FIT_ROUTE_ECOSYSTEMS) {
+    const route = fit.routes[ecosystem] as
+      | SponsoredFitRoute<string>
+      | undefined
+    if (!route) continue
+    const table = SPONSORED_FIT_ECOSYSTEMS[ecosystem].frameworks as Record<
+      string,
+      readonly string[]
+    >
+    const deps = [
+      ...(route.frameworks ?? []).flatMap((name) => table[name] ?? []),
+      ...(route.anyDependency ?? []),
+    ]
+    out.push({
+      ecosystem,
+      dependencies: deps.length ? [...new Set(deps)] : null,
+    })
+  }

+function escapeRegex(text: string): string {
+  return text.replace(/[.*+?^${}()|[\]\\/]/g, '\\$&')

+/** A PyPI name as a regex: case-insensitive by flag, `-`, `_` and `.` interchangeable (PEP 503). */
+function pypiNameRegex(name: string): string {

+/** Start of the file, whatever the `m` flag says. */

+/** At a line start, in a file that is not a JSON object (so never package.json). */
+const PYTHON_LINE = `^(?<=${FILE_START}(?!\\s*\\{)[\\s\\S]*)[ \\t]*["']?`
+/** After a Python requirement's name: a version, extra, marker, comment, separator or the end of the line. */
+const PYTHON_NAME_END = `["']?(?=[ \\t]*(?:[<>=!~;\\[\\],#@]|$))`

+/**
+ * The regex alternative a route compiles to, matched per manifest file:
+ * - node: the dependency is a JSON key (`"next":`).
+ * - python: a line of a non-JSON manifest starts with the requirement
+ *   (`fastapi>=0.1`, `"fastapi[all]",`, `fastapi = "^0.110"`).
+ * - a route with no dependency list: any manifest of the ecosystem
+ *   (package.json is a JSON object; a Python manifest is not).
+ */
+function routePattern(route: {
+  ecosystem: SponsoredFitRouteEcosystem
+  dependencies: string[] | null
+}): string {
+  if (route.ecosystem === 'node') {
+    return route.dependencies
+      ? `"(?:${route.dependencies.map(escapeRegex).join('|')})"\\s*:`
+      : `${FILE_START}\\s*\\{`
+  }
+  return route.dependencies
+    ? `${PYTHON_LINE}(?:${route.dependencies.map(pypiNameRegex).join('|')})${PYTHON_NAME_END}`
+    : `${FILE_START}(?!\\s*\\{)`
+}

+/** Each declared route's manifest basenames, one brace glob for every depth. */
+export function sponsoredFitManifestGlob(fit: SponsoredSkillFit): string {
+  const names = sponsoredFitRoutes(fit).flatMap(
+    (route) => SPONSORED_FIT_ECOSYSTEMS[route.ecosystem].manifests,
+  )
+  return `**/{${names.join(',')}}`

+/**
+ * THE fit block as preflight checks, run ahead of the skill's own
+ * `preflight` by the step driver (`sponsoredSkillPreflightChecks`):
+ * 1. `fit-manifest`: a manifest of a declared route exists;
+ * 2. `fit-route`: one such manifest declares a route's dependency (dropped
+ *    when every route is unconstrained);
+ * 3. `fit-must-not-exist`: none of `mustNotExist` exists (even empty).
+ * The OS rides `requirements.os`, which the step driver already checks.
+ * Generated, never hand-edited, so the regex cap on authored checks does not
+ * apply.
+ */
+export function compileSponsoredSkillFit(
+  fit: SponsoredSkillFit,
+): SponsoredSkillCheck[] {
+  const routes = sponsoredFitRoutes(fit)
+  const path = sponsoredFitManifestGlob(fit)
+  const ecosystems = routes.map((route) =>
+    route.ecosystem === 'node' ? 'a package.json' : 'a Python manifest',
+  )
+  const checks: SponsoredSkillCheck[] = [
+    {
+      id: SPONSORED_FIT_CHECK_IDS.manifest,
+      label: `the project has ${ecosystems.join(' or ')}`,
+      kind: 'file_exists',
+      path,
+    },
+  ]
+  if (routes.some((route) => route.dependencies)) {
+    checks.push({
+      id: SPONSORED_FIT_CHECK_IDS.route,
+      label: `the project uses ${routes
+        .map((route) =>
+          route.dependencies
+            ? route.dependencies.slice(0, 4).join(', ') +
+              (route.dependencies.length > 4 ? ' …' : '')
+            : `any ${route.ecosystem} project`,
+        )
+        .join(' or ')}`,
+      kind: 'file_contains',
+      path,
+      pattern: routes.map(routePattern).join('|'),
+      flags: 'im',
+      changedOnly: false,
+    })
+  }
+  if (fit.mustNotExist?.length) {
+    checks.push({
+      id: SPONSORED_FIT_CHECK_IDS.mustNotExist,
+      label: `the project has no ${fit.mustNotExist.join(' or ')} yet`,
+      kind: 'file_contains',
+      path:
+        fit.mustNotExist.length === 1
+          ? fit.mustNotExist[0]!
+          : `{${fit.mustNotExist.join(',')}}`,
+      // `^` matches every file, empty ones too: any match is a failure.
+      pattern: '^',
+      absent: true,
+      changedOnly: false,
+    })
+  }
+  return checks

+export async function sponsoredRepoFacts(input: {
+  paths: Iterable<string>
+  complete: boolean

+}): Promise<SponsoredRepoFacts> {
+  let complete = input.complete
+  const found = new Set<string>()
+  for (const raw of input.paths) {

+    if (!path || path.split('/').includes('node_modules')) continue

+    paths = paths.slice(0, MAX_REPO_FACT_PATHS)
+    complete = false
+  }

+    manifestEcosystem(path.split('/').at(-1) ?? ''),

+/**
+ * The dependency names one manifest declares. Never throws: a manifest that

+export function manifestDependencyNames(
+  basename: string,
+  text: string,
+): { names: string[]; complete: boolean } {
+  const names = new Set<string>()
+  let complete = true
+  const add = (name: string | null) => {

+        return { names: [], complete: false }
+      for (const field of [
+        'dependencies',
+        'devDependencies',
+        'peerDependencies',
+        'optionalDependencies',
+      ]) {
+        const deps = (parsed as Record<string, unknown>)[field]
+        if (deps && typeof deps === 'object' && !Array.isArray(deps))
+          for (const key of Object.keys(deps)) add(key)
+      }
+    } catch {
+      return { names: [], complete: false }
+    }
+    return { names: [...names], complete }
+  }
+  const lines = text.replace(/^﻿/, '').split(/\r?\n/)
+  if (/^requirements[^/]*\.txt$/.test(basename)) {
+    for (const line of lines) add(requirementName(line.replace(/\s#.*$|^#.*$/, '')))
+    return { names: [...names], complete }
+  }
+  if (basename === 'Pipfile' || basename === 'pyproject.toml') {
+    let table = ''
+    let arrayOpen = false
+    for (const rawLine of lines) {
+      const line = rawLine.replace(/\s+#.*$/, '').trim()
+      if (!line || line.startsWith('#')) continue
+      const header = /^\[\[?([^\]]+)\]\]?$/.exec(line)
+      if (header) {
+        table = header[1]!.trim()
+        arrayOpen = false
+        continue
+      }
+      if (arrayOpen) {
+        for (const literal of line.matchAll(/["']([^"']+)["']/g))
+          add(requirementName(literal[1]!))
+        if (outsideStrings(line).includes(']')) arrayOpen = false
+        continue
+      }
+      const assign = /^([A-Za-z0-9_.\-"']+)\s*=\s*(.*)$/.exec(line)
+      if (!assign) continue
+      const key = assign[1]!.replace(/["']/g, '')
+      const value = assign[2]!
+      const keyTable =
+        table === 'packages' ||
+        table === 'dev-packages' ||
+        /^tool\.poetry(?:\.group\.[^.]+)?\.(?:dev-)?dependencies$/.test(table)
+      if (keyTable) {
+        if (key !== 'python') add(requirementName(key))
+        continue
+      }
+      const arrayTable =
+        (table === 'project' && key === 'dependencies') ||
+        table === 'project.optional-dependencies' ||
+        table === 'dependency-groups' ||
+        (table === 'tool.uv' && key === 'dev-dependencies')
+      if (arrayTable && value.startsWith('[')) {
+        for (const literal of value.matchAll(/["']([^"']+)["']/g))
+          add(requirementName(literal[1]!))
+        if (!outsideStrings(value).slice(1).includes(']')) arrayOpen = true
+      }
+    }
+    return { names: [...names], complete }
+  }
+  if (basename === 'setup.cfg') {
+    let collecting = false
+    let table = ''
+    for (const rawLine of lines) {
+      const header = /^\s*\[([^\]]+)\]\s*$/.exec(rawLine)
+      if (header) {
+        table = header[1]!.trim()
+        collecting = false
+        continue
+      }
+      const assign = /^([A-Za-z0-9_.-]+)\s*=\s*(.*)$/.exec(rawLine)
+      if (assign) {
+        collecting =
+          (table === 'options' && assign[1] === 'install_requires') ||
+          table === 'options.extras_require'
+        if (collecting) add(requirementName(assign[2]!.replace(/#.*$/, '')))
+        continue
+      }
+      if (collecting && /^\s+\S/.test(rawLine))
+        add(requirementName(rawLine.replace(/#.*$/, '')))
+    }
+    return { names: [...names], complete }
+  }
+  if (basename === 'setup.py') {
+    // A setup.py is code: read the literal install_requires / extras_require
+    // lists, and say the answer may be partial.
+    complete = false
+    for (const block of text.matchAll(
+      /(?:install_requires|extras_require)\s*=\s*([[{][\s\S]*?[\]}])\s*[,)]/g,
+    )) {
+      for (const literal of block[1]!.matchAll(/["']([^"']+)["']/g))
+        add(requirementName(literal[1]!))
+    }
+    return { names: [...names], complete }
+  }
+  return { names: [], complete: false }

+/** One preflight check's three-valued verdict over the facts. */
+export function evaluateSponsoredFitCheck(
+  check: SponsoredSkillCheck,
+  facts: Facts | null | undefined,
+): SponsoredFitCheckVerdict {
+  // Before the first edit nothing has changed: these pass in every preflight.
+  if (check.kind === 'no_privileged_secrets' || check.kind === 'changes_within')
+    return 'pass'
+  if (!facts) return 'unknown'
+  switch (check.kind) {
+    case 'file_exists':
+      if (pathsMatching(facts, check.path).length) return 'pass'
+      return provenAbsent(facts, check.path) ? 'fail' : 'unknown'
+    case 'file_contains': {
+      // A changed-only check reads the run's own changes; none exist yet.
+      if ((check.changedOnly ?? true) !== false) return 'unknown'
+      let re: RegExp
+      try {
+        re = new RegExp(check.pattern, check.flags)
+      } catch {
+        return 'unknown'
+      }
+      const files = pathsMatching(facts, check.path)
+      if (check.absent) {
+        let unknown = false
+        for (const path of files) {
+          if (facts.empty?.includes(path)) {
+            if (re.test('')) return 'fail'
+            continue
+          }
+          const manifest = manifestAt(facts, path)
+          const isMarker = (SPONSORED_FIT_MARKER_NAMES as readonly string[]).includes(
+            path.split('/').at(-1) ?? '',
+          )
+          // A present marker that is not empty has a non-blank character:
+          // a pattern any such text matches fails it.
+          if (isMarker && (check.pattern === '\\S' || check.pattern === '^'))
+            return 'fail'
+          if (manifest && re.test(sponsoredManifestProjection(manifest))) return 'fail'
+          if (check.pattern === '^') return 'fail'
+          unknown = true
+        }
+        if (unknown) return 'unknown'
+        if (files.length === 0 && !provenAbsent(facts, check.path)) return 'unknown'
+        return 'pass'
+      }
+      let unknown = false
+      for (const path of files) {
+        const manifest = manifestAt(facts, path)
+        if (facts.empty?.includes(path)) {
+          if (re.test('')) return 'pass'
+          continue
+        }
+        if (!manifest) {
+          unknown = true
+          continue
+        }
+        if (re.test(sponsoredManifestProjection(manifest))) return 'pass'
+        if (!manifest.complete) unknown = true
+      }
+      if (unknown) return 'unknown'
+      if (files.length === 0 && !provenAbsent(facts, check.path)) return 'unknown'
+      return 'fail'
+    }
+    case 'package_declared': {
+      const root = manifestAt(facts, 'package.json')
+      if (root) {
+        if (root.names.includes(check.name)) return 'pass'
+        return root.complete ? 'fail' : 'unknown'
+      }
+      if (facts.paths.includes('package.json')) return 'unknown'
+      return provenAbsent(facts, 'package.json') ? 'fail' : 'unknown'
+    }
+    default:
+      return 'unknown'
+  }
+}

+function routeVerdict(
+  facts: Facts,
+  ecosystem: SponsoredFitRouteEcosystem,
+  dependencies: string[] | null,
+): SponsoredFitCheckVerdict {
+  const glob = `**/{${SPONSORED_FIT_ECOSYSTEMS[ecosystem].manifests.join(',')}}`
+  const files = pathsMatching(facts, glob)
+  if (!files.length) return provenAbsent(facts, glob) ? 'fail' : 'unknown'
+  if (!dependencies) return 'pass'

+/** A format-2 fit block over the facts: the routes ORed, then `mustNotExist`. */
+function fitBlockVerdict(
+  skill: SponsoredSkill,
+  facts: Facts | null | undefined,
+): { verdict: SponsoredFitCheckVerdict; reason?: SponsoredFitMisfitReason } {
+  const fit = skill.fit
+  if (!fit) return { verdict: 'pass' }
+  if (!facts) return { verdict: 'unknown' }
+  const routes = sponsoredFitRoutes(fit)
+  const verdicts = routes.map((route) =>
+    routeVerdict(facts, route.ecosystem, route.dependencies),
+  )
+  let route: SponsoredFitCheckVerdict = verdicts.includes('pass')
+    ? 'pass'
+    : verdicts.includes('unknown')
+      ? 'unknown'
+      : 'fail'
+  if (route === 'fail') {
+    const anyManifest = routes.some(

+        pathsMatching(
+          facts,
+          `**/{${SPONSORED_FIT_ECOSYSTEMS[r.ecosystem].manifests.join(',')}}`,
+        ).length > 0,
+    )
+    return { verdict: 'fail', reason: anyManifest ? 'no_route' : 'no_manifest' }

+      return { verdict: 'fail', reason: 'file_present' }
+    if (!provenAbsent(facts, glob)) route = route === 'pass' ? 'unknown' : route

+/**
+ * THE fit verdict of one skill for one repository and device. `fits` only
+ * when every fit check passes; `misfit` as soon as one is proven to fail
+ * (the first, in the order the step driver runs them); `unknown` otherwise.

+export function evaluateSponsoredSkillFit(
+  skill: SponsoredSkill,
+  input: SponsoredFitInput,
+): SponsoredSkillFitVerdict {
+  const unknownChecks: string[] = []
+  const misfit = (
+    reason: SponsoredFitMisfitReason,
+    checkId: string,
+  ): SponsoredSkillFitVerdict => ({
+    fit: 'misfit',
+    reason,
+    checkId,
+    proof: 'known',
+    unknownChecks,
+  })
+  if (input.web) {
+    const web = sponsoredSkillWebRefusal(skill)
+    if (web) return misfit(web, SPONSORED_FIT_WEB_CHECK_ID)
+  }
+  const supported = skill.requirements?.os ?? []
+  if (supported.length) {
+    if (!input.os) unknownChecks.push(SPONSORED_FIT_OS_CHECK_ID)
+    else if (!supported.includes(input.os))
+      return misfit('os', SPONSORED_FIT_OS_CHECK_ID)
+  }
+  if (skill.fit) {
+    const block = fitBlockVerdict(skill, input.repo)

+      return misfit(block.reason ?? 'no_route', SPONSORED_FIT_BLOCK_CHECK_ID)
+    if (block.verdict === 'unknown') unknownChecks.push(SPONSORED_FIT_BLOCK_CHECK_ID)
+  }

+    if (verdict === 'unknown') unknownChecks.push(check.id)

 /** Semver order (prerelease below its release), for picking the latest version. */
 export function compareSponsoredSkillVersions(a: string, b: string): number {
   const parse = (version: string) => {
     const [core = '', pre] = version.split('-', 2)
     return { parts: core.split('.').map(Number), pre }
   }
   const left = parse(a)
   const right = parse(b)
   for (let i = 0; i < 3; i++) {
     const diff = (left.parts[i] ?? 0) - (right.parts[i] ?? 0)
     if (diff) return diff
   }
   if (left.pre === right.pre) return 0
   if (left.pre === undefined) return 1
   if (right.pre === undefined) return -1
   return left.pre < right.pre ? -1 : 1
 }
