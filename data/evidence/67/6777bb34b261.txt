export type SponsoredRunSurface = SponsoredAcceptSurface

/** What the mid-run account step does: open a tracked link, or only ask for keys. */
export type SponsoredRunDeferredSignup = 'link' | 'credentials'

/** "How to carry it out here" for a surface. Only Desktop has a worktree flow. */
export function sponsoredRunGuidance(
  surface: SponsoredRunSurface,
  inPlace: boolean,
): string {
  if (surface === 'cli') return SPONSORED_RUN_CLI_GUIDANCE
  if (surface === 'web') return SPONSORED_RUN_WEB_GUIDANCE
  return inPlace
    ? SPONSORED_RUN_DESKTOP_IN_PLACE_GUIDANCE
    : SPONSORED_RUN_DESKTOP_WORKTREE_GUIDANCE
}

/**
 * Which account step the run has, as each surface's account step decides it.
 * Desktop and the CLI open a signup link when `sponsoredSignupStep` finds one
 * (a reviewed `bfcid={bfcid}` link, else the Accept's tracked link when the
 * procedure needs an account), else ask only for declared credentials. The
 * Cloud account step opens the Accept's tracked link whenever there is one
 * (`sponsoredCloudAccountStep`).
 */
export function sponsoredRunDeferredSignup(input: {
  surface: SponsoredRunSurface
  procedure: string
  advertiserLink: string | null
  skill: SponsoredSkill | null
}): SponsoredRunDeferredSignup | undefined {
  const hasLink =
    input.surface === 'web'
      ? Boolean(input.advertiserLink)
      : sponsoredSignupStep(
          input.procedure,
          { advertiserLink: input.advertiserLink },
          input.skill,
        ) !== undefined
  if (hasLink) return 'link'
  return declaredRunCredentials(input.procedure).length > 0
    ? 'credentials'
    : undefined
}

/**

export function buildSponsoredRunPromptTemplate(input: {
  surface: SponsoredRunSurface
  inPlace: boolean
  procedure: string
  advertiserLink: string | null
  deferredSignup?: SponsoredRunDeferredSignup
  /** The client has the `install_package` tool (COD-965). */
  installTool?: boolean

  const guidance = sponsoredRunGuidance(input.surface, input.inPlace)
  // ORDER IS THE FIX. The task before the procedure, the rules after both,

    })}`,

    `How to carry it out here:\n${
      input.installTool
        ? guidance.replace(PINNED_INSTALLS_BULLET, INSTALL_TOOL_BULLET)
        : guidance

    ...(input.deferredSignup === 'link'
      ? [SPONSORED_RUN_DEFERRED_SIGNUP_LINK]
      : input.deferredSignup === 'credentials'
        ? [SPONSORED_RUN_DEFERRED_SIGNUP_CREDENTIALS]

    SPONSORED_RUN_RESULT_REPORT,
  ].join('\n\n')

/**

 * own messages, oldest first, numbered. Empty for no messages.
 */
export function renderSponsoredRunTaskContext(
  messages: readonly string[],
): string {
  return messages
    .map((message, index) => `User message ${index + 1}:\n${message}`)
    .join('\n\n')

/**
 * The run spec the funded Accept returns to a client that sent
 * `runSpecVersion`. `skill` must be the server-resolved skill the procedure
 * renders (`sponsoredSkillMatchesProcedure`), or null for a free-text
 * procedure: a caller whose procedure names a skill it could not resolve
 * refuses the Accept instead of building a spec with no packages.
 */
export function buildSponsoredRunSpec(input: {
  procedure: string
  skill: SponsoredSkill | null
  surface: SponsoredRunSurface
  inPlace: boolean
  advertiserLink: string | null
  /**
   * The client sent `installToolVersion` (COD-965): installs go through its
   * `install_package` tool. Absent or false leaves the prompt byte-identical.
   */
  installTool?: boolean
}): SponsoredRunSpec {
  return {
    v: SPONSORED_RUN_SPEC_VERSION,
    procedure: input.procedure,
    procedureSha256: sha256Hex(input.procedure),
    skill: input.skill,
    declaredPackages: input.skill?.packages ?? [],
    // Registered long-form skills keep the frame they were written against.
    promptTemplate: isLayer2Skill(input.skill)
      ? buildSponsoredRunLayer1PromptTemplate(input)
      : buildSponsoredRunPromptTemplate({
          surface: input.surface,
          inPlace: input.inPlace,
          procedure: input.procedure,
          advertiserLink: input.advertiserLink,
          deferredSignup: sponsoredRunDeferredSignup(input),
          ...(input.installTool ? { installTool: true } : {}),
        }),
  }

/** `posthog` → `Posthog`, `open-router` → `Open Router`: a label, never parsed. */
function advertiserLabel(advertiser: string): string {
  return advertiser
    .split('-')
    .map((word) => word.charAt(0).toUpperCase() + word.slice(1))
    .join(' ')
}

/** `POSTHOG_PROJECT_TOKEN` → `project token`, `EXA_API_KEY` → `API key`. */
function keyLabel(env: string, advertiser: string): string {
  const prefix = `${advertiser.replace(/-/g, '_').toUpperCase()}_`
  const rest = env.startsWith(prefix) ? env.slice(prefix.length) : env
  return rest
    .split('_')
    .map((word) => (word === 'API' ? word : word.toLowerCase()))
    .join(' ')
}

/** `a Posthog`, `an Archil`. */
function withArticle(noun: string): string {
  return `${/^[aeiou]/i.test(noun) ? 'an' : 'a'} ${noun}`
}

function escapeRegex(text: string): string {
  return text.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
}

const ENV_EXAMPLE_PATH =
  '**/{.env.example,.env.local.example,.env.sample,.env.template}'
/** The source files a verify check reads. */
const SOURCE_FILES = '**/*.{ts,tsx,js,jsx,mjs,cjs,mts,cts,py}'
const PYTHON_MANIFESTS =
  '**/{pyproject.toml,requirements*.txt,setup.py,setup.cfg,Pipfile}'

/**
 * The format-1, fit-free skill a Layer 2 record stands for (COD-965: no
 * project gate, so any folder can take the offer). Deterministic: the same
 * source and version always give the same bytes, and nothing optional is
 * emitted empty or as a default, so the hash only moves when a field does.
 */
export function compileLayer2Skill(
  source: SponsoredSkillLayer2,
  version: string,
): SponsoredSkill {
  const npm = source.packages.filter((pkg) => pkg.ecosystem === 'npm')
  const pypi = source.packages.filter((pkg) => pkg.ecosystem === 'pypi')
  const envKeys = source.envKeys ?? []
  const label = advertiserLabel(source.advertiser)

  // Format 1 has no fit block, so `frameworks` must be non-empty even for a
  // Python-only skill; it is informational there, like the archil skill's.
  const requirements: SponsoredSkill['requirements'] = {
    os: ['darwin', 'linux', 'win32'],
    runtimes: [
      // Native HTTP skills have no packages; retain a valid informational runtime.
      ...(npm.length || !pypi.length ? [{ name: 'node' as const }] : []),
      ...(pypi.length ? [{ name: 'python' as const }] : []),
    ],
    frameworks: ['nodejs'],
    packageManagers: ['npm', 'pnpm', 'bun'],
  }

  const packages: SponsoredSkillPackage[] = source.packages.map((pkg) => ({
    ecosystem: pkg.ecosystem,
    name: pkg.name,
    version: pkg.version,
  }))

  // A package check only when one ecosystem is declared: in a mixed skill a
  // Node project would fail the Python check and the other way round.
  const packageChecks: SponsoredSkillCheck[] =
    npm.length && !pypi.length
      ? npm.map((pkg, index) => ({
          id: `package-${index + 1}`,
          kind: 'package_declared',
          label: `${pkg.name} is declared in package.json`,
          name: pkg.name,
        }))
      : pypi.length && !npm.length
        ? pypi.map((pkg, index) => ({
            id: `package-${index + 1}`,
            kind: 'file_contains',
            label: `${pkg.name} is declared in a Python manifest`,
            path: PYTHON_MANIFESTS,
            pattern: `(^|[\\s"'\\[,])${escapeRegex(pkg.name)}\\b`,
            flags: 'im',
          }))
        : []

  const envChecks: SponsoredSkillCheck[] = envKeys.map((key, index) => ({
    id: `env-${index + 1}`,
    kind: 'env_key_declared',
    label: `${key} is declared, empty, in an example env file`,
    path: ENV_EXAMPLE_PATH,
    key,
  }))

  const verifyChecks: SponsoredSkillCheck[] = (source.verify ?? []).map(
    (item, index) => ({
      id: `verify-${index + 1}`,
      kind: 'file_contains',
      label: item.label,
      path: SOURCE_FILES,
      pattern: item.pattern,
      flags: 'i',
    }),
  )

  // Guidance the agent adapts to the project, not a script; the checks are
  // what it must meet. No wording here may read as user work
  // (`NEEDS_USER_PATTERNS` in sponsored-procedure-steps.ts).
  const envLine = envKeys.length
    ? `\nRead ${envKeys.join(' and ')} from the environment; declare ${
        envKeys.length === 1 ? 'it' : 'each'
      } empty in the example env file.`
    : ''
  const integrate: SponsoredSkillStep = {
    id: 'integrate',
    kind: 'agent',
    title: `Build the ${label} integration in this project`,
    // Layer 1 already says fit the project, fewest changes, read the docs;
    // the renderer says a step is done when its checks pass.
    instructions: `${source.goal}\nDocs: ${source.docsUrl}\nIn an empty folder, scaffold a minimal project first.${envLine}`,
    checks: [
      ...verifyChecks,
      ...packageChecks,
      ...envChecks,
      {
        id: 'no-secrets',
        kind: 'no_privileged_secrets',
        label: 'changed files carry no secrets',
        paths: ['**'],
      },
    ],
  }

  const userSteps: SponsoredSkillStep[] = envKeys.length
    ? envKeys.map((env, index) => {
        const credential: SponsoredRunCredential = {
          env,
          label: `${label} ${keyLabel(env, source.advertiser)}`.slice(0, 60),
          getUrl: source.signupUrl,
        }
        return {
          id: `connect-${index + 1}`,
          kind: 'user',
          title: `Ask the user to create ${withArticle(`${label} ${keyLabel(env, source.advertiser)}`)} and paste it`,
          instructions: `The harness saves the value; never read, write or print it or an env file. Once ${env} is set, run the feature once and report whether ${label} answered.`,
          checks: [
            {
              id: `credential-${index + 1}`,
              kind: 'credential_provided',
              label: `the ${label} ${keyLabel(env, source.advertiser)} was provided`,
              envKey: env,
            },
          ],
          credential,
        }
      })
    : [
        {
          id: 'connect',
          kind: 'user',
          title: `Ask the user to sign up for ${label}`,
          instructions: `Open ${source.signupUrl} so the user can create an account, then say what is left to finish the integration.`,
          checks: [
            {
              id: 'signed-up',
              kind: 'user_confirmed',
              label: `the user finished signing up for ${label}`,
            },
          ],
        },
      ]

  return {
    format: 1,
    id: source.id,
    version,
    changelog: LAYER2_SKILL_CHANGELOG,
    advertiser: source.advertiser,
    goal: source.goal,
    fits: source.goal,
    consent: { title: source.consent.title, summary: source.consent.summary },
    requirements,
    packages,
    tools: [],
    // The docs link is in the integrate step.
    knowledge: [],
    preflight: [],
    steps: [integrate, ...userSteps],
    outcomes: envKeys.length ? ['api_key_issued'] : [],
    ...(source.conversion ? { conversion: source.conversion } : {}),
  }
}

/** Zod issues as the skill parser's `{ path, message }` errors. */
function zodErrors(error: z.ZodError): SponsoredSkillParseError[] {
  return error.issues.map((issue) => ({
    path: issue.path.join('.') || '(root)',
    message: issue.message,
  }))
}

/**
 * Validate a Layer 2 record and compile it: the schema, then the full skill
 * parser every client runs, then the tracked-link rule (COD-882). Any failure
 * comes back as errors an admin can act on.
 */
export function buildLayer2Skill(
  value: unknown,
  version: string,
): Layer2BuildResult {
  const parsed = sponsoredSkillLayer2Schema.safeParse(value)
  if (!parsed.success) return { ok: false, errors: zodErrors(parsed.error) }
  const compiled = parseSponsoredSkill(compileLayer2Skill(parsed.data, version))
  if (!compiled.ok) return { ok: false, errors: compiled.errors }
  if (!sponsoredSkillTrackedLink(compiled.skill)) {
    return {
      ok: false,
      errors: [
        { path: 'signupUrl', message: SPONSORED_SKILL_TRACKED_LINK_ERROR },
      ],
    }
  }
  return { ok: true, source: parsed.data, skill: compiled.skill }
}
