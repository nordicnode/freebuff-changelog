// Users paste MCP server addresses the way they type them into a browser: `mcp.example.com/sse`,
// `example.com/mcp`, `localhost:3000/mcp`. `new URL` rejects the first two and misreads the last
// as a URL with the scheme `localhost:`, so every surface failed with "not a valid URL".
//
// This only supplies a missing scheme. It never removes or changes one the user typed, so each
// surface's own scheme, host and SSRF checks run unchanged on the result: `javascript:`, `file:`
// and `ftp:` URLs come back as given and are rejected there, as before.

```
import { z } from 'zod/v4'

import { normalizeMcpServerUrl } from '../mcp/server-url'

export const CUSTOM_MCP_LIMIT = 20
export const customMcpInputSchema = z
  .object({
    name: z
      .string()
      .trim()
      .min(1)
      .max(80)
      .regex(/^[\p{L}\p{N} _.-]+$/u),
    url: z
      .string()
      // Supply the scheme people leave off (`example.com/mcp`) before every check below.
      .overwrite(normalizeMcpServerUrl)
      .url()
      .max(2048)
      // Checked first so it is the message shown. Wording only: the public-address check that
      // actually keeps Cloud off private networks is the transport's, and runs regardless.
      .refine((value) => {
        if (!URL.canParse(value)) return true
        const host = new URL(value).hostname
        return !(
          host === 'localhost' ||
          host.endsWith('.localhost') ||
          host === '[::1]' ||
          host === '0.0.0.0' ||
          /^127\./.test(host)
        )
      }, 'Cloud can’t reach a server on your own computer. In Freebuff Desktop, add it under This computer instead, or use a public HTTPS URL.')
      .refine((value) => {
        // Zod still runs this after `.url()` fails; a throw here would escape `safeParse`.
        if (!URL.canParse(value)) return false
        const url = new URL(value)
        return (
          url.protocol === 'https:' &&
          !url.username &&
          !url.password &&
          !url.hash &&
          !url.search &&
          (!url.port || url.port === '443')
        )
      }, 'Use a public HTTPS URL without credentials or query parameters'),
    token: z
      .string()
      .trim()
      .max(8192)
      .regex(/^[^\r\n]*$/)
      .optional(),
  })
  .strict()
export const customMcpApprovalSchema = z
  .object({
    manifestHash: z.string().regex(/^sha256:[a-f0-9]{64}$/),
    tools: z.array(z.string().min(1).max(200)).max(200),
  })
  .strict()

export type CustomMcpTool = {
  name: string
  description: string
  inputSchema: Record<string, unknown>
  annotations?: { readOnlyHint?: boolean; destructiveHint?: boolean }
}
export type CustomMcpServer = {
  id: string
  name: string
  url: string
  enabled: boolean
  manifestHash: string
  tools: CustomMcpTool[]
  approvedTools: string[]
}
export type CustomMcpInput = z.infer<typeof customMcpInputSchema>
```

```
// Users paste MCP server addresses the way they type them into a browser: `mcp.example.com/sse`,
// `example.com/mcp`, `localhost:3000/mcp`. `new URL` rejects the first two and misreads the last
// as a URL with the scheme `localhost:`, so every surface failed with "not a valid URL".
//
// This only supplies a missing scheme. It never removes or changes one the user typed, so each
// surface's own scheme, host and SSRF checks run unchanged on the result: `javascript:`, `file:`
// and `ftp:` URLs come back as given and are rejected there, as before.

const LOOPBACK_HOSTS = new Set(['localhost', '127.0.0.1', '[::1]'])

/** `name:` followed by digits and then the end, a path, a query or a fragment is a port, not a scheme. */
const HOST_WITH_PORT = /^[^\s/:?#]+:\d+(?:[/?#]|$)/
const SCHEME = /^[a-z][a-z0-9+.-]*:/i

/**
 * Whether one field that takes either a command line or a server URL holds a URL: one with an
 * http(s) scheme, or one typed without it (`localhost:3845/mcp`, `127.0.0.1:3845`, `[::1]:3845`).
 * A command never starts with `name:<digits>`.
 */
export function isMcpServerAddress(raw: string): boolean {
  const value = raw.trim()
  return (
    /^https?:\/\//i.test(value) ||
    HOST_WITH_PORT.test(value) ||
    /^\[[0-9a-f:.]+\]:\d+(?:[/?#]|$)/i.test(value) ||
    /^localhost(?:[/?#]|$)/i.test(value)
  )
}

/**
 * Trims `raw` and, when it has no scheme, prefixes `https://` — or `http://` for a loopback host,
 * the only place plain HTTP is allowed and where local MCP servers almost always listen.
 * A value that already has a scheme is returned trimmed and otherwise untouched.
 */
export function normalizeMcpServerUrl(raw: string): string {
  const value = raw.trim()
  if (!value) return value
  if (value.startsWith('//')) return `https:${value}`
  if (SCHEME.test(value) && !HOST_WITH_PORT.test(value)) return value

  let hostname: string
  try {
    hostname = new URL(`https://${value}`).hostname
  } catch {
    // Not a URL with any scheme; leave it for the caller's validation to report.
    return value
  }
  return `${LOOPBACK_HOSTS.has(hostname) ? 'http' : 'https'}://${value}`
}
```
