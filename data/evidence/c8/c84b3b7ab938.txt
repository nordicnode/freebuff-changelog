/**
 * Native app client events (COD-1719) — the wire contract of
 * `POST /api/v1/client-events`, shared by the `web` route and the iOS and
 * Android apps (COD-1726, COD-1725).
 *
 * The apps batch these in memory and send them on the same host and bearer they
 * use for `GET /api/v1/ads/policy`, with `X-Freebuff-Platform: ios|android` and
 * `X-Freebuff-App-Version`. The server stamps platform, app version and the
 * authenticated account id itself and forwards each event to PostHog; nothing
 * in a body is trusted as an identity.
 *
 * The NAME is an allowlist: an unknown name rejects the whole batch (400), so a
 * typo in an app shows up in its own logs rather than as a silent new event.
 * PROPS are filtered, never rejected: a key a name does not declare, a value of
 * the wrong type or enum, or an oversize string is dropped, and the event still
 * counts. Props never carry prompts, message text, file paths, URLs, titles or
 * artifact ids — only the bounded fields below.
 *
 * Add a name or a prop here, then in both apps; never change what an existing
 * value means (recorded PostHog history reads it).
 */

type EnterAction = { label: string; run: () => void }

/** What Enter, and the button beside Esc, does for this admission. */
function enterAction(
  admission: ChatAdmission,
  takeover: boolean,
  waitingOnTurn: boolean,
): EnterAction | null {
  const counted = (label: string, run: () => unknown): EnterAction => ({
    label,
    run: () => {
      captureCliFunnel('admission', 'clicked', { model: admission.model })
      void run()
    },
  })
  if (takeover) return counted('Enter: take over', takeOverFreebuffSession)
  if (admission.phase === 'confirm')
    return counted('Enter: confirm and send', () =>
      beginFreebuffChatAdmission(admission),
    )
  // Retrying a refusal for too few hours only repeats it.
  if (admission.phase === 'failed' && admission.chooseModel)
    return { label: 'Enter: choose another model', run: openFreebuffModelPicker }
  if (admission.phase === 'failed')
    return counted('Enter: retry', requestFreebuffChatAdmission)
  // A message queued behind a running turn: nothing is starting yet.
  if (waitingOnTurn)
    return { label: 'Enter: change model', run: openFreebuffModelPicker }
  return null
}

export function FreebuffChatControls() {
  const theme = useTheme()
  const { terminalHeight } = useTerminalDimensions()
  const pickerOpen = useFreebuffChatStore((s) => s.pickerOpen)
  const pickerInitialView = useFreebuffChatStore((s) => s.pickerInitialView)
  const admission = useFreebuffChatStore((s) => s.admission)
  const nextModel = useFreebuffChatStore((s) => s.nextModel)
  const model = useFreebuffModelStore((s) => s.selectedModel)
  const session = useFreebuffSessionStore((s) => s.session)
  const { clearQueue, queuePaused, turnRunning } = useChatRuntime()
  const takeover = session?.status === 'takeover_prompt'
  const waitingOnTurn =
    admission?.phase === 'requested' && !queuePaused && turnRunning
  const enter = admission
    ? enterAction(admission, takeover, waitingOnTurn)
    : null
  useEffect(() => {
    if (pickerOpen) {
      captureCliFunnel('model_picker', 'shown', { model: nextModel ?? model })
      void refreshFreebuffSessionMetadata().catch(() => {})
    }
  }, [pickerOpen])
  useEffect(() => {
    if (!admission) return
    const state =
      admission.phase === 'failed'
        ? 'failed'
        : admission.phase === 'starting'
          ? 'started'
          : 'shown'
    // A status the schema refuses would drop the whole event: send only a valid one.
    const status =
      state === 'failed' && /^[a-z_]{1,40}$/.test(admission.status ?? '')
        ? admission.status
        : undefined
    captureCliFunnel('admission', state, {
      model: admission.model,
      ...(status ? { admission_status: status } : {}),
    })
  }, [admission?.phase, admission?.model])

  const cancel = useCallback(async () => {
    if (pickerOpen) {
      captureCliFunnel('model_picker', 'cancelled')
      useFreebuffChatStore.setState({ pickerOpen: false })
      return
    }
    const current = useFreebuffChatStore.getState().admission
    if (!current) return
    // Abort/release an in-flight claim before returning to the composer. The
    // queue remains held until DELETE confirms, including a lost POST reply.
    if (
      current.phase === 'starting' ||
      (current.phase === 'failed' &&
        current.previousSession !== undefined &&
        useFreebuffSessionStore.getState().pendingAdmission)
    ) {
      try {
        await returnToFreebuffLanding({ preserveQueue: true })
      } catch {
        return
      }
    }
    captureCliFunnel('admission', 'cancelled', { model: current.model })
    const queued = clearQueue()
    const chat = useChatStore.getState()
    const text = [...queued.map((m) => m.content), chat.inputValue]
      .filter(Boolean)
      .join('\n\n')
    useChatStore.setState((s) => {
      s.pendingAttachments = [
        ...queued.flatMap((m) => m.attachments),
        ...s.pendingAttachments,
      ]
    })
    chat.setInputValue({
      text,
      cursorPosition: text.length,
      lastEditDueToNav: false,
    })
    useFreebuffChatStore.setState({ admission: null })
  }, [pickerOpen, clearQueue])

  useKeyboard(
    useCallback(
      (key) => {
        if ((!pickerOpen && key.name === 'escape') || (key.ctrl && key.name === 'c')) {
          key.preventDefault?.()
          key.stopPropagation?.()
          void cancel()
        } else if (!pickerOpen && isPlainEnterKey(key)) {
          key.preventDefault?.()
          key.stopPropagation?.()
          enter?.run()
        }
      },
      [cancel, enter, pickerOpen],
    ),
  )

  if (pickerOpen)
    return (
      <box style={{ flexDirection: 'column', paddingLeft: 1, paddingRight: 1 }}>
        <FreebuffModelSelector
          maxHeight={Math.max(4, Math.floor(terminalHeight * 0.65) - 2)}
          selectedModelOverride={nextModel ?? model}
          initialView={pickerInitialView}
          onSelectModel={(selected) => {
            selectFreebuffChatModel(selected)
            if (!useFreebuffChatStore.getState().pickerOpen)
              captureCliFunnel('model_selection', 'selected', {
                model: selected,
                previous_model: nextModel ?? model,
                selection_kind:
                  selected === (nextModel ?? model) ? 'reselected' : 'changed',
              })
          }}
          onCancel={() => {
            captureCliFunnel('model_picker', 'cancelled')
            useFreebuffChatStore.setState({ pickerOpen: false })
          }}
        />
      </box>
    )
  if (!admission) return null
  return (
    <box
      style={{
        border: true,
        borderColor: theme.border,
        paddingLeft: 1,
        paddingRight: 1,
        flexDirection: 'column',
      }}
    >
      <text style={{ fg: theme.foreground, wrapMode: 'word' }}>
        {takeover
          ? (session.message ??
            'Freebuff is already running elsewhere. Take over that session?')
          : (admission.message ??
            (admission.phase === 'requested' && queuePaused
              ? FREEBUFF_ADMISSION_QUEUE_PAUSED_MESSAGE
              : waitingOnTurn
                ? 'Your message is saved and will send after the current turn.'
                : 'Starting your model session… Your message is saved.'))}
      </text>

import {
  formatHoursWithUnit,
  freebucksShortfallLine,
} from '@codebuff/common/util/freebuff-hours'
import { formatFreebuffHardBlockedPrivacySignals } from '@codebuff/common/util/freebuff-privacy'

import {
  resolveFreebuffModelPickForSession,
  refreshFreebuffSessionMetadata,
  startFreebuffSession,
} from './use-freebuff-session'
import {
  useFreebuffChatStore,
  type ChatAdmission,
} from '../state/freebuff-chat-store'
import { getFreebuffModelDirectory } from '../state/freebuff-catalog-store'
import { useFreebuffModelStore } from '../state/freebuff-model-store'
import {
  useFreebuffSessionStore,
  type FreebuffSessionFailure,
} from '../state/freebuff-session-store'
import { freebucksOf, freebucksRowIntent } from '../utils/freebucks'
import type { FreebuffSessionResponse } from '../types/freebuff-session'
import type { StreamStatus } from './use-message-queue'

export function freebuffAdmissionNotice(
  session: FreebuffSessionResponse | null,
): string | null {
  if (!session) return null
  switch (session.status) {
    case 'banned':
      return 'This account is suspended. To appeal, visit https://freebuff.com/account?tab=standing. If this is a mistake, contact support@freebuff.com.'
    case 'country_blocked':
      return session.countryBlockReason === 'anonymous_network'
        ? `Freebuff detected ${formatFreebuffHardBlockedPrivacySignals(session.ipPrivacySignals)} traffic. Disable VPN, proxy or Tor traffic and try again.`
        : session.countryCode === 'UNKNOWN'
          ? 'Freebuff could not verify your location. Check your VPN or proxy settings and try again.'
          : `Freebuff is unavailable in ${session.countryCode}. Use /byok to use your own API key.`
    case 'rate_limited':
      return session.freebucksShortfall
        ? `${
            freebucksShortfallLine({
              ...session.freebucksShortfall,
              modelName: getFreebuffModelDirectory().get(session.model)
                .displayName,
            }).line
          } Choose another model or visit https://freebuff.com/plans.`
        : 'Session limit reached. Choose another model or visit https://freebuff.com/plans.'
    case 'spend_limited':
      return session.message
    case 'ip_capped':
      return 'Too many Freebuff sessions on this network. Try again after another session finishes.'
    case 'superseded':
      return 'This session was taken over elsewhere. Send again to start a new session.'
    default:
      return null
  }
}

/** Identity-check every continuation: a cancelled send must never restart itself. */
export async function beginFreebuffChatAdmission(admission: ChatAdmission) {
  if (useFreebuffChatStore.getState().admission !== admission) return
  if (admission.phase === 'confirm') {
    const intentFor = (session: FreebuffSessionResponse | null | undefined) =>
      freebucksRowIntent(
        freebucksOf(session),
        admission.model,
        session?.status === 'active' ? session.model : undefined,
      )
    // A price/balance change must update the question before its answer can
    // authorize spending. This re-check is the only guard: the server takes
    // no wallet ceiling, so a stale answer would spend unasked.
    if (
      JSON.stringify(intentFor(admission.previousSession)) !==
      JSON.stringify(intentFor(useFreebuffSessionStore.getState().session))
    ) {
      useFreebuffChatStore.setState({
        admission: { phase: 'requested', model: admission.model },
      })
      return
    }
  }
  const starting: ChatAdmission = {
    ...admission,
    phase: 'starting',
    message: undefined,
    previousSession: useFreebuffSessionStore.getState().session,
  }
  useFreebuffSessionStore.getState().setFailure(null)
  useFreebuffChatStore.setState({ admission: starting })
  try {
    await startFreebuffSession(starting.model, {
      preserveQueue: true,
      persistSelection: false,
    })
  } catch (error) {
    if (useFreebuffChatStore.getState().admission !== starting) return
    useFreebuffChatStore.setState({
      admission: {
        ...starting,
        phase: 'failed',
        message: error instanceof Error ? error.message : String(error),
        status: 'start_error',
      },
    })
  }
}

/** A session request's failure as `admission_status` (`^[a-z_]+$`, no HTTP code). */
function sessionFailureStatus(failure: FreebuffSessionFailure): string {
  if (failure.type === 'timeout') return 'request_timeout'
  if (failure.type === 'http')
    return failure.statusCode >= 500 ? 'server_error' : 'request_refused'
  return 'request_failed'
}

/** A turn is running: its stream is live or its chain is still in progress. */
export function freebuffTurnRunning(queue: {
  chainInProgress: boolean
  streamStatus: StreamStatus
}): boolean {
  return queue.chainInProgress || queue.streamStatus !== 'idle'
}

/**
 * Admission buys an hour for the queue to spend, so it waits until the queue
 * can spend it. A user-paused queue sends nothing: admitting for one bought
 * an hour no turn started, which the server refunds after five minutes.
 */
export function freebuffChatAdmissionMayStart(queue: {
  turnRunning: boolean
  queuedCount: number
  queuePaused: boolean
}): boolean {
  return !queue.turnRunning && queue.queuedCount > 0 && !queue.queuePaused
}

/** Shown while a requested admission waits on a paused queue. */
export const FREEBUFF_ADMISSION_QUEUE_PAUSED_MESSAGE =
  '[freebuff product prompt text omitted]'

/** The queue holds the submitted text and attachments while admission runs. */
export function useFreebuffChatAdmission(enabled: boolean) {
  const checking = useRef<ChatAdmission | null>(null)
  const admission = useFreebuffChatStore((s) => s.admission)
  const pickerOpen = useFreebuffChatStore((s) => s.pickerOpen)
  const session = useFreebuffSessionStore((s) => s.session)
  const failure = useFreebuffSessionStore((s) => s.failure)

  useEffect(() => {
    if (!enabled || !admission) return
    if (admission.phase === 'starting') {
      if (failure && !failure.retry) {
        useFreebuffChatStore.setState({
          admission: {
            ...admission,
            phase: 'failed',
            message: failure.message,
            status: sessionFailureStatus(failure),
          },
        })
      } else if (session === admission.previousSession) {
        return
      } else if (session?.status === 'active') {
        useFreebuffChatStore.setState({ admission: null, nextModel: null })
      } else {
        const message = freebuffAdmissionNotice(session)
        if (message)
          useFreebuffChatStore.setState({
            admission: {
              ...admission,
              phase: 'failed',
              message,
              chooseModel: session?.status === 'rate_limited',
              status: session?.status ?? 'no_session',
            },
          })
      }
      return
    }
    // A pick in progress replaces this request; don't price the old model.
    if (admission.phase !== 'requested' || pickerOpen) return
    if (!session) {
      if (failure && !failure.retry)
        useFreebuffChatStore.setState({
          admission: {
            ...admission,
            phase: 'failed',
            message: failure.message,
            status: sessionFailureStatus(failure),
          },
        })
      return
    }
    if (!admission.metadataChecked) {
      if (checking.current === admission) return
      checking.current = admission
      void refreshFreebuffSessionMetadata()
        .then(() => {
          if (useFreebuffChatStore.getState().admission === admission)
            useFreebuffChatStore.setState({
              admission: { ...admission, metadataChecked: true },
            })
        })
        .catch((error) => {
          if (useFreebuffChatStore.getState().admission === admission)
            useFreebuffChatStore.setState({
              admission: {
                ...admission,
                phase: 'failed',
                message: error instanceof Error ? error.message : String(error),
                status: 'metadata_error',
              },
            })
        })
      return
    }
    if (session.status === 'banned' || session.status === 'country_blocked') {
      useFreebuffChatStore.setState({
        admission: {
          ...admission,
          phase: 'failed',
          message: freebuffAdmissionNotice(session)!,
          status: session.status,
        },
      })
      return
    }
    // A send before the first session response captured the launch seed.
    const { hasExplicitPick, selectedModel } = useFreebuffModelStore.getState()
    const model = resolveFreebuffModelPickForSession(
      hasExplicitPick ? admission.model : selectedModel,
      session,
    )
    const freebucks = freebucksOf(session)
    const intent = freebucksRowIntent(
      freebucks,
      model,
      session.status === 'active' ? session.model : undefined,
    )
    const resolved = { ...admission, model }
    if (intent.kind === 'paywall') {
      useFreebuffChatStore.setState({
        admission: {
          ...resolved,
          phase: 'failed',
          status: 'paywall',
          message: `Not enough hours for ${getFreebuffModelDirectory().get(model).displayName}: an hour on it takes ${formatHoursWithUnit(intent.price)}. Choose another model or visit https://freebuff.com/plans.`,
          chooseModel: true,
        },
      })
    } else if (
      intent.kind === 'confirm' ||
      // Full access is never asked, not even to end the live session.
      (!freebucks?.spendWithoutAsking &&
        session.status === 'active' &&
        session.model !== model)
    ) {
      const cost =
        intent.price === undefined
          ? freebucksOf(session) === null
            ? 'may spend wallet hours'
            : 'starts a new session'
          : `takes ${formatHoursWithUnit(intent.price)} from your balance`
      const wallet =
        intent.kind === 'confirm' && intent.walletSpend
          ? ` Uses ${formatHoursWithUnit(intent.walletSpend)} from your wallet.`
          : ''
      useFreebuffChatStore.setState({
        admission: {
          ...resolved,
          phase: 'confirm',
          previousSession: session,
          message: `${getFreebuffModelDirectory().get(model).displayName} ${cost}.${wallet}${session.status === 'active' ? ' This ends your current model session; your conversation is kept.' : ''}${intent.kind === 'confirm' && 'claimEarned' in intent && intent.claimEarned ? ' Earned hours will be claimed on admission.' : ''}`,
        },
      })
    } else {
      // Keep the exact object used by the asynchronous admission continuation.
      useFreebuffChatStore.setState({ admission: resolved })
      void beginFreebuffChatAdmission(resolved)
    }
  }, [enabled, admission, pickerOpen, session, failure])
}

export const useFreebuffChatStore = create<{

  pickerInitialView: 'model' | 'reasoning'
  nextModel: string | null
  admission: ChatAdmission | null

  pickerOpen: false,
  pickerInitialView: 'model',

  admission: null,

/** Only an admission POST in flight holds the model. One waiting on a busy
 *  queue, a confirmation or a refusal can still change it. */
function admissionInFlight() {
  return useFreebuffChatStore.getState().admission?.phase === 'starting'
}

function openFreebuffPicker(initialView: 'model' | 'reasoning') {
  if (admissionInFlight()) return
  useFreebuffChatStore.setState({
    pickerOpen: true,
    pickerInitialView: initialView,
  })
}

export function openFreebuffModelPicker() {
  openFreebuffPicker('model')
}

export function openFreebuffReasoningPicker() {
  openFreebuffPicker('reasoning')
}

export function selectFreebuffChatModel(model: string) {
  if (admissionInFlight()) return
  persistFreebuffModelPick(model)
  const waiting = useFreebuffChatStore.getState().admission !== null
  useFreebuffChatStore.setState({
    nextModel: model,
    pickerOpen: false,
    // The held message goes out on the new pick, asked about afresh.
    ...(waiting ? { admission: { phase: 'requested', model } } : {}),
  })
}

export function freebuffChatModel() {
  return useFreebuffChatStore.getState().nextModel ?? getSelectedFreebuffModel()
}

export function freebuffChatNeedsAdmission() {
  const { session } = useFreebuffSessionStore.getState()
  return (
    session?.status !== 'active' ||
    (session?.status === 'active' && session.model !== freebuffChatModel())
  )
}

export function requestFreebuffChatAdmission() {
  const { admission } = useFreebuffChatStore.getState()
  if (admission && admission.phase !== 'failed') return
  useFreebuffChatStore.setState({
    admission: { phase: 'requested', model: freebuffChatModel() },
  })
}

/** Strip SDK URL/person enrichment before either capture or the log mirror. */
export function sanitizeProductFunnelCapture<
  T extends {
    event: string
    properties: Record<string, unknown>
  },
>(event: T, projectToken: string): T | null {
  if (event.event !== PRODUCT_FUNNEL_EVENT) return event
  const fields: Record<string, unknown> = {}
  for (const key of productFunnelEventSchema.keyof().options) {
    if (event.properties[key] !== undefined) fields[key] = event.properties[key]
  }
  const parsed = productFunnelEventSchema.safeParse(fields)
  if (!parsed.success) return null
  // posthog-js adds this public ingest key before before_send; transport does
  // not restore it after sanitization. Never copy an untrusted event token.
  const properties: Record<string, unknown> = {
    ...parsed.data,
    token: projectToken,
  }
  for (const key of [
    'distinct_id',
    '$insert_id',
    '$session_id',
    '$is_identified',
  ]) {
    const value = event.properties[key]
    if (typeof value === 'string' || typeof value === 'boolean')
      properties[key] = value
  }
  const sanitized = { ...event, properties }
  Reflect.deleteProperty(sanitized, '$set')
  Reflect.deleteProperty(sanitized, '$set_once')
  return sanitized
}

/** One NDJSON line per event; the stream ends after `done` or `error`. */
export function sitePublishResponse(
  run: (
    onEvent: (event: SitePublishEvent) => void,
    signal: AbortSignal,
  ) => Promise<unknown>,
  signal: AbortSignal,
): Response {
  const encoder = new TextEncoder()
  const cancelled = new AbortController()
  const stop = AbortSignal.any([signal, cancelled.signal])
  const stream = new ReadableStream<Uint8Array>({
    start(sink) {
      let open = true
      const write = (event: SitePublishEvent) => {
        if (!open) return
        try {
          sink.enqueue(encoder.encode(JSON.stringify(event) + '\n'))
        } catch {
          open = false
        }
      }
      const beat = setInterval(
        () => write({ type: 'heartbeat' }),
        SITE_PUBLISH_HEARTBEAT_MS,
      )
      void run(write, stop)
        .catch(() =>
          write({
            type: 'error',
            code: 'unknown',
            message: 'Publishing failed. Try again.',
          }),
        )
        .finally(() => {
          clearInterval(beat)
          if (!open) return
          open = false
          try {
            sink.close()
          } catch {
            /* Already cancelled. */
          }
        })
    },
    cancel() {
      cancelled.abort()
    },
  })
  return new Response(stream, {
    headers: {
      'Content-Type': 'application/x-ndjson',
      'Cache-Control': 'private, no-store',
      'X-Accel-Buffering': 'no',
    },
  })
}

function parseEvent(line: string): SitePublishEvent | null {
  try {
    const value = JSON.parse(line) as SitePublishEvent
    if (value.type === 'stage' && value.stage in SITE_PUBLISH_STAGE_LABELS)
      return value
    if (
      value.type === 'done' &&
      typeof value.siteId === 'string' &&
      typeof value.url === 'string' &&
      ['preview', 'production'].includes(value.environment)
    )
      return value
    if (value.type === 'error' && typeof value.message === 'string') {
      const replace = value.replace
      return {
        type: 'error',
        message: value.message.slice(0, 4096),
        ...(typeof value.code === 'string' && FAILURE_CODES.has(value.code)
          ? { code: value.code }
          : {}),
        ...(replace &&
        typeof replace.chatId === 'string' &&
        typeof replace.chatTitle === 'string' &&
        typeof replace.url === 'string'
          ? {
              replace: {
                chatId: replace.chatId.slice(0, 160),
                chatTitle: replace.chatTitle.slice(0, 200),
                url: replace.url.slice(0, 2048),
              },
            }
          : {}),
      }
    }
    if (value.type === 'heartbeat') return value
  } catch {
    /* A partial or foreign line is not an event. */
  }
  return null
}

/** Reads a publish stream to its result; a refused request is an error result. */
export async function readSitePublishStream(
  response: Response,
  onEvent: (event: SitePublishEvent) => void = () => {},
): Promise<SitePublishResult> {
  if (
    !response.ok ||
    !response.body ||
    !response.headers.get('content-type')?.includes('application/x-ndjson')
  ) {
    const body = (await response.json().catch(() => null)) as {
      error?: unknown
      message?: unknown
    } | null
    const message =
      typeof body?.message === 'string'
        ? body.message
        : typeof body?.error === 'string'
          ? body.error
          : 'Could not start publishing.'
    return { type: 'error', code: 'refused', message }
  }
  const reader = response.body.pipeThrough(new TextDecoderStream()).getReader()
  let buffer = ''
  let result: SitePublishResult | null = null
  try {
    while (!result) {
      const { done, value } = await reader.read()
      if (done) break
      buffer += value
      const lines = buffer.split('\n')
      buffer = lines.pop() ?? ''
      for (const line of lines) {
        const event = line.trim() ? parseEvent(line) : null
        if (!event) continue
        onEvent(event)
        if (event.type === 'done' || event.type === 'error') {
          result = event
          break
        }
      }
    }
  } catch {
    return STOPPED
  } finally {
    reader.cancel().catch(() => {})
  }
  return result ?? STOPPED
}

/** Receipts keep only codes; say what each means without vendor prose. */
export function operationFailureMessage(operation: Operation): string {
  const result = operation.result
  if (operation.state === 'unknown')
    return 'Cloudflare did not confirm the deployment. Publish again in 10 minutes: Freebuff first checks with Cloudflare whether this one landed.'
  // Settled later from Cloudflare's state (web reconcile.ts) without landing.
  if (result?.error === 'cloudflare_outcome_unknown')
    return 'Cloudflare did not confirm the deployment. Publish again.'
  // The server stopped before recording a result, then settled it the same way.
  if (result?.error === 'request_interrupted')
    return 'Freebuff was interrupted before it recorded the deployment. Publish again.'
  // No upload starts 15 minutes into a deployment (web reconcile.ts).
  if (result?.error === 'operation_deadline')
    return 'The deployment ran out of time and was stopped before it went live. Publish again.'
  const provider = result?.provider
  if (provider?.stage === 'worker_upload' && provider.codes?.includes(10021))
    return 'Cloudflare could not start the built Worker: it failed startup validation. This usually means code that cannot run on Workers, such as a Node.js-only module.'
  if (provider?.status === 429)
    return 'Cloudflare is rate limiting deployments. Try again in a minute.'
  if (result?.error === 'opennext_manifest_required')
    return 'This environment hosts a Next.js app. Publish the Next.js project to it.'
  return 'Cloudflare rejected the deployment.'
}

export async function publishSite(options: {
  environment: 'preview' | 'production'
  /** The site to update; absent creates one for this project. */
  siteId?: string
  siteName?: string
  /** Runs one Sites tool with the caller's credentials and workspace. */
  tool: (name: string, args: Record<string, unknown>) => Promise<unknown>
  prepare: () => Promise<SitePreparationResult>
  onEvent: (event: SitePublishEvent) => void
  id?: () => string
  /** Waits between polls of a deployment still running. */
  sleep?: (ms: number) => Promise<void>
  /** A caller's own user-facing errors (Cloud workspace states); others stay generic. */
  errorMessage?: (error: unknown) => string | undefined
}): Promise<SitePublishResult> {
  const id = options.id ?? randomUUID
  const { environment } = options
  const sleep =
    options.sleep ?? ((ms: number) => new Promise((r) => setTimeout(r, ms)))
  const fail = (message: string, code: SitePublishFailureCode) => {
    const event = { type: 'error' as const, message, code }
    options.onEvent(event)
    return event
  }
  const call = async <T>(name: string, args: Record<string, unknown>) => {
    const value = await options.tool(name, args)
    const refused = refusal(value)
    const owner = refused?.deployedFromChat
    if (
      refused?.error === 'site_deployed_from_another_chat' &&
      typeof owner?.id === 'string' &&
      typeof refused.url === 'string'
    )
      throw new OtherChatDeployError({
        chatId: owner.id,
        chatTitle: owner.title ?? '',
        url: refused.url,
      })
    if (refused)
      throw new SitesClientError(
        400,
        refused.error,
        refused.message ?? 'The Sites request failed.',
      )
    return value as T
  }
  try {
    options.onEvent({ type: 'stage', stage: 'building' })
    const prepared = await options.prepare()
    let manifestPath = prepared.manifestPath
    if (prepared.kind === 'next')
      manifestPath = (
        await call<{ manifestPath: string }>('sites_prepare_next', {})
      ).manifestPath
    if (!manifestPath) return fail('The build did not produce a deployment.', 'build_failed')

    options.onEvent({ type: 'stage', stage: 'deploying' })
    let siteId = options.siteId
    if (!siteId) {
      siteId = id()
      const name =
        (options.siteName ?? prepared.name ?? '').trim().slice(0, 80) ||
        'My site'
      await call('sites_create', { siteId, name })
    }
    const target = { siteId, environment }
    const inspect = () =>
      call<{ environment: SiteEnvironment }>('sites_inspect', target)
    const inspected = await inspect()
    const details = inspected.environment
    if (prepared.usesAuth && !details.auth)
      await call('sites_enable_auth', target)
    const existing = new Set((details.bindings ?? []).map((b) => b.name))
    for (const kind of ['d1', 'r2'] as const)
      for (const name of prepared.storage[kind]) {
        if (existing.has(name)) continue
        const added = await call<{ operation?: Operation }>(
          'sites_add_storage',
          { ...target, requestId: id(), kind, name },
        )
        if (added.operation && added.operation.state !== 'succeeded')
          return fail(
            `Could not create the ${kind === 'd1' ? 'database' : 'storage bucket'} ${name}. Check deployment activity in Manage site.`,
            'deploy_rejected',
          )
      }

    const requestId = id()
    let operation = (
      await call<{ operation: Operation }>('sites_deploy', {
        ...target,
        requestId,
        manifestPath,
      })
    ).operation
    // Deployments normally settle in the request; a slow one is polled by key.
    for (let i = 0; operation.state === 'running' && i < 30; i++) {
      await sleep(2000)
      const activity = (
        await call<{
          activity?: { operations?: Operation[] }
        }>('sites_inspect', target)
      ).activity
      operation =
        activity?.operations?.find((op) => op.id === operation.id) ?? operation
    }
    if (operation.state === 'running')
      return fail(
        'The deployment is still running. Check deployment activity in Manage site.',
        'timeout',
      )
    if (operation.state !== 'succeeded')
      return fail(
        operationFailureMessage(operation),
        operationFailureCode(operation),
      )

    // A successful upload is not publication. Republish only when this request
    // started with an unpublished environment; preserve a newer Unpublish.
    let current = (await inspect()).environment
    if (!isPublished(current) && details.unpublished_at) {
      options.onEvent({ type: 'stage', stage: 'publishing' })
      const publication = await call<{ operation: Operation }>(
        'sites_set_published',
        {
          ...target,
          published: true,
          requestId: id(),
        },
      )
      if (publication.operation.state !== 'succeeded')
        return fail(
          'Publication was not confirmed. Check deployment activity in Manage site.',
          'deploy_rejected',
        )
      current = (await inspect()).environment
    }
    if (!isPublished(current))
      return fail(
        'The deployment succeeded, but this environment is unpublished. Check Publication in Manage site.',
        'deploy_rejected',
      )
    const event = {
      type: 'done' as const,
      siteId,
      environment,
      url: operation.result?.url ?? details.url,
    }
    options.onEvent(event)
    return event
  } catch (error) {
    if (error instanceof OtherChatDeployError) {
      const event = {
        type: 'error' as const,
        message: error.message,
        code: 'other_chat_deploy' as const,
        replace: error.replace,
      }
      options.onEvent(event)
      return event
    }
    if (error instanceof SitesClientError)
      return fail(
        error.message,
        error.code === 'site_prepare_failed'
          ? preparationFailureCode(error.message)
          : 'deploy_rejected',
      )
    const own = options.errorMessage?.(error)
    return own !== undefined
      ? fail(own, 'workspace_unavailable')
      : fail(
          'Publishing failed. Try again, or ask the agent to publish.',
          'unknown',
        )
  }
}
