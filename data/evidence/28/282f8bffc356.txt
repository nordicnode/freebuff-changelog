import { FREEBUFF_STREAK_REWARD_INTERVAL_DAYS } from './freebuff-models'

/**
 * Every appearance preset Freebuff Desktop ships, by id. Desktop reports which
 * one an install is running (never a custom theme's name) and the web API
 * counts them, so both sides validate against this one list.
 */
export const DESKTOP_APPEARANCE_PRESET_IDS = [
  'system', 'dark', 'light', 'shine',
  'alpine', 'dune', 'tide', 'blossom', 'orbit', 'after-rain', 'ember', 'kanagawa',
  'aurora', 'violet-hour', 'phosphor', 'mirage', 'nebula', 'frost', 'peach', 'midnight',
  'lilac', 'dusk', 'seafoam', 'graphite', 'daybreak', 'sherbet', 'paper', 'carbon',
  'streak-fire', 'gold-streak-fire', 'white-streak-fire', 'blue-streak-fire',
  'plasma-streak-fire', 'rainbow-streak-fire',
] as const

/**
 * The theme a 7-day streak unlocks, for good: the account reached the streak
 * once (a `freebuff_streak_reward` row exists). Neither a plan nor referrals
 * unlock it; it is earned by using Freebuff seven days in a row.
 */
export const STREAK_THEME_PRESET_ID = 'streak-fire'
/** The first streak display experiment (docs/experiments.md, "Streaks"),
 *  replaced by `STREAK_CLASSIC_EXPERIMENT_KEY`: its `streak` arm saw the
 *  activity grid and keeps the Streak Fire offer. */
export const STREAK_EXPERIMENT_KEY = 'streak_display_2026_11'
/** The streak experiment (docs/experiments.md, "Classic streak"): its
 *  `classic` arm is shown the streak (the classic seven-day UI on web and
 *  Desktop) and offered Streak Fire; `control` sees no streak. */
export const STREAK_CLASSIC_EXPERIMENT_KEY = 'streak_classic_2026_11'
/** The streak that unlocks it: the reward interval, whose grant is the unlock. */
export const STREAK_THEME_REQUIRED_DAYS = FREEBUFF_STREAK_REWARD_INTERVAL_DAYS

/**
 * The streak themes, in the order they unlock: a flame burning hotter (orange,
 * gold, white, blue), then past fire (plasma, a prism). Streak Fire is the
 * 7-day one above; each later one unlocks when the account's streak reaches
 * its `days` after they launched (2026-11-12), counting the streak already
 * running then, so 100 days in a row unlock them all at once and an older
 * streak unlocks nothing. All are kept for good (`freebuff_streak_theme_unlock`).
 */
export const STREAK_THEME_TIERS = [
  { id: STREAK_THEME_PRESET_ID, name: 'Streak Fire', days: STREAK_THEME_REQUIRED_DAYS },
  { id: 'gold-streak-fire', name: 'Gold Streak Fire', days: 14 },
  { id: 'white-streak-fire', name: 'White Streak Fire', days: 21 },
  { id: 'blue-streak-fire', name: 'Blue Streak Fire', days: 28 },
  { id: 'plasma-streak-fire', name: 'Plasma Streak Fire', days: 50 },
  { id: 'rainbow-streak-fire', name: 'Rainbow Streak Fire', days: 100 },
] as const
export type StreakThemeTier = (typeof STREAK_THEME_TIERS)[number]

/** The highest streak theme a streak of `days` has reached, or undefined
 *  below 7: the streak's own colours (`workspace-streak-classic.css`). */
export function streakThemeTierForDays(days: number): StreakThemeTier | undefined {
  return [...STREAK_THEME_TIERS].reverse().find((tier) => days >= tier.days)
}

/** The streak theme `id` names, or undefined for any other preset. */
export function streakThemeTier(id: string): StreakThemeTier | undefined {
  return STREAK_THEME_TIERS.find((tier) => tier.id === id)
}

/** What Desktop reports: a shipped preset, or `custom` for any saved custom theme. */
export const DESKTOP_APPEARANCE_REPORTABLE = [...DESKTOP_APPEARANCE_PRESET_IDS, 'custom'] as const
export type DesktopAppearanceLabel = (typeof DESKTOP_APPEARANCE_REPORTABLE)[number]

/** Every install starts on one of these, so a usage count on them says nothing. */
export const DESKTOP_DEFAULT_APPEARANCE_IDS: readonly string[] = ['system', 'dark', 'light']

/** An install counts toward its appearance while it has reported within this window. */
export const DESKTOP_APPEARANCE_ACTIVE_DAYS = 30

export function isDesktopAppearanceLabel(value: unknown): value is DesktopAppearanceLabel {
  return typeof value === 'string' && (DESKTOP_APPEARANCE_REPORTABLE as readonly string[]).includes(value)
}

export interface FreebuffStreakResponse {
  streak: number
  todayUsed: boolean
  /** Latest Pacific usage date (`YYYY-MM-DD`), for history copy. */
  lastUsageDate: string | null
  /**
   * The account's daily reset timezone, whose days the streak counts (since
   * 2026-10-01; Pacific before, and still Pacific for an account whose
   * devices have not reported one).
   */
  timeZone: string
  /**
   * Freebucks a day of a 7+ day streak adds to this account's daily allowance
   * (never the wallet; it expires at the allowance's reset), or null when the
   * account is not on the Freebucks meter and the streak still pays sessions.
   * Absent from older servers, which clients read as null. Server-set so the
   * copy a client draws matches what the meter will do.
   */
  freebucksDailyBonus?: number | null
  /**
   * When the current streak day ends and the next one (and its +15) begins,
   * as an ISO instant: the daily allowance's reset, normally the next
   * midnight in `timeZone`. Since 2026-10-01 the streak day IS the allowance
   * day, so this is the reader's own midnight at home; before, it was the
   * next Pacific midnight (3 PM in Singapore). Clients render it in the
   * reader's zone. Absent from older servers.
   */
  nextResetAt?: string
  /**
   * Whether today's streak bonus has been added to the daily allowance. It is
   * granted with the first free-mode message of a streak day, not at sign-in.
   * Null when the account is off the meter or the lookup failed; absent from
   * older servers (which meant "in the wallet").
   */
  todayCredited?: boolean | null
  /**
   * When today's granted bonus leaves the allowance: the allowance's own reset
   * (midnight in the account's reset timezone), as an ISO instant — since
   * 2026-10-01 always equal to `nextResetAt`. Null when that reset has
   * already passed and the bonus is gone (older servers, whose streak day was
   * Pacific). Absent when nothing was granted, and from older servers.
   */
  bonusExpiresAt?: string | null
  /**
   * The days the account was active in the last
   * `FREEBUFF_STREAK_ACTIVITY_WINDOW_DAYS`, as days before today (0 is today),
   * ascending: the activity grid a client draws beside the count. It reaches
   * back past the current run, so a streak that broke still shows where it
   * was. Absent from older servers; a client then draws the current run alone.
   */
  activeDaysAgo?: number[]
  /**
   * Whether the account has ever reached a 7-day streak, which unlocks the
   * Streak Fire theme for good (`STREAK_THEME_PRESET_ID`). Absent from older
   * servers, which clients read as false.
   */
  streakThemeUnlocked?: boolean
}

/** Referred friends it takes to unlock themes, at limited access or when the tier is unknown. */
export const THEME_REFERRAL_REQUIREMENT = 3 as const
/**
 * The same at full access: one friend. Full-access accounts get a lower bar
 * on every referral reward (docs/freebuff-referral-rewards.md).
 */
export const THEME_REFERRAL_REQUIREMENT_FULL_ACCESS = 1 as const

/** The unlock threshold for an account's access tier; unknown reads as limited. */
export function themeReferralRequirement(
  accessTier: 'full' | 'limited' | null | undefined,
): number {
  return accessTier === 'full'
    ? THEME_REFERRAL_REQUIREMENT_FULL_ACCESS
    : THEME_REFERRAL_REQUIREMENT
}

/** Account entitlement, calculated by the server from a subscription or referred signups. */
export interface ThemeReferralAccess {
  /** 1 at full access, 3 otherwise (`themeReferralRequirement`). */
  requiredReferrals: number
  /** Progress is capped at the unlock threshold; no referred identities leave the server. */
  qualifyingReferrals: number
  unlocked: boolean
  /** The account once reached a 7-day streak, which unlocks the Streak Fire
   *  theme (`STREAK_THEME_PRESET_ID`) on its own: `unlocked` does not include
   *  it, and it does not unlock the rest. Absent from servers before
   *  2026-11-11, read as false. */
  streakThemeUnlocked?: boolean
  /** The longest streak, in days, the account has reached since the later
   *  streak themes launched (or its current one), which unlocks them for
   *  good (`STREAK_THEME_TIERS`). 0 for an account not offered them or
   *  without Streak Fire. Absent from older servers, read as 0. */
  streakThemeDays?: number
  /** The account is shown the streak (`streak_classic_2026_11`'s `classic`
   *  arm, or `streak_display_2026_11`'s `streak` arm before it), so clients
   *  show its Streak reward section: the streak themes earned and the next one
   *  to earn. Everyone else never sees it. Absent from older servers, read as
   *  false. */
  streakThemeOffered?: boolean
  referralCode: string | null
  /** When the account was created (ISO), for the browser workspace's first-visit
   *  welcome. Absent from servers before 2026-11-07. */
  accountCreatedAt?: string | null
}

export interface ThemeReferralPromptResponse {
  show: boolean
  access: ThemeReferralAccess
  /**
   * The account's timing-experiment arm, or null when it is not in the
   * experiment (or the server predates it). A deferred arm answers
   * `show: false` at the first send and is claimed later with its stage.
   */
  arm?: string | null
}

/** What the first-prompt invitation reports: `shown` once per mount, then each copy or dismissal. */
export type ThemeReferralPromptAction =
  | 'shown'
  | 'copied'
  | 'copy_failed'
  | 'dismissed'

/**
 * What every invite placement reports on `freebuff.theme_referral_prompt`:
 * the card's actions, plus `opened` for a placement that opens on a click and
 * the email invite's answer.
 */
export type ReferralInviteAction =
  | ThemeReferralPromptAction
  | 'opened'
  | 'invite_sent'
  | 'invite_failed'
