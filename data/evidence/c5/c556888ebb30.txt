/** A branch name, not a revision expression or shell fragment. */
export function validGitHubBranch(value: string): boolean {
  return (
    value.length > 0 &&
    value.length <= 200 &&
    !/^[.-]|[\s~^:?*\[\\\x00-\x1f\x7f]|\.\.|@\{|\/\/|[/.]$/.test(value) &&
    value !== '@' &&
    !value.startsWith('/') &&
    value
      .split('/')
      .every((part) => !part.startsWith('.') && !part.endsWith('.lock'))
  )
}
