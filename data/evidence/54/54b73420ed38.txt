export function createCloudHistoryClient(
  options: {
    baseUrl?: string
    getToken?: () => string | null | Promise<string | null>
    fetch?: (url: string, init?: RequestInit) => Promise<Response>
  } = {},
) {
  const base = (options.baseUrl ?? '/api/v1/cloud/history').replace(/\/$/, '')
  async function get<T>(
    path: string,
    page: CloudHistoryOptions = {},
    projectId?: string,
    update?: CloudHistoryChatUpdate | CloudHistoryProjectUpdate | CloudHistoryChatFork | CloudHistoryChatRollback | CloudHistoryAttentionAck,
    search?: string,
    method: 'GET' | 'PATCH' | 'POST' = update ? 'PATCH' : 'GET',
    archived?: boolean,
  ): Promise<T> {
    const query = new URLSearchParams()
    if (search) query.set('search', search)
    if (archived) query.set('archived', 'true')
    if (page.cursor != null) query.set('cursor', page.cursor)
    if (page.limit !== undefined) query.set('limit', String(page.limit))
    if (projectId !== undefined) query.set('projectId', projectId)
    const token = await options.getToken?.()
    const res = await (options.fetch ?? fetch)(
      `${base}/${path}${query.size ? `?${query}` : ''}`,
      {
        credentials: 'same-origin',
        cache: 'no-store',
        signal: page.signal,
        method,
        body: update ? JSON.stringify(update) : undefined,
        headers: update
          ? {
              'Content-Type': 'application/json',
              ...(token ? bearer(token) : {}),
            }
          : token
            ? bearer(token)
            : undefined,
      },
    )
    if (!res.ok) {
      const body = await res.json().catch(() => null)
      throw new CloudHistoryError(
        res.status,
        body?.error ?? 'Cloud history unavailable',
      )
    }
    return res.json() as Promise<T>
  }
  const id = encodeURIComponent
  return {
    listProjects: (page: CloudHistoryOptions & CloudHistoryArchivedFilter = {}) =>
      get<CloudHistoryPage<CloudHistoryProject>>('projects', page, undefined, undefined, undefined, 'GET', page.archived),
    getProject: (projectId: string, signal?: AbortSignal) =>
      get<CloudHistoryProject>(`projects/${id(projectId)}`, { signal }),
    updateProject: (
      projectId: string,
      update: CloudHistoryProjectUpdate,
      signal?: AbortSignal,
    ) =>
      get<CloudHistoryProject>(
        `projects/${id(projectId)}`,
        { signal },
        undefined,
        update,
      ),
    /** `archived`: the chats Settings > Archived lists, archived alone or with their project. */
    listChats: (page: CloudHistoryOptions & CloudHistoryArchivedFilter & { projectId?: string; search?: string } = {}) =>
      get<CloudHistoryPage<CloudHistoryChat>>('chats', page, page.projectId, undefined, page.search, 'GET', page.archived),
    getChat: (chatId: string, signal?: AbortSignal) =>
      get<CloudHistoryChat>(`chats/${id(chatId)}`, { signal }),
    forkChat: (chatId: string, fork: CloudHistoryChatFork, signal?: AbortSignal) =>
      get<CloudHistoryChat>(`chats/${id(chatId)}/fork`, { signal }, undefined, fork, undefined, 'POST'),
    openSideChat: (chatId: string, side: CloudHistorySideChat, signal?: AbortSignal) =>
      get<CloudHistoryChat>(`chats/${id(chatId)}/side`, { signal }, undefined, side, undefined, 'POST'),
    rollbackChat: (chatId: string, rollback: CloudHistoryChatRollback, signal?: AbortSignal) =>
      get<CloudHistoryChatRolledBack>(`chats/${id(chatId)}/rollback`, { signal }, undefined, rollback, undefined, 'POST'),
    getAttachment: (chatId: string, signal?: AbortSignal) =>
      get<CloudHistoryAttachment>(`chats/${id(chatId)}/attachment`, { signal }),
    updateChat: (
      chatId: string,
      update: CloudHistoryChatUpdate,
      signal?: AbortSignal,
    ) =>
      get<CloudHistoryChat>(
        `chats/${id(chatId)}`,
        { signal },
        undefined,
        update,
      ),
    acknowledgeAttention: (
      chatId: string,
      throughRevision: number,
      signal?: AbortSignal,
    ) =>
      get<CloudHistoryChat>(

/** One top-level folder a project checkpoint would hold, with its size before compression. */
export type CheckpointFolder = { name: string; bytes: number }

/**
 * A Cloud project's latest checkpoint that failed for being over the 4 GiB
 * limit: when, and its largest folders (none when the size scan could not
 * finish). Kept on the project's sandbox until a checkpoint succeeds
 * (`freebuff_cloud_sandbox.checkpoint_too_large`).
 */
export type CheckpointTooLarge = { at: string; largest: CheckpointFolder[] }

const size = (bytes: number) =>
  bytes >= 1024 ** 3
    ? `${(bytes / 1024 ** 3).toFixed(1)} GB`
    : `${Math.max(1, Math.round(bytes / 1024 ** 2))} MB`

/** What a user is told when a project's checkpoint is over the limit even without what can be rebuilt. */
export function checkpointTooLargeMessage(largest: readonly CheckpointFolder[] = []) {
  const folders = largest.length
    ? ` Its largest folders, before compression: ${largest.map((f) => `“${f.name}” (${size(f.bytes)})`).join(', ')}.`
    : ''
  return `Cloud can’t save this project: even without dependencies, build output, Python virtual environments and caches, its files are over the 4 GiB limit for a saved copy.${folders} Delete large generated or data files you no longer need, or move them out of this project (for example to cloud storage or a separate GitHub repository) and download them only when you need them. Until the project fits, Cloud can’t archive it or back up its newest changes.`
}
