/** Strip SDK URL/person enrichment before either capture or the log mirror. */
export function sanitizeProductFunnelCapture<
  T extends {
    event: string
    properties: Record<string, unknown>
  },
>(event: T, projectToken: string): T | null {
  if (event.event !== PRODUCT_FUNNEL_EVENT) return event
  const fields: Record<string, unknown> = {}
  for (const key of productFunnelEventSchema.keyof().options) {
    if (event.properties[key] !== undefined) fields[key] = event.properties[key]
  }
  const parsed = productFunnelEventSchema.safeParse(fields)
  if (!parsed.success) return null
  // posthog-js adds this public ingest key before before_send; transport does
  // not restore it after sanitization. Never copy an untrusted event token.
  const properties: Record<string, unknown> = {
    ...parsed.data,
    token: projectToken,
  }
  for (const key of [
    'distinct_id',
    '$insert_id',
    '$session_id',
    '$is_identified',
  ]) {
    const value = event.properties[key]
    if (typeof value === 'string' || typeof value === 'boolean')
      properties[key] = value
  }
  const sanitized = { ...event, properties }
  Reflect.deleteProperty(sanitized, '$set')
  Reflect.deleteProperty(sanitized, '$set_once')
  return sanitized
}

/**
 * Which store owns the Desktop to phone mirror (the `mirror_*` tables) — the
 * checked-in switch for moving the mirror off platform Convex (COD-742 batch
 * 5, track D1; docs/freebuff-mirror-postgres.md).
 *
 * LIGHT tier, so there is no shadow mode: the Desktop's local SQLite is the
 * source of truth and re-pushes every open thread on connect, so the
 * Postgres copy rebuilds itself after a flip and a flip back loses nothing.
 *
 * - `convex`   Desktop is told to use Convex, every
 *              `/api/fn/*` mirror entry answers `unknown_function`, the
 *              `mirror.*` jobs no-op and the mobile config keeps phones on
 *              Convex.
 * - `postgres` (since 2026-10-17): freebuff-web advertises `mirror_backend: 'postgres'` to
 *              Desktop and `mirrorBackend: 'postgres'` to the phones, the
 *              registry entries serve, and the APNs push and retention sweep
 *              run as graphile jobs.
 *
 * It lives in `common` because freebuff-web (advertisement, registry) and
 * the `freebuff-jobs` worker (push, sweep) both read it. A flip is a reviewed
 * one-line diff here, never an env var.
 */
export type MirrorBackend = 'convex' | 'postgres'
