import { FREEBUFF_BASE3_FAST_WORKER_AGENT_ID } from '@codebuff/common/constants/free-agents'

import { createBase3, createBase3CliRoot } from './base3'
import { publisher } from './constants'

import type { CompactionPolicy } from '@codebuff/common/constants/compaction-policy'
import type { SecretAgentDefinition } from './types/secret-agent-definition'

/**
 * base3-fast: base3's loop with a fan-out on top.
 *
 * The root is a base3 CLI root plus `spawn_agents` and two things to spawn:
 * `code-searcher` for context, in parallel, and the worker below for the one
 * shape of work worth splitting — a large mechanical change
 * repeated across many independent files. The root implements everything
 * else itself: across five 62-task runs the tasks it split were the ones it
 * did worst on and no faster than base3 (docs/benchmarks/base3-fast/). Every
 * base3 lever stays as it is (`windowedFileReads`, mechanical compaction, no
 * `instructionsPrompt`), and the root reconciles and verifies the result.
 *
 * The worker runs the root's model with `inheritParentSystemPrompt` and
 * `includeMessageHistory`, so the runtime sends it the root's system prompt,
 * tool schemas and message history byte-for-byte with only the assignment
 * appended (packages/agent-runtime/src/run-agent-step.ts). On a prefix-cached
 * provider that request is a cache hit except its last few hundred tokens,
 * which is what makes N workers affordable. It therefore has no
 * `systemPrompt` and no `compactContext` of its own; either would diverge its
 * prefix from the root's. Its last message is its report to the root and
 * names the files it edited; the root reads those files to reconcile the
 * slices, so no git round-trip (and no git) is needed.
 *
 * A worker must be pinned to its root's model or the session gate rejects it
 * mid-run, so the worker id names the model. Which models run this harness is
 * FREEBUFF_FAST_MODE_MODEL_IDS (common/src/constants/freebuff-model-ids.ts).
 */

/** Everything the root may spawn: searchers for context, the worker for
 *  slices. The code-searcher is programmatic — a batch of ripgrep queries and
 *  no model call — so it needs no free-mode allowlist entry. No file-picker
 *  since 2026-10-16: the 62-task bench without it (2026-10-05) was a wash. */
export const BASE3_FAST_SPAWNABLE_AGENTS = [
  'code-searcher',
  FREEBUFF_BASE3_FAST_WORKER_AGENT_ID,
] as const

/** The line every worker assignment opens with. The worker inherits a prompt
 *  that says "you coordinate workers", so its own role is the first sentence
 *  it reads and is repeated in its instructions. */
export const BASE3_FAST_WORKER_ASSIGNMENT_OPENING =
  '[freebuff product prompt text omitted]'

/** What a worker may not run of base3's own tools: the terminal, because
 *  verification is the root's, once, for every slice together, and a worker
 *  with a terminal runs the suite N times in parallel; and the todo list,
 *  which is the root's plan. Everything else base3 has, the worker has. */
const WORKER_WITHHELD_TOOLS: ReadonlySet<string> = new Set([
  'run_terminal_command',
  'write_todos',
])

/**
 * Appended to base3's prompt, never prepended: the chat-completions gate
 * requires base3's canonical opening at byte 0. Parameterized on the worker id
 * so the buffbench arm, whose worker is pinned to the ordinary Flash id, can
 * name its own worker; production uses BASE3_FAST_GUIDANCE below.
 *
 * Deliberately short and mostly suggestions: the agent answers questions,
 * explores as long as it needs to, and decides when to split. What it is told
 * firmly is the one thing it would otherwise skip (spawning helpers in
 * parallel) and the one failure the fan-out invites (incoherent slices).
 */
export function buildBase3FastGuidance(
  workerAgentId: string = FREEBUFF_BASE3_FAST_WORKER_AGENT_ID,
): string {
  return `
# Fast mode

[freebuff product prompt text omitted]

[freebuff product prompt text omitted]

**Implementing.** Make the change yourself. A change you hold in your head comes out coherent, and the slow part of any delegation is reading, reconciling and repairing what came back. Workers exist for one shape of work: a large, mechanical change repeated across many independent files, where each slice can be written without seeing the others, such as a rename across a tree or the same migration applied to dozens of modules. Only then spawn one ${workerAgentId} per slice, all in one spawn_agents call, after writing any piece the slices must agree on (a new type, signature or helper) yourself. Give each worker a short assignment that opens with "${BASE3_FAST_WORKER_ASSIGNMENT_OPENING}", then names the files it owns, the shared names it must use, and what belongs to another worker. Never give two workers the same file.

[freebuff product prompt text omitted]

**Verifying.** Verify after the last worker has returned, and verify properly: typecheck each package you touched, run the tests that cover the change (adding focused tests where the change warrants them), and when no test exercises the requested behaviour, exercise it yourself. Fix what fails and re-run what you changed until it is green. What is not worth repeating is a check that already passed on files you have not edited since, or the whole repository's suite when the covering tests exist. Run each check as a plain command so its exit status is yours to read.
`.trim()
}

export const BASE3_FAST_GUIDANCE = buildBase3FastGuidance()

/**
 * The Freebuff CLI root for a fast-mode model: `createBase3CliRoot` plus the
 * fan-out. Its toolset is the CLI root's plus `spawn_agents`, assembled rather
 * than restated; the shipped-agents CI guard imports the definition and
 * checks the list it becomes.
 */
export function createBase3FastCliRoot(
  model: SecretAgentDefinition['model'],
  options: {
    /** Drop the tools that address a human, as createBase3CliRoot does for
     *  the eval harness: an ask_user call would stall a run. */
    noAskUser?: boolean
    /** Drop the web tools, as createBase3CliRoot does for the eval harness. */
    noWeb?: boolean
    /** The worker this root fans out to. Production roots take the default;
     *  the buffbench arm names a worker pinned to the ordinary Flash id. */
    workerAgentId?: string
    /** See createBase3CliRoot: the CLI's server-catalog root runs on an
     *  opaque handle and passes the row's name and compaction policy. */
    modelLabel?: string
    compaction?: CompactionPolicy & { maxContextLength?: number }
  } = {},
): Omit<SecretAgentDefinition, 'id'> {
  const {
    noAskUser = false,
    noWeb = false,
    workerAgentId = FREEBUFF_BASE3_FAST_WORKER_AGENT_ID,
    modelLabel,
    compaction,
  } = options
  const root = createBase3CliRoot({
    model,
    isFreebuff: true,
    noAskUser,
    noWeb,
    extraSystemPrompt: buildBase3FastGuidance(workerAgentId),
    modelLabel,
    compaction,
  })
  return {
    ...root,
    displayName: 'Buffy (fast mode)',
    spawnerPrompt:
      'Fast-mode coding agent: gathers context with parallel code-searchers, implements the change itself, and can split a large mechanical change across parallel prompt-cached workers that report which files they changed',
    toolNames: [...root.toolNames!, 'spawn_agents'],
    spawnableAgents: BASE3_FAST_SPAWNABLE_AGENTS.map((id) =>
      id === FREEBUFF_BASE3_FAST_WORKER_AGENT_ID ? workerAgentId : id,
    ),
  }
}

/**
 * The worker a fast-mode root fans its implementation out to. It shares the
 * root's prompt, tools and history (see the file comment) and may run only
 * base3's own tools minus WORKER_WITHHELD_TOOLS: it SEES the root's full tool
 * schema, but the runtime executes just these and answers anything else with
 * a tool error (packages/agent-runtime/src/tools/tool-executor.ts), and tells
 * the worker which tools are its own.
 *
 * `windowedFileReads` is set even though the schema comes from the parent:
 * the read_files and glob handlers read the flag off the CHILD's template.
 */
export function createBase3FastWorker(
  model: SecretAgentDefinition['model'],
): Omit<SecretAgentDefinition, 'id'> {
  const base3 = createBase3(model)
  return {
    publisher,
    model,
    providerOptions: base3.providerOptions,
    displayName: 'Buffy Worker',
    spawnerPrompt:
      'Implements ONE slice of the current task in parallel with sibling workers by reading and editing files; it has no terminal and no web. It sees this whole conversation and inherits your system prompt (its request is a prompt-cache hit), so brief it rather than re-explaining. Its report names every file it edited.',
    inputSchema: {
      prompt: {
        type: 'string',
        description:
          'The assignment: open with the worker-N-of-M line, then the goal, the files this worker owns, the shared names and signatures it must use, and what belongs to another worker. Brief — it already has the context above; never paste file contents or finished code.',
      },
    },
    outputMode: 'last_message',
    includeMessageHistory: true,
    inheritParentSystemPrompt: true,
    windowedFileReads: true,
    toolNames: base3.toolNames!.filter(
      (name) => !WORKER_WITHHELD_TOOLS.has(name),
    ),
    spawnableAgents: [],
    instructionsPrompt: `[freebuff product prompt text omitted]

- Implement your assignment. Edit only the files it names; if the slice needs a change elsewhere, leave it and say so in your report.
- Read a file before editing it; search only to confirm a name or usage you were not given. Do not re-explore what the conversation above already established.
-[freebuff product prompt text omitted]
- Work in as few steps as the edits allow, and stop as soon as your slice is done. Your last message is your report: every file you edited and what changed in each, plus anything left undone or worth checking, in a few sentences.`,
  }
}

          requestId,
          signal,
        )

        message:
          error instanceof z.ZodError
            ? rejectedArguments(error)
            : 'Could not prepare the Sites request. Check arguments and workspace files.',
      }
    }
  }

export function parseSitePreparationOutput(
  stdout: string,

    .split('\n')
    .findLast((line) => line.startsWith(SITE_PREPARE_RESULT_PREFIX))

  const value = JSON.parse(
    line.slice(SITE_PREPARE_RESULT_PREFIX.length),

    Array.isArray(list) &&

    400,
    'site_prepare_failed',
    message ??
      'The site build failed. Inspect freebuff-sites-build/build.log in the project.',
  )

/** Run the preparation program in a local workspace, as `prepareNextSite` does. */
export async function prepareSiteLocally(
  cwd: string,
  signal?: AbortSignal,
): Promise<SitePreparationResult> {
  const stdout = await new Promise<string>((resolve, reject) => {
    const buildSignal = AbortSignal.any([
      AbortSignal.timeout(15 * 60 * 1000),
      ...(signal ? [signal] : []),
    ])
    const child = spawn(
      'node',
      ['--input-type=module', '-e', sitePreparationScript()],
      {
        cwd,
        signal: buildSignal,
        detached: process.platform !== 'win32',
        env: nextPreparationEnvironment(process.env),
        stdio: ['ignore', 'pipe', 'pipe'],
      },
    )
    const stop = () => {
      if (process.platform !== 'win32' && child.pid) {
        try {
          process.kill(-child.pid, 'SIGTERM')
        } catch {
          /* Already exited. */
        }
      }
    }
    buildSignal.addEventListener('abort', stop, { once: true })
    let output = ''
    child.stdout.on('data', (data: Buffer) => {
      output = (output + data.toString()).slice(-64 * 1024)
    })
    child.stderr.on('data', () => {})
    child.once('error', () =>
      reject(
        preparationFailed(
          buildSignal.aborted
            ? 'Publishing was cancelled or the build exceeded its 15-minute limit.'
            : 'Publishing needs Node.js 20.19 or newer in the workspace PATH.',
        ),
      ),
    )
    child.once('close', (code) => {
      buildSignal.removeEventListener('abort', stop)
      if (code === 0) resolve(output)
      else reject(preparationFailed(parseSitePreparationError(output)))
    })
  })
  try {
    return parseSitePreparationOutput(stdout)
  } catch {
    throw preparationFailed()
  }
}

type ToolRefusal = { error: string; message?: string }
function refusal(value: unknown): ToolRefusal | null {
  const error = (value as { error?: unknown } | null)?.error
  return typeof error === 'string' ? (value as ToolRefusal) : null
}
type Operation = {
  id: string
  state: 'running' | 'succeeded' | 'failed' | 'unknown'
  result?: {
    url?: string
    error?: string
    provider?: { codes?: number[]; stage?: string; status?: number }
  } | null
}
type SiteEnvironment = {
  url: string
  published?: boolean
  deployed_at?: string | null
  unpublished_at?: string | null
  auth?: unknown
  bindings?: { name: string; kind: string }[]
}

// Older backends have the timestamps but no explicit publication field.
function isPublished(environment: SiteEnvironment): boolean {
  return (
    environment.published ??
    Boolean(environment.deployed_at && !environment.unpublished_at)
  )
}

/** Receipts keep only codes; say what each means without vendor prose. */
export function operationFailureMessage(operation: Operation): string {
  const result = operation.result
  if (operation.state === 'unknown' || result?.error === 'cloudflare_outcome_unknown')
    return 'Cloudflare did not confirm the deployment. Check deployment activity in Manage site before publishing again.'
  const provider = result?.provider
  if (provider?.stage === 'worker_upload' && provider.codes?.includes(10021))
    return 'Cloudflare could not start the built Worker: it failed startup validation. This usually means code that cannot run on Workers, such as a Node.js-only module.'
  if (provider?.status === 429)
    return 'Cloudflare is rate limiting deployments. Try again in a minute.'
  if (result?.error === 'opennext_manifest_required')
    return 'This environment hosts a Next.js app. Publish the Next.js project to it.'
  return 'Cloudflare rejected the deployment.'
}

export async function publishSite(options: {
  environment: 'preview' | 'production'
  /** The site to update; absent creates one for this project. */
  siteId?: string
  siteName?: string
  /** Runs one Sites tool with the caller's credentials and workspace. */
  tool: (name: string, args: Record<string, unknown>) => Promise<unknown>
  prepare: () => Promise<SitePreparationResult>
  onEvent: (event: SitePublishEvent) => void
  id?: () => string
  /** Waits between polls of a deployment still running. */
  sleep?: (ms: number) => Promise<void>
  /** A caller's own user-facing errors (Cloud workspace states); others stay generic. */
  errorMessage?: (error: unknown) => string | undefined
}): Promise<SitePublishResult> {
  const id = options.id ?? randomUUID
  const { environment } = options
  const sleep =
    options.sleep ?? ((ms: number) => new Promise((r) => setTimeout(r, ms)))
  const fail = (message: string) => {
    const event = { type: 'error' as const, message }
    options.onEvent(event)
    return event
  }
  const call = async <T>(name: string, args: Record<string, unknown>) => {
    const value = await options.tool(name, args)
    const refused = refusal(value)
    if (refused)
      throw new SitesClientError(
        400,
        refused.error,
        refused.message ?? 'The Sites request failed.',
      )
    return value as T
  }
  try {
    options.onEvent({ type: 'stage', stage: 'building' })
    const prepared = await options.prepare()
    let manifestPath = prepared.manifestPath
    if (prepared.kind === 'next')
      manifestPath = (
        await call<{ manifestPath: string }>('sites_prepare_next', {})
      ).manifestPath
    if (!manifestPath) return fail('The build did not produce a deployment.')

    options.onEvent({ type: 'stage', stage: 'deploying' })
    let siteId = options.siteId
    if (!siteId) {
      siteId = id()
      const name =
        (options.siteName ?? prepared.name ?? '').trim().slice(0, 80) ||
        'My site'
      await call('sites_create', { siteId, name })
    }
    const target = { siteId, environment }
    const inspect = () =>
      call<{ environment: SiteEnvironment }>('sites_inspect', target)
    const inspected = await inspect()
    const details = inspected.environment
    if (prepared.usesAuth && !details.auth)
      await call('sites_enable_auth', target)
    const existing = new Set((details.bindings ?? []).map((b) => b.name))
    for (const kind of ['d1', 'r2'] as const)
      for (const name of prepared.storage[kind]) {
        if (existing.has(name)) continue
        const added = await call<{ operation?: Operation }>(
          'sites_add_storage',
          { ...target, requestId: id(), kind, name },
        )
        if (added.operation && added.operation.state !== 'succeeded')
          return fail(
            `Could not create the ${kind === 'd1' ? 'database' : 'storage bucket'} ${name}. Check deployment activity in Manage site.`,
          )
      }

    const requestId = id()
    let operation = (
      await call<{ operation: Operation }>('sites_deploy', {
        ...target,
        requestId,
        manifestPath,
      })
    ).operation
    // Deployments normally settle in the request; a slow one is polled by key.
    for (let i = 0; operation.state === 'running' && i < 30; i++) {
      await sleep(2000)
      const activity = (
        await call<{
          activity?: { operations?: Operation[] }
        }>('sites_inspect', target)
      ).activity
      operation =
        activity?.operations?.find((op) => op.id === operation.id) ?? operation
    }
    if (operation.state === 'running')
      return fail(
        'The deployment is still running. Check deployment activity in Manage site.',
      )
    if (operation.state !== 'succeeded')
      return fail(operationFailureMessage(operation))

    // A successful upload is not publication. Republish only when this request
    // started with an unpublished environment; preserve a newer Unpublish.
    let current = (await inspect()).environment
    if (!isPublished(current) && details.unpublished_at) {
      options.onEvent({ type: 'stage', stage: 'publishing' })
      const publication = await call<{ operation: Operation }>(
        'sites_set_published',
        {
          ...target,
          published: true,
          requestId: id(),
        },
      )
      if (publication.operation.state !== 'succeeded')
        return fail(
          'Publication was not confirmed. Check deployment activity in Manage site.',
        )
      current = (await inspect()).environment
    }
    if (!isPublished(current))
      return fail(
        'The deployment succeeded, but this environment is unpublished. Check Publication in Manage site.',
      )
    const event = {
      type: 'done' as const,
      siteId,
      environment,
      url: operation.result?.url ?? details.url,
    }
    options.onEvent(event)
    return event
  } catch (error) {
    return fail(
      error instanceof SitesClientError
        ? error.message
        : (options.errorMessage?.(error) ??
            'Publishing failed. Try again, or ask the agent to publish.'),
    )
  }
}

+): Promise<Uint8Array> {
+  let response: Response
+  try {
+    response = await (options.fetcher ?? fetch)(template.archive.url, {
+      redirect: 'error',
+      signal: options.signal ?? AbortSignal.timeout(60_000),
+    })
+  } catch (error) {
+    throw new TemplateCatalogError('Could not download the template', 503, { cause: error })
+  }
+  if (!response.ok) throw new TemplateCatalogError(`Template download returned ${response.status}`)
+  const bytes = new Uint8Array(await response.arrayBuffer())
+  if (bytes.byteLength !== template.archive.bytes)
+    throw new TemplateCatalogError('Template download is not the size the catalogue lists')
+  const digest = options.digest ? await options.digest(bytes) : await sha256Hex(bytes)
+  if (digest !== template.archive.sha256) throw new TemplateCatalogError('Template download failed its checksum')
+  return bytes
+}
+
+async function sha256Hex(bytes: Uint8Array): Promise<string> {
+  const buffer = bytes.buffer.slice(bytes.byteOffset, bytes.byteOffset + bytes.byteLength) as ArrayBuffer
+  const hash = await crypto.subtle.digest('SHA-256', buffer)
+  return [...new Uint8Array(hash)].map((b) => b.toString(16).padStart(2, '0')).join('')
