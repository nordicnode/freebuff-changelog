# Agent Template Override System

This system allows users to customize agent behavior by placing override configuration files in their project's `.agents/templates/` directory.

## How It Works

1. **File Detection**: The system automatically detects `.agents/templates/*.json` and `.md` files and loads them into a dedicated `agentTemplates` field in the project file context
2. **Override Processing**: When an agent is spawned, the system checks for matching override files and applies them to the base template
3. **Dynamic Application**: Overrides are applied at runtime, so changes take effect immediately without restarting

## Override File Format

Override files should be JSON files with the following structure:

```json
{
  "override": {
    "type": "CodebuffAI/reviewer",
    "version": "0.1.7",
    "model": "anthropic/claude-sonnet-4",
    "systemPrompt": {
      "type": "append",
      "content": "Additional system instructions"
    },
    "spawnableAgents": {
      "type": "append",
      "content": ["thinker"]
    }
  }
}
```

## Configuration Options

### Basic Properties

- `type`: Agent type identifier (e.g., "CodebuffAI/reviewer" maps to "reviewer")
- `version`: Version identifier (currently informational)
- `model`: Override the model used by the agent

### Prompt Overrides

Each prompt type (`systemPrompt`, `instructionsPrompt`, `stepPrompt`) supports:

- `type`: How to apply the override
  - `"append"`: Add content after the base prompt
  - `"prepend"`: Add content before the base prompt
  - `"replace"`: Replace the entire base prompt
- `content`: Inline content to use
- `path`: Path to external file (relative to the override file)

### Array Overrides

For arrays like `spawnableAgents` and `toolNames`:

- `type`: How to apply the override
  - `"append"`: Add items to the existing array
  - `"replace"`: Replace the entire array
- `content`: String or array of strings to add/replace

/**
 * The CLI's own base3 roots — Codebuff's DEFAULT and LITE modes, and every
 * Freebuff model the picker offers.
 *
 * Kept separate from `createBase3` rather than folded into it, because Freebuff
 * Desktop derives its toolset from `createBase3().toolNames` and would silently
 * inherit whatever is added here (THREAD_AGENT_TOOLS in
 * freebuff-desktop/.../thread-agent.ts unions that array with its own extras).
 * The bare eight are the harness; what follows is CLI product surface.
 *
 * One thing is load-bearing, for the same reason it is on the Web roots
 * (docs/freebuff-base3-harness.md): the appendix is APPENDED. The root prompt
 * must keep its canonical opening first (`hasFreebuffRootSystemPromptOpening`).
 */
export function createBase3CliRoot(
  options: {
    model?: SecretAgentDefinition['model']
    /** Freebuff branding and meta-information instead of Codebuff's. */
    isFreebuff?: boolean
    /** Drop the tools that address a human. For the eval harness, where an
     *  ask_user call would stall the run rather than gather anything. */
    noAskUser?: boolean
    /** Drop web_search and read_url. For the eval harness: the evaluated
     *  repositories are public, and a search result that quotes their
     *  current code is the answer key (36% of base3's search results in the
     *  2026-10-04 runs cited the repository under evaluation). */
    noWeb?: boolean
    /** Harness guidance that belongs BETWEEN base3's prompt and the CLI
     *  appendix (base3-fast's fan-out section). Appended, never prepended,
     *  for the reason the appendix is: the canonical opening stays at byte 0. */
    extraSystemPrompt?: string
    /** The model's human name, for the meta-information line. A server
     *  catalog root's `model` is an opaque handle, which must not be what the
     *  prompt tells the model it is running on. */
    modelLabel?: string
    /** See createBase3. */
    compaction?: RootCompactionPolicy
    /** See createBase3. */
    compiledModelId?: string
    /** See createBase3. On for CLI roots; the sponsored and fast-mode roots
     *  turn it off. */
    batching?: boolean
  } = {},
): Omit<SecretAgentDefinition, 'id'> {
  const {
    model = OPUS_MODEL,
    isFreebuff = false,
    noAskUser = false,
    noWeb = false,
    extraSystemPrompt,
  } = options
  const base3 = createBase3(model, {
    compaction: options.compaction,
    compiledModelId: options.compiledModelId,
    batching: options.batching ?? true,
  })

  const root: Omit<SecretAgentDefinition, 'id'> = {
    ...base3,
    // base3's own eight, then the CLI surface: `web_search`/`read_url`
    // replace the researcher subagents base2 spawned; the last five drive the
    // ask-user panel, the followup cards, service discovery, rendered UI and
    // skills. (The shipped-agents CI guard checks this toolset by importing
    // the definitions, so it may be assembled rather than restated.)
    toolNames: [
      ...base3.toolNames!,
      'web_search',
      'read_url',
      'ask_user',
      'suggest_followups',
      'gravity_index',
      'render_ui',
      'skill',
    ],
    systemPrompt: `${base3.systemPrompt}
${extraSystemPrompt ? `${extraSystemPrompt}\n` : ''}${buildCliAppendix({
      isFreebuff,
      model: options.modelLabel ?? model,
      noAskUser,
    })}`,
  }

  const withheld = new Set<string>([
    ...(noAskUser ? HUMAN_TOOL_NAMES : []),
    ...(noWeb ? WEB_TOOL_NAMES : []),
  ])
  if (withheld.size === 0) return root
  return {
    ...root,
    toolNames: root.toolNames?.filter((name) => !withheld.has(name)),
  }
}

/**
 * Returns whether trace writing is enabled: always in dev, opt-in via

export function isTraceEnabled(): boolean {
  if (IS_DEV) return true
  // Same accepted values as CODEBUFF_FULL_TELEMETRY (see analytics-sampling)
  const flag = getCliEnv().CODEBUFF_TRACE
  return flag === '1' || flag === 'true' || flag === 'yes'
}

  try {
    return IS_DEV
      ? path.join(getProjectRoot(), 'debug', TRACE_FILENAME)
      : path.join(getCurrentChatDir(), TRACE_FILENAME)
  } catch {
    return null // No project root set yet
  }
}

 * re-traced — acceptable for a debug trace.
 */

): TraceWriter | undefined {
  if (!isTraceEnabled()) {
    return undefined

  const agentStates = new Map<string, AgentTraceState>()
  let ensuredDir: string | undefined

  return {
    recordStep: ({
      agentId,
      agentType,
      runId,
      userInputId,
      step,
      system,
      messages,
    }) => {
      const state = agentStates.get(agentId) ?? {
        writtenRoles: [],
        system: undefined,
        stepPrompt: undefined,
      }
      const base = { agentId, agentType, runId, userInputId, step }
      const timestamp = new Date().toISOString()
      const lines: string[] = []
      const appendLine = (record: Record<string, unknown>): void => {
        lines.push(JSON.stringify({ timestamp, ...record }))
      }

      // An agent's step prompt ends each step's request and is gone from the
      // next (the runtime drops it every step): diffed with the history it
      // would read as a rewrite at every step. It is recorded like the system
      // prompt instead, once and again only when it changes.
      const history = messages.filter((m) => !m.tags?.includes('STEP_PROMPT'))
      const stepPrompt = messages.find((m) => m.tags?.includes('STEP_PROMPT'))

      const rewritten =
        history.length < state.writtenRoles.length ||
        state.writtenRoles.some((role, i) => history[i]?.role !== role)
      if (rewritten) {
        appendLine({
          ...base,
          type: 'history_rewritten',
          previousMessageCount: state.writtenRoles.length,
          messageCount: history.length,
        })
        state.writtenRoles = []
      }

      if (system !== undefined && system !== state.system) {
        appendLine({ ...base, type: 'system', system })
        state.system = system
      }

      const stepPromptText = stepPrompt && JSON.stringify(stepPrompt.content)
      if (stepPromptText && stepPromptText !== state.stepPrompt) {
        appendLine({ ...base, type: 'step_prompt', message: stepPrompt })
        state.stepPrompt = stepPromptText
      }

      for (let i = state.writtenRoles.length; i < history.length; i++) {
        const message = history[i] as Message
        appendLine({
          ...base,
          type: 'message',
          index: i,
          message,
        })
      }

      state.writtenRoles = history.map((m) => m.role)
      agentStates.set(agentId, state)

      if (lines.length === 0) return
      // Resolve the path per step (not cached for the writer's lifetime: the
      // current chat directory changes when the user starts a new chat).
      const filePath = resolveTraceFilePath()
      if (!filePath) return
      try {
        const dir = dirname(filePath)
        if (ensuredDir !== dir) {
          mkdirSync(dir, { recursive: true })
          ensuredDir = dir
        }
        appendFileSync(filePath, lines.join('\n') + '\n')
      } catch {
        // Tracing must never break the run
      }
    },
  }
}

import { BASE3_BATCHING_STEP_PROMPT } from '../constants/base3-step-prompt'

/**
 * How the agent runtime frames an agent's step prompt: a user message of its
 * own, wrapped so the model does not read it as the user speaking again
 * (packages/agent-runtime/src/templates/strings.ts).
 */
export function withSystemReminderTags(text: string): string {
  return `<system_reminder>${text}</system_reminder>`
}

/** Our roots' step reminders, as they arrive. */
const ROOT_STEP_REMINDERS: ReadonlySet<string> = new Set([
  withSystemReminderTags(BASE3_BATCHING_STEP_PROMPT),
])

/**
 * A user message that is exactly one of our roots' step reminders: the last
 * message of every request on base3 batching (agents/base3.ts), and never the
 * user's own words. Matched by its whole text, never by its tags alone, so a
 * prompt cannot pass for one by wrapping itself in them.
 */
export function isRootStepReminderMessage(message: unknown): boolean {
  const { role, content } = (message ?? {}) as {
    role?: unknown
    content?: unknown
  }
  if (role !== 'user') return false
  let text: string
  if (typeof content === 'string') {
    text = content
  } else if (Array.isArray(content)) {
    const parts = content as { type?: unknown; text?: unknown }[]
    if (!parts.every((part) => part?.type === 'text')) return false
    text = parts.map((part) => String(part.text ?? '')).join('')
  } else {
    return false
  }
  return ROOT_STEP_REMINDERS.has(text.trim())
}
