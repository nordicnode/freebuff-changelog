```
/**
 * Compaction by structured summary, prepared in the background.
 *
 * The goals: a summary is ready before it is needed (low latency), it keeps
 * decisions, requirements and next steps (accurate), and a user who returns
 * after the prompt cache expired sends a small request (no cold start).
 *
 * 1. When the context reaches FIRST_SUMMARY_TOKENS, and then each time it
 *    grows by SUMMARY_GROWTH_TOKENS since the last summary, the runtime asks
 *    the agent's own model for a summary of the whole conversation. The
 *    request is the agent's normal request plus one instruction, so the
 *    provider reads the history from its warm cache. It runs in the
 *    background: after a step, or when the turn ends. A new summary replaces
 *    the old one.
 * 2. The full history is still sent while the cache is warm.
 * 3. The summary is used only when the user returns after SUMMARY_IDLE_MS, or
 *    when the context reaches its limit. Then the request is the summary, the
 *    messages after it, and the new message.
 *
 * The summary waits on the agent state (`compactionSummary`), outside the
 * history. A summary that finishes after its run ended reaches the next run
 * of the same conversation through an in-process store.
 */
```

```
/**
 * The instruction a root agent appends to its own history to ask its model
 * for a structured summary of the conversation (packages/agent-runtime/src/
 * conversation-summary.ts). It is the request's last message, after the same
 * system prompt, tools and history the agent's own requests send, so the
 * provider serves the history from its prompt cache.
 *
 * The section format follows Factory's anchored iterative summary
 * (https://factory.com/news/evaluating-compression) as OpenCode words it.
 */
export const COMPACTION_SUMMARY_PROMPT_OPENING =
  'Stop working on the task. Write a structured summary of this whole conversation.'

/** `codebuff_metadata` key that marks a summary request. */
export const COMPACTION_SUMMARY_METADATA_KEY = 'compaction_summary'

export function buildCompactionSummaryPrompt(targetTokens: number): string {
  return `${COMPACTION_SUMMARY_PROMPT_OPENING}
The summary replaces the conversation history. Another agent continues the work from it, with the same system prompt and tools, so it must be able to continue without the original messages.

Summarize what the user and the assistant said and did. Do not restate the system prompt, repository instructions or environment details: the next agent gets them separately. If the conversation starts with an earlier summary, merge it into this one. Newer messages take precedence over it.

Use exactly this format. Write "(none)" for an empty section.

## Objective
- [one or two sentences: what the user wants to accomplish]

## Requirements
- [constraints, preferences and scope limits the user stated; keep their exact wording when it matters]

## Decisions
- [decisions made and why]

## Work State
### Completed
- [finished goals]
### Active
- [goals in progress, and how far they got]
### Blocked
- [what blocks progress, and why]

## Next Move
1. [the next actions, in order]

## Relevant Files
- \`path\`: [why it matters, what changed in it]

## Important Context
- [facts the next agent cannot easily find again: exact commands, error messages, values, identifiers, URLs, and test or build results]

## Latest Exchange
- [the user's latest message, and what the assistant last did or answered]

Rules:
- Use short single-line bullets, not paragraphs.
- Prefer short references over restating details. Leave out what the next agent can get again from the code or files.
- Keep exact file paths, symbols, commands, error text and identifiers.
- Keep user requests that are still open, with their exact wording. Drop the ones newer messages resolved.
- Keep the state of changes: uncommitted, committed, pushed, in review or merged.
- Keep every heading exactly as written here, in English. Write the text under the headings in the language the user writes in.
- Keep the summary under about ${targetTokens} tokens.
-[freebuff product prompt text omitted]
}
```

```
function sanitizeAdClickIvtScoredFields(
  record: Record<string, unknown>,
): AxiomOnlyLogEvent['data'] {
  const data: AxiomOnlyLogEvent['data'] = sanitizeAllowlistedFields(
    record,
    AD_CLICK_IVT_SCORED_FIELDS,
  )
  // Unknown is a feature value, not false or zero. Keep explicit nulls on
  // this event only; other operational allowlists retain their scalar rules.
  for (const key of Object.keys(AD_CLICK_IVT_SCORED_FIELDS)) {
    if (record[key] === null) data[key] = null
  }
  if (
    Array.isArray(record.reasons) &&
    record.reasons.length <= 32 &&
    record.reasons.every(
      (reason) => typeof reason === 'string' && /^[a-z_]{1,64}$/.test(reason),
    )
  ) {
    data.reasons = [...record.reasons]
  }
  return data
}

/**
 * A client attempt to acknowledge a first-party unit it mounted. This is an
 * operational transport census, never an attribution record: opaque tokens,
 * identifiers, URLs, bodies, and raw errors are rejected rather than redacted.
 */
export type FirstPartyViewAckTelemetry = FirstPartyViewAckObservation

const FIRST_PARTY_VIEW_ACK_FIELDS = [
  'surface',
  'placement_id',
  'outcome',
  'attempt',
  'duration_ms',
  'client_family',
] as const

/**
 * Fields a NEWER client may add to the exact set above (COD-365). Optional
 * on the check, not appended to the exact set, because a released CLI or
 * Desktop binary still sends the six-field shape and its events must stay
 * valid. Each is still validated when present -- a malformed value rejects
 * the whole event exactly as a malformed required field does.
 */
const FIRST_PARTY_VIEW_ACK_OPTIONAL_FIELDS = [
  'client_event_id',
  'sample_rate',
] as const
const FIRST_PARTY_VIEW_ACK_CLIENT_EVENT_ID_RE = /^[A-Za-z0-9._:-]{1,128}$/

const FIRST_PARTY_VIEW_ACK_PLACEMENTS = new Map<string, string>(
  PLACEMENT_SLOTS.map((slot) => [slot.id, slot.surface]),
)

/**
 * Validate the only telemetry payload a client may send for view
 * acknowledgement. Returning null rejects the entire event: dropping an
 * unsafe field but retaining the count would make malformed client input look
 * like a real rendering signal.
 */
export function createFirstPartyViewAckTelemetry(
  input: unknown,
): FirstPartyViewAckTelemetry | null {
  if (input == null || typeof input !== 'object' || Array.isArray(input)) {
    return null
  }
  const record = input as Record<string, unknown>
  const keys = Object.keys(record)
  const required = new Set<string>(FIRST_PARTY_VIEW_ACK_FIELDS)
  const optional = new Set<string>(FIRST_PARTY_VIEW_ACK_OPTIONAL_FIELDS)
  if (
    keys.some((key) => !required.has(key) && !optional.has(key)) ||
    FIRST_PARTY_VIEW_ACK_FIELDS.some((key) => !(key in record))
  ) {
    return null
  }
  const clientEventId = record.client_event_id
  const sampleRate = record.sample_rate
  if (
    (clientEventId !== undefined &&
      (typeof clientEventId !== 'string' ||
        !FIRST_PARTY_VIEW_ACK_CLIENT_EVENT_ID_RE.test(clientEventId))) ||
    (sampleRate !== undefined &&
      (typeof sampleRate !== 'number' ||
        !Number.isInteger(sampleRate) ||
        sampleRate < 1))
  ) {
    return null
  }

  const surface = record.surface
  const placementId = record.placement_id
  const outcome = record.outcome
  const attempt = record.attempt
  const durationMs = record.duration_ms
  const clientFamily = record.client_family
  if (
    typeof surface !== 'string' ||
    typeof placementId !== 'string' ||
    FIRST_PARTY_VIEW_ACK_PLACEMENTS.get(placementId) !== surface ||
    !FIRST_PARTY_VIEW_ACK_OUTCOMES.includes(
      outcome as FirstPartyViewAckOutcome,
    ) ||
    typeof attempt !== 'number' ||
    !Number.isInteger(attempt) ||
    attempt < 1 ||
    attempt > 3 ||
    typeof durationMs !== 'number' ||
    !Number.isFinite(durationMs) ||
    durationMs < 0 ||
    durationMs > FIRST_PARTY_VIEW_ACK_MAX_DURATION_MS ||
    !FIRST_PARTY_VIEW_ACK_CLIENT_FAMILIES.includes(
      clientFamily as FirstPartyViewAckClientFamily,
    )
  ) {
    return null
  }
  return {
    surface,
    placement_id: placementId,
    outcome: outcome as FirstPartyViewAckOutcome,
    attempt: attempt as 1 | 2 | 3,
    duration_ms: durationMs,
    client_family: clientFamily as FirstPartyViewAckClientFamily,
    ...(clientEventId !== undefined
      ? { client_event_id: clientEventId as string }
      : {}),
    ...(sampleRate !== undefined ? { sample_rate: sampleRate as number } : {}),
  }
}

/** Keep only the allowlisted keys whose value matches the declared type
 *  (strings truncated); everything else is dropped. */
function sanitizeAllowlistedFields(
  record: Record<string, unknown>,
  fields: AxiomOnlyFieldSchema,
): Record<string, string | number | boolean> {
  const sanitized: Record<string, string | number | boolean> = {}
  for (const [key, expectedType] of Object.entries(fields)) {
    const value = record[key]
    if (typeof value !== expectedType) continue
    if (typeof value === 'string') {
      sanitized[key] = value.slice(0, 200)
    } else if (typeof value === 'number' && Number.isFinite(value)) {
      sanitized[key] = value
    } else if (typeof value === 'boolean') {
      sanitized[key] = value
    }
  }
  return sanitized
}

/**
 * Return a sanitized Axiom-only event, or null for ordinary logger payloads.
 * The event name comes from `data.axiomEvent` (the in-process marker set at
 * the log call site) or the `event` param (the wire-format field a caller
 * already extracted, e.g. the server-side sink re-checking a persisted
 * `LogRow`). Unknown keys and unexpected value types are deliberately
 * discarded.
 *
 * Matched by exact equality (not a lookup keyed on the caller-supplied name)
 * so a value like 'constructor' can't resolve through an object's prototype.
 */
export function getAxiomOnlyLogEvent(
  data: unknown,
  event?: string | null,
): AxiomOnlyLogEvent | null {
  const record =
    data != null && typeof data === 'object' && !Array.isArray(data)
      ? (data as Record<string, unknown>)
      : {}
  const eventName =
    typeof record.axiomEvent === 'string' ? record.axiomEvent : event

  if (eventName === CONTEXT_PRUNING_COMPLETED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, CONTEXT_PRUNING_FIELDS),
    }
  }
  if (eventName === STREAM_RECOVERY_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, STREAM_RECOVERY_FIELDS),
    }
  }
  if (eventName === FILE_WRITE_GUARD_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, FILE_WRITE_GUARD_FIELDS),
    }
  }
  if (eventName === MECHANICAL_COMPACTION_SKIPPED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(
        record,
        MECHANICAL_COMPACTION_SKIPPED_FIELDS,
      ),
    }
  }
  if (eventName === CONTEXT_COMPACTION_COMPLETED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(
        record,
        CONTEXT_COMPACTION_COMPLETED_FIELDS,
      ),
    }
  }
  if (eventName === COMPACTION_FOLLOWUP_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, COMPACTION_FOLLOWUP_FIELDS),
    }
  }
  if (eventName === COMPACTION_SUMMARY_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, COMPACTION_SUMMARY_FIELDS),
    }
  }
  if (eventName === COMPACTION_REQUEST_USAGE_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, COMPACTION_REQUEST_USAGE_FIELDS),
    }
  }
  if (eventName === COMPACTION_ATTEMPT_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, COMPACTION_ATTEMPT_FIELDS),
    }
  }
  if (eventName === COMPACTION_FIRST_OUTPUT_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, {
        ...COMPACTION_CORRELATION_FIELDS,
        agent_run_id: 'string',
        model: 'string',
        duration_ms: 'number',
        idle_return: 'boolean',
      }),
    }
  }
  if (eventName === FILE_EDIT_LOOP_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, FILE_EDIT_LOOP_FIELDS),
    }
  }
  if (eventName === TODO_PROGRESS_REMINDER_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, TODO_PROGRESS_REMINDER_FIELDS),
    }
  }
  if (eventName === GRAVITY_REPORT_NUDGE_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, GRAVITY_REPORT_NUDGE_FIELDS),
    }
  }
  if (eventName === ADS_FETCH_COMPLETED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, ADS_FETCH_COMPLETED_FIELDS),
    }
  }
  if (eventName === ADS_FIRST_PARTY_DECISION_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, ADS_FIRST_PARTY_DECISION_FIELDS),
    }
  }
  if (eventName === ADS_FIRST_PARTY_SETTLEMENT_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(
        record,
        ADS_FIRST_PARTY_SETTLEMENT_FIELDS,
      ),
    }
  }
  if (eventName === ADS_FIRST_PARTY_VIEW_ACK_EVENT) {
    // `axiomEvent` is the in-process marker only; it is not telemetry data.
    const { axiomEvent: _axiomEvent, ...payload } = record
    const telemetry = createFirstPartyViewAckTelemetry(payload)
    return telemetry ? { event: eventName, data: { ...telemetry } } : null
  }
  if (
    eventName === ADS_FIRST_PARTY_CLICK_RECORDED_EVENT ||
    eventName === ADS_FIRST_PARTY_IMPRESSION_RECORDED_EVENT
  ) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, ADS_FIRST_PARTY_TRACKING_FIELDS),
    }
  }
  if (eventName === ADS_REQUEST_REJECTED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, ADS_REQUEST_REJECTED_FIELDS),
    }
  }
  if (eventName === ADS_SHOWCASE_PRESENTED_EVENT) {
    const data =
      createGravityShowcasePresentationTelemetry(record) ??
      createWebShowcasePresentationTelemetry(record)
    // Keep the event inside the restricted branch even when malformed. The
    // sink treats `null` as an ordinary row and would otherwise retain the raw
    // client payload that failed validation.
    return { event: eventName, data: data ?? {} }
  }
  if (
    eventName === ADS_BREAK_SHOWN_EVENT ||
    eventName === ADS_BREAK_CLOSED_EVENT ||
    eventName === ADS_BREAK_CLICKED_EVENT
  ) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, ADS_SPONSOR_BREAK_FIELDS),
    }
  }
  if (eventName === ADS_EXTERNAL_CONVERSION_POSTBACK_EVENT) {
    return {
      event: eventName,
      data: sanitizeExternalConversionPostbackFields(record),
    }
  }
  if (eventName === ADS_CAMPAIGN_INGRESS_EVIDENCE_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(
        record,
        ADS_CAMPAIGN_INGRESS_EVIDENCE_FIELDS,
      ),
    }
  }
  if (eventName === ADS_ADVERTISER_REPORTING_READ_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(
        record,
        ADS_ADVERTISER_REPORTING_READ_FIELDS,
      ),
    }
  }
  if (eventName === ADS_MCP_TOOL_CALL_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, ADS_MCP_TOOL_CALL_FIELDS),
    }
  }
  if (eventName === AD_CLICK_IVT_SCORED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAdClickIvtScoredFields(record),
    }
  }
  if (eventName === PAID_API_REQUEST_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, PAID_API_REQUEST_FIELDS),
    }
  }
  return null
}

/**
 * COD-365 hygiene contract, exported for the CI guard in
 * `__tests__/axiom-only-log.test.ts`: every ads event allowlist this module
 * owns for CLIENT-emitted events must carry these keys, and an unsampled
```

```
/**
 * Compaction by structured summary, prepared in the background.
 *
 * The goals: a summary is ready before it is needed (low latency), it keeps
 * decisions, requirements and next steps (accurate), and a user who returns
 * after the prompt cache expired sends a small request (no cold start).
 *
 * 1. When the context reaches FIRST_SUMMARY_TOKENS, and then each time it
 *    grows by SUMMARY_GROWTH_TOKENS since the last summary, the runtime asks
 *    the agent's own model for a summary of the whole conversation. The
 *    request is the agent's normal request plus one instruction, so the
 *    provider reads the history from its warm cache. It runs in the
 *    background: after a step, or when the turn ends. A new summary replaces
 *    the old one.
 * 2. The full history is still sent while the cache is warm.
 * 3. The summary is used only when the user returns after SUMMARY_IDLE_MS, or
 *    when the context reaches its limit. Then the request is the summary, the
 *    messages after it, and the new message.
 *
 * The summary waits on the agent state (`compactionSummary`), outside the
 * history. A summary that finishes after its run ended reaches the next run
 * of the same conversation through an in-process store.
 */

/**
 * The idle gap after which the summary replaces the history. Five minutes is
 * the shortest prompt cache measured on our lanes (Anthropic via Respan,
 * prod 2026-10-26); Pareto keeps it 10-15 minutes, OpenAI about 30.
 */
export const SUMMARY_IDLE_MS = 5 * 60 * 1000
export const FIRST_SUMMARY_TOKENS = 40_000
export const SUMMARY_GROWTH_TOKENS = 30_000
/** Small windows use shares of the window instead of the fixed sizes. */

const SUMMARY_GROWTH_SHARE = 0.15

const SUMMARY_TARGET_SHARE = 0.1

const SUMMARY_NOTE =
  'Historical memory only: it replaces the earlier conversation. Messages after it are real messages. The latest user request can appear both in the memory and after it: the memory then records work already done on it, so continue that work, do not start over. If no user message follows, continue the latest request in the memory with real tool calls.'

export function summaryThresholds(maxContextLength: number): {
  first: number
  growth: number
} {
  return {
    first: Math.min(
      FIRST_SUMMARY_TOKENS,
      Math.floor(maxContextLength * FIRST_SUMMARY_SHARE),
    ),
    growth: Math.min(
      SUMMARY_GROWTH_TOKENS,
      Math.floor(maxContextLength * SUMMARY_GROWTH_SHARE),
    ),
  }
}

/** How long a summary is asked to be: small windows get a shorter one. */

  return Math.min(
    SUMMARY_TARGET_TOKENS,
    Math.floor(maxContextLength * SUMMARY_TARGET_SHARE),
  )
}

/** Whether the context grew enough since `latest` to need a new summary. */
export function summaryIsDue(params: {
  contextTokens: number
  maxContextLength: number
  /** The newest summary made or in progress, as its context size. */
  latestContextTokens: number | undefined
}): boolean {
  const { first, growth } = summaryThresholds(params.maxContextLength)
  if (params.latestContextTokens === undefined)
    return params.contextTokens >= first
  return params.contextTokens - params.latestContextTokens >= growth
}

/**
 * Whether the live prompt arrived after the conversation sat idle longer than
 * the shortest prompt cache. Only true before the assistant answers it.
 */
export function isIdleReturn(messages: Message[], now = Date.now()): boolean {
  const promptIndex = messages.findLastIndex((m) =>
    m.tags?.includes('USER_PROMPT'),
  )
  if (promptIndex <= 0) return false
  if (messages.slice(promptIndex + 1).some((m) => m.role === 'assistant'))
    return false
  const prompt = messages[promptIndex]
  const sentAt = prompt.sentAt ?? now
  for (let i = promptIndex - 1; i >= 0; i--) {
    const message = messages[i]
    if (message.role !== 'assistant') continue
    return (
      message.sentAt !== undefined && sentAt - message.sentAt > SUMMARY_IDLE_MS
    )
  }
  return false
}

  return dropUnansweredToolCalls(expireMessages(messages, 'agentStep'))
}

function isEphemeral(message: Message): boolean {
  return (
    message.timeToLive !== undefined ||
    message.tags?.includes('STEP_PROMPT') === true

}

/** A short, stable id for a message's content (FNV-1a). */
export function messageFingerprint(message: Message): string {
  const text =
    message.role +
    '\u0000' +
    (message.role === 'tool' ? message.toolCallId : '') +
    '\u0000' +
    JSON.stringify(message.content)
  let hash = 0x811c9dc5
  for (let i = 0; i < text.length; i++) {
    hash ^= text.charCodeAt(i)
    hash = Math.imul(hash, 0x01000193)
  }
  return `${(hash >>> 0).toString(16)}:${text.length}`
}

/**
 * Where the summary's covered part ends in `messages`, or -1 when the last
 * covered message is gone (an edit removed it). Searched from its recorded
 * position first, then from the end: dropped scaffolding can move it.
 */
function anchorIndex(
  messages: Message[],
  summary: ConversationSummary,
): number {
  const expected = summary.coveredMessages - 1
  if (
    messages[expected] &&
    messageFingerprint(messages[expected]) === summary.anchor
  )
    return expected
  for (let i = Math.min(expected, messages.length - 1); i >= 0; i--) {
    if (messageFingerprint(messages[i]) === summary.anchor) return i
  }
  return -1
}

export function summaryMatchesHistory(
  messages: Message[],
  summary: ConversationSummary,
): boolean {
  return anchorIndex(messages, summary) >= 0
}

export function buildConversationSummaryMessage(

/**
 * The history with the summary in place of what it covers, and the summary
 * re-anchored on its own message (it now covers itself). Covered messages
 * that `keep` accepts (the current turn's own user messages) stay as real
 * messages right after it: a summary made at the limit, or during a turn,
 * covers the request it is answering, and that request must reach the model
 * word for word. Its size is unset until the provider reports the next
 * request's. Null when the summary no longer matches.
 */
export function applyConversationSummary(params: {
  messages: Message[]
  summary: ConversationSummary
  keep?: (message: Message) => boolean
  now?: number
}): { messages: Message[]; summary: ConversationSummary } | null {
  const index = anchorIndex(params.messages, params.summary)
  if (index < 0) return null
  const head = buildConversationSummaryMessage(params.summary.text, params.now)
  const { contextTokens: _, ...summary } = params.summary
  const kept = params.keep
    ? params.messages.slice(0, index + 1).filter(params.keep)
    : []
  return {
    messages: [head, ...kept, ...params.messages.slice(index + 1)],
    summary: {
      ...summary,
      coveredMessages: 1,
      anchor: messageFingerprint(head),
    },
  }
}

/**
 * The summary in a model's reply, or '' when the reply is not one. It starts
 * at its first heading; a preamble before it is dropped. The format's headings
 * are in English, but a model can translate them with the rest (GLM 5.3 Flash
 * did on 12% of its summaries, prod 2026-10-30, and each one was thrown away),
 * so a reply with the format's shape counts without the name.
 */
export function summaryFromReply(reply: string): string {
  const text = reply.replace(/<think>[\s\S]*?<\/think>/g, '')
  const english = text.indexOf('## Objective')
  if (english >= 0) return text.slice(english).trim()
  const sections = [...text.matchAll(/^## /gm)]
  return sections.length >= MIN_TRANSLATED_SECTIONS
    ? text.slice(sections[0].index).trim()
    : ''
}

/**
 * Asks the model for the summary. `history` must be what the agent's next
 * request sends (`summaryRequestHistory`), or the cache misses.
 */
export async function generateConversationSummary(params: {
  summaryId?: string
  history: Message[]
  system: string
  /** The provider's count when the summary was requested, if known. */
  contextTokens?: number
  targetTokens: number
  maxOutputTokens?: number
  stream: SummaryStream
  signal: AbortSignal
  onUsage?: (usage: ModelUsageData) => void
}): Promise<{ summary: ConversationSummary; finishReason?: string }> {
  const { history } = params
  const summaryId = params.summaryId ?? crypto.randomUUID()
  const createdAt = Date.now()
  const anchor = history.findLastIndex((m) => !isEphemeral(m))
  if (anchor < 0)
    throw new SummaryError('no_history', 'No history to summarize')
  let finishReason: string | undefined
  const stream = params.stream({
    messages: [
      systemMessage(params.system),
      ...history,
      // Tagged so the server's prompt classifier skips it as product text.
      userMessage({
        content: withSystemInstructionTags(
          buildCompactionSummaryPrompt(params.targetTokens),
        ),
      }),
    ],
    maxOutputTokens:
      params.maxOutputTokens !== undefined && params.maxOutputTokens > 0
        ? Math.min(SUMMARY_MAX_OUTPUT_TOKENS, params.maxOutputTokens)
        : SUMMARY_MAX_OUTPUT_TOKENS,
    signal: params.signal,
    extraCodebuffMetadata: {
      [COMPACTION_SUMMARY_METADATA_KEY]: '1',
      compaction_summary_id: summaryId,
      compaction_request_id: summaryId,
      compaction_request_group: 'summary',
    },
    onFinishReason: (reason) => {
      finishReason = reason
    },
    onUsageReceived: (usage) => params.onUsage?.(usage),
  })
  let text = ''
  for (;;) {
    const next = await stream.next()
    if (next.done) {
      if (next.value.aborted) params.signal.throwIfAborted()
      break
    }
    const chunk = next.value
    if (chunk.type === 'error')
      throw new SummaryError('provider_error', chunk.message)
    if (chunk.type === 'text') text += chunk.text
  }
  params.signal.throwIfAborted()
  if (finishReason === 'length')
    throw new SummaryError('output_limit', 'The summary hit the output limit')
  const summaryText = summaryFromReply(text)
  if (summaryText.length < MIN_SUMMARY_CHARS)
    throw new SummaryError(
      text.trim().length < MIN_SUMMARY_CHARS ? 'empty' : 'unrecognized',
      'The model returned no summary',
    )
  return {
    finishReason,
    summary: {
      id: summaryId,
      text: summaryText,
      coveredMessages: anchor + 1,
      anchor: messageFingerprint(history[anchor]),
      ...(params.contextTokens === undefined
        ? {}
        : { contextTokens: params.contextTokens }),
      createdAt,
    },
  }
}

type Entry = {
  latest?: ConversationSummary
  pending?: {
    promise: Promise<ConversationSummary | null>
    contextTokens: number
  }
}

const MAX_CONVERSATIONS = 200
const conversations = new Map<string, Entry>()

function entry(key: string): Entry {
  let value = conversations.get(key)
  if (value) {
    conversations.delete(key)
  } else {
    value = {}
  }
  conversations.set(key, value)
  while (conversations.size > MAX_CONVERSATIONS) {
    const oldest = conversations.keys().next().value
    if (oldest === undefined) break
    conversations.delete(oldest)
  }
  return value
}

/** The newest of the stored summary and the one in `state`. */
function newer(
  a: ConversationSummary | undefined,
  b: ConversationSummary | undefined,
): ConversationSummary | undefined {
  if (!a) return b
  if (!b) return a
  return b.createdAt > a.createdAt ? b : a
}

/**
 * One conversation's summaries in this process: the newest finished one and
 * the one in progress. Keyed by conversation and agent, so a summary that
 * finishes after its run ended reaches the next turn.
 */
export function conversationSummaries(key: string) {
  return {
    /** The newest summary that still matches `messages`. */
    latest(
      fromState: ConversationSummary | undefined,
      messages: Message[],
    ): ConversationSummary | undefined {
      const summary = newer(fromState, entry(key).latest)
      return summary && summaryMatchesHistory(messages, summary)
        ? summary
        : undefined
    },
    pending() {
      return entry(key).pending
    },
    /** Records a summary the history now carries (after it was applied). */
    replace(summary: ConversationSummary | undefined) {
      entry(key).latest = summary
    },
    /** Starts a summary unless one is running. */
    start(
      contextTokens: number,
      run: () => Promise<ConversationSummary | null>,
    ): Promise<ConversationSummary | null> {
      const current = entry(key)
      if (current.pending) return current.pending.promise
      const promise = run()
        .then((summary) => {
          if (summary) {
            const target = entry(key)
            target.latest = newer(target.latest, summary)
          }
          return summary
        })
        .catch(() => null)
        .finally(() => {
          const target = conversations.get(key)
          if (target?.pending?.promise === promise) target.pending = undefined
        })
      current.pending = { promise, contextTokens }
      return promise
    },
  }
}

    runId: string

  },

    runId,
    model,
    mode,
    midTurn,

    telemetry,
    ...summaryParams
  } = params
  const started = Date.now()
  const summaryId = params.summaryId ?? crypto.randomUUID()
  let usage: ModelUsageData | undefined
  const log = (data: Record<string, unknown>, message: string) => {
    try {
      logger.info(
        {
          axiomEvent: COMPACTION_SUMMARY_EVENT,
          ...telemetry,
          summary_id: summaryId,
          agent_run_id: runId,
          model,
          mode,
          mid_turn: midTurn,
          context_token_count: params.contextTokens,
          duration_ms: Date.now() - started,
          ...(usage
            ? {
                input_tokens: usage.inputTokens,
                cached_input_tokens: usage.cachedInputTokens,
                output_tokens: usage.outputTokens,
              }
            : {}),
          ...data,
        },
        message,
      )
    } catch {
      // Logging must never fail a summary.
    }
  }
  try {
    const { summary, finishReason } = await generateConversationSummary({
      ...summaryParams,
      summaryId,
      onUsage: (received) => {
        usage = received
        summaryParams.onUsage?.(received)
      },
    })
    log(
      {
        outcome: 'completed',
        finish_reason: finishReason,
        summary_tokens: countTokens(summary.text),
      },
      'Conversation summary completed',
    )
    return summary
  } catch (error) {
    log(
      {
        outcome: cancelSignal?.aborted ? 'cancelled' : 'failed',
        error_kind:
          error instanceof SummaryError
            ? error.kind
            : isAbortError(error) || summaryParams.signal.aborted
              ? 'timeout'
              : 'provider_error',
        // Not allowlisted for Axiom; local/debug logs only.
        error: getErrorObject(error),
      },
      'Conversation summary failed',
    )
    if (cancelSignal?.aborted) throw error
    return null
  }
}

/** Waits for `promise`, or rejects when `signal` aborts first. */
export function untilAborted<T>(
  promise: Promise<T>,
  signal: AbortSignal,
): Promise<T> {
  signal.throwIfAborted()
  return new Promise<T>((resolve, reject) => {
    const onAbort = () => reject(new AbortError())
    signal.addEventListener('abort', onAbort, { once: true })
    promise.then(resolve, reject).finally(() => {
      signal.removeEventListener('abort', onAbort)
    })
  })
}
```
