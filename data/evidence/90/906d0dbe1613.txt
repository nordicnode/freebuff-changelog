// common; this module re-exports it and adds the terminal rendering and layout

export {
  FREEBUFF_STREAK_WEEK,
  getFreebuffStreakBonusNote,
} from '@codebuff/common/util/freebuff-streak-line'
export type { FreebuffStreakLine } from '@codebuff/common/util/freebuff-streak-line'

import {
  getFreebuffStreakBonusNote,
  getFreebuffStreakLine as getSharedFreebuffStreakLine,
} from '@codebuff/common/util/freebuff-streak-line'

import type { FreebuffStreakLine } from '@codebuff/common/util/freebuff-streak-line'

const FREEBUFF_STREAK_BONUS_MIN_HEIGHT = 30

/** Progress glyphs for a terminal — the same ●/○ pair the shared module and
 *  the desktop app use. Bullet and middle dot were tried here because U+25CF
 *  is missing from a few terminal fonts and lands as a tofu box, but • and ·
 *  differ only in size: at a glance a partial week and a full one look alike,
 *  which is the whole point of the row. Filled-vs-hollow reads instantly, and
 *  the CLI already bets on ●/○ for its agent status indicators, so a font that
 *  can't draw them is already visibly broken elsewhere.
 *
 *  If a font ever fails these, █/░ (Block Elements) is the fallback pair: the
 *  ASCII logo and the progress bar are built from them, so anything that
 *  renders the CLI at all renders those. */
const TERMINAL_DOT_CHARS = { filled: '●', empty: '○' }

/** The streak line as the CLI draws it. */
export function getFreebuffStreakLine(
  streak: number,
): FreebuffStreakLine | null {
  return getSharedFreebuffStreakLine(streak, TERMINAL_DOT_CHARS)
}

/** Returns the earned perk note only when the landing layout can show it
 * without crowding the picker or wrapping onto additional rows. */
export function getFreebuffStreakBonusNoteForLayout(params: {
  streak: number
  terminalHeight: number
  availableWidth: number
}): string | null {
  if (params.terminalHeight < FREEBUFF_STREAK_BONUS_MIN_HEIGHT) return null

  const note = getFreebuffStreakBonusNote(params)
  if (!note || note.length > params.availableWidth) return null

  return note
}

import { FREEBUFF_STREAK_REWARDS_ENABLED } from '../constants/freebuff-models'

/** Days in a streak "week" — the milestone the progress dots fill toward. */
export const FREEBUFF_STREAK_WEEK = 7

export interface FreebuffStreakLine {
  /** Count label, e.g. "2 day streak". */
  label: string
  /** A week's worth of progress dots toward the 7-day milestone, e.g.
   *  "●●○○○○○". Fills to "●●●●●●●" at 7, then gains a trailing "+"
   *  ("●●●●●●●+") for any streak beyond the week so long runs read as
   *  "earned and still going" rather than just maxed out. */
  dots: string
  /** The same progress as counts, for a surface that draws its own dots rather
   *  than glyphs (the desktop app draws CSS circles). `total` is always
   *  FREEBUFF_STREAK_WEEK, carried so a renderer that can't import this module
   *  knows how many slots to draw; `beyond` is the trailing "+". */
  progress: { filled: number; total: number; beyond: boolean }
}

/** One slot of the week a graphical surface draws: a streak day, today's
 *  (still unsent) or ahead. */
export type FreebuffStreakSlot = 'done' | 'today' | 'empty'

/**
 * The week toward the 7-day milestone as slots, for the workspace's streak
 * card and popup (the phone apps mirror it): one per streak day, filling to
 * seven and staying full past it as the dots do. While today is still
 * unsent, the next slot is today's.
 */
export function getFreebuffStreakWeekSlots(
  streak: number,
  todayUsed: boolean,
): FreebuffStreakSlot[] {
  const filled = Math.min(Math.max(streak, 0), FREEBUFF_STREAK_WEEK)
  return Array.from({ length: FREEBUFF_STREAK_WEEK }, (_, index) =>
    index < filled ? 'done' : index === filled && !todayUsed ? 'today' : 'empty',
  )
}

/** Glyph pair used to draw the progress dots. */
export interface FreebuffStreakDotChars {
  filled: string
  empty: string
}

/**
 * Pure presentation logic for the streak line shown on the CLI landing screen
 * and the desktop account popover: a plain count plus a week of filled/empty
 * progress dots. Returns null for streak <= 0 so the caller hides the row
 * entirely — new / lapsed users should be nudged to start using the product,
 * not shown an empty streak.
 *
 * The glyphs default to ●/○, which is what a real UI font renders best; a
 * surface whose font can't be trusted with those passes its own pair.
 */
export function getFreebuffStreakLine(
  streak: number,
  chars: FreebuffStreakDotChars = { filled: '●', empty: '○' },
): FreebuffStreakLine | null {
  if (streak <= 0) return null

  // Fill toward the 7-day milestone, then stay full — a 19-day streak should
  // read as fully earned, not roll back over into a partial second week. Past
  // the week, a trailing "+" marks that the streak has run beyond the row.
  const filled = Math.min(streak, FREEBUFF_STREAK_WEEK)
  const beyond = streak > FREEBUFF_STREAK_WEEK
  const dots =
    chars.filled.repeat(filled) +
    chars.empty.repeat(FREEBUFF_STREAK_WEEK - filled) +
    (beyond ? '+' : '')

  // "day" stays singular — it's a compound modifier ("7 day streak"), not a
  // count of days on its own.
  return {
    label: `${streak} day streak`,
    dots,
    progress: { filled, total: FREEBUFF_STREAK_WEEK, beyond },
  }
}

/**
 * The streak's perk note: every streak day earns XP on the Rewards page
 * (docs/freebuff-xp.md). Null with no streak at all (the streak row is hidden
 * then too). Until 2026-11-24 a 7+ day streak added Freebucks or a session to
 * each day's allowance; that bonus is retired, so the server no longer sends
 * `freebucksDailyBonus` and the note no longer reads it.
 */
export function getFreebuffStreakBonusNote(params: {
  streak: number
}): string | null {
  if (!FREEBUFF_STREAK_REWARDS_ENABLED) return null
  if (params.streak <= 0) return null
  return '🎁 Every streak day earns XP toward rewards'
}

/**
 * How XP is earned (the Rewards page's XP system). One definition for the page
 * (`freebuff/web/src/lib/xp-rewards.ts`) and the server that counts it
 * (`packages/auth/src/xp.ts`).
 */

/** A friend you invited who counts (`countingReferralFriendsSql`). */
export const XP_PER_FRIEND = 1_000

/** Signing up through a friend's invite, once. */
export const XP_INVITED_BONUS = 1_000

/** The XP that unlocks custom themes, like a friend or a subscription does. */
export const XP_THEMES_UNLOCK = 1_000

/** One-time XP, each awarded once per account. */
export const XP_ONE_TIME = {
  discord: 500,
  site: 500,
  github: 250,
} as const
export type XpOneTimeKey = keyof typeof XP_ONE_TIME

/**
 * The XP one day of a streak earns, for the `streakDay`th day (1-based):
 * 5 a day in week 1, 5 more each week after, 25 a day from week 5 on.
 * A broken streak starts again at day 1.
 */
export function dailyStreakXp(streakDay: number): number {
  if (streakDay < 1) return 0
  const week = Math.ceil(streakDay / 7)
  return Math.min(week * 5, 25)
}

/**
 * Limited access earns a tenth of every XP amount here, rounded up to a
 * whole XP (a first-week streak day is 1); each level needs the same XP at
 * either tier. Applied to each amount as it is earned, never to a sum, and at
 * the tier of the request that reads it.
 */
export function xpAtTier(
  xp: number,
  tier: 'full' | 'limited' | null | undefined,
): number {
  return tier === 'limited' ? Math.ceil(xp / 10) : xp
}

/** One-time XP for reaching a streak of `days`. */
export const XP_STREAK_MILESTONES = [

/** One recorded account day (`freebuff_streak_day`). */
export interface XpStreakAccountDay {
  startsAt: Date
  resetsAt: Date
  firstAt: Date
}

/**
 * The gap between two account days under which they are still consecutive:
 * the streak's own rule (`MIN_STREAK_BONUS_DAY_MS`,
 * `web/src/server/free-session/freebucks-streak-day.ts`). A real missed day
 * is at least 23 hours.
 */
const STREAK_GAP_MS = 20 * 60 * 60 * 1000

/**
 * Every day the account kept a streak, oldest first, as its place in that
 * streak (1 for a streak's first day). The same days and the same rule the
 * shown streak counts (`calculateAccountDayStreak`): account days, two of
 * them consecutive when the newer starts less than 20 hours after the older
 * resets; before the first account day (the 2026-10-01 switch), the Pacific
 * usage dates, which the first account day continues when the Pacific date
 * before its first message was used.
 */
export function streakDayPositions(params: {
  accountDays: readonly XpStreakAccountDay[]
  pacificUsageDates: readonly string[]
}): number[] {
  const days = [...params.accountDays].sort(
    (a, b) => a.startsAt.getTime() - b.startsAt.getTime(),
  )
  const switchDate = days[0] ? getFreebuffUsageDateKey(days[0].firstAt) : null
  const pacific = [...new Set(params.pacificUsageDates)]
    .filter((date) => switchDate === null || date < switchDate)
    .sort()

  const positions: number[] = []
  let position = 0
  let lastDate: string | null = null
  for (const date of pacific) {
    position =
      lastDate !== null && addDaysToDateKey(lastDate, 1) === date
        ? position + 1
        : 1
    positions.push(position)
    lastDate = date
  }
  let lastDay: XpStreakAccountDay | null = null
  for (const day of days) {
    const continues = lastDay
      ? day.startsAt.getTime() - lastDay.resetsAt.getTime() < STREAK_GAP_MS
      : lastDate !== null && addDaysToDateKey(lastDate, 1) === switchDate
    position = continues ? position + 1 : 1
    positions.push(position)
    lastDay = day
  }
  return positions
}

/**
 * The streak XP of a history (`streakDayPositions`): each day's daily XP,
 * plus each milestone the longest streak reached, once, each at `tier`.
 */
export function streakXp(
  positions: readonly number[],
  tier?: 'full' | 'limited' | null,
): number {
  let xp = 0
  let longest = 0
  for (const position of positions) {
    xp += xpAtTier(dailyStreakXp(position), tier)
    longest = Math.max(longest, position)
  }
  for (const milestone of XP_STREAK_MILESTONES)
    if (longest >= milestone.days) xp += xpAtTier(milestone.xp, tier)
  return xp
}

+import {
+  XP_STREAK_MILESTONES,
+  dailyStreakXp,
+  xpAtTier,
+} from '../constants/freebuff-xp'
+import { addDaysToDateKey, getFreebuffUsageDateKey } from './freebuff-streak'
+
+/** One recorded account day (`freebuff_streak_day`). */
+export interface XpStreakAccountDay {
+  startsAt: Date
+  resetsAt: Date
+  firstAt: Date
+}
+
+/**
+ * The gap between two account days under which they are still consecutive:
+ * the streak's own rule (`MIN_STREAK_BONUS_DAY_MS`,
+ * `web/src/server/free-session/freebucks-streak-day.ts`). A real missed day
+ * is at least 23 hours.
+ */
+const STREAK_GAP_MS = 20 * 60 * 60 * 1000
+
+/**
+ * Every day the account kept a streak, oldest first, as its place in that
+ * streak (1 for a streak's first day). The same days and the same rule the
+ * shown streak counts (`calculateAccountDayStreak`): account days, two of
+ * them consecutive when the newer starts less than 20 hours after the older
+ * resets; before the first account day (the 2026-10-01 switch), the Pacific
+ * usage dates, which the first account day continues when the Pacific date
+ * before its first message was used.
+ */
+export function streakDayPositions(params: {
+  accountDays: readonly XpStreakAccountDay[]
+  pacificUsageDates: readonly string[]
+}): number[] {
+  const days = [...params.accountDays].sort(
+    (a, b) => a.startsAt.getTime() - b.startsAt.getTime(),
+  )
+  const switchDate = days[0] ? getFreebuffUsageDateKey(days[0].firstAt) : null
+  const pacific = [...new Set(params.pacificUsageDates)]
+    .filter((date) => switchDate === null || date < switchDate)
+    .sort()
+
+  const positions: number[] = []
+  let position = 0
+  let lastDate: string | null = null
+  for (const date of pacific) {
+    position =
+      lastDate !== null && addDaysToDateKey(lastDate, 1) === date
+        ? position + 1
+        : 1
+    positions.push(position)
+    lastDate = date
+  }
+  let lastDay: XpStreakAccountDay | null = null
+  for (const day of days) {
+    const continues = lastDay
+      ? day.startsAt.getTime() - lastDay.resetsAt.getTime() < STREAK_GAP_MS
+      : lastDate !== null && addDaysToDateKey(lastDate, 1) === switchDate
+    position = continues ? position + 1 : 1
+    positions.push(position)
+    lastDay = day
+  }
+  return positions
+}

+/**
+ * The streak XP of a history (`streakDayPositions`): each day's daily XP,
+ * plus each milestone the longest streak reached, once, each at `tier`.
+ */
+export function streakXp(
+  positions: readonly number[],
+  tier?: 'full' | 'limited' | null,
+): number {
+  let xp = 0
+  let longest = 0
+  for (const position of positions) {
+    xp += xpAtTier(dailyStreakXp(position), tier)
+    longest = Math.max(longest, position)
+  }
+  for (const milestone of XP_STREAK_MILESTONES)
+    if (longest >= milestone.days) xp += xpAtTier(milestone.xp, tier)
+  return xp
+}
