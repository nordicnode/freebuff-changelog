 ## Freebuff 的工作原理
 
 Freebuff 使用专业化智能体，而不是把所有任务都交给同一个模型和同一条提示词。根据任务需要，智能体会收集上下文、制定计划、编辑或研究、运行工具并审查结果。
 
 - **代码库上下文** —— 文件查找智能体会在编辑前定位项目中的相关部分。
 - **实现与审查** —— 智能体可以拆分工作、修改文件、运行命令并检查结果。
 - **研究与浏览器操作** —— 智能体可以查阅文档，并在真实浏览器中测试应用。
 - **本地并行工作** —— Desktop 会将并发智能体隔离在各自的工作区中。

+export function createCliRunFunnel(deps: {
+  capture(
+    stage: ProductFunnelStage,
+    state: ProductFunnelState,
+    details?: {
+      output_kind?: 'text' | 'tool'
+      has_output?: boolean
+      reason?: 'error' | 'empty'
+      duration_ms?: number
+    },

+  now?: () => number
+}) {
+  const now = deps.now ?? Date.now
+  let startedAt = 0
+  let accepted = false
+  let started = false
+  let ended = false
+  const outputKinds = new Set<'text' | 'tool'>()
+  const emit: typeof deps.capture = (...args) => {
+    try {
+      deps.capture(...args)
+    } catch {
+      /* Never interrupt a run. */
+    }
+  }
+  const output = (kind: 'text' | 'tool') => {
+    if (!accepted || ended || outputKinds.has(kind)) return
+    outputKinds.add(kind)
+    emit('response', 'received', { output_kind: kind })
+  }
+  return {
+    accept() {
+      if (accepted || ended) return
+      accepted = true
+      startedAt = now()
+      emit('send', 'accepted')
+    },
+    event(event: PrintModeEvent) {
+      if (!accepted || ended) return
+      if (event.type === 'start' && !started) {
+        started = true
+        emit('turn', 'started')
+      }
+      // Match ToolBranch: hidden file/terminal work still renders collapsed;
+      // agents can only suppress their own bookkeeping. Control tools render
+      // elsewhere and do not establish a useful root-tool response.
+      if (
+        event.type === 'tool_call' &&
+        !event.parentAgentId &&
+        !(
+          event.includeToolCall === false &&
+          canAgentHideToolCall(event.toolName)
+        ) &&
+        ![
+          'end_turn',
+          'ask_user',
+          'spawn_agents',
+          'spawn_agent_inline',
+        ].includes(event.toolName)
+      ) {
+        output('tool')
+      }
+    },
+    chunk(chunk: unknown) {
+      // Root text is delivered as a string. Structured chunks are subagent
+      // output or reasoning and must not make a first-turn success.
+      if (typeof chunk === 'string' && chunk.trim()) output('text')
+    },
+    finish(success: boolean) {
+      if (!accepted || ended) return
+      ended = true
+      const hasOutput = outputKinds.size > 0
+      emit('turn', success && hasOutput ? 'succeeded' : 'failed', {
+        has_output: hasOutput,
+        ...(!success
+          ? { reason: 'error' as const }
+          : !hasOutput
+            ? { reason: 'empty' as const }
+            : {}),
+        duration_ms: Math.min(
+          7 * 24 * 60 * 60 * 1000,
+          Math.max(0, now() - startedAt),
+        ),
+      })
+    },
+    cancel() {
+      if (ended) return
+      ended = true
+      if (!accepted) return
+      emit('turn', 'cancelled', { has_output: outputKinds.size > 0 })

/**
 * THE fit oracle: whether a sponsored skill's preflight would refuse a
 * repository, decided from the repository's FACTS before anything is
 * offered. Pure, no I/O: serving (freebuff-web), Desktop, the Web worker, the
 * CLI and the bench all call it, over facts built by the one builder here
 * ({@link sponsoredRepoFacts}).
 *
 * Each preflight check gets a three-valued verdict, ANDed across checks:
 * - `requirements.os` against the device OS;
 * - `file_exists` when its glob can only match fit files
 *   (`SPONSORED_FIT_FILE_NAMES`), so the facts' listing decides it;
 * - `file_contains changedOnly:false` against the manifests' dependency names
 *   (projected back into manifest-shaped text and matched with the check's
 *   own regex);
 * - `package_declared` against the root package.json's names;
 * - an `absent` check on a marker file (`.coderabbit.yaml`);
 * - a format-2 `fit` block directly.
 * Command and script checks are never fit checks: always `unknown`.
 *
 * Only a KNOWN misfit refuses ({@link sponsoredSkillFitRefusal}): an unknown
 * verdict serves, as today, and the skill's own preflight decides in the run.
 * Doc: `docs/ads/agentic/sponsored-skills.md`, "Fit".

/**
 * The step driver: the harness, not the model, decides when a sponsored
 * skill's steps are done (piece 3 of COD-822; the format is
 * `docs/ads/agentic/sponsored-skills.md`).
 *
 * After the run's turn has done its work, the driver walks the skill's steps
 * in order and runs each step's checks through THE skill evaluator
 * (`evaluateSponsoredSkillChecks`), the same one the sponsored eval scores
 * with. A step with no failing check is done, whatever the model said. A
 * failing `agent` or `check` step goes back to the agent with the failed
 * checks (`sponsoredSkillCheckFeedback`), to fix and stop; the checks run
 * again after each fix. The step's `maxAttempts` counts the first try (the
 * turn's own work) plus fixes, and the run as a whole gets at most
 * {@link SPONSORED_STEP_MAX_FIXES_PER_RUN}. Then the step is BLOCKED, and the
 * walk ends there: later steps usually build on it, and prompting them would
 * spend the run's compute on work that cannot pass. They are still checked
 * once, so the result says what is true on disk.
 *
 * - A `user` step is never prompted or retried. It is done only once its
 *   checks pass (the user handed over the credential), else it needs the user.
 * - A step that passed before a later fix is checked again at the end: a fix
 *   for step 4 can break step 2.
 *
 * DELIVERED means every `agent` and `check` step passed. That is the run's
 * completion, not the model writing "Done:".
 *
 * Pure: the caller hands in the check context (its worktree, its sandbox) and
 * how to prompt the agent, so the whole loop runs in tests without a model,
 * a sandbox or a disk.

/**
 * THE fit oracle: whether a sponsored skill's preflight would refuse a
 * repository, decided from the repository's FACTS before anything is
 * offered. Pure, no I/O: serving (freebuff-web), Desktop, the Web worker, the
 * CLI and the bench all call it, over facts built by the one builder here
 * ({@link sponsoredRepoFacts}).
 *
 * Each preflight check gets a three-valued verdict, ANDed across checks:
 * - `requirements.os` against the device OS;
 * - `file_exists` when its glob can only match fit files
 *   (`SPONSORED_FIT_FILE_NAMES`), so the facts' listing decides it;
 * - `file_contains changedOnly:false` against the manifests' dependency names
 *   (projected back into manifest-shaped text and matched with the check's
 *   own regex);
 * - `package_declared` against the root package.json's names;
 * - an `absent` check on a marker file (`.coderabbit.yaml`);
 * - a format-2 `fit` block directly.
 * Command and script checks are never fit checks: always `unknown`.
 *
 * Only a KNOWN misfit refuses ({@link sponsoredSkillFitRefusal}): an unknown
 * verdict serves, as today, and the skill's own preflight decides in the run.
 * Doc: `docs/ads/agentic/sponsored-skills.md`, "Fit".
 */

/**

 * {@link sponsoredRepoFacts} on the client (or from legacy capability facts on

 */
export type SponsoredRepoFacts = {
  v: 1
  /** Every fit-file path found (basename in SPONSORED_FIT_FILE_NAMES), repo-relative, sorted. */
  paths: string[]
  /**
   * The listing covered the whole project and nothing was cut, so a fit file
   * NOT in `paths` is absent. False makes every absence unknown.
   */
  complete: boolean
  /** Exact paths known to be absent even when `complete` is false (legacy facts). */
  absent?: string[]
  manifests: SponsoredRepoManifest[]
  /** Fit marker files (`.coderabbit.yaml`) that are empty or whitespace-only. */
  empty?: string[]
}

type MaybePromise<T> = T | Promise<T>

/**
 * THE builder of repository facts. `paths` is the project's file listing,
 * repo-relative (either separator; tracked and untracked, ignored files
 * excluded, like the step driver's `files()`); `complete` says that listing
 * is the whole project. `read` returns a file's text, or null when it is
 * missing or unreadable. Reads only manifests (≤16, ≤256 KiB each) and fit
 * marker files.
 */
export async function sponsoredRepoFacts(input: {
  paths: Iterable<string>
  complete: boolean
  read: (path: string) => MaybePromise<string | null>
}): Promise<SponsoredRepoFacts> {
  let complete = input.complete
  const found = new Set<string>()
  for (const raw of input.paths) {
    const path = raw.replace(/\\/g, '/').replace(/^\.\//, '')
    if (!path || path.split('/').includes('node_modules')) continue
    const basename = path.split('/').at(-1) ?? ''
    if (isSponsoredFitFileName(basename)) found.add(path)
  }
  let paths = [...found].sort(byDepthThenPath)
  if (paths.length > MAX_REPO_FACT_PATHS) {
    paths = paths.slice(0, MAX_REPO_FACT_PATHS)
    complete = false
  }

  const manifests: SponsoredRepoManifest[] = []
  let budget = MAX_REPO_FACT_DEPENDENCY_NAMES
  const manifestPaths = paths.filter((path) =>
    manifestEcosystem(path.split('/').at(-1) ?? ''),
  )
  for (const path of manifestPaths.slice(0, MAX_REPO_FACT_MANIFESTS)) {
    const basename = path.split('/').at(-1)!
    const ecosystem = manifestEcosystem(basename)!
    const text = await safeRead(input.read, path)
    const parsed =
      text === null || text.length > MAX_REPO_FACT_MANIFEST_BYTES
        ? { names: [], complete: false }
        : manifestDependencyNames(basename, text)
    const names = parsed.names.slice(0, Math.max(0, budget))
    budget -= names.length
    manifests.push({
      path,
      ecosystem,
      names,
      complete: parsed.complete && names.length === parsed.names.length,
    })
  }

  const empty: string[] = []
  for (const path of paths) {
    const basename = path.split('/').at(-1) ?? ''
    if (!(SPONSORED_FIT_MARKER_NAMES as readonly string[]).includes(basename))
      continue
    const text = await safeRead(input.read, path)
    if (text !== null && !/\S/.test(text)) empty.push(path)
  }

  return {
    v: 1,
    paths: paths.sort(),
    complete,
    manifests: manifests.sort((a, b) => a.path.localeCompare(b.path)),
    ...(empty.length ? { empty } : {}),
  }
}

async function safeRead(
  read: (path: string) => MaybePromise<string | null>,
  path: string,
): Promise<string | null> {
  try {
    return await read(path)
  } catch {
    return null
  }
}

function byDepthThenPath(a: string, b: string): number {
  const depth = a.split('/').length - b.split('/').length
  return depth || a.localeCompare(b)
}

const PYTHON_MANIFEST_REGEXES = SPONSORED_FIT_ECOSYSTEMS.python.manifests.map(
  (glob) => new RegExp(`^${glob.replace(/\./g, '\\.').replace(/\*/g, '[^/]*')}$`),
)

  basename: string,

  if (basename === 'package.json') return 'npm'

const PYTHON_NAME = /^[A-Za-z0-9][A-Za-z0-9._-]*/

/** A requirement string's project name (`fastapi[all]>=0.1; python_version>"3"` → fastapi). */
function requirementName(requirement: string): string | null {
  const text = requirement.trim().replace(/^["']|["']$/g, '')
  if (!text || /^(?:-|git\+|https?:|file:|\.|\/)/.test(text)) return null
  const match = PYTHON_NAME.exec(text)
  return match ? match[0] : null
}

/** A TOML line with its string literals removed (`"fastapi[all]"` holds a `]`). */
function outsideStrings(line: string): string {
  return line.replace(/"[^"]*"|'[^']*'/g, '""')
}

/**
 * The dependency names one manifest declares. Never throws: a manifest that
 * does not parse is `complete: false` with whatever was read.
 */
export function manifestDependencyNames(
  basename: string,
  text: string,
): { names: string[]; complete: boolean } {
  const names = new Set<string>()
  let complete = true
  const add = (name: string | null) => {
    if (name && name.length <= MAX_DEPENDENCY_NAME_CHARS) names.add(name)
  }
  if (basename === 'package.json') {
    try {
      const parsed = JSON.parse(text.replace(/^﻿/, '')) as unknown
      if (!parsed || typeof parsed !== 'object' || Array.isArray(parsed))
        return { names: [], complete: false }
      for (const field of [
        'dependencies',
        'devDependencies',
        'peerDependencies',
        'optionalDependencies',
      ]) {
        const deps = (parsed as Record<string, unknown>)[field]
        if (deps && typeof deps === 'object' && !Array.isArray(deps))
          for (const key of Object.keys(deps)) add(key)
      }
    } catch {
      return { names: [], complete: false }
    }
    return { names: [...names], complete }
  }
  const lines = text.replace(/^﻿/, '').split(/\r?\n/)
  if (/^requirements[^/]*\.txt$/.test(basename)) {
    for (const line of lines) add(requirementName(line.replace(/\s#.*$|^#.*$/, '')))
    return { names: [...names], complete }
  }
  if (basename === 'Pipfile' || basename === 'pyproject.toml') {
    let table = ''
    let arrayOpen = false
    for (const rawLine of lines) {
      const line = rawLine.replace(/\s+#.*$/, '').trim()
      if (!line || line.startsWith('#')) continue
      const header = /^\[\[?([^\]]+)\]\]?$/.exec(line)
      if (header) {
        table = header[1]!.trim()
        arrayOpen = false
        continue
      }
      if (arrayOpen) {
        for (const literal of line.matchAll(/["']([^"']+)["']/g))
          add(requirementName(literal[1]!))
        if (outsideStrings(line).includes(']')) arrayOpen = false
        continue
      }
      const assign = /^([A-Za-z0-9_.\-"']+)\s*=\s*(.*)$/.exec(line)
      if (!assign) continue
      const key = assign[1]!.replace(/["']/g, '')
      const value = assign[2]!
      const keyTable =
        table === 'packages' ||
        table === 'dev-packages' ||
        /^tool\.poetry(?:\.group\.[^.]+)?\.(?:dev-)?dependencies$/.test(table)
      if (keyTable) {
        if (key !== 'python') add(requirementName(key))
        continue
      }
      const arrayTable =
        (table === 'project' && key === 'dependencies') ||
        table === 'project.optional-dependencies' ||
        table === 'dependency-groups' ||
        (table === 'tool.uv' && key === 'dev-dependencies')
      if (arrayTable && value.startsWith('[')) {
        for (const literal of value.matchAll(/["']([^"']+)["']/g))
          add(requirementName(literal[1]!))
        if (!outsideStrings(value).slice(1).includes(']')) arrayOpen = true
      }
    }
    return { names: [...names], complete }
  }
  if (basename === 'setup.cfg') {
    let collecting = false
    let table = ''
    for (const rawLine of lines) {
      const header = /^\s*\[([^\]]+)\]\s*$/.exec(rawLine)
      if (header) {
        table = header[1]!.trim()
        collecting = false
        continue
      }
      const assign = /^([A-Za-z0-9_.-]+)\s*=\s*(.*)$/.exec(rawLine)
      if (assign) {
        collecting =
          (table === 'options' && assign[1] === 'install_requires') ||
          table === 'options.extras_require'
        if (collecting) add(requirementName(assign[2]!.replace(/#.*$/, '')))
        continue
      }
      if (collecting && /^\s+\S/.test(rawLine))
        add(requirementName(rawLine.replace(/#.*$/, '')))
    }
    return { names: [...names], complete }
  }
  if (basename === 'setup.py') {
    // A setup.py is code: read the literal install_requires / extras_require
    // lists, and say the answer may be partial.
    complete = false
    for (const block of text.matchAll(
      /(?:install_requires|extras_require)\s*=\s*([[{][\s\S]*?[\]}])\s*[,)]/g,
    )) {
      for (const literal of block[1]!.matchAll(/["']([^"']+)["']/g))
        add(requirementName(literal[1]!))
    }
    return { names: [...names], complete }
  }
  return { names: [], complete: false }
}

function pathsMatching(facts: Facts, glob: string): string[] {
  const re = sponsoredSkillGlob(glob)

/** Whether no fit path the glob could match exists, as far as the facts prove. */
function provenAbsent(facts: Facts, glob: string): boolean {
  if (facts.complete) return sponsoredFitGlobDecidable(glob)
  // Without a full listing, only a glob of exact paths the facts name absent.
  const literals = expandSponsoredGlobBraces(glob)
  return (
    literals.every((literal) => !/[*?]/.test(literal)) &&
    literals.every((literal) => facts.absent?.includes(literal))
  )
}

function manifestAt(facts: Facts, path: string): SponsoredRepoManifest | null {
  return facts.manifests.find((manifest) => manifest.path === path) ?? null
}

/**
 * A manifest as text a check's regex can read: package.json as a JSON
 * object of its dependencies, a Python manifest as one requirement per line.
 * What the facts carry of the real file, never more.
 */
export function sponsoredManifestProjection(
  manifest: SponsoredRepoManifest,
): string {
  if (manifest.ecosystem === 'npm')
    return `{\n${manifest.names.map((name) => `  "${name}": "*"`).join(',\n')}\n}\n`
  return `${manifest.names.join('\n')}\n`
}

/** Whether a check can ever be decided from repository facts. */
export function sponsoredFitCheckDecidable(check: SponsoredSkillCheck): boolean {
  switch (check.kind) {
    case 'file_exists':
      return sponsoredFitGlobDecidable(check.path)
    case 'file_contains':
      return (check.changedOnly ?? true) === false && sponsoredFitGlobDecidable(check.path)
    case 'package_declared':
    case 'no_privileged_secrets':
    case 'changes_within':
      return true
    default:
      return false
  }
}

/** One preflight check's three-valued verdict over the facts. */
export function evaluateSponsoredFitCheck(
  check: SponsoredSkillCheck,
  facts: Facts | null | undefined,
): SponsoredFitCheckVerdict {
  // Before the first edit nothing has changed: these pass in every preflight.
  if (check.kind === 'no_privileged_secrets' || check.kind === 'changes_within')
    return 'pass'
  if (!facts) return 'unknown'
  switch (check.kind) {
    case 'file_exists':
      if (pathsMatching(facts, check.path).length) return 'pass'
      return provenAbsent(facts, check.path) ? 'fail' : 'unknown'
    case 'file_contains': {
      // A changed-only check reads the run's own changes; none exist yet.
      if ((check.changedOnly ?? true) !== false) return 'unknown'
      let re: RegExp
      try {
        re = new RegExp(check.pattern, check.flags)
      } catch {
        return 'unknown'
      }
      const files = pathsMatching(facts, check.path)
      if (check.absent) {
        let unknown = false
        for (const path of files) {
          if (facts.empty?.includes(path)) {
            if (re.test('')) return 'fail'
            continue
          }
          const manifest = manifestAt(facts, path)
          const isMarker = (SPONSORED_FIT_MARKER_NAMES as readonly string[]).includes(
            path.split('/').at(-1) ?? '',
          )
          // A present marker that is not empty has a non-blank character:
          // a pattern any such text matches fails it.
          if (isMarker && (check.pattern === '\\S' || check.pattern === '^'))
            return 'fail'
          if (manifest && re.test(sponsoredManifestProjection(manifest))) return 'fail'
          if (check.pattern === '^') return 'fail'
          unknown = true
        }
        if (unknown) return 'unknown'
        if (files.length === 0 && !provenAbsent(facts, check.path)) return 'unknown'
        return 'pass'
      }
      let unknown = false
      for (const path of files) {
        const manifest = manifestAt(facts, path)
        if (facts.empty?.includes(path)) {
          if (re.test('')) return 'pass'
          continue
        }
        if (!manifest) {
          unknown = true
          continue
        }
        if (re.test(sponsoredManifestProjection(manifest))) return 'pass'
        if (!manifest.complete) unknown = true
      }
      if (unknown) return 'unknown'
      if (files.length === 0 && !provenAbsent(facts, check.path)) return 'unknown'
      return 'fail'
    }
    case 'package_declared': {
      const root = manifestAt(facts, 'package.json')
      if (root) {
        if (root.names.includes(check.name)) return 'pass'
        return root.complete ? 'fail' : 'unknown'
      }
      if (facts.paths.includes('package.json')) return 'unknown'
      return provenAbsent(facts, 'package.json') ? 'fail' : 'unknown'
    }
    default:
      return 'unknown'
  }
}

function routeVerdict(
  facts: Facts,
  ecosystem: SponsoredFitRouteEcosystem,
  dependencies: string[] | null,
): SponsoredFitCheckVerdict {
  const glob = `**/{${SPONSORED_FIT_ECOSYSTEMS[ecosystem].manifests.join(',')}}`
  const files = pathsMatching(facts, glob)
  if (!files.length) return provenAbsent(facts, glob) ? 'fail' : 'unknown'
  if (!dependencies) return 'pass'
  const wanted = new Set(
    ecosystem === 'python' ? dependencies.map(normalizePypiName) : dependencies,
  )
  let unknown = false
  for (const path of files) {
    const manifest = manifestAt(facts, path)
    if (!manifest) {
      unknown = true
      continue
    }
    const names =
      ecosystem === 'python' ? manifest.names.map(normalizePypiName) : manifest.names
    if (names.some((name) => wanted.has(name))) return 'pass'
    if (!manifest.complete) unknown = true
  }
  return unknown ? 'unknown' : 'fail'
}

/** A format-2 fit block over the facts: the routes ORed, then `mustNotExist`. */
function fitBlockVerdict(
  skill: SponsoredSkill,
  facts: Facts | null | undefined,
): { verdict: SponsoredFitCheckVerdict; reason?: SponsoredFitMisfitReason } {
  const fit = skill.fit
  if (!fit) return { verdict: 'pass' }
  if (!facts) return { verdict: 'unknown' }
  const routes = sponsoredFitRoutes(fit)
  const verdicts = routes.map((route) =>
    routeVerdict(facts, route.ecosystem, route.dependencies),
  )
  let route: SponsoredFitCheckVerdict = verdicts.includes('pass')
    ? 'pass'
    : verdicts.includes('unknown')
      ? 'unknown'
      : 'fail'
  if (route === 'fail') {
    const anyManifest = routes.some(
      (r) =>
        pathsMatching(
          facts,
          `**/{${SPONSORED_FIT_ECOSYSTEMS[r.ecosystem].manifests.join(',')}}`,
        ).length > 0,
    )
    return { verdict: 'fail', reason: anyManifest ? 'no_route' : 'no_manifest' }
  }
  for (const glob of fit.mustNotExist ?? []) {
    if (pathsMatching(facts, glob).length)
      return { verdict: 'fail', reason: 'file_present' }
    if (!provenAbsent(facts, glob)) route = route === 'pass' ? 'unknown' : route
  }
  return { verdict: route }
}

function checkMisfitReason(check: SponsoredSkillCheck): SponsoredFitMisfitReason {
  switch (check.kind) {
    case 'file_exists':
      return 'file_missing'
    case 'package_declared':
      return 'package_missing'
    case 'file_contains':
      return check.absent ? 'file_present' : 'file_content'
    default:
      return 'file_content'
  }
}

/**
 * THE fit verdict of one skill for one repository and device. `fits` only
 * when every fit check passes; `misfit` as soon as one is proven to fail
 * (the first, in the order the step driver runs them); `unknown` otherwise.
 */
export function evaluateSponsoredSkillFit(
  skill: SponsoredSkill,
  input: SponsoredFitInput,
): SponsoredSkillFitVerdict {
  const unknownChecks: string[] = []
  const misfit = (
    reason: SponsoredFitMisfitReason,
    checkId: string,
  ): SponsoredSkillFitVerdict => ({
    fit: 'misfit',
    reason,
    checkId,
    proof: 'known',
    unknownChecks,
  })
  if (input.web) {
    const web = sponsoredSkillWebRefusal(skill)
    if (web) return misfit(web, SPONSORED_FIT_WEB_CHECK_ID)
  }
  const supported = skill.requirements?.os ?? []
  if (supported.length) {
    if (!input.os) unknownChecks.push(SPONSORED_FIT_OS_CHECK_ID)
    else if (!supported.includes(input.os))
      return misfit('os', SPONSORED_FIT_OS_CHECK_ID)
  }
  if (skill.fit) {
    const block = fitBlockVerdict(skill, input.repo)
    if (block.verdict === 'fail')
      return misfit(block.reason ?? 'no_route', SPONSORED_FIT_BLOCK_CHECK_ID)
    if (block.verdict === 'unknown') unknownChecks.push(SPONSORED_FIT_BLOCK_CHECK_ID)
  }
  for (const check of skill.preflight ?? []) {
    const verdict = evaluateSponsoredFitCheck(check, input.repo)
    if (verdict === 'fail') return misfit(checkMisfitReason(check), check.id)
    if (verdict === 'unknown') unknownChecks.push(check.id)
  }
  return unknownChecks.length ? { fit: 'unknown', unknownChecks } : { fit: 'fits' }
}

export type SponsoredSkillFitRefusal = {
  reason: SponsoredFitMisfitReason
  checkId: string
}

/**
 * Why `skill` must not be offered for this repository and device, or null.
 * Refuses ONLY a known misfit: an unknown verdict serves.
 */
export function sponsoredSkillFitRefusal(
  skill: SponsoredSkill,
  input: SponsoredFitInput,
): SponsoredSkillFitRefusal | null {
  const verdict = evaluateSponsoredSkillFit(skill, input)
  return verdict.fit === 'misfit' && verdict.proof === 'known'
    ? { reason: verdict.reason, checkId: verdict.checkId }
    : null
}

/**
 * The repository's ecosystems, for logs: which manifests exist, never what
 * they declare. `root` counts only manifests at the repository root.
 */
export function sponsoredRepoFactsSummary(facts: SponsoredRepoFacts | null | undefined): {
  ecosystems: string[]
  rootEcosystems: string[]
  complete: boolean | null
  empty: boolean | null
  monorepo: boolean | null
} {
  if (!facts)
    return { ecosystems: [], rootEcosystems: [], complete: null, empty: null, monorepo: null }
  const ecosystemOf = (basename: string): string | null => {
    if (basename === 'package.json' || basename === 'deno.json' || basename === 'deno.jsonc')
      return basename === 'package.json' ? 'node' : 'deno'
    if (manifestEcosystem(basename) === 'pypi') return 'python'
    if (basename === 'go.mod') return 'go'
    if (basename === 'Cargo.toml') return 'rust'
    if (basename === 'composer.json') return 'php'
    if (basename === 'Gemfile') return 'ruby'
    if (basename === 'pom.xml' || basename.startsWith('build.gradle')) return 'java'
    return null
  }
  const all = new Set<string>()
  const root = new Set<string>()
  let packageJsons = 0
  for (const path of facts.paths) {
    const parts = path.split('/')
    const ecosystem = ecosystemOf(parts.at(-1) ?? '')
    if (!ecosystem) continue
    all.add(ecosystem)
    if (parts.length === 1) root.add(ecosystem)
    if (parts.at(-1) === 'package.json') packageJsons++
  }
  return {
    ecosystems: [...all].sort(),
    rootEcosystems: [...root].sort(),
    complete: facts.complete,
    empty: facts.complete ? facts.paths.length === 0 : null,
    monorepo:
      packageJsons > 1 ||
      facts.paths.includes('pnpm-workspace.yaml') ||
      (all.size > 1 && root.size < all.size),
  }
}

 * evaluator can read ({@link sponsoredStaticCheckContext}):
 * - each listed manifest, rendered from its dependency names (package.json
 *   as a `dependencies` object, a Python manifest one name per line, as

 * - and NOTHING the facts do not list.

 * (`runSponsoredSkillPreflightOverFacts` in `./sponsored-step-driver`): a
 * check that needs a file or a dependency the facts do not show fails. Only
 * a caller that has decided to hold the facts to that should use it; the

export function sponsoredRepoFactsFiles(

  for (const path of facts.paths) {

    const ecosystem = manifest?.ecosystem ?? manifestEcosystem(basename)
    if (ecosystem === 'npm') {
      const dependencies = Object.fromEntries(

      files[path] = manifest ? sponsoredManifestProjection(manifest) : ''
    } else {
      files[path] = facts.empty?.includes(path) ? '' : LISTED_FILE_PLACEHOLDER

}

/**
 * The step driver: the harness, not the model, decides when a sponsored
 * skill's steps are done (piece 3 of COD-822; the format is
 * `docs/ads/agentic/sponsored-skills.md`).
 *
 * After the run's turn has done its work, the driver walks the skill's steps
 * in order and runs each step's checks through THE skill evaluator
 * (`evaluateSponsoredSkillChecks`), the same one the sponsored eval scores
 * with. A step with no failing check is done, whatever the model said. A
 * failing `agent` or `check` step goes back to the agent with the failed
 * checks (`sponsoredSkillCheckFeedback`), to fix and stop; the checks run
 * again after each fix. The step's `maxAttempts` counts the first try (the
 * turn's own work) plus fixes, and the run as a whole gets at most
 * {@link SPONSORED_STEP_MAX_FIXES_PER_RUN}. Then the step is BLOCKED, and the
 * walk ends there: later steps usually build on it, and prompting them would
 * spend the run's compute on work that cannot pass. They are still checked
 * once, so the result says what is true on disk.
 *
 * - A `user` step is never prompted or retried. It is done only once its
 *   checks pass (the user handed over the credential), else it needs the user.
 * - A step that passed before a later fix is checked again at the end: a fix
 *   for step 4 can break step 2.
 *
 * DELIVERED means every `agent` and `check` step passed. That is the run's
 * completion, not the model writing "Done:".
 *
 * Pure: the caller hands in the check context (its worktree, its sandbox) and
 * how to prompt the agent, so the whole loop runs in tests without a model,
 * a sandbox or a disk.
 */

/**

 * the text is exactly that skill's rendering: this build's registry first,

async function check(
  deps: Pick<SponsoredStepDriverDeps, 'ctx' | 'skill'>,
  step: Pick<SponsoredSkillStep, 'checks'>,
): Promise<{ passed: boolean; results: SponsoredSkillCheckResult[] }> {
  try {
    return await evaluateSponsoredSkillChecks(step.checks, deps.ctx, deps.skill)
  } catch (error) {
    // The evaluator never throws for a check's own failure; this is the
    // context failing (a disk error). A failure, not a pass.
    return {
      passed: false,
      results: [
        {
          id: 'check-error',
          label: 'Run the checks',
          status: 'fail',
          detail: error instanceof Error ? error.message : String(error),
        },
      ],
    }
  }
}

/** Walk the steps, check each, and have the agent fix what fails. */
export async function driveSponsoredSkillSteps(
  deps: SponsoredStepDriverDeps,
): Promise<SponsoredStepDriveResult> {
  const maxPerRun = deps.maxFixesPerRun ?? SPONSORED_STEP_MAX_FIXES_PER_RUN
  const total = deps.skill.steps.length
  const results: SponsoredStepResult[] = []
  // The fix count at which each step last passed: a later fix may undo it.
  const passedAt = new Map<number, number>()
  let fixes = 0
  let walking = true
  let stoppedBecause: SponsoredStepDriveResult['stoppedBecause']

  for (const [index, step] of deps.skill.steps.entries()) {
    const number = index + 1
    const base = { number, id: step.id, title: step.title, kind: step.kind }
    try {
      deps.onStep?.({ number, total, title: step.title })
    } catch {
      // Progress is display only; it never stops the walk.
    }
    let outcome = await check(deps, step)

    if (step.kind === 'user') {
      results.push({
        ...base,
        status: outcome.passed ? 'passed' : 'needs_user',
        fixes: 0,
        checks: outcome.results,
      })
      continue
    }

    const walkingAtEntry = walking
    // The turn's own work was the first attempt.
    const maxFixes = (step.maxAttempts ?? DEFAULT_SKILL_MAX_ATTEMPTS) - 1
    let stepFixes = 0
    while (walking && !outcome.passed) {
      if (stepFixes >= maxFixes || fixes >= maxPerRun) break
      if (!deps.canContinue()) {
        walking = false
        stoppedBecause = 'out_of_time'
        break
      }
      stepFixes += 1
      fixes += 1
      const completed = await deps.prompt(
        sponsoredStepFixPrompt(
          step,
          number,
          total,
          outcome.results,
          stepFixes,
          maxFixes,
        ),
      )
      if (!completed) {
        walking = false
        stoppedBecause = 'turn_failed'
      }
      outcome = await check(deps, step)
    }

    if (outcome.passed) passedAt.set(number, fixes)
    // Blocked: the driver tried, or would have but the fix budget is spent.
    // Not reached: the walk had already stopped (or ran out of time) first.
    const blocked = stepFixes > 0 || (walkingAtEntry && walking)
    results.push({
      ...base,
      status: outcome.passed ? 'passed' : blocked ? 'blocked' : 'not_reached',
      fixes: stepFixes,
      checks: outcome.results,
    })
    if (!outcome.passed) walking = false
  }

  for (const result of results) {
    const at = passedAt.get(result.number)
    if (at === undefined || at === fixes) continue
    const again = await check(deps, deps.skill.steps[result.number - 1]!)
    result.checks = again.results
    if (!again.passed) result.status = 'blocked'
  }

  return {
    skill: skillKey(deps.skill),
    steps: results,
    delivered: results.every(
      (result) => result.kind === 'user' || result.status === 'passed',
    ),
    fixes,
    ...(stoppedBecause ? { stoppedBecause } : {}),
  }
}

export type SponsoredStepLoopDeps<State> = {
  skill: SponsoredSkill
  ctx: SponsoredSkillCheckContext
  /** One more segment of the run's turn; resolves null when it did not complete. */
  resume: (prompt: string, previous: State) => Promise<State | null>
  /** False once the run is aborted or out of time. */
  canContinue: () => boolean
  /** Takes the run's credentials out of command output before the agent reads it. */
  redact?: (text: string) => string
  /** A `report_friction` blocker, sent for each step left blocked. */
  onFriction?: (input: SponsoredFrictionInput) => void
  /** Persist checked completion before the optional final report segment. */
  onChecksComplete?: (result: SponsoredStepDriveResult) => void
  /** Progress only (see {@link SponsoredStepDriverDeps.onStep}). */
  onStep?: (step: SponsoredStepProgress) => void
}

/**
 * The driver run inside a turn, for any runtime (Desktop's harness, the Cloud
 * runner): fixes are more segments of the SAME run (same transcript), and the
 * state it resolves is the last segment that COMPLETED, so a fix that errored
 * leaves the turn where it was rather than failing work already on disk.
 */
export async function runSponsoredStepLoop<State>(
  initial: State,
  deps: SponsoredStepLoopDeps<State>,
): Promise<{ state: State; result: SponsoredStepDriveResult }> {
  const redact = deps.redact ?? identity
  let state = initial

  const result = await driveSponsoredSkillSteps({
    skill: deps.skill,
    ctx: redactedCheckContext(deps.ctx, redact),
    canContinue: deps.canContinue,
    ...(deps.onStep ? { onStep: deps.onStep } : {}),
    prompt: async (text) => {
      const next = await deps.resume(text, state)
      if (next === null) return false
      state = next
      return true
    },
  })

  deps.onChecksComplete?.(result)

  // The report the user reads must match the checks, not the model's memory
  // of what it tried. Only after a fix: otherwise the first segment's report
  // already described the work the checks then confirmed.
  if (result.fixes > 0 && deps.canContinue()) {
    const next = await deps.resume(sponsoredStepReportPrompt(result), state)
    if (next !== null) state = next
  }

  for (const step of result.steps) {
    if (step.kind === 'user' || step.status !== 'blocked') continue
    deps.onFriction?.({
      blocking: true,
      category: step.checks.some(
        (check) =>
          check.status === 'fail' &&
          /\bexited\b|could not run/.test(check.detail),
      )
        ? 'command_failed'
        : 'other',
      detail: `Step check: step ${step.number} still failed its checks after ${step.fixes} fix${
        step.fixes === 1 ? '' : 'es'
      } by the agent, so Freebuff marked it blocked.`,
      step: step.number,
    })
  }
  return { state, result }
}

const identity = (text: string) => text

/** Command output reaches the agent and the report (COD-665): redacted first. */
function redactedCheckContext(
  ctx: SponsoredSkillCheckContext,
  redact: (text: string) => string,
): SponsoredSkillCheckContext {
  return {
    ...ctx,
    runScript: async (script, timeout) => {
      const out = await ctx.runScript(script, timeout)
      return out === 'missing' ? out : { ...out, output: redact(out.output) }
    },
    runCommand: async (argv, timeout) => {
      const out = await ctx.runCommand(argv, timeout)
      return { ...out, output: redact(out.output) }
    },
  }
}

export type SponsoredSkillPreflightOutcome =
  | { passed: true; checks: SponsoredSkillCheckResult[] }
  | {
      passed: false
      checks: SponsoredSkillCheckResult[]
      /** Names each failed preflight (and a missing path tool's install hint): the run's error. */
      reason: string
      /** The run's step result: no step reached, stopped by the preflight. */
      result: SponsoredStepDriveResult
    }

const PREFLIGHT_DETAIL_CHARS = 200

/** A failed preflight check, as the run's error and friction name it. */
function preflightFailureLine(
  skill: SponsoredSkill,
  failed: SponsoredSkillCheckResult,
): string {
  // One line: the runs' diagnostics keep only an error's first line.
  const oneLine = failed.detail.replace(/\s+/g, ' ').trim()
  const detail =
    oneLine.length > PREFLIGHT_DETAIL_CHARS
      ? `${oneLine.slice(0, PREFLIGHT_DETAIL_CHARS - 1)}…`
      : oneLine
  const declared = sponsoredSkillPreflightChecks(skill).find(
    (
      check,
    ): check is Extract<SponsoredSkillCheck, { kind: 'command_succeeds' }> =>
      check.id === failed.id && check.kind === 'command_succeeds',
  )
  const tool = declared
    ? (skill.tools ?? []).find((candidate) => candidate.name === declared.argv[0])
    : undefined
  const hint =
    tool?.source === 'path'
      ? ` Install \`${tool.name}\`: ${tool.installHint}`
      : ''
  return `"${failed.label}" (${detail}).${hint}`
}

/**
 * THE PREFLIGHT, run before the agent's first turn: the skill's
 * `requirements.os` against the platform the checks run on, then its
 * `preflight` checks through the same evaluator as the steps. On a failure no
 * agent turn runs and nothing is edited: the caller ends the run with
 * `reason`, and it is reported the way a blocked step is (the
 * `report_friction` blocker is sent here; `result` is the step report, every
 * step not reached). An unsupported OS runs no command: the skill's tool
 * check says nothing more on a platform the skill does not support.
 */
export async function runSponsoredSkillPreflight(
  deps: SponsoredSkillPreflightDeps,
): Promise<SponsoredSkillPreflightOutcome> {
  const { skill } = deps
  const checks: SponsoredSkillCheckResult[] = []
  const os = sponsoredSkillOs(deps.platform)
  const supported = skill.requirements?.os ?? []
  if (os && supported.length > 0) {
    const ok = supported.includes(os)
    checks.push({
      id: SPONSORED_PREFLIGHT_OS_CHECK_ID,
      label: `Runs on ${supported.join(' or ')}`,
      status: ok ? 'pass' : 'fail',
      detail: ok
        ? `this run is on ${os}`
        : `this skill supports ${supported.join(', ')}; this run is on ${os}`,
    })
  }
  if (checks.every((result) => result.status !== 'fail')) {
    const outcome = await check(
      { skill, ctx: redactedCheckContext(deps.ctx, deps.redact ?? identity) },
      { checks: sponsoredSkillPreflightChecks(skill) },
    )
    checks.push(...outcome.results)
  }
  const failed = checks.filter((result) => result.status === 'fail')
  if (failed.length === 0) return { passed: true, checks }

  const reason = `Preflight failed before any edit: ${failed
    .map((result) => preflightFailureLine(skill, result))
    .join(' ')}`
  // A wrong OS or a file/package check is the project not fitting the skill;
  // only a command or script check is a missing toolchain.
  const declaredChecks = sponsoredSkillPreflightChecks(skill)
  const toolchainOnly = failed.every((result) => {
    const declared = declaredChecks.find((c) => c.id === result.id)
    return (
      declared?.kind === 'command_succeeds' ||
      declared?.kind === 'script_passes'
    )
  })
  deps.onFriction?.({
    blocking: true,
    category: toolchainOnly ? 'missing_toolchain' : 'project_mismatch',
    // The server keeps the first SPONSORED_FRICTION_DETAIL_MAX characters.
    detail: reason.slice(0, SPONSORED_FRICTION_DETAIL_MAX),
  })
  return {
    passed: false,
    checks,
    reason,
    result: {
      skill: skillKey(skill),
      steps: skill.steps.map((step, index) => ({
        number: index + 1,
        id: step.id,
        title: step.title,
        kind: step.kind,
        status: 'not_reached',
        fixes: 0,
        checks: [],
      })),
      delivered: false,
      fixes: 0,
      stoppedBecause: 'preflight_failed',
      preflight: checks,
    },
  }
}

/** The run error's opening, as {@link runSponsoredSkillPreflight} writes it. */
export const SPONSORED_PREFLIGHT_FAILED_PREFIX = 'Preflight failed before any edit:'

/**
 * What a preflight refusal was, read back from its run error (also its
 * SCRUBBED funnel copy, `scrubSponsoredDiagnostic`, which keeps these
 * phrases): `os` when the skill does not support the OS, `fit` when a
 * file or package check refused the project (the skill does not fit it),
 * `command` when only a command or script check failed (a missing
 * toolchain). Null for anything that is not a preflight refusal. The phrases
 * are the evaluator's own details (`sponsored-skill-checks.ts`).
 */
export function sponsoredPreflightRefusalKind(
  text: string | null | undefined,
): 'os' | 'fit' | 'command' | null {
  if (!text) return null
  const lower = text.toLowerCase()
  const at = lower.indexOf(SPONSORED_PREFLIGHT_FAILED_PREFIX.toLowerCase())
  if (at < 0) return null
  const body = lower.slice(at)
  if (/this skill supports [^;]*; this run is on/.test(body)) return 'os'
  if (
    /no file matches |no (?:changed )?file matching .*? contains |no (?:changed )?file matches |, which must not appear|package\.json is missing|package\.json does not declare|declares [a-z0-9_]+=/.test(
      body,
    )
  )
    return 'fit'
  if (/could not run |exited -?(?:\d+|n)\b|has no [a-z]+ script/.test(body))
    return 'command'
  // A preflight refusal whose detail was cut: the project did not pass a
  // check before any edit, which is a fit question first.
  return 'fit'
}

}

+ * - each listed manifest, rendered from its dependency names (package.json
+ *   as a `dependencies` object, a Python manifest one name per line, as
+ *   {@link sponsoredManifestProjection}); a listed manifest with no names is
+ *   an empty one (`{}`, or nothing);
+ * - an empty marker empty, and any other listed file a non-blank line;
+ * - and NOTHING the facts do not list.
+ *
+ * That last point is an assumption about what the facts leave out, and it is
+ * what lets the run's own preflight answer over facts
+ * (`runSponsoredSkillPreflightOverFacts` in `./sponsored-step-driver`): a
+ * check that needs a file or a dependency the facts do not show fails. Only
+ * a caller that has decided to hold the facts to that should use it; the
+ * offer route does it for a Desktop V1 snapshot (docs/freebuff-agentic-ads.md,
+ * "Skill fit on the offer route").
+ */
+export function sponsoredRepoFactsFiles(
+  facts: SponsoredRepoFacts,
+): Record<string, string> {
+  const files: Record<string, string> = {}
+  for (const path of facts.paths) {
+    const basename = path.split('/').at(-1) ?? ''
+    const manifest = manifestAt(facts, path)
+    const ecosystem = manifest?.ecosystem ?? manifestEcosystem(basename)
+    if (ecosystem === 'npm') {
+      const dependencies = Object.fromEntries(
+        (manifest?.names ?? []).map((name) => [name, '*']),
+      )
+      files[path] = `${JSON.stringify({ dependencies }, null, 2)}\n`
+    } else if (ecosystem === 'pypi') {
+      files[path] = manifest ? sponsoredManifestProjection(manifest) : ''
+    } else {
+      files[path] = facts.empty?.includes(path) ? '' : LISTED_FILE_PLACEHOLDER
+    }
+  }
+  return files
+}
