export const DEV_SERVER_STARTUP_GUIDANCE = `Dev-server startup: inspect package.json (engines, packageManager, scripts), the lockfile, and any script wrapper or execution-profile selector before choosing a command. Follow the project's install/bootstrap procedure and preserve the lockfile. Keep installation logs and the real exit status; piping an install into tail can hide failure unless pipefail is enabled.

Check the installed CLI's --help for bind and port flags instead of guessing from the underlying framework: Vite uses --host, while Vinext uses --hostname even though it runs Vite internally. A wrapper may override or ignore flags. Do not switch execution profiles merely to change the bind address: profiles can also change authentication, storage bindings, and build behavior.

Reuse an existing healthy server. Otherwise start a persistent process with stdin detached and stdout/stderr redirected to a log, and record its PID. Inspect the actual listener address and port, then poll HTTP with a bounded timeout and inspect the response and logs; a startup banner or an open port alone does not prove readiness. If startup fails, diagnose the first useful error before retrying. Stop only the identified server PID/process tree after checking ownership: kill the recorded PID, or its children with pkill -P <pid>. Never stop servers by name pattern (pkill -f, pgrep -f | xargs kill, killall): your own agent process's command line can contain the same words, so a pattern can terminate you mid-turn, along with the cleanup shell or unrelated processes.`

/**
 * Parse ONE argument, keeping its name on the issue.
 *
 * `schema.parse(args.name)` produces an issue with an empty path, so a refusal
 * could only ever name the failing CHECK and not the argument — the half of
 * this that a model cannot work around.
 */
function field<T>(schema: z.ZodType<T>, argName: string, value: unknown): T {
  return z.object({ [argName]: schema }).parse({ [argName]: value })[
    argName
  ] as T
}

/**
 * Name the arguments a Zod check rejected, and nothing else.
 *
 * The opaque fallback below exists because a filesystem or validator error may
 * carry file contents or a secret value. A Zod issue cannot: this reports only
 * the argument's path and the kind of check that failed, never the value that
 * failed it and never the issue's own message (`invalid_value` carries the
 * received literal). Without this an argument the model can fix reads as
 * "check arguments and workspace files", so it retries the same call.
 */
function rejectedArguments(error: z.ZodError): string {
  const named = [
    ...new Set(
      error.issues.map((issue) => {
        const path = issue.path.map(String).join('.')
        return path ? `${path} (${issue.code})` : issue.code
      }),
    ),
  ]
  return named.length > 0
    ? `Sites rejected these arguments: ${named.join(', ')}. Read the tool's input schema and send values that match it.`
    : 'Could not prepare the Sites request. Check arguments and workspace files.'
}

async function readBounded(path: string, limit: number): Promise<Buffer> {
  const file = await open(path, 'r')
  try {
    const stat = await file.stat()
    if (!stat.isFile() || stat.size > limit)
      throw invalid(
        'Site file exceeds the upload limit or is not a regular file.',
      )
    const buffer = Buffer.alloc(Math.min(stat.size + 1, limit + 1))
    let used = 0
    while (used < buffer.length) {
      const { bytesRead } = await file.read(
        buffer,
        used,
        buffer.length - used,
        null,
      )
      if (!bytesRead) break
      used += bytesRead
    }
    if (used > limit || used > stat.size)
      throw invalid('Site file changed during upload. Rebuild and retry.')
    return buffer.subarray(0, used)
  } finally {
    await file.close()
  }
}

export async function bundleSite(cwd: string, manifestPath: string) {
  const root = await realpath(cwd)
  const path = await localPath(root, manifestPath)
  const parsed = manifestSchema.safeParse(
    JSON.parse((await readBounded(path, 512 * 1024)).toString('utf8')),
  )
  if (!parsed.success)
    throw invalid(
      'Invalid deploy manifest. entrypoint must match an ESM module name (not its local path). Supply compatibilityDate and modules with name/type/path; optional assetsDirectory and spa.',
    )
  const manifest = parsed.data
  let remaining = (manifest.runtime === 'opennext' ? 32 : 10) * 1024 * 1024
  const read = async (file: string) => {
    const data = await readBounded(file, remaining)
    remaining -= data.length
    return data.toString('base64')
  }
  const modules = []
  for (const module of manifest.modules) {
    const file = await localPath(root, resolve(dirname(path), module.path))
    if (
      relative(root, file).split(/[\\/]/).some(unsafeAsset) ||
      !['.js', '.mjs', '.txt', '.json', '.html', '.wasm'].includes(
        extname(file),
      )
    )
      throw invalid(
        'Only built JS, text, JSON and WASM modules may be uploaded.',
      )
    modules.push({
      name: module.name,
      type: module.type,
      contentBase64: await read(file),
    })
  }
  let assets: { path: string; contentBase64: string }[] | undefined
  if (manifest.assetsDirectory !== undefined) {
    const directory = await localPath(
      root,
      resolve(dirname(path), manifest.assetsDirectory),
    )
    if (relative(root, directory).split(/[\\/]/).some(unsafeAsset))
      throw invalid(
        'Use a dedicated public build directory, not a hidden or private directory.',
      )
    assets = []
    const walk = async (
      folder: string,
      prefix: string,
      depth = 0,
    ): Promise<void> => {
      if (depth > 20) throw invalid('Static assets are nested too deeply.')
      // Name order, like Cloud's bundler (`sorted(os.listdir())` in
      // web/src/server/desktop-cloud/sites-files.ts): readdir's order is the
      // filesystem's, so the same build could otherwise bundle differently.
      const entries = (await readdir(folder, { withFileTypes: true })).sort(
        (a, b) => (a.name < b.name ? -1 : a.name > b.name ? 1 : 0),
      )
      for (const entry of entries) {
        if (unsafeAsset(entry.name) || entry.isSymbolicLink())
          throw invalid(
            'Assets must contain only public build output: no hidden files, source maps, private keys, dependencies, or symlinks.',
          )
        const file = await localPath(root, resolve(folder, entry.name))
        const assetPath = `${prefix}/${entry.name}`
        if (entry.isDirectory()) await walk(file, assetPath, depth + 1)
        else {
          if (assets!.length >= 1000)
            throw invalid('A deployment may have at most 1,000 assets.')
          assets!.push({ path: assetPath, contentBase64: await read(file) })
        }
      }
    }
    await walk(directory, '')
  }
  const cache = manifest.cache
    ? ([] as { key: string; contentBase64: string }[])
    : undefined
  for (const entry of manifest.cache ?? []) {
    const file = await localPath(root, resolve(dirname(path), entry.path))
    if (
      relative(root, file).split(/[\\/]/).some(unsafeAsset) ||
      extname(file) !== '.json'
    )
      throw invalid(
        'OpenNext cache entries must be dedicated private build JSON files.',
      )
    cache!.push({ key: entry.key, contentBase64: await read(file) })
  }
  return {
    ...(manifest.runtime ? { runtime: manifest.runtime } : {}),
    entrypoint: manifest.entrypoint,
    compatibilityDate: manifest.compatibilityDate,
    compatibilityFlags: manifest.compatibilityFlags ?? [],
    modules,
    ...(assets !== undefined ? { assets } : {}),
    ...(cache !== undefined ? { cache } : {}),
    spa: manifest.spa ?? false,
  }
}

export type BoatBrowserSize = { width: number; height: number }

// Chrome's window minimum is 500px. CVT modes need widths divisible by 8;
// bounded CSS-pixel sizes keep the software encoder's work independent of DPR.
export function boatBrowserSize(
  width: number,
  height: number,
): BoatBrowserSize | null {
  if (
    !Number.isFinite(width) ||
    !Number.isFinite(height) ||
    width <= 0 ||
    height <= 0
  )
    return null
  const scale = Math.min(
    Math.max(1, 512 / width, 128 / height),
    2560 / width,
    2160 / height,
  )
  return {
    width: Math.max(512, Math.min(2560, Math.round((width * scale) / 8) * 8)),
    height: Math.max(128, Math.min(2160, Math.round((height * scale) / 8) * 8)),
  }
}

export function isBoatBrowserSize(value: unknown): value is BoatBrowserSize {
  if (!value || typeof value !== 'object') return false
  const v = value as BoatBrowserSize
  return (
    Object.keys(v).length === 2 &&
    Number.isInteger(v.width) &&
    v.width >= 512 &&
    v.width <= 2560 &&
    v.width % 8 === 0 &&
    Number.isInteger(v.height) &&
    v.height >= 128 &&
    v.height <= 2160 &&
    v.height % 8 === 0
  )
}

export function sameBoatBrowserSize(
  a: BoatBrowserSize | null,
  b: BoatBrowserSize | null,
) {
  return a?.width === b?.width && a?.height === b?.height
}

+export type BoatBrowserSize = { width: number; height: number }
+
+// Chrome's window minimum is 500px. CVT modes need widths divisible by 8;
+// bounded CSS-pixel sizes keep the software encoder's work independent of DPR.
+export function boatBrowserSize(
+  width: number,
+  height: number,
+): BoatBrowserSize | null {
+  if (
+    !Number.isFinite(width) ||
+    !Number.isFinite(height) ||
+    width <= 0 ||
+    height <= 0
+  )
+    return null
+  const scale = Math.min(
+    Math.max(1, 512 / width, 128 / height),
+    2560 / width,
+    2160 / height,
+  )
+  return {
+    width: Math.max(512, Math.min(2560, Math.round((width * scale) / 8) * 8)),
+    height: Math.max(128, Math.min(2160, Math.round((height * scale) / 8) * 8)),
+  }
+}
+
+export function isBoatBrowserSize(value: unknown): value is BoatBrowserSize {
+  if (!value || typeof value !== 'object') return false
+  const v = value as BoatBrowserSize
+  return (
+    Object.keys(v).length === 2 &&
+    Number.isInteger(v.width) &&
+    v.width >= 512 &&
+    v.width <= 2560 &&
+    v.width % 8 === 0 &&
+    Number.isInteger(v.height) &&
+    v.height >= 128 &&
+    v.height <= 2160 &&
+    v.height % 8 === 0
+  )
+}

+/** Only fixed package-manager commands enter cmd.exe; project paths never do. */
+export function nextInstallInvocation(manager: string, platform: string) {
+  if (!['npm', 'pnpm', 'yarn', 'bun'].includes(manager))
+    throw new Error('Unsupported package manager')
+  const packages = [
+    `@opennextjs/cloudflare@${NEXT_ADAPTER_VERSION}`,
+    `wrangler@${NEXT_WRANGLER_VERSION}`,
+  ]
+  const args =
+    manager === 'npm'
+      ? ['install', '--save-dev', '--save-exact', ...packages]
+      : ['yarn', 'bun'].includes(manager)
+        ? ['add', '--dev', '--exact', ...packages]
+        : ['add', '--save-dev', '--save-exact', ...packages]
+  if (platform !== 'win32' || manager === 'bun')
+    return { command: manager, args, windowsVerbatimArguments: false }
+  // Node cannot launch npm/pnpm/yarn's .cmd shims directly. These allowlisted
+  // words contain no whitespace or shell metacharacters, so no customer data
+  // participates in the command line parsed by cmd.exe.
+  return {
+    command: 'cmd.exe',
+    args: ['/d', '/s', '/c', `${manager}.cmd ${args.join(' ')}`],
+    windowsVerbatimArguments: true,
+  }
+}

+export function parseNextPreparationOutput(
+  stdout: string,
+): NextPreparationResult {
+  const line = stdout
+    .split('\n')
+    .findLast((line) => line.startsWith(NEXT_RESULT_PREFIX))
+  if (!line)
+    throw new Error('Next.js preparation did not return a build manifest.')
+  const value = JSON.parse(
+    line.slice(NEXT_RESULT_PREFIX.length),
+  ) as NextPreparationResult
+  if (
+    value.runtime !== 'opennext' ||
+    ![
+      'manifestPath',
+      'localWranglerConfigPath',
+      'logPath',
+      'nextVersion',
+      'adapterVersion',
+    ].every(
+      (key) => typeof value[key as keyof NextPreparationResult] === 'string',
+    )
+  )
+    throw new Error('Invalid Next.js preparation result.')
+  return value
+}

+/** Only the bounded, program-generated message belongs in a tool response. */
+export function parseNextPreparationError(output: string): string | undefined {
+  return output
+    .split('\n')
+    .findLast((line) => line.startsWith(NEXT_ERROR_PREFIX))
+    ?.slice(NEXT_ERROR_PREFIX.length, NEXT_ERROR_PREFIX.length + 4096)
+    .trim()
+}

+/**
+ * Exactly the same credential-free preparation program runs in a local workspace
+ * and in a Cloud VM. Child build output goes to a private local log, never into
+ * the tool response. No Cloudflare API or account credentials are used here.
+ */
+export function nextPreparationScript(
+  options: { projectDirectory?: string } = {},
+): string {
+  const installCommands = Object.fromEntries(
+    ['win32', 'posix'].map((platform) => [
+      platform,
+      Object.fromEntries(
+        ['npm', 'pnpm', 'yarn', 'bun'].map((manager) => [
+          manager,
+          nextInstallInvocation(manager, platform),
+        ]),
+      ),
+    ]),
+  )
+  return (
+    `const input = ${JSON.stringify({ projectDirectory: options.projectDirectory ?? '.', adapterVersion: NEXT_ADAPTER_VERSION, wranglerVersion: NEXT_WRANGLER_VERSION, installCommands })};\n` +
+    NEXT_PREPARATION_PROGRAM
+  )
+}

+export async function prepareNextSite(
+  cwd: string,
+  projectDirectory = '.',
+  signal?: AbortSignal,
+): Promise<NextPreparationResult> {
+  const stdout = await new Promise<string>((resolve, reject) => {
+    const buildSignal = AbortSignal.any([
+      AbortSignal.timeout(15 * 60 * 1000),
+      ...(signal ? [signal] : []),
+    ])
+    const child = spawn(
+      'node',
+      [
+        '--input-type=module',
+        '-e',
+        nextPreparationScript({ projectDirectory }),
+      ],
+      {
+        cwd,
+        signal: buildSignal,
+        detached: process.platform !== 'win32',
+        env: nextPreparationEnvironment(process.env),
+        stdio: ['ignore', 'pipe', 'pipe'],
+      },
+    )
+    const stop = () => {
+      if (process.platform !== 'win32' && child.pid) {
+        try {
+          process.kill(-child.pid, 'SIGTERM')
+        } catch {
+          /* Already exited. */
+        }
+      }
+    }
+    buildSignal.addEventListener('abort', stop, { once: true })
+    let output = ''
+    let errorOutput = ''
+    child.stdout.on('data', (data: Buffer) => {
+      output = (output + data.toString()).slice(-64 * 1024)
+    })
+    child.stderr.on('data', (data: Buffer) => {
+      errorOutput = (errorOutput + data.toString()).slice(-4096)
+    })
+    child.once('error', () =>
+      reject(
+        new SitesClientError(
+          400,
+          'next_prepare_failed',
+          buildSignal.aborted
+            ? 'Next.js preparation was cancelled or exceeded its 15-minute limit.'
+            : 'Next.js preparation needs Node.js 20.19 or newer in the workspace PATH.',
+        ),
+      ),
+    )
+    child.once('close', (code) => {
+      buildSignal.removeEventListener('abort', stop)
+      code === 0
+        ? resolve(output)
+        : reject(
+            new SitesClientError(
+              400,
+              'next_prepare_failed',
+              parseNextPreparationError(errorOutput) ??
+                'Next.js preparation failed. Inspect freebuff-sites-build/build.log in the project.',
+            ),
+          )
+    })
+  })
+  return parseNextPreparationOutput(stdout)
+}

+export const SITE_PUBLISH_HEARTBEAT_MS = 15_000

+  type: 'error',

+/** One NDJSON line per event; the stream ends after `done` or `error`. */
+export function sitePublishResponse(
+  run: (
+    onEvent: (event: SitePublishEvent) => void,
+    signal: AbortSignal,
+  ) => Promise<unknown>,
+  signal: AbortSignal,
+): Response {
+  const encoder = new TextEncoder()
+  const cancelled = new AbortController()
+  const stop = AbortSignal.any([signal, cancelled.signal])
+  const stream = new ReadableStream<Uint8Array>({
+    start(sink) {
+      let open = true
+      const write = (event: SitePublishEvent) => {
+        if (!open) return
+        try {
+          sink.enqueue(encoder.encode(JSON.stringify(event) + '\n'))
+        } catch {
+          open = false
+        }
+      }
+      const beat = setInterval(
+        () => write({ type: 'heartbeat' }),
+        SITE_PUBLISH_HEARTBEAT_MS,
+      )
+      void run(write, stop)
+        .catch(() =>
+          write({ type: 'error', message: 'Publishing failed. Try again.' }),
+        )
+        .finally(() => {
+          clearInterval(beat)
+          if (!open) return
+          open = false
+          try {
+            sink.close()
+          } catch {
+            /* Already cancelled. */
+          }
+        })
+    },
+    cancel() {
+      cancelled.abort()
+    },
+  })
+  return new Response(stream, {
+    headers: {
+      'Content-Type': 'application/x-ndjson',
+      'Cache-Control': 'private, no-store',
+      'X-Accel-Buffering': 'no',
+    },
+  })
+}

+function parseEvent(line: string): SitePublishEvent | null {
+  try {
+    const value = JSON.parse(line) as SitePublishEvent
+    if (value.type === 'stage' && value.stage in SITE_PUBLISH_STAGE_LABELS)
+      return value
+    if (
+      value.type === 'done' &&
+      typeof value.siteId === 'string' &&
+      typeof value.url === 'string' &&
+      ['preview', 'production'].includes(value.environment)
+    )
+      return value
+    if (value.type === 'error' && typeof value.message === 'string')
+      return { type: 'error', message: value.message.slice(0, 4096) }
+    if (value.type === 'heartbeat') return value
+  } catch {
+    /* A partial or foreign line is not an event. */
+  }
+  return null
+}

+/** Reads a publish stream to its result; a refused request is an error result. */
+export async function readSitePublishStream(
+  response: Response,
+  onEvent: (event: SitePublishEvent) => void = () => {},
+): Promise<SitePublishResult> {
+  if (
+    !response.ok ||
+    !response.body ||
+    !response.headers.get('content-type')?.includes('application/x-ndjson')
+  ) {
+    const body = (await response.json().catch(() => null)) as {
+      error?: unknown
+      message?: unknown
+    } | null
+    const message =
+      typeof body?.message === 'string'
+        ? body.message
+        : typeof body?.error === 'string'
+          ? body.error
+          : 'Could not start publishing.'
+    return { type: 'error', message }
+  }
+  const reader = response.body.pipeThrough(new TextDecoderStream()).getReader()
+  let buffer = ''
+  let result: SitePublishResult | null = null
+  try {
+    while (!result) {
+      const { done, value } = await reader.read()
+      if (done) break
+      buffer += value
+      const lines = buffer.split('\n')
+      buffer = lines.pop() ?? ''
+      for (const line of lines) {
+        const event = line.trim() ? parseEvent(line) : null
+        if (!event) continue
+        onEvent(event)
+        if (event.type === 'done' || event.type === 'error') {
+          result = event
+          break
+        }
+      }
+    }
+  } catch {
+    return STOPPED
+  } finally {
+    reader.cancel().catch(() => {})
+  }
+  return result ?? STOPPED
+}

+  siteId?: string
+  siteName?: string

+  errorMessage?: (error: unknown) => string | undefined

+    return event

+    const prepared = await options.prepare()

+    if (prepared.kind === 'next')
+      manifestPath = (
+        await call<{ manifestPath: string }>('sites_prepare_next', {})
+      ).manifestPath
+    if (!manifestPath) return fail('The build did not produce a deployment.')

+    if (!siteId) {
+      siteId = id()

+        'My site'

+    }

+    const inspected = await call<{
+      environment: {
+        url: string
+        unpublished_at?: string | null
+        auth?: unknown
+        bindings?: { name: string; kind: string }[]
+      }
+    }>('sites_inspect', target)
+    const details = inspected.environment
+    if (prepared.usesAuth && !details.auth)
+      await call('sites_enable_auth', target)

+    for (const kind of ['d1', 'r2'] as const)

+          )

+        ...target,
+        requestId,

+      })
