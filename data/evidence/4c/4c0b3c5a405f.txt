/**
 * What a sponsored run's OUTCOME row in `ad_agentic_funnel_event` carries
 * beside it (COD-665): `run_failed`, `run_committed` and `run_delivered`.
 *
 * Before this those rows had `metadata = null`, and the only record of WHY a
 * run failed was `failure_reason` / `diagnostic_reason` on the Convex
 * proposal row -- so the Postgres funnel could count failures and say nothing
 * about them, and every "why did these runs fail" question was a Convex
 * export joined by hand. This is the smallest closed shape that answers it from
 * Postgres alone.
 *
 * TELEMETRY, NEVER MONEY. Nothing reads this to bill, refund or settle: the
 * `accepted` row is the one billable stage and it has its own producer. A
 * test asserts this module imports nothing from billing, and the
 * never-started refund path does not read the stored metadata either.
 *
 * CLOSED AND BOUNDED ON PURPOSE. Scalar facts are enums or booleans; structured
 * check evidence has bounded IDs, step numbers and categories. The diagnostic
 * is SCRUBBED (`scrubSponsoredDiagnostic`)
 * and capped at `SPONSORED_FUNNEL_DIAGNOSTIC_MAX` characters. The full,
 * unscrubbed text stays on the Convex row, which is joined by `proposal_id`
 * when an operator needs it; the funnel is for counting, and a counting table
 * is not a place for a user's paths, prompts or repository text.
 *
 * ONE BUILDER FOR EVERY PRODUCER. The funnel row is insert-once per
 * `(campaign, <type>:<proposalId>)`, so whichever producer wins the race is
 * the row analytics reads. Both producers of these three events -- the
 * Convex outbox and the off-Cloud state route -- take the metadata from
 * `buildSponsoredRunFunnelMetadata`, computed ONCE inside the Convex
 * transition and handed to the route, so the winner is the same row either way.

 * NO ZOD HERE. The wire schemas a receiver validates with are in
 * `./sponsored-run-funnel-metadata`, which re-exports this module. Convex's

/**
 * The step driver: the harness, not the model, decides when a sponsored
 * skill's steps are done (piece 3 of COD-822; the format is
 * `docs/ads/agentic/sponsored-skills.md`).
 *
 * After the run's turn has done its work, the driver walks the skill's steps
 * in order and runs each step's checks through THE skill evaluator
 * (`evaluateSponsoredSkillChecks`), the same one the sponsored eval scores
 * with. A step with no failing check is done, whatever the model said. A
 * failing `agent` or `check` step goes back to the agent with the failed
 * checks (`sponsoredSkillCheckFeedback`), to fix and stop; the checks run
 * again after each fix. The step's `maxAttempts` counts the first try (the
 * turn's own work) plus fixes, and the run as a whole gets at most
 * {@link SPONSORED_STEP_MAX_FIXES_PER_RUN}. Then the step is BLOCKED, and the
 * walk ends there: later steps usually build on it, and prompting them would
 * spend the run's compute on work that cannot pass. They are still checked
 * once, so the result says what is true on disk.
 *
 * - A `user` step is never prompted or retried. It is done only once its
 *   checks pass (the user handed over the credential), else it needs the user.
 * - A step that passed before a later fix is checked again at the end: a fix
 *   for step 4 can break step 2.
 *
 * DELIVERED means every `agent` and `check` step passed. That is the run's
 * completion, not the model writing "Done:".
 *
 * Pure: the caller hands in the check context (its worktree, its sandbox) and
 * how to prompt the agent, so the whole loop runs in tests without a model,
 * a sandbox or a disk.
 */

import type { SponsoredStepDriveResult } from './sponsored-step-driver'
import type { SponsoredTurnEnding } from './sponsored-in-place'

/** Only final human steps remain; setup that precedes implementation is not verification. */
export function sponsoredUserVerificationPending(
  result: SponsoredStepDriveResult,
): boolean {
  const firstPending = result.steps.findIndex(
    (step) => step.status === 'needs_user',
  )
  return (
    result.delivered &&
    !result.stoppedBecause &&
    firstPending > 0 &&
    result.steps.some((step) => step.kind === 'agent') &&
    result.steps.every((step, index) =>
      index < firstPending
        ? step.status === 'passed'
        : step.kind === 'user' &&
          (step.status === 'passed' || step.status === 'needs_user'),
    )
  )
}

/** Closed telemetry only: no paths, labels, command output, or credential values. */
export const SPONSORED_BLOCKER_CATEGORIES = [
  'scope_check',
  'needs_api_key',
  'needs_login',
  'command_failed',
  'verification_failed',
] as const
export type SponsoredBlockerCategory =
  (typeof SPONSORED_BLOCKER_CATEGORIES)[number]
export const SPONSORED_RUN_ENDINGS = [
  'completed',
  'stopped',
  'closed',
  'interrupted',
  'error',
  'grant_expired',
  'runner_restarted',
] as const
export type SponsoredRunEvidence = {
  version: 1
  ending: (typeof SPONSORED_RUN_ENDINGS)[number]
  checks: Array<{
    step: number
    checkId: string
    category: SponsoredBlockerCategory
  }>
  /** At least one evaluator result survived; absent on ending-only reports. */
  checksRecorded?: true
  /** Implementation checks passed; only final human verification remains. Optional for old clients. */
  verificationPending?: true
  truncated?: true
}
export const SPONSORED_EVIDENCE_CHECK_LIMIT = 20

/** Also used by Convex: keep this module free of runtime/zod dependencies. */
export function parseSponsoredRunEvidence(
  value: unknown,
): SponsoredRunEvidence | undefined {
  if (!value || typeof value !== 'object' || Array.isArray(value))
    return undefined
  const v = value as Record<string, unknown>
  if (
    Object.keys(v).some(
      (k) =>
        ![
          'version',
          'ending',
          'checks',
          'checksRecorded',
          'verificationPending',
          'truncated',
        ].includes(k),
    ) ||
    v.version !== 1 ||
    !(SPONSORED_RUN_ENDINGS as readonly unknown[]).includes(v.ending) ||
    (v.verificationPending !== undefined &&
      (v.verificationPending !== true ||
        !['completed', 'closed', 'stopped', 'interrupted'].includes(
          v.ending as string,
        ))) ||
    (v.checksRecorded !== undefined && v.checksRecorded !== true) ||
    (v.truncated !== undefined && v.truncated !== true) ||
    !Array.isArray(v.checks) ||
    v.checks.length > SPONSORED_EVIDENCE_CHECK_LIMIT
  )
    return undefined
  const checks: SponsoredRunEvidence['checks'] = []
  for (const raw of v.checks) {
    if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return undefined
    const c = raw as Record<string, unknown>
    if (
      Object.keys(c).some(
        (k) => !['step', 'checkId', 'category'].includes(k),
      ) ||
      !Number.isInteger(c.step) ||
      (c.step as number) < 0 ||
      (c.step as number) > 100 ||
      typeof c.checkId !== 'string' ||
      !/^[a-zA-Z0-9_-]{1,64}$/.test(c.checkId) ||
      !(SPONSORED_BLOCKER_CATEGORIES as readonly unknown[]).includes(c.category)
    )
      return undefined
    checks.push({
      step: c.step as number,
      checkId: c.checkId,
      category: c.category as SponsoredBlockerCategory,
    })
  }
  return {
    version: 1,
    ending: v.ending as SponsoredRunEvidence['ending'],
    checks,
    ...(v.verificationPending ? { verificationPending: true as const } : {}),
    ...(v.checksRecorded ? { checksRecorded: true } : {}),
    ...(v.truncated ? { truncated: true } : {}),
  }
}

/** Recover structured evidence on idempotent retries without a new proposal column. */
export function sponsoredEvidenceFromReport(
  payload: string | null | undefined,
): SponsoredRunEvidence | undefined {
  try {
    return payload
      ? parseSponsoredRunEvidence(JSON.parse(payload).runEvidence)
      : undefined
  } catch {
    return undefined
  }
}

export function sponsoredRunEvidence(
  ending: SponsoredTurnEnding,
  steps?: SponsoredStepDriveResult | null,
  errorText?: string | null,
): SponsoredRunEvidence {
  const checks: SponsoredRunEvidence['checks'] = []
  for (const step of [
    { number: 0, checks: steps?.preflight ?? [] },
    ...(steps?.steps ?? []),
  ]) {
    for (const check of step.checks) {
      if (check.status !== 'fail') continue
      const category: SponsoredBlockerCategory =
        check.kind === 'changes_within'
          ? 'scope_check'
          : check.kind === 'credential_provided'
            ? 'needs_api_key'
            : check.kind === 'user_confirmed'
              ? 'needs_login'
              : check.kind === 'command_succeeds' ||
                  check.kind === 'script_passes'
                ? 'command_failed'
                : 'verification_failed'
      // Skill ids are bounded telemetry identifiers, never arbitrary output.
      checks.push({
        step: step.number,
        checkId: /^[a-zA-Z0-9_-]{1,64}$/.test(check.id) ? check.id : 'unknown',
        category,
      })
    }
  }
  return {
    version: 1,
    ending: /cloud runner restarted/i.test(errorText ?? '')
      ? 'runner_restarted'
      : ending,
    ...(steps &&
    sponsoredUserVerificationPending(steps) &&
    ['completed', 'closed', 'stopped', 'interrupted'].includes(ending) &&
    !/cloud runner restarted/i.test(errorText ?? '')
      ? { verificationPending: true as const }
      : {}),
    ...(steps?.preflight?.length ||
    steps?.steps.some((step) => step.checks.length)
      ? { checksRecorded: true as const }
      : {}),
    checks: checks.slice(0, SPONSORED_EVIDENCE_CHECK_LIMIT),
    ...(checks.length > SPONSORED_EVIDENCE_CHECK_LIMIT
      ? { truncated: true }
      : {}),
  }
}

import {
  parseSponsoredRunEvidence,
  type SponsoredBlockerCategory,
} from './sponsored-run-evidence'
import { scrubSponsoredDiagnostic } from './sponsored-run-funnel-builder'

export const SPONSORED_BLOCKER_LABELS: Record<
  SponsoredBlockerCategory,
  string
> = {
  scope_check: 'Changes outside the allowed scope failed verification',
  needs_api_key: 'An API key is still required',
  needs_login: 'Account connection or approval is still required',
  command_failed: 'A verification command failed',
  verification_failed: 'A verification check failed',
}

export type SponsoredFailureFriction = {
  blocking: boolean
  category: string
  detail?: string | null
  step?: number | null
}

/** Historical scope reports have no check id. Interpret the category, never publish their text. */
export function sponsoredFrictionBlocker(
  report: SponsoredFailureFriction,
): SponsoredBlockerCategory | undefined {
  if (!report.blocking) return undefined
  if (
    /scope check|scope-check|file-scope|in-scope.files|out-of-scope/i.test(
      report.detail ?? '',
    )
  )
    return 'scope_check'
  if (report.category === 'needs_api_key') return 'needs_api_key'
  if (
    report.category === 'needs_login' ||
    report.category === 'needs_human_action'
  )
    return 'needs_login'
  if (report.category === 'command_failed') return 'command_failed'
  return undefined
}

/** Never infer that the user deliberately cancelled from a generic closed/interrupted event. */
export function sponsoredFailureExplanation(input: {
  metadata: Record<string, unknown> | null
  friction?: readonly SponsoredFailureFriction[]
  assessedCause?: string | null
}): { label: string; detail: string } {
  const m = input.metadata ?? {}
  const evidence = parseSponsoredRunEvidence(m.run_evidence)
  const diagnostic =
    typeof m.diagnostic_reason === 'string' ? m.diagnostic_reason : ''
  const code = typeof m.failure_code === 'string' ? m.failure_code : ''
  const text = diagnostic.toLowerCase()
  const ending = evidence?.ending
  let ended: string
  if (ending === 'runner_restarted' || text.includes('cloud runner restarted'))
    ended = 'The Cloud runner restarted.'
  else if (text.includes('app-quit') || code.includes('app_quit'))
    ended = 'The app quit.'
  else if (text.includes('provider failed'))
    ended = 'The model provider failed.'
  else if (
    /enotfound|econnreset|econnrefused|network request failed|fetch failed|connection timed out|no data was received/.test(
      text,
    )
  )
    ended = 'The run encountered a network error.'
  else if (ending === 'grant_expired' || code === 'grant_expired')
    ended = 'The compute grant expired.'
  else if (code === 'timed_out')
    ended = 'The run stopped reporting and was marked timed out.'
  else if (ending === 'closed' || text.startsWith('turn closed'))
    ended = 'The session closed.'
  else if (ending === 'stopped' || text.startsWith('turn stopped'))
    ended = 'The run was stopped.'
  else if (ending === 'interrupted' || text.startsWith('turn interrupted'))
    ended = 'The run was interrupted; the initiator was not recorded.'
  else if (ending === 'completed' || text.startsWith('turn completed')) {
    ended =
      code === 'no_edits'
        ? 'The agent turn completed without recorded file edits.'
        : code === 'partial_edits'
          ? 'The agent turn completed, but verification did not pass.'
          : 'The agent turn completed, but delivery could not be verified.'
  } else {
    const legacy = m.failure_reason ?? m.reason
    const reason = diagnostic || (typeof legacy === 'string' ? legacy : '')
    ended =
      scrubSponsoredDiagnostic(reason) ??
      'The run failed; no specific reason was recorded.'
  }
  if (evidence?.verificationPending) {
    return {
      label: 'Verification incomplete',
      detail: `Implementation finished, but user verification is incomplete.${ending === 'completed' ? '' : ` ${ended}`}`,
    }
  }
  const blockers: string[] = []
  const categories = new Set<SponsoredBlockerCategory>()
  for (const check of evidence?.checks ?? []) {
    categories.add(check.category)
    blockers.push(
      `${check.step === 0 ? 'Preflight' : `Step ${check.step}`} (${check.checkId}): ${SPONSORED_BLOCKER_LABELS[check.category]}.`,
    )
  }
  // Structured checks are authoritative where present. Old friction is supporting evidence.
  for (const report of evidence?.checksRecorded ? [] : (input.friction ?? [])) {
    const category = sponsoredFrictionBlocker(report)
    if (!category || categories.has(category)) continue
    categories.add(category)
    blockers.push(`Agent reported: ${SPONSORED_BLOCKER_LABELS[category]}.`)
  }
  if (
    !evidence?.checksRecorded &&
    !categories.size &&
    (input.assessedCause === 'needs_api_key' ||
      input.assessedCause === 'needs_login' ||
      input.assessedCause === 'needs_human_action')
  ) {
    blockers.push(
      `Run assessment: ${SPONSORED_BLOCKER_LABELS[input.assessedCause === 'needs_api_key' ? 'needs_api_key' : 'needs_login']}.`,
    )
  }
  return {
    label:
      categories.has('scope_check') ||
      (code === 'partial_edits' &&
        (ending === 'completed' || text.startsWith('turn completed')))
        ? 'Verification failed'
        : 'Run failed',
    detail: [
      ended,
      ...blockers,
      ...(evidence?.truncated
        ? ['Additional failed checks were omitted.']
        : []),
    ]
      .join('\n')
      .slice(0, 2_000),
  }
}

/**
 * The closed code for a failure.
 *
 * A Cloud `refusal` is looked up, and one this list does not name (a runner
 * status added later, say) is `other`; an off-Cloud failure is classified from the leading clause of its
 * diagnostic, which our own client code writes. ORDER MATTERS below: the
 * more specific clauses are tested before the general `turn …` ones.
 */
export function sponsoredRunFailureCode(args: {
  refusal?: string | null
  diagnosticReason?: string | null
}): SponsoredRunFailureCode {
  const refusal = args.refusal?.trim()
  if (refusal) {
    return (SPONSORED_RUN_FAILURE_CODES as readonly string[]).includes(refusal)
      ? (refusal as SponsoredRunFailureCode)
      : 'other'
  }
  const text = (args.diagnosticReason ?? '').trim().toLowerCase()
  if (!text) return 'unclassified'
  if (text.startsWith(SPONSORED_NEVER_STARTED_PREFIX)) {
    return neverStartedCode(text)
  }
  if (text.startsWith(SPONSORED_FUNDING_REFUSED_PREFIX)) {
    return 'funding_refused'
  }
  if (text.startsWith('accept-identity-changed'))
    return 'accept_identity_changed'
  if (text.startsWith('accept-failed')) return 'accept_failed'
  if (text.startsWith('containment-mismatch')) return 'containment_mismatch'
  if (text.startsWith('resume-declined')) return 'resume_declined'
  if (text.startsWith('stale-sweep')) return 'timed_out'
  if (text.startsWith('app-quit')) return 'app_quit'
  if (text.startsWith('interrupted')) return 'interrupted'
  if (text.includes('the verdict could not be decided')) {
    return 'verdict_undecided'
  }
  if (text.includes('commit follow-up')) return 'commit_follow_up_failed'
  // Before every `turn …` branch: a turn that ended early with edits is its
  // own outcome whatever ended it -- which also reclassifies the diagnostics
  // shipped Desktop builds already write.
  if (text.includes(SPONSORED_PARTIAL_EDITS_MARKER)) return 'partial_edits'
  if (text.startsWith('turn grant-expired')) return 'grant_expired'
  if (
    text.startsWith('turn interrupted') ||
    text.startsWith('turn stopped') ||
    text.startsWith('turn closed')
  ) {
    return 'interrupted'
  }
  // An errored turn is `turn_error` whatever it left behind: "no file edits"
  // after an error is how a session that never started reads, too.
  if (text.startsWith('turn error')) return 'turn_error'
  if (text.startsWith('turn completed')) {
    return text.includes('recorded no file edits') ? 'no_edits' : 'no_commit'
  }
  return 'unclassified'
}

/** The code of a (lower-cased) `never-started: <token>…` diagnostic. */
function neverStartedCode(text: string): SponsoredRunFailureCode {
  const rest = text.slice(SPONSORED_NEVER_STARTED_PREFIX.length)
  if (!rest.startsWith(': ')) return 'never_started_other'
  const body = rest.slice(2)
  // Desktop's rewind / close / delete paths wrote a sentence, not a token.
  if (body.startsWith('the conversation was')) {
    return 'never_started_inputs_dropped'
  }
  const token = body.split(/[:;\s]/, 1)[0] ?? ''
  return NEVER_STARTED_CODE_BY_REASON[token] ?? 'never_started_other'
}

/**
 * The scrubbed, capped copy of a diagnostic for the funnel.
 *
 * Operator text is authored by our code, but its CAUSE clause is whatever the
 * failing link said -- an exception's first line, a git error, a provider
 * message -- and any of those can name a user's home directory, a file in
 * their repository, a URL or a token. So: first line only, control characters
 * dropped, then every span that could carry user content replaced by a
 * placeholder naming what was removed, then capped. What survives is the
 * SHAPE of the failure (`turn error: <path> could not be opened; HEAD is
 * still the base commit`), which is what a count needs. The unscrubbed text is
 * still on the Convex row.
 *
 * `null` when nothing is left, so the key is omitted rather than empty.
 */
export function scrubSponsoredDiagnostic(
  raw: string | null | undefined,
): string | null {
  if (!raw) return null
  let text = raw.split(/\r?\n/, 1)[0] ?? ''
  // Control characters (C0 + DEL), as a scan-free replace.
  text = text.replace(/[\u0000-\u001f\u007f]/g, ' ')
  text = text
    // URLs of any scheme, before paths, so `https://x/y` is one placeholder.
    .replace(/\b[a-z][a-z0-9+.-]*:\/\/\S+/gi, '<url>')
    .replace(/[\w.+-]+@[\w-]+(?:\.[\w-]+)+/g, '<email>')
    // Quoted spans are where an error echoes its input (a file name, a
    // command, a model's words). The run state in backticks is ours, but it
    // costs nothing to lose: `from_state` carries it.
    .replace(/`[^`]*`/g, '<quoted>')
    .replace(/"[^"]*"/g, '<quoted>')
    // Single quotes only where one OPENS a quote, so "can't" is not a quote.
    .replace(/(^|[\s(=:])'[^']*'/g, '$1<quoted>')
    // Windows absolute paths and UNC shares.
    .replace(/\b[a-z]:[\\/][^\s;,)]*/gi, '<path>')
    .replace(/\\\\[^\s;,)]+/g, '<path>')
    // POSIX absolute and home-relative paths. `/dev/null` is kept: it names no
    // user and it is the exact string one real macOS failure turned on.
    .replace(/(^|[\s(=:])~?\/(?!dev\/null\b)[^\s;,)]+/g, '$1<path>')
    // Relative paths that name a file (`src/app.ts`); a model id such as
    // `deepseek/deepseek-v4-flash` has no extension and is kept.
    .replace(/\b[\w.-]+(?:\/[\w.-]+)+\.[a-z0-9]{1,8}\b/gi, '<path>')
    // Long opaque tokens: hashes, keys, ids.
    .replace(/\b[a-f0-9]{16,}\b/gi, '<hex>')
    .replace(/\b[A-Za-z0-9_-]{32,}\b/g, '<token>')
    // Last: every remaining number, so `95 minutes` and `96 minutes` count as
    // one shape. After the hex and token passes, which need the digits.
    .replace(/\d+/g, 'N')
    .replace(/\s+/g, ' ')
    .trim()
  if (!text) return null
  return text.length > SPONSORED_FUNNEL_DIAGNOSTIC_MAX
    ? `${text.slice(0, SPONSORED_FUNNEL_DIAGNOSTIC_MAX - 1)}…`
    : text
}

export function buildSponsoredRunFunnelMetadata(args: {

  refusal?: string | null
}): SponsoredRunFunnelMetadata {
  const { row } = args

  const client = (SPONSORED_RUN_CLIENTS as readonly string[]).includes(

    : undefined
  // A Freebuff Web run executes on OUR Linux VM, in the accepting chat's
  // worktree, and its row carries no Convex `project_id` (that key belongs to
  // the retired /cloud executor). It is Cloud execution all the same: the
  // worker reports it, not a user's machine.
  const webRun = !row.project_id && client === 'web'
  const executionMode: SponsoredRunExecutionMode =
    row.project_id || webRun
      ? 'cloud'
      : row.in_place_execution === true
        ? 'in_place'
        : 'worktree'

    ? row.execution_surface
    : undefined

  const os = surface ? OS_BY_SURFACE[surface] : webRun ? 'linux' : undefined
  const fromState = (SPONSORED_RUN_FROM_STATES as readonly string[]).includes(

    typeof clientVersion === 'string' &&
    SPONSORED_CLIENT_VERSION_PATTERN.test(clientVersion)
      ? clientVersion
      : undefined

  let failureCode: SponsoredRunFailureCode | undefined
  let llmCalled: boolean | undefined
  if (args.eventType === 'run_failed') {
    failureCode = evidence?.verificationPending
      ? 'verification_incomplete'
      : sponsoredRunFailureCode({
          refusal: args.refusal,
          diagnosticReason: row.diagnostic_reason,
        })
    if (NO_TURN_CODES.has(failureCode)) llmCalled = false
    else if (TURN_RAN_CODES.has(failureCode)) llmCalled = true

    // web row never started its turn: the web worker starts one only after

    // advisory `running` report lost, so it stays unknown.

      executionMode === 'cloud' &&
      fromState === 'accepted'

      llmCalled = false
    }

    llmCalled = true
  }

    ...(evidence ? { run_evidence: evidence } : {}),
    execution_mode: executionMode,
    ...(executionSurface ? { execution_surface: executionSurface } : {}),
    ...(client ? { client } : {}),
    ...(os ? { os } : {}),
    ...(row.acceptance?.containment === 'floor'
      ? { containment: 'floor' as const }
      : {}),
    ...(fromState ? { from_state: fromState } : {}),
    ...(failureCode ? { failure_code: failureCode } : {}),
    ...(diagnostic ? { diagnostic_reason: diagnostic } : {}),
    ...(llmCalled !== undefined ? { llm_called: llmCalled } : {}),
    ...(validClientVersion ? { client_version: validClientVersion } : {}),
  }
}

/**
 * The step driver: the harness, not the model, decides when a sponsored
 * skill's steps are done (piece 3 of COD-822; the format is
 * `docs/ads/agentic/sponsored-skills.md`).
 *
 * After the run's turn has done its work, the driver walks the skill's steps
 * in order and runs each step's checks through THE skill evaluator
 * (`evaluateSponsoredSkillChecks`), the same one the sponsored eval scores
 * with. A step with no failing check is done, whatever the model said. A
 * failing `agent` or `check` step goes back to the agent with the failed
 * checks (`sponsoredSkillCheckFeedback`), to fix and stop; the checks run
 * again after each fix. The step's `maxAttempts` counts the first try (the
 * turn's own work) plus fixes, and the run as a whole gets at most
 * {@link SPONSORED_STEP_MAX_FIXES_PER_RUN}. Then the step is BLOCKED, and the
 * walk ends there: later steps usually build on it, and prompting them would
 * spend the run's compute on work that cannot pass. They are still checked
 * once, so the result says what is true on disk.
 *
 * - A `user` step is never prompted or retried. It is done only once its
 *   checks pass (the user handed over the credential), else it needs the user.
 * - A step that passed before a later fix is checked again at the end: a fix
 *   for step 4 can break step 2.
 *
 * DELIVERED means every `agent` and `check` step passed. That is the run's
 * completion, not the model writing "Done:".
 *
 * Pure: the caller hands in the check context (its worktree, its sandbox) and
 * how to prompt the agent, so the whole loop runs in tests without a model,
 * a sandbox or a disk.
 */

const NO_SKILL: SponsoredSkillSelection = { skill: null, source: 'none' }

/**
 * The skill an accepted procedure names, and where it came from. Only when
 * the text is exactly that skill's rendering: this build's registry first,
 * then the skill the Accept response offered (`offered`, untrusted JSON, held
 * to the procedure by `deliveredSponsoredSkillForProcedure`),
 * else none, and a legacy or unmatched procedure runs as free text.

export function selectSponsoredProcedureSkill(

  offered?: unknown,

async function check(

/** Walk the steps, check each, and have the agent fix what fails. */
export async function driveSponsoredSkillSteps(
  deps: SponsoredStepDriverDeps,
): Promise<SponsoredStepDriveResult> {
  const maxPerRun = deps.maxFixesPerRun ?? SPONSORED_STEP_MAX_FIXES_PER_RUN
  const total = deps.skill.steps.length
  const results: SponsoredStepResult[] = []
  // The fix count at which each step last passed: a later fix may undo it.
  const passedAt = new Map<number, number>()
  let fixes = 0
  let walking = true
  let stoppedBecause: SponsoredStepDriveResult['stoppedBecause']

  for (const [index, step] of deps.skill.steps.entries()) {
    const number = index + 1
    const base = { number, id: step.id, title: step.title, kind: step.kind }
    try {
      deps.onStep?.({ number, total, title: step.title })
    } catch {
      // Progress is display only; it never stops the walk.
    }
    let outcome = await check(deps, step)

    if (step.kind === 'user') {
      results.push({
        ...base,
        status: outcome.passed ? 'passed' : 'needs_user',
        fixes: 0,
        checks: outcome.results,
      })
      continue
    }

    const walkingAtEntry = walking
    // The turn's own work was the first attempt.
    const maxFixes = (step.maxAttempts ?? DEFAULT_SKILL_MAX_ATTEMPTS) - 1
    let stepFixes = 0
    while (walking && !outcome.passed) {
      if (stepFixes >= maxFixes || fixes >= maxPerRun) break
      if (!deps.canContinue()) {
        walking = false
        stoppedBecause = 'out_of_time'
        break
      }
      stepFixes += 1
      fixes += 1
      const completed = await deps.prompt(
        sponsoredStepFixPrompt(
          step,
          number,
          total,
          outcome.results,
          stepFixes,
          maxFixes,
        ),
      )
      if (!completed) {
        walking = false
        stoppedBecause = 'turn_failed'
      }
      outcome = await check(deps, step)
    }

    if (outcome.passed) passedAt.set(number, fixes)
    // Blocked: the driver tried, or would have but the fix budget is spent.
    // Not reached: the walk had already stopped (or ran out of time) first.
    const blocked = stepFixes > 0 || (walkingAtEntry && walking)
    results.push({
      ...base,
      status: outcome.passed ? 'passed' : blocked ? 'blocked' : 'not_reached',
      fixes: stepFixes,
      checks: outcome.results,
    })
    if (!outcome.passed) walking = false
  }

  for (const result of results) {
    const at = passedAt.get(result.number)
    if (at === undefined || at === fixes) continue
    const again = await check(deps, deps.skill.steps[result.number - 1]!)
    result.checks = again.results
    if (!again.passed) result.status = 'blocked'
  }

  return {
    skill: skillKey(deps.skill),
    steps: results,
    delivered: results.every(
      (result) => result.kind === 'user' || result.status === 'passed',
    ),
    fixes,
    ...(stoppedBecause ? { stoppedBecause } : {}),
  }
}

export type SponsoredStepLoopDeps<State> = {
  skill: SponsoredSkill
  ctx: SponsoredSkillCheckContext
  /** One more segment of the run's turn; resolves null when it did not complete. */
  resume: (prompt: string, previous: State) => Promise<State | null>
  /** False once the run is aborted or out of time. */
  canContinue: () => boolean
  /** Takes the run's credentials out of command output before the agent reads it. */
  redact?: (text: string) => string
  /** A `report_friction` blocker, sent for each step left blocked. */
  onFriction?: (input: SponsoredFrictionInput) => void
  /** Persist checked completion before the optional final report segment. */
  onChecksComplete?: (result: SponsoredStepDriveResult) => void
  /** Progress only (see {@link SponsoredStepDriverDeps.onStep}). */
  onStep?: (step: SponsoredStepProgress) => void
}

/**
 * The driver run inside a turn, for any runtime (Desktop's harness, the Cloud
 * runner): fixes are more segments of the SAME run (same transcript), and the
 * state it resolves is the last segment that COMPLETED, so a fix that errored
 * leaves the turn where it was rather than failing work already on disk.
 */
export async function runSponsoredStepLoop<State>(
  initial: State,
  deps: SponsoredStepLoopDeps<State>,
): Promise<{ state: State; result: SponsoredStepDriveResult }> {
  const redact = deps.redact ?? identity
  let state = initial

  const result = await driveSponsoredSkillSteps({
    skill: deps.skill,
    ctx: redactedCheckContext(deps.ctx, redact),
    canContinue: deps.canContinue,
    ...(deps.onStep ? { onStep: deps.onStep } : {}),
    prompt: async (text) => {
      const next = await deps.resume(text, state)
      if (next === null) return false
      state = next
      return true
    },
  })

  deps.onChecksComplete?.(result)

  // The report the user reads must match the checks, not the model's memory
  // of what it tried. Only after a fix: otherwise the first segment's report
  // already described the work the checks then confirmed.
  if (result.fixes > 0 && deps.canContinue()) {
    const next = await deps.resume(sponsoredStepReportPrompt(result), state)
    if (next !== null) state = next
  }

  for (const step of result.steps) {
    if (step.kind === 'user' || step.status !== 'blocked') continue
    deps.onFriction?.({
      blocking: true,
      category: step.checks.some(
        (check) =>
          check.status === 'fail' &&
          /\bexited\b|could not run/.test(check.detail),
      )
        ? 'command_failed'
        : 'other',
      detail: `Step check: step ${step.number} still failed its checks after ${step.fixes} fix${
        step.fixes === 1 ? '' : 'es'
      } by the agent, so Freebuff marked it blocked.`,
      step: step.number,
    })
  }
  return { state, result }
}

const identity = (text: string) => text

/** Command output reaches the agent and the report (COD-665): redacted first. */
function redactedCheckContext(
  ctx: SponsoredSkillCheckContext,
  redact: (text: string) => string,
): SponsoredSkillCheckContext {
  return {
    ...ctx,
    runScript: async (script, timeout) => {
      const out = await ctx.runScript(script, timeout)
      return out === 'missing' ? out : { ...out, output: redact(out.output) }
    },
    runCommand: async (argv, timeout) => {
      const out = await ctx.runCommand(argv, timeout)
      return { ...out, output: redact(out.output) }
    },
  }
}

const PREFLIGHT_DETAIL_CHARS = 200

/** A failed preflight check, as the run's error and friction name it. */
function preflightFailureLine(
  skill: SponsoredSkill,
  failed: SponsoredSkillCheckResult,
): string {
  // One line: the runs' diagnostics keep only an error's first line.
  const oneLine = failed.detail.replace(/\s+/g, ' ').trim()
  const detail =
    oneLine.length > PREFLIGHT_DETAIL_CHARS
      ? `${oneLine.slice(0, PREFLIGHT_DETAIL_CHARS - 1)}…`
      : oneLine
  const declared = sponsoredSkillPreflightChecks(skill).find(
    (
      check,
    ): check is Extract<SponsoredSkillCheck, { kind: 'command_succeeds' }> =>
      check.id === failed.id && check.kind === 'command_succeeds',
  )
  const tool = declared
    ? (skill.tools ?? []).find((candidate) => candidate.name === declared.argv[0])
    : undefined
  const hint =
    tool?.source === 'path'
      ? ` Install \`${tool.name}\`: ${tool.installHint}`
      : ''
  return `"${failed.label}" (${detail}).${hint}`
}

/**
 * THE PREFLIGHT, run before the agent's first turn: the skill's
 * `requirements.os` against the platform the checks run on, then its
 * `preflight` checks through the same evaluator as the steps. On a failure no
 * agent turn runs and nothing is edited: the caller ends the run with
 * `reason`, and it is reported the way a blocked step is (the
 * `report_friction` blocker is sent here; `result` is the step report, every
 * step not reached). An unsupported OS runs no command: the skill's tool
 * check says nothing more on a platform the skill does not support.
 */
export async function runSponsoredSkillPreflight(
  deps: SponsoredSkillPreflightDeps,
): Promise<SponsoredSkillPreflightOutcome> {
  const { skill } = deps
  const checks: SponsoredSkillCheckResult[] = []
  const os = sponsoredSkillOs(deps.platform)
  const supported = skill.requirements?.os ?? []
  if (os && supported.length > 0) {
    const ok = supported.includes(os)
    checks.push({
      id: SPONSORED_PREFLIGHT_OS_CHECK_ID,
      label: `Runs on ${supported.join(' or ')}`,
      status: ok ? 'pass' : 'fail',
      detail: ok
        ? `this run is on ${os}`
        : `this skill supports ${supported.join(', ')}; this run is on ${os}`,
    })
  }
  if (checks.every((result) => result.status !== 'fail')) {
    const outcome = await check(
      { skill, ctx: redactedCheckContext(deps.ctx, deps.redact ?? identity) },
      { checks: sponsoredSkillPreflightChecks(skill) },
    )
    checks.push(...outcome.results)
  }
  const failed = checks.filter((result) => result.status === 'fail')
  if (failed.length === 0) return { passed: true, checks }

  const reason = `Preflight failed before any edit: ${failed
    .map((result) => preflightFailureLine(skill, result))
    .join(' ')}`
  // A wrong OS or a file/package check is the project not fitting the skill;
  // only a command or script check is a missing toolchain.
  const declaredChecks = sponsoredSkillPreflightChecks(skill)
  const toolchainOnly = failed.every((result) => {
    const declared = declaredChecks.find((c) => c.id === result.id)
    return (
      declared?.kind === 'command_succeeds' ||
      declared?.kind === 'script_passes'
    )
  })
  deps.onFriction?.({
    blocking: true,
    category: toolchainOnly ? 'missing_toolchain' : 'project_mismatch',
    // The server keeps the first SPONSORED_FRICTION_DETAIL_MAX characters.
    detail: reason.slice(0, SPONSORED_FRICTION_DETAIL_MAX),
  })
  return {
    passed: false,
    checks,
    reason,
    result: {
      skill: skillKey(skill),
      steps: skill.steps.map((step, index) => ({
        number: index + 1,
        id: step.id,
        title: step.title,
        kind: step.kind,
        status: 'not_reached',
        fixes: 0,
        checks: [],
      })),
      delivered: false,
      fixes: 0,
      stoppedBecause: 'preflight_failed',
      preflight: checks,
    },
  }
}

/** The run error's opening, as {@link runSponsoredSkillPreflight} writes it. */
export const SPONSORED_PREFLIGHT_FAILED_PREFIX = 'Preflight failed before any edit:'

/**
 * What a preflight refusal was, read back from its run error (also its
 * SCRUBBED funnel copy, `scrubSponsoredDiagnostic`, which keeps these
 * phrases): `os` when the skill does not support the OS, `fit` when a
 * file or package check refused the project (the skill does not fit it),
 * `command` when only a command or script check failed (a missing
 * toolchain). Null for anything that is not a preflight refusal. The phrases
 * are the evaluator's own details (`sponsored-skill-checks.ts`).
 */
export function sponsoredPreflightRefusalKind(
  text: string | null | undefined,
): 'os' | 'fit' | 'command' | null {
  if (!text) return null
  const lower = text.toLowerCase()
  const at = lower.indexOf(SPONSORED_PREFLIGHT_FAILED_PREFIX.toLowerCase())
  if (at < 0) return null
  const body = lower.slice(at)
  if (/this skill supports [^;]*; this run is on/.test(body)) return 'os'
  if (
    /no file matches |no (?:changed )?file matching .*? contains |no (?:changed )?file matches |, which must not appear|package\.json is missing|package\.json does not declare|declares [a-z0-9_]+=/.test(
      body,
    )
  )
    return 'fit'
  if (/could not run |exited -?(?:\d+|n)\b|has no [a-z]+ script/.test(body))
    return 'command'
  // A preflight refusal whose detail was cut: the project did not pass a
  // check before any edit, which is a fit question first.
  return 'fit'
}
