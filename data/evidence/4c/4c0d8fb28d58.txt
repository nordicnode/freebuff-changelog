/**
 * Report an injected Bun preload, and decide in code whether to run anyway.
 *
 * `BUN_OPTIONS=--preload <file>` runs caller code inside this process before
 * the first line of the shipped entry, which is how a capture bridge takes a
 * sponsored grant without ever sending a request (poc/claude-code-bridge;
 * `poc/cli-env-inheritance` measured the channel on the published binaries).
 *
 * This is a SPEED BUMP, not a barrier, and it says so on purpose: a preload
 * runs before this check, so it can rewrite `process.execArgv` and drop itself
 * from `require.cache`. What it buys is that capture stops being one
 * environment variable and becomes a patched binary — a different act, with a
 * different ban story.
 *
 * Every module the compiled binary ships lives under `/$bunfs/`; a path
 * outside it was not loaded by this build.
 */

/** The prefix of every module inside a compiled Bun binary. */
const BUNDLE_PREFIX = '/$bunfs/'

/** The flags Bun honours as "load this file before the entry point". */
const PRELOAD_FLAGS = ['--preload', '-r', '--require'] as const

function foreignPath(candidate: string | undefined): string | null {
  if (typeof candidate !== 'string') return null
  const trimmed = candidate.trim()
  if (!trimmed) return null
  return trimmed.startsWith(BUNDLE_PREFIX) ? null : trimmed
}

/**
 * The first injected preload path, or null. Pure: the caller supplies both
 * channels, so this is testable without a Bun process.
 *
 * `execArgv` is the channel measured on the published binary. `loadedModules`
 * (the keys of `require.cache`) is best-effort: a compiled binary reports its
 * own modules under the bundle prefix, and may report none at all.
 *
 * `cwd` excludes that module channel under the working directory, and it is
 * there for one measured reason: Bun also reads `preload` from a `bunfig.toml`
 * in the current directory, so the CLI run inside a user's own repository
 * would otherwise see that repository's setup file in the cache and refuse to
 * start — a false stop on ordinary use. Measured 2026-10-27: a `bunfig.toml`
 * preload executes in this process and never appears in `execArgv`, so this
 * check cannot tell a repository's own setup file from an injection anyway;
 * excluding `cwd` states that instead of pretending otherwise.
 */
export function foreignPreloadFrom(input: {
  execArgv: readonly string[]
  loadedModules?: readonly string[]
  cwd?: string
}): string | null {
  const argv = input.execArgv
  for (let index = 0; index < argv.length; index++) {
    const arg = argv[index] as string
    if ((PRELOAD_FLAGS as readonly string[]).includes(arg)) {
      const next = foreignPath(argv[index + 1])
      if (next) return next
      continue
    }
    for (const flag of PRELOAD_FLAGS) {
      const prefix = `${flag}=`
      if (!arg.startsWith(prefix)) continue
      const value = foreignPath(arg.slice(prefix.length))
      if (value) return value
    }
  }
  const cwd = input.cwd?.replace(/\/+$/, '')
  for (const modulePath of input.loadedModules ?? []) {
    if (cwd && (modulePath === cwd || modulePath.startsWith(`${cwd}/`)))
      continue
    const foreign = foreignPath(modulePath)
    if (foreign) return foreign
  }
  return null
}

/**
 * Whether a build stops on an injected preload.
 *
 * A build-time decision, not a runtime one: asking a user to set an environment
 * variable to get a security control is how controls end up unset, and asking
 * them to unset it is how a legitimate instrumentation hook strands a whole
 * machine. The build ships one policy; the user launches the CLI and nothing
 * else. A hardened distribution flips this to `true` in its own build.
 *
 * Default is `false` on purpose: the preload has already run by the time this
 * check executes, so stopping cannot undo it - it can only remove the CLI from
 * a user whose toolchain loads a module this way.
 */
export const REFUSE_FOREIGN_PRELOAD = false

/**
 * One line for stderr. Names the path; does not explain how to evade it, and
 * asks the user for nothing.
 *
 * It never says "refused to start" unless this build actually refuses - a
 * message that claims a stop that did not happen sends whoever reads it after
 * the wrong thing (that wording shipped once, in `describeForeignPreload`'s
 * `enforced` branch, and is why the mode is no longer a variable).
 */
export function describeForeignPreload(preloadPath: string): string {
  return [
    `freebuff warning: code was injected into this process before it began (${preloadPath}).`,
    'It has already run; the CLI continues.',
  ].join('\n')
}

// ------------------------------------------------------------------ clock

/** Past this, Desktop skips turn-start probes whose 25 s deadline already "passed" (PR #9473). */
const CLOCK_SKEW_WARN_MS = 20_000
/** Past this, commands (5 min deadline), sign-in and certificate checks break too. */
const CLOCK_SKEW_ERROR_MS = 5 * 60_000

export type ClockReading = { skewMs: number; rttMs: number } | { error: string }

/**
 * Compares this computer's clock with the server's `Date` header. The header has one-second
 * precision, so the server time is taken as the middle of that second and the local time as the
 * middle of the round trip; the error is under a second plus half the round trip.
 */
export async function readClockSkew(url: string, options: { fetch?: Fetch; now?: () => number; timeoutMs?: number } = {}): Promise<ClockReading> {
  const fetcher = options.fetch ?? fetch
  const now = options.now ?? Date.now
  const started = now()
  try {
    const response = await fetcher(url, { method: 'HEAD', redirect: 'manual', signal: AbortSignal.timeout(options.timeoutMs ?? 6_000) })
    const ended = now()
    await response.body?.cancel().catch(() => {})
    const header = response.headers.get('date')
    const server = header ? Date.parse(header) : NaN
    if (!Number.isFinite(server)) return { error: 'no-date' }
    return { skewMs: Math.round((started + ended) / 2 - (server + 500)), rttMs: ended - started }
  } catch (error) {
    return { error: classifyNetworkError(error) }
  }
}

function clockSyncSteps(platform: string): string[] {
  if (platform === 'win32')
    return ['Open Settings > Time & language > Date & time.', 'Turn on "Set time automatically", then click "Sync now".', 'Restart Freebuff.']
  if (platform === 'darwin')
    return ['Open System Settings > General > Date & Time.', 'Turn on "Set time and date automatically".', 'Restart Freebuff.']
  return ['Turn on automatic time sync: `timedatectl set-ntp true` on most distributions.', 'Restart Freebuff.']
}

export function clockFinding(reading: ClockReading, platform: string, product: 'desktop' | 'cli'): DoctorFinding {
  if ('error' in reading)
    return finding('clock.unknown', 'info', 'clock', 'Could not compare your clock with Freebuff',
      'The server did not answer, so the clock check was skipped. See the network findings.', [])
  const seconds = Math.round(Math.abs(reading.skewMs) / 1000)
  const direction = reading.skewMs > 0 ? 'ahead' : 'behind'
  const amount = seconds >= 120 ? `${Math.round(seconds / 60)} min` : `${seconds} s`
  if (Math.abs(reading.skewMs) <= CLOCK_SKEW_WARN_MS)
    return finding('clock.ok', 'ok', 'clock', 'Your clock is in sync', `Within ${seconds} s of Freebuff's servers.`)
  const effect = product === 'desktop'
    ? 'Freebuff gives each piece of work it sends to this computer a deadline. With a clock this far off, Desktop treats that work as already expired and silently skips it, so turns can start without your files or terminal.'
    : 'Sign-in tokens and secure connections carry time limits, and a clock this far off can make them fail.'
  return finding('clock.skew', Math.abs(reading.skewMs) > CLOCK_SKEW_ERROR_MS ? 'error' : 'warning', 'clock',
    `Your computer's clock is ${amount} ${direction}`,
    `${effect} Set your clock to sync automatically.`, clockSyncSteps(platform))
}

export type NetworkErrorKind = 'dns' | 'refused' | 'timeout' | 'reset' | 'tls' | 'offline' | 'other'

const TLS_ERROR = /CERT|certificate|SSL|TLS|self[- ]signed|UNABLE_TO_(?:GET|VERIFY)/i

export function classifyNetworkError(error: unknown): NetworkErrorKind {
  const e = error as { code?: unknown; name?: unknown; message?: unknown; cause?: { code?: unknown; message?: unknown } } | undefined
  const text = [e?.code, e?.name, e?.message, e?.cause?.code, e?.cause?.message].filter(v => typeof v === 'string').join(' ')
  if (TLS_ERROR.test(text)) return 'tls'
  if (/ENOTFOUND|EAI_AGAIN|getaddrinfo|DNS|resolve host/i.test(text)) return 'dns'
  if (/ECONNREFUSED|ConnectionRefused/i.test(text)) return 'refused'
  if (/ETIMEDOUT|Timeout|timed out|AbortError/i.test(text)) return 'timeout'
  if (/ECONNRESET|socket hang up|ConnectionClosed|EPIPE/i.test(text)) return 'reset'
  if (/ENETUNREACH|EHOSTUNREACH|FailedToOpenSocket|network is unreachable/i.test(text)) return 'offline'
  return 'other'
}

export type Endpoint = { id: string; label: string; url: string }
export type EndpointResult = { endpoint: Endpoint; ms: number } & ({ ok: true; status: number } | { ok: false; error: NetworkErrorKind })

export async function probeEndpoint(endpoint: Endpoint, options: { fetch?: Fetch; timeoutMs?: number; now?: () => number } = {}): Promise<EndpointResult> {
  const now = options.now ?? Date.now
  const started = now()
  try {
    const response = await (options.fetch ?? fetch)(endpoint.url, { method: 'GET', redirect: 'manual', signal: AbortSignal.timeout(options.timeoutMs ?? 6_000) })
    await response.body?.cancel().catch(() => {})
    // Any HTTP answer, even a 404, means the network path works.
    return { endpoint, ms: now() - started, ok: true, status: response.status }
  } catch (error) {
    return { endpoint, ms: now() - started, ok: false, error: classifyNetworkError(error) }
  }
}

// Issuers that only appear when something on this computer or network re-signs HTTPS traffic.
const INTERCEPTORS = /Avast|AVG|Kaspersky|ESET|Bitdefender|Norton|Symantec|McAfee|Sophos|Trend ?Micro|Webroot|F-Secure|Malwarebytes|Zscaler|Netskope|Fortinet|FortiGate|Palo Alto|Cisco Umbrella|Blue ?Coat|Forcepoint|Barracuda|Check ?Point|Cloudflare Gateway|mitmproxy|Charles Proxy|Fiddler|DO_NOT_TRUST|Proxy/i

export type TlsInspection =
  | { authorized: boolean; authorizationError: string | null; issuers: string[] }
  | { error: NetworkErrorKind }

export function inspectTls(host: string, options: { connect?: typeof tlsConnect; timeoutMs?: number } = {}): Promise<TlsInspection> {
  return new Promise(resolve => {
    let socket: TLSSocket | undefined
    const timer = setTimeout(() => { socket?.destroy(); resolve({ error: 'timeout' }) }, options.timeoutMs ?? 6_000)
    try {
      // rejectUnauthorized: false so a rejected chain can still be described; nothing is sent.
      const opened: TLSSocket = (options.connect ?? tlsConnect)({ host, port: 443, servername: host, rejectUnauthorized: false }, () => {
        clearTimeout(timer)
        const issuers: string[] = []
        try {
          let cert: PeerCertificate | undefined = opened.getPeerCertificate(true)
          const seen = new Set<string>()
          while (cert && Object.keys(cert).length && issuers.length < 6) {
            const name = [cert.issuer?.O, cert.issuer?.CN].flat().filter(Boolean).join(' / ')
            if (name) issuers.push(name)
            const next: PeerCertificate | undefined = (cert as PeerCertificate & { issuerCertificate?: PeerCertificate }).issuerCertificate
            if (!next || seen.has(cert.fingerprint256) || next.fingerprint256 === cert.fingerprint256) break
            seen.add(cert.fingerprint256)
            cert = next
          }
        } catch { /* a runtime without chain access still reports authorization */ }
        const authorizationError = opened.authorizationError ? String(opened.authorizationError) : null
        resolve({ authorized: opened.authorized, authorizationError, issuers })
        opened.end()
      })
      socket = opened
      opened.on('error', error => { clearTimeout(timer); resolve({ error: classifyNetworkError(error) }) })
    } catch (error) {
      clearTimeout(timer)
      resolve({ error: classifyNetworkError(error) })
    }
  })
}

export function tlsFinding(host: string, inspection: TlsInspection): DoctorFinding {
  if ('error' in inspection)
    return finding('tls.unknown', 'info', 'network', `Could not inspect the secure connection to ${host}`,
      'The connection failed before its certificate could be read. See the network findings.', [])
  const interceptor = inspection.issuers.find(name => INTERCEPTORS.test(name))
  if (!inspection.authorized || interceptor) {
    const by = interceptor ? ` The certificate was issued by "${interceptor}", not by a public authority.` : ''
    return finding('tls.intercepted', inspection.authorized ? 'warning' : 'error', 'network',
      'HTTPS to Freebuff is being intercepted',
      `Something between this computer and ${host} replaces Freebuff's certificate.${by} Antivirus HTTPS scanning and corporate proxies do this, and it makes Freebuff fail with "Can't reach Freebuff".`,
      NETWORK_STEPS.tls.steps)
  }
  return finding('tls.ok', 'ok', 'network', 'Secure connections are not intercepted', `${host} presented a certificate from a public authority.`)
}

const FREEBUFF_DOMAINS = ['freebuff.com', 'codebuff.com']

const LOCAL_ADDRESS = /^(?:0\.0\.0\.0|127\.|10\.|192\.168\.|172\.(?:1[6-9]|2\d|3[01])\.|169\.254\.|::1?$|::ffff:127\.|f[cd][0-9a-f]{2}:|fe80:)/i

function isLocalAddress(address: string) {
  return LOCAL_ADDRESS.test(address)
}

  (await lookup(host, { all: true })).map(entry => entry.address)): Promise<DoctorFinding[]> {

export function hostsFilePath(platform: string, env: Record<string, string | undefined>) {
  return platform === 'win32' ? `${env.SystemRoot ?? env.windir ?? 'C:\\Windows'}\\System32\\drivers\\etc\\hosts` : '/etc/hosts'

/** Hosts-file lines that name one of our domains (or a subdomain), comments ignored. */
export function hostsFileEntries(text: string, domains = FREEBUFF_DOMAINS): Array<{ address: string; host: string }> {
  const entries: Array<{ address: string; host: string }> = []
  for (const raw of text.split(/\r?\n/)) {
    const line = raw.replace(/#.*/, '').trim()
    if (!line) continue
    const [address, ...names] = line.split(/\s+/)
    for (const name of names) {
      const host = name.toLowerCase()
      if (domains.some(domain => host === domain || host.endsWith(`.${domain}`))) entries.push({ address: address!, host })
    }
  }
  return entries
}

  return finding('hosts.pinned', 'error', 'network', 'Your hosts file redirects Freebuff',
    `It has ${entries.length} entr${entries.length === 1 ? 'y' : 'ies'} for ${[...new Set(entries.map(e => e.host))].join(', ')}. Freebuff's addresses change, so a pinned or blocked entry breaks it.`,
    ['Open the hosts file as an administrator and delete those lines.', 'Restart Freebuff.'])
}

const PROXY_VARS = ['HTTPS_PROXY', 'https_proxy', 'HTTP_PROXY', 'http_proxy', 'ALL_PROXY', 'all_proxy']

export function configuredProxies(env: Record<string, string | undefined>) {
  const seen = new Map<string, { variable: string; host: string; port: number }>()
  for (const variable of PROXY_VARS) {
    const value = env[variable]
    if (!value) continue
    try {
      const url = new URL(value.includes('://') ? value : `http://${value}`)
      const port = Number(url.port) || (url.protocol === 'https:' ? 443 : 80)
      // Only host and port: a proxy URL can carry a user name and password.
      const key = `${url.hostname}:${port}`
      if (!seen.has(key)) seen.set(key, { variable, host: url.hostname, port })
    } catch {
      seen.set(`invalid:${variable}`, { variable, host: '', port: 0 })
    }
  }
  return [...seen.values()]
}

function tcpReachable(host: string, port: number, timeoutMs = 3_000, connect = netConnect): Promise<boolean> {
  return new Promise(resolve => {
    const socket = connect({ host, port })
    const done = (ok: boolean) => { clearTimeout(timer); socket.destroy(); resolve(ok) }
    const timer = setTimeout(() => done(false), timeoutMs)
    socket.once('connect', () => done(true))
    socket.once('error', () => done(false))
  })
}

export async function proxyFindings(env: Record<string, string | undefined>, reachable: typeof tcpReachable = tcpReachable): Promise<DoctorFinding[]> {
  const proxies = configuredProxies(env)
  const out: DoctorFinding[] = []
  for (const proxy of proxies) {
    if (!proxy.host) {
      out.push(finding(`proxy.invalid.${proxy.variable.toLowerCase()}`, 'warning', 'network', `${proxy.variable} is not a valid proxy address`,
        'Freebuff sends its requests through this setting, and it cannot be read.', [`Fix or remove the ${proxy.variable} environment variable.`]))
      continue
    }
    if (!(await reachable(proxy.host, proxy.port)))
      out.push(finding(`proxy.down.${proxy.variable.toLowerCase()}`, 'error', 'network', `The proxy in ${proxy.variable} does not answer`,
        `${proxy.variable} sends Freebuff's requests through ${proxy.host}:${proxy.port}, which refused or ignored a connection.`,
        ['Start the proxy, or remove the variable if you no longer use it.', 'Restart Freebuff (and your terminal) after changing it.']))
  }
  if (proxies.length && !out.length)
    out.push(finding('proxy.ok', 'info', 'network', 'A proxy is configured', `Requests go through ${proxies.map(p => `${p.host}:${p.port}`).join(', ')}, which answers. If it inspects HTTPS, see the secure-connection finding.`))
  return out
}

// ---------------------------------------------------------------- disk

const LOW_DISK_BYTES = 2 * 1024 ** 3
const CRITICAL_DISK_BYTES = 512 * 1024 ** 2

const gb = (bytes: number) => `${(bytes / 1024 ** 3).toFixed(bytes < 10 * 1024 ** 3 ? 1 : 0)} GB`

export async function diskFinding(label: string, dir: string, redact: DoctorRedactor,
  statfs: (path: string) => Promise<Pick<StatsFs, 'bavail' | 'bsize'>> | Pick<StatsFs, 'bavail' | 'bsize'> = statfsAsync): Promise<DoctorFinding> {
  let free: number
  try {
    const stats = await statfs(dir)
    free = Number(stats.bavail) * Number(stats.bsize)
  } catch {
    return finding('disk.unknown', 'info', 'disk', `Could not read free space for ${label}`, `The drive holding ${redact.path(dir)} did not report its free space.`)
  }
  if (free < CRITICAL_DISK_BYTES)
    return finding('disk.critical', 'error', 'disk', `Almost no disk space left (${gb(free)})`,
      `The drive holding ${label} has ${gb(free)} free. Chats, worktrees and updates fail or are left half written when it fills.`,
      ['Free up several gigabytes on this drive (empty the trash, remove old downloads or unused node_modules folders).', 'Restart Freebuff.'])
  if (free < LOW_DISK_BYTES)
    return finding('disk.low', 'warning', 'disk', `Disk space is low (${gb(free)})`,
      `The drive holding ${label} has ${gb(free)} free. Worktrees and dependency installs need room.`, ['Free up a few gigabytes on this drive.'])
  return finding('disk.ok', 'ok', 'disk', 'Enough disk space', `${gb(free)} free on the drive holding ${label}.`)
}

const SYNC_PROVIDERS: Array<[RegExp, string]> = [
  [/(?:^|[\\/])OneDrive(?: - [^\\/]+)?(?:[\\/]|$)/i, 'OneDrive'],
  [/(?:^|[\\/])Dropbox(?: \([^\\/]+\))?(?:[\\/]|$)/i, 'Dropbox'],

/** The sync client that owns a path, or null. Recognises the default folder names only. */
export function syncProviderFor(path: string): string | null {
  for (const [pattern, name] of SYNC_PROVIDERS) if (pattern.test(path)) return name
  return null
}

export async function tempDirFinding(dir = tmpdir(), redact?: DoctorRedactor): Promise<DoctorFinding> {

/** macOS on Apple silicon running an x64 build under Rosetta: slower, and native modules mismatch. */
export function architectureFinding(platform = process.platform, arch = process.arch, cpuModel = cpus()[0]?.model ?? '',
  translated?: boolean): DoctorFinding | null {
  const rosetta = translated ?? (platform === 'darwin' && arch === 'x64' && /Apple/i.test(cpuModel))
  if (rosetta)
    return finding('system.rosetta', 'warning', 'system', 'Running the Intel build on an Apple silicon Mac',

/** Counts files under `roots` whose full path passes Windows' 260-character limit. Bounded. */
export async function longPathFinding(roots: string[], platform = process.platform, maxEntries = 20_000,
  list: (dir: string) => Promise<Array<{ name: string; isDirectory(): boolean }>> | Array<{ name: string; isDirectory(): boolean }> = dir => readdir(dir, { withFileTypes: true })): Promise<DoctorFinding | null> {
  if (platform !== 'win32') return null
  let seen = 0
  let over = 0
  const stack = [...roots]
  while (stack.length && seen < maxEntries) {
    const dir = stack.pop()!
    let entries: Array<{ name: string; isDirectory(): boolean }>
    try { entries = await list(dir) } catch { continue }
    for (const entry of entries) {
      seen++
      const full = `${dir}${sep}${entry.name}`
      if (full.length >= 260) over++
      if (entry.isDirectory() && entry.name !== 'node_modules' && entry.name !== '.git') stack.push(full)
    }
  }
  if (!over) return null
  return finding('system.long-paths', 'warning', 'system', `${over} file path${over === 1 ? ' is' : 's are'} longer than Windows allows by default`,
    'Paths over 260 characters fail in many Windows tools unless long paths are enabled.',
    ['Enable long paths: run `git config --global core.longpaths true`, and turn on "Enable Win32 long paths" in Group Policy (or the LongPathsEnabled registry value).', 'Keep projects in a short folder such as C:\\code.'],
    { learnMore: LEARN_MORE.windowsLongPaths })
}

import type { DoctorFinding, DoctorReport, DoctorSeverity } from './types'

const LABEL: Record<DoctorSeverity, string> = { error: 'Problem', warning: 'Warning', info: 'Note', ok: 'OK' }

function doctorCounts(report: Pick<DoctorReport, 'findings'>) {
  const counts: Record<DoctorSeverity, number> = { error: 0, warning: 0, info: 0, ok: 0 }
  for (const item of report.findings) counts[item.severity]++
  return counts
}

export function doctorSummary(report: Pick<DoctorReport, 'findings'>) {
  const { error, warning } = doctorCounts(report)
  if (!error && !warning) return 'No problems found.'
  const parts = [
    error ? `${error} problem${error === 1 ? '' : 's'}` : null,
    warning ? `${warning} warning${warning === 1 ? '' : 's'}` : null,
  ].filter(Boolean)
  return `${parts.join(' and ')} found.`
}

function header(report: DoctorReport) {
  const product = report.product === 'desktop' ? 'Freebuff Desktop' : 'Freebuff CLI'
  return [
    `${product} ${report.appVersion ?? '(unknown version)'} on ${report.platform} ${report.arch}`,
    `Scanned ${report.generatedAt} in ${(report.durationMs / 1000).toFixed(1)} s` +
      `${report.options.deep ? ', deep scan' : ''}${report.options.includePaths ? ', full paths included' : ', paths redacted'}`,
  ]
}

function steps(item: DoctorFinding, bullet: string) {
  return item.steps.map((step, index) => `${bullet}${index + 1}. ${step}`)
}

/** The "Copy report" text: Markdown for a support ticket or Discord. Already redacted by the scan. */
export function doctorReportMarkdown(report: DoctorReport): string {
  const lines = ['## Freebuff diagnostics', '', ...header(report).map(line => `- ${line}`), `- ${doctorSummary(report)}`, '']
  const shown = report.findings.filter(item => item.severity !== 'ok')
  for (const item of shown) {
    lines.push(`### ${LABEL[item.severity]}: ${item.title}`, '', `\`${item.id}\` (${item.area})`, '', item.detail, '')
    if (item.steps.length) lines.push(...steps(item, ''), '')
    if (item.learnMore) lines.push(`More: ${item.learnMore}`, '')
  }
  const passed = report.findings.filter(item => item.severity === 'ok')
  if (passed.length) lines.push('### Passed', '', ...passed.map(item => `- ${item.title}`), '')
  const missing = report.checks.filter(check => check.outcome === 'timeout' || check.outcome === 'failed')
  if (missing.length) lines.push(`Checks that did not finish: ${missing.map(check => `${check.id} (${check.outcome})`).join(', ')}`, '')
  return lines.join('\n').trimEnd() + '\n'
}

/** Plain text for a terminal. */
export function doctorReportText(report: DoctorReport, options: { verbose?: boolean } = {}): string {
  const lines = ['Freebuff doctor', ...header(report), '']
  for (const item of report.findings) {
    if (item.severity === 'ok' && !options.verbose) continue
    lines.push(`[${LABEL[item.severity].toUpperCase()}] ${item.title}`, `  ${item.detail}`)
    lines.push(...steps(item, '  '))
    if (item.learnMore) lines.push(`  More: ${item.learnMore}`)
    lines.push('')
  }
  const passed = report.findings.filter(item => item.severity === 'ok').length
  lines.push(doctorSummary(report) + (passed && !options.verbose ? ` ${passed} checks passed.` : ''))
  return lines.join('\n') + '\n'
}
