/**
 * The refusals that mean only "no bot-check proof reached the server" (or one
 * failed). Freebuff's login pages never show these as a refusal: they re-run
 * the checks in the background and continue the same sign-in once they bank
 * (`signup-challenge-client/challenge-recovery.ts`). The copy above remains
 * for surfaces that do not recover.
 */
export const BOT_CHECK_REFUSALS = [
  'captcha_missing',
  'captcha_invalid',
  'recaptcha_missing',
  'recaptcha_invalid',
] as const satisfies readonly SignupBlockReason[]

export type BotCheckRefusal = (typeof BOT_CHECK_REFUSALS)[number]

export function isBotCheckRefusal(
  code: string | null | undefined,
): code is BotCheckRefusal {
  return (BOT_CHECK_REFUSALS as readonly string[]).includes(code ?? '')
}

/** True when `code` is a reason this module has copy for. Narrows an untrusted
 *  query-string value without a cast at every call site. */
export function isSignupBlockReason(
  code: string | null | undefined,
): code is SignupBlockReason {
  return Boolean(code) && code! in SIGNUP_BLOCK_MESSAGES
}

/**
 * AWS slots serve the API on `N.staging.codebuff.com` and auth/web on
 * `N.staging.freebuff.com`; `slotOrigins()` in `infra/staging/instance.ts` derives both
 * from the slot number. One reviewed rule rather than one entry per slot, and it still
 * pairs a single number: a session cannot cross slots.
 */
const SLOT_API_ORIGIN = /^https:\/\/([1-9][0-9]?)\.staging\.codebuff\.com$/
const SLOT_AUTH_ORIGIN = /^https:\/\/([1-9][0-9]?)\.staging\.freebuff\.com$/

export function isStagingSlotOriginPair(apiOrigin: string, authOrigin: string): boolean {
  const api = SLOT_API_ORIGIN.exec(apiOrigin)
  const auth = SLOT_AUTH_ORIGIN.exec(authOrigin)
  return api !== null && auth !== null && api[1] === auth[1]
}

/** A slot's auth origin from its API origin (`N.staging.codebuff.com` => `N.staging.freebuff.com`). */
export function stagingSlotAuthOrigin(apiOrigin: string): string | null {
  const api = SLOT_API_ORIGIN.exec(apiOrigin)
  return api ? `https://${api[1]}.staging.freebuff.com` : null
}

   return (BOT_CHECK_REFUSALS as readonly string[]).includes(code ?? '')
 }
