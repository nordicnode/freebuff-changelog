// Users paste MCP server addresses the way they type them into a browser: `mcp.example.com/sse`,
// `example.com/mcp`, `localhost:3000/mcp`. `new URL` rejects the first two and misreads the last
// as a URL with the scheme `localhost:`, so every surface failed with "not a valid URL".
//
// This only supplies a missing scheme. It never removes or changes one the user typed, so each
// surface's own scheme, host and SSRF checks run unchanged on the result: `javascript:`, `file:`
// and `ftp:` URLs come back as given and are rejected there, as before.

/** Capability detection is presentation only; native IPC checks its sender/account. */
export function hostedWorkspaceHost(
  value: unknown,
): HostedWorkspaceHost | null {
  if (!value || typeof value !== 'object') return null
  const host = value as Partial<HostedWorkspaceHost>
  return host.protocolVersion === HOSTED_WORKSPACE_PROTOCOL_VERSION &&
    typeof host.platform === 'string' &&
    typeof host.appVersion === 'string' &&
    ['status', 'openLocalProject', 'disconnect', 'returnToDesktop'].every(
      (name) => typeof host[name as keyof HostedWorkspaceHost] === 'function',
    )
    ? (host as HostedWorkspaceHost)
    : null
}

import { z } from 'zod/v4'

/**
 * Local MCP servers: stdio commands and localhost URLs that run on one of the account's computers.
 * Desktop owns their configuration (`~/.agents/mcp.json`); Cloud only ever sees each computer's
 * published tool list, and reaches a tool through that computer's device connection
 * (docs/mcp-desktop/04-local-mcp-everywhere.md).
 */

export const LOCAL_MCP_SERVER_LIMIT = 30
export const LOCAL_MCP_TOOL_LIMIT = 200
/** Published lists larger than this are trimmed by Desktop, tool by tool, before they are sent. */
export const LOCAL_MCP_MANIFEST_BYTES = 256 * 1024
/** Desktop republishes well inside this; a computer that stops is forgotten after it. */
export const LOCAL_MCP_MANIFEST_TTL_MS = 3 * 60_000
export const LOCAL_MCP_REPUBLISH_MS = 60_000

/** `device:<deviceId>:<name>`: the id the agent passes back to call a local server's tool. */
export const LOCAL_MCP_SERVER_ID = /^device:([0-9a-f-]{36}):(.{1,80})$/

export const localMcpServerName = z
  .string()
  .trim()
  .min(1)
  .max(80)
  .regex(/^[\p{L}\p{N} _.@-]+$/u, 'Use letters, numbers, spaces, dots, dashes or underscores')

  .object({
    name: z.string().min(1).max(200),
    description: z.string().max(4000),
    inputSchema: z.record(z.string(), z.unknown()),
    annotations: z
      .object({
        readOnlyHint: z.boolean().optional(),
        destructiveHint: z.boolean().optional(),
      })

// Users paste MCP server addresses the way they type them into a browser: `mcp.example.com/sse`,
// `example.com/mcp`, `localhost:3000/mcp`. `new URL` rejects the first two and misreads the last
// as a URL with the scheme `localhost:`, so every surface failed with "not a valid URL".
//
// This only supplies a missing scheme. It never removes or changes one the user typed, so each
// surface's own scheme, host and SSRF checks run unchanged on the result: `javascript:`, `file:`
// and `ftp:` URLs come back as given and are rejected there, as before.

const LOOPBACK_HOSTS = new Set(['localhost', '127.0.0.1', '[::1]'])

/** `name:` followed by digits and then the end, a path, a query or a fragment is a port, not a scheme. */
const HOST_WITH_PORT = /^[^\s/:?#]+:\d+(?:[/?#]|$)/
const SCHEME = /^[a-z][a-z0-9+.-]*:/i

/**
 * Whether one field that takes either a command line or a server URL holds a URL: one with an
 * http(s) scheme, or one typed without it (`localhost:3845/mcp`, `127.0.0.1:3845`, `[::1]:3845`).
 * A command never starts with `name:<digits>`.
 */
export function isMcpServerAddress(raw: string): boolean {
  const value = raw.trim()
  return (
    /^https?:\/\//i.test(value) ||
    HOST_WITH_PORT.test(value) ||
    /^\[[0-9a-f:.]+\]:\d+(?:[/?#]|$)/i.test(value) ||
    /^localhost(?:[/?#]|$)/i.test(value)
  )
}

/**
 * Trims `raw` and, when it has no scheme, prefixes `https://` — or `http://` for a loopback host,
 * the only place plain HTTP is allowed and where local MCP servers almost always listen.
 * A value that already has a scheme is returned trimmed and otherwise untouched.
 */
export function normalizeMcpServerUrl(raw: string): string {
  const value = raw.trim()
  if (!value) return value
  if (value.startsWith('//')) return `https:${value}`
  if (SCHEME.test(value) && !HOST_WITH_PORT.test(value)) return value

  let hostname: string
  try {
    hostname = new URL(`https://${value}`).hostname
  } catch {
    // Not a URL with any scheme; leave it for the caller's validation to report.
    return value
  }
  return `${LOOPBACK_HOSTS.has(hostname) ? 'http' : 'https'}://${value}`
}
