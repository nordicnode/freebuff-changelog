export type SponsoredRunSurface = SponsoredAcceptSurface

/** What the mid-run account step does: open a tracked link, or only ask for keys. */
export type SponsoredRunDeferredSignup = 'link' | 'credentials'

/** "How to carry it out here" for a surface. Only Desktop has a worktree flow. */
export function sponsoredRunGuidance(
  surface: SponsoredRunSurface,
  inPlace: boolean,
): string {
  if (surface === 'cli') return SPONSORED_RUN_CLI_GUIDANCE
  if (surface === 'web') return SPONSORED_RUN_WEB_GUIDANCE
  return inPlace
    ? SPONSORED_RUN_DESKTOP_IN_PLACE_GUIDANCE
    : SPONSORED_RUN_DESKTOP_WORKTREE_GUIDANCE
}

/**
 * Which account step the run has, as each surface's account step decides it.
 * Desktop and the CLI open a signup link when `sponsoredSignupStep` finds one
 * (a reviewed `bfcid={bfcid}` link, else the Accept's tracked link when the
 * procedure needs an account), else ask only for declared credentials. The
 * Cloud account step opens the Accept's tracked link whenever there is one
 * (`sponsoredCloudAccountStep`).
 */
export function sponsoredRunDeferredSignup(input: {
  surface: SponsoredRunSurface
  procedure: string
  advertiserLink: string | null
  skill: SponsoredSkill | null
}): SponsoredRunDeferredSignup | undefined {
  const hasLink =
    input.surface === 'web'
      ? Boolean(input.advertiserLink)
      : sponsoredSignupStep(
          input.procedure,
          { advertiserLink: input.advertiserLink },
          input.skill,
        ) !== undefined
  if (hasLink) return 'link'
  return declaredRunCredentials(input.procedure).length > 0
    ? 'credentials'
    : undefined
}

/**

  advertiserLink: string | null

/**
 * The task context slot's content, as every client renders it: the user's
 * own messages, oldest first, numbered. Empty for no messages.
 */
export function renderSponsoredRunTaskContext(
  messages: readonly string[],
): string {
  return messages
    .map((message, index) => `User message ${index + 1}:\n${message}`)
    .join('\n\n')
}

/**
 * The run spec the funded Accept returns to a client that sent
 * `runSpecVersion`. `skill` must be the server-resolved skill the procedure
 * renders (`sponsoredSkillMatchesProcedure`), or null for a free-text
 * procedure: a caller whose procedure names a skill it could not resolve
 * refuses the Accept instead of building a spec with no packages.
 */
export function buildSponsoredRunSpec(input: {
  procedure: string
  skill: SponsoredSkill | null
  surface: SponsoredRunSurface
  inPlace: boolean
  advertiserLink: string | null
}): SponsoredRunSpec {
  return {
    v: SPONSORED_RUN_SPEC_VERSION,
    procedure: input.procedure,
    procedureSha256: sha256Hex(input.procedure),
    skill: input.skill,
    declaredPackages: input.skill?.packages ?? [],
    // Registered long-form skills keep the frame they were written against.
    promptTemplate: isLayer2Skill(input.skill)
      ? buildSponsoredRunLayer1PromptTemplate(input)
      : buildSponsoredRunPromptTemplate({
          surface: input.surface,
          inPlace: input.inPlace,
          procedure: input.procedure,
          advertiserLink: input.advertiserLink,
          deferredSignup: sponsoredRunDeferredSignup(input),
        }),
  }
}

*
 * Before this, each client compiled its own copy of the run prompt (the
 * consent framing, how-to-carry-it-out rules, the deferred-account rules and
 * the final report) and resolved the skill, and so the packages it may
 * install, from its own bundled registry first. A skill version or prompt
 * change then needed a client release, and a build that could not resolve a
 * new skill refused that skill's own pinned install (Exa 2.0.0, 2026-10-13).
 *
 * The server builds it with `buildSponsoredRunSpec`
 * (`./sponsored-run-prompt.ts`).
 *
 * With a spec the server is the one source of truth: the client runs
 * `promptTemplate` with its two slots filled, drives `skill`'s steps and
 * installs only `declaredPackages`. It never substitutes a bundled skill or a

* ## The consent contract this must not break
 *
 * The advertiser's reviewed procedure is hashed (SHA-256) and the user consents
 * to that exact template hash: Desktop previews it over
 * `GET /api/v1/ads/proposal/{id}/accept`, the native dialog shows it, and the
 * `POST` carries the hash back — a changed procedure is 409 and cannot inherit
 * the earlier consent. Keep the stored procedure, accept response and hash
 * unchanged. Only the execution copy resolves the explicit `{bfcid}` slot
 * using the signed token minted by Accept. No URLs or instructions are
 * otherwise rewritten. Legacy runtime values still travel as a separate
 * section after the procedure.
 *
 * ## `advertiserLink`
 *
 * The advertiser CTA URL, the campaign's landing
 * URL carrying the signed `bfcid` conversion token
 * ({@link ./sponsored-proposal-cta.ts}). A procedure that wants it — to leave
 * it in the `.env.example` comment block it writes, or in the pull request
 * body template — DECLARES that by containing the placeholder
 * `{{advertiserLink}}`. A procedure that does not contain the placeholder gets
 * no URL: an unrequested URL in the prompt is an invitation for the
 * model to paste it somewhere the reviewer never approved.
 *
 * The section is rendered even when the link is UNAVAILABLE, provided the
 * procedure declared it: settlement runs beside the accept rather than
 * before it, so a run can start before the token exists. In that case the
 * model is told to omit the line rather than leave the literal placeholder in
 * a committed file.
 *
 * `{{advertiserClickId}}` separately opts into the same link's opaque bfcid
 * for command-scoped CLI attribution. It never introduces another token
 * source or changes the reviewed procedure. Missing attribution skips the
 * dependent command; it must not silently turn into an unattributed conversion.
 */

export const ADVERTISER_CLICK_ID_PLACEHOLDER = '{{advertiserClickId}}'

  inputs: SponsoredProcedureRuntimeInputs,

  if (!text.match(BFCID_SLOT)) return text
