 * The chart spec shared by artifact kinds: the JSON a Docs ```chart block holds,
 * and the `chart` of a slide. Pure (zod only), so the server validates a body
 * with exactly the rules the app draws by.
 *
 *   { "type": "bar" | "line" | "area", "title"?: string, "x": "<row key for the x axis>",
 *     "series": [{ "key": "<row key>", "label": "<legend label>" }],   // 1-8, unique, not x
 *     "data": [{ "<x>": "Week 1", "<series key>": 40 }, ...] }           // 1-500 rows
 *
 * Every row has the x key; every value is a string or a finite number, and
 * series values are numbers. Docs strips legacy metadata on series entries before
 * validating fenced JSON with this schema. Stored slide charts use the strict schema directly.
 *
 * Sheet charts (a range of cells) and data charts (a dataset's columns) are not
 * this spec: sheet.ts shares `CHART_TYPES`, data.ts the series limit.
 */

/**
 * Freebuff Data: the content model of a data artifact (dashboards, metrics pages, KPI trackers,
 * scorecards, data stories), version 1.
 *
 * ===== For the agent writing one =====
 *
 * A data artifact is ONE JSON object. It is declarative: no HTML, no scripts, no formulas as
 * code. Every number on the page is computed by Freebuff from the rows you put in `datasets`,
 * and every tile and chart names the source it came from. Unknown keys are rejected everywhere.
 *
 * {
 *   "version": 1,
 *   "title": "Weekly product metrics",
 *   "subtitle": "Freebuff, all platforms",           // optional, one line
 *   "asOf": "2026-11-14",                           // optional: ISO date or date-time; when the data was pulled
 *   "layout": "grid",                               // "grid" (dashboard) or "story" (one column, text between charts)
 *   "currency": "USD",                              // optional ISO 4217 code for currency values (default USD)
 *   "sources": [                                    // where numbers come from. At least one.
 *     { "id": "wh", "label": "Warehouse events table", "detail": "how it was pulled (query, file, export)",
 *       "window": "Sep 21 – Nov 14, 2026 (PT)" }   // detail and window optional
 *   ],
 *   "datasets": [                                   // the rows, inline. At most 2000 rows per dataset.
 *     { "id": "weekly", "sourceId": "wh",
 *       "columns": [ { "key": "week", "label": "Week", "type": "date" },
 *                    { "key": "users", "label": "Active users", "type": "number" } ],
 *       "rows": [ { "week": "2026-11-09", "users": 48210 } ] }
 *   ],
 *   "blocks": [ ... in page order, see below ... ],
 *   "filters": [ ... optional, see below ... ]
 * }
 *
 * Column types: "string", "number", "date" (an ISO "YYYY-MM-DD" string), "currency" (a number in
 * major units: 12.5 = $12.50), "percent" (a FRACTION: 0.125 = 12.5%). A cell may be null. Row
 * keys must be declared columns; a row may leave a column out (read as null).
 *
 * Blocks (each has a unique `id` and a `kind`):
 *  - kpi: a big number.
 *    { "kind": "kpi", "id": "wau", "label": "Weekly active users", "datasetId": "weekly",
 *      "value": { "column": "users", "agg": "sum" },       // agg: sum | avg | min | max | last | count
 *      "sparkline": "week",                                // optional: a column to trend over (see below)
 *      "delta": { "vs": "previous", "format": "percent" }, // optional
 *      "format": "compact",                                // number | integer | compact | currency | percent
 *      "sourceId": "wh" }
 *    `value` may instead be { "literal": 1234 } (no datasetId needed; sourceId says where it came
 *    from) or a derived rate { "ratio": { "numerator": {column, agg}, "denominator": {column, agg} } }.
 *    With `sparkline`, rows are grouped by that column (sorted ascending): the tile shows the
 *    LATEST group's value, a sparkline of every group, and a delta against the group before it.
 *    Without it, the value is taken over all (filtered) rows.
 *    delta.vs: "previous" (needs `sparkline`) or a number (a target or baseline, e.g. a goal).
 *    delta.format: "percent" (relative change) or "absolute" (difference, in the tile's format;
 *    percentage points for percent tiles). Optional delta.label ("vs last week", "vs goal") and
 *    delta.good: "up" (default) or "down" (for numbers where lower is better, like latency).
 *  - chart: { "kind": "chart", "id", "title", "type": "bar" | "line" | "area" | "stacked-bar" | "pie" | "scatter",
 *      "datasetId", "x": "week", "series": [ { "key": "users", "label": "Active users" } ],
 *      "sourceId", "note"?: "one line under the chart",
 *      "split"?: "platform",   // one series drawn once per distinct value of this column (stacked by platform, etc.)
 *      "agg"?: "sum",          // how rows sharing an x are combined (default sum)
 *      "size"?: "half" | "full" }  // grid width (default half; story layout is always one column)
 *    Rows are grouped by `x`. pie: `x` names the slices, the first series sizes them. scatter: `x`
 *    and the first series are both numeric and rows are plotted as-is.

/**
 * Components. A MAIN component is a frame with `component: true`, anywhere in the design. An
 * INSTANCE is a frame with `instanceOf: <main id>` and no children of its own: it draws a copy of
 * the main, so editing the main restyles every instance. An instance keeps its own place and
 * size (x, y, w, h, grow, hidden, name, linkTo) and may override CONTENT only: a text's `text`,
 * a button's `label`, an image's `src`/`placeholder`/`alt`, keyed by the main's node id. A stale
 * override (its node gone from the main) is ignored, never an error.
 *
 * In the drawn copy every inner id is `<instance id>~<main node id>`: `~` is never in a stored
 * id, so the canvas can tell an instance's inside from real nodes and select the instance.
 */

 *                   "#0f172a80") or "transparent". Prefer tokens: editing a token restyles every
 *                   board at once. The renderer's defaults read these names when present:

 *                   rounded, serif, didone, mono). Headings (display, h1-h3) use `heading`.

 *   Board  { id, title, x, y, w, h, background?, root }   root is a frame that fills the board.

 *          row of boards, a "note" for a sticky comment. Notes are never part of a board.

 *       "cells": {                              // SPARSE: only cells that have something
 *         "A1": { "v": "Item", "bold": true, "bg": "gray" },
 *         "B2": { "v": 1200, "fmt": "currency" },
 *         "C2": { "v": 0.125, "fmt": "percent" },     // percent is a fraction: 0.125 shows 12.5%
 *         "D2": { "v": "2026-11-20", "fmt": "date" }, // dates are ISO yyyy-mm-dd strings

 *     }
 *   ]

 * - `bold`: true. `align`: left | center | right (default: numbers right, text left).
 * - `bg`: gray | green | yellow | red | blue | brand — a theme-aware tint, not a hex.

  // and never sampled. `distinct_id` is the canonical codebuff Postgres user
  // id on every surface, so unique-users of this event gives accurate
  // per-surface DAU (filter on the `surface` property) and a combined DAU (no
  // filter). The `surface` property is one of: cli, web, chat, desktop, cloud
  // (web = the freebuff.com builder, cloud = connected-repo builder projects).
  // A native app's send reports its platform instead, ios or android, from
  // any screen (messageSentSurface).
  // Emission points: cli client analytics; chat's stream route (server-side);
  // desktop's analytics module; web/cloud via the Convex send mutation
  // (PostHog + Axiom, both direct from Convex — see convex/analytics.ts).
  MESSAGE_SENT = 'message_sent',

/**
 * The chart spec shared by artifact kinds: the JSON a Docs ```chart block holds,
 * and the `chart` of a slide. Pure (zod only), so the server validates a body
 * with exactly the rules the app draws by.
 *
 *   { "type": "bar" | "line" | "area", "title"?: string, "x": "<row key for the x axis>",
 *     "series": [{ "key": "<row key>", "label": "<legend label>" }],   // 1-8, unique, not x
 *     "data": [{ "<x>": "Week 1", "<series key>": 40 }, ...] }           // 1-500 rows
 *
 * Every row has the x key; every value is a string or a finite number, and
 * series values are numbers. Docs strips legacy metadata on series entries before
 * validating fenced JSON with this schema. Stored slide charts use the strict schema directly.
 *
 * Sheet charts (a range of cells) and data charts (a dataset's columns) are not
 * this spec: sheet.ts shares `CHART_TYPES`, data.ts the series limit.
 */

import { z } from 'zod/v4'

export const CHART_TYPES = ['bar', 'line', 'area'] as const
export type ChartSpecType = (typeof CHART_TYPES)[number]

export const CHART_LIMITS = { series: 8, rows: 500 } as const

export type ChartSpec = {
  type: ChartSpecType
  title?: string
  /** The key in each row that labels the x axis. */
  x: string
  series: Array<{ key: string; label: string }>
  data: Array<Record<string, string | number>>
}

const chartCell = z.union([z.string(), z.number().finite()])

export const chartSpecSchema = z
  .strictObject({
    type: z.enum(CHART_TYPES),
    title: z.string().optional(),
    x: z.string().refine((v) => v.trim() !== '', 'x must name a row key'),
    series: z
      .array(
        z.strictObject({
          key: z.string().refine((v) => v.trim() !== '', 'series key is empty'),
          label: z.string(),
        }),
      )
      .min(1)
      .max(CHART_LIMITS.series),
    data: z.array(z.record(z.string(), chartCell)).min(1).max(CHART_LIMITS.rows),
  })
  .superRefine((spec, ctx) => {
    const keys = new Set<string>()
    spec.series.forEach((s, i) => {
      if (s.key === spec.x || keys.has(s.key)) {
        ctx.addIssue({
          code: 'custom',
          path: ['series', i, 'key'],
          message: 'series keys must be unique and differ from x',
        })
      }
      keys.add(s.key)
    })
    spec.data.forEach((row, i) => {
      if (!(spec.x in row)) {
        ctx.addIssue({ code: 'custom', path: ['data', i], message: `row lacks "${spec.x}"` })
      }
      for (const key of keys) {
        if (key in row && typeof row[key] !== 'number') {
          ctx.addIssue({
            code: 'custom',
            path: ['data', i, key],
            message: 'series values must be numbers',
          })
        }
      }
    })
  })

// The schema's output is exactly the spec type.
type _SpecFits = z.infer<typeof chartSpecSchema> extends ChartSpec ? true : never
const _specFits: _SpecFits = true
void _specFits

    (f) => f.datasetIds.includes(dataset.id) && isFilterActive(state[f.id]),
  )
  const cross = crossFilter(state)
  const crossColumn = cross && dataset.columns.some((c) => c.key === cross.column) ? cross : null
  if (active.length === 0 && !crossColumn) return dataset.rows
  return dataset.rows.filter((row) => {
    if (crossColumn) {
      const cell = row[crossColumn.column]
      if (cell === null || cell === undefined || String(cell) !== crossColumn.value) return false
    }
    return active.every((filter) => {
      const choice = state[filter.id]
      const cell = row[filter.column]
      if (cell === null || cell === undefined) return false
      if (choice.kind === 'select') return String(cell) === choice.value
      if (choice.kind === 'multi') return choice.values.includes(String(cell))
      if (choice.kind !== 'date-range') return true
      const value = String(cell)
      if (choice.from !== null && value < choice.from) return false
      if (choice.to !== null && value > choice.to) return false
      return true
    })
  })

const DAY_MS = 86_400_000

/** An ISO day moved by `days` (negative goes back). */
export function addDays(iso: string, days: number): string {
  const date = utcDate(iso)
  if (!date) return iso
  return new Date(date.getTime() + days * DAY_MS).toISOString().slice(0, 10)
}

/** Days from `from` to `to`, both counted. */
export function daySpan(from: string, to: string): number {
  const a = utcDate(from)
  const b = utcDate(to)
  if (!a || !b) return 0
  return Math.round((b.getTime() - a.getTime()) / DAY_MS) + 1
}

export const DATE_PRESETS = [
  { id: 'last-7', label: 'Last 7 days' },
  { id: 'last-28', label: 'Last 28 days' },
  { id: 'last-90', label: 'Last 90 days' },
  { id: 'month', label: 'This month' },
  { id: 'quarter', label: 'This quarter' },
  { id: 'ytd', label: 'Year to date' },
] as const
export type DatePreset = (typeof DATE_PRESETS)[number]['id']

/**
 * A preset's range, counted back from the LATEST date in the data (a dashboard is a snapshot:
 * "last 28 days" means the 28 days the data ends with, not the 28 before today).
 */
export function presetRange(preset: DatePreset, bounds: { min: string; max: string }): { from: string; to: string } {
  const to = bounds.max
  const [y, m] = to.split('-')
  let from = to
  switch (preset) {
    case 'last-7':
      from = addDays(to, -6)
      break
    case 'last-28':
      from = addDays(to, -27)
      break
    case 'last-90':
      from = addDays(to, -89)
      break
    case 'month':
      from = `${y}-${m}-01`
      break
    case 'quarter': {
      const q = Math.floor((Number(m) - 1) / 3) * 3 + 1
      from = `${y}-${String(q).padStart(2, '0')}-01`
      break
    }
    case 'ytd':
      from = `${y}-01-01`
      break
  }
  return { from: from < bounds.min ? bounds.min : from, to }
}

/** Which preset a range is, if any. */
export function matchPreset(range: { from: string | null; to: string | null }, bounds: { min: string; max: string } | null): DatePreset | null {
  if (!bounds || range.from === null) return null
  for (const p of DATE_PRESETS) {
    const r = presetRange(p.id, bounds)
    if (r.from === range.from && (range.to ?? bounds.max) === r.to) return p.id
  }
  return null
}

/**
 * The window just before a dataset's active date range, the same number of days long, as a
 * state to measure against; null when no date range on that dataset is narrowed.
 */
export function previousPeriod(
  dashboard: Dashboard,
  datasetId: string | undefined,
  state: FilterState,
): { state: FilterState; from: string; to: string; column: string; days: number } | null {
  if (!datasetId) return null
  for (const filter of dashboard.filters ?? []) {
    const choice = state[filter.id]
    if (filter.type !== 'date-range' || !filter.datasetIds.includes(datasetId)) continue
    if (!choice || choice.kind !== 'date-range' || !isFilterActive(choice)) continue
    const bounds = dateBounds(dashboard, filter)
    const from = choice.from ?? bounds?.min
    const to = choice.to ?? bounds?.max
    if (!from || !to) continue
    const length = daySpan(from, to)
    if (length <= 0) continue
    const prevTo = addDays(from, -1)
    const prevFrom = addDays(from, -length)
    return {
      state: { ...state, [filter.id]: { kind: 'date-range', from: prevFrom, to: prevTo }, [COMPARE_KEY]: { kind: 'compare', on: false } },
      from: prevFrom,
      to: prevTo,
      column: filter.column,
      days: length,
    }
  }
  return null
}

export const compareOn = (state: FilterState) => {
  const choice = state[COMPARE_KEY]
  return choice?.kind === 'compare' && choice.on
}

export function datasetRows(dashboard: Dashboard, datasetId: string | undefined, state: FilterState): DataRow[] {
  const dataset = findDataset(dashboard, datasetId)
  return dataset ? filterRows(dataset, dashboard.filters, state) : []
}

/** A select filter's choices: the distinct values of its column across its datasets, in order seen. */
export function filterOptions(dashboard: Dashboard, filter: DataFilter): string[] {
  const seen = new Set<string>()
  for (const id of filter.datasetIds) {
    for (const row of findDataset(dashboard, id)?.rows ?? []) {
      const cell = row[filter.column]
      if (cell !== null && cell !== undefined) seen.add(String(cell))
    }
  }
  return [...seen]
}

/** The earliest and latest date in a date-range filter's column, across its datasets. */
export function dateBounds(dashboard: Dashboard, filter: DataFilter): { min: string; max: string } | null {
  let min: string | null = null
  let max: string | null = null
  for (const id of filter.datasetIds) {
    for (const row of findDataset(dashboard, id)?.rows ?? []) {
      const cell = row[filter.column]
      if (typeof cell !== 'string') continue
      if (min === null || cell < min) min = cell
      if (max === null || cell > max) max = cell
    }
  }
  return min !== null && max !== null ? { min, max } : null
}

// ---- Aggregation

/** Combine a column's cells. Nulls are skipped; nothing to combine is `null`, never 0. */
export function aggregate(cells: Cell[], agg: Agg): number | null {
  if (agg === 'count') return cells.filter((c) => c !== null && c !== undefined).length
  const values = cells.filter((c): c is number => typeof c === 'number' && Number.isFinite(c))
  if (values.length === 0) return null
  switch (agg) {
    case 'sum':
      return values.reduce((a, b) => a + b, 0)
    case 'avg':
      return values.reduce((a, b) => a + b, 0) / values.length
    case 'min':
      return Math.min(...values)
    case 'max':
      return Math.max(...values)
    case 'last':
      return values[values.length - 1]
  }
}

export function measure(rows: DataRow[], m: Measure): number | null {
  return aggregate(
    rows.map((row) => row[m.column] ?? null),
    m.agg,
  )
}

export function kpiValue(rows: DataRow[], value: KpiValue): number | null {
  if ('literal' in value) return value.literal
  if ('ratio' in value) {
    const numerator = measure(rows, value.ratio.numerator)
    const denominator = measure(rows, value.ratio.denominator)
    if (numerator === null || denominator === null || denominator === 0) return null
    return numerator / denominator
  }
  return measure(rows, value)
}

export type Group = { key: string | number; rows: DataRow[] }

/**
 * Rows grouped by one column. `sorted` orders groups ascending (numbers numerically, strings
 * and ISO dates lexically); otherwise groups keep the order their key first appears in.
 * Rows with no value in the column are left out.
 */
export function groupRows(rows: DataRow[], column: string, sorted: boolean): Group[] {
  const groups = new Map<string | number, DataRow[]>()
  for (const row of rows) {
    const key = row[column]
    if (key === null || key === undefined) continue
    const bucket = groups.get(key)
    if (bucket) bucket.push(row)
    else groups.set(key, [row])
  }
  const out = [...groups].map(([key, groupRowsList]) => ({ key, rows: groupRowsList }))
  if (sorted) out.sort((a, b) => compareCells(a.key, b.key))
  return out
}

function compareCells(a: Cell, b: Cell): number {
  if (typeof a === 'number' && typeof b === 'number') return a - b
  return String(a).localeCompare(String(b), LOCALE)
}

// ---- KPIs

export type DeltaResult = {
  /** Relative change (0.062 = +6.2%) for `percent`, the difference for `absolute`. */
  change: number
  direction: 'up' | 'down' | 'flat'
  sentiment: 'good' | 'bad' | 'neutral'
  /** "+6.2%", "−1,204", "+1.2 pp". */
  text: string
  /** "vs Nov 2", "vs goal". */
  label: string
}

export type KpiResult = {
  value: number | null
  /** The comparison value: the previous period's, or the target. */
  base: number | null
  delta: DeltaResult | null
  /** One point per sparkline group, oldest first. Empty without a sparkline. */
  trend: Array<{ key: string | number; value: number | null }>
  /** The latest group's key, when the tile shows one period. */
  period: string | number | null
  /** How many rows went into the value. */
  rowCount: number
}

export function kpiResult(dashboard: Dashboard, block: KpiBlock, state: FilterState): KpiResult {
  const rows = datasetRows(dashboard, block.datasetId, state)
  const dataset = findDataset(dashboard, block.datasetId)
  const sparkType = block.sparkline ? findColumn(dataset, block.sparkline)?.type : undefined

  let value: number | null
  let previous: number | null = null
  let previousKey: string | number | null = null
  let period: string | number | null = null
  let rowCount = rows.length
  let trend: KpiResult['trend'] = []

  if (block.sparkline && !('literal' in block.value)) {
    const groups = groupRows(rows, block.sparkline, true)
    trend = groups.map((g) => ({ key: g.key, value: kpiValue(g.rows, block.value) }))
    const last = groups[groups.length - 1]
    const before = groups[groups.length - 2]
    value = last ? kpiValue(last.rows, block.value) : null
    period = last?.key ?? null
    rowCount = last?.rows.length ?? 0
    if (before) {
      previous = kpiValue(before.rows, block.value)
      previousKey = before.key
    }
  } else {
    value = kpiValue(rows, block.value)
  }

  let base: number | null = null
  let defaultLabel = ''
  if (block.delta) {
    if (block.delta.vs === 'previous') {
      base = previous
      defaultLabel =
        previousKey === null ? '' : `vs ${formatKey(previousKey, sparkType ?? 'string', true)}`
    } else {
      base = block.delta.vs
      defaultLabel = `vs ${formatValue(block.delta.vs, block.format, dashboard.currency, block.percentDigits)}`
    }
  }

  const delta =
    block.delta && value !== null && base !== null
      ? computeDelta(value, base, {
          format: block.delta.format,
          good: block.delta.good ?? 'up',
          valueFormat: block.format,
          currency: dashboard.currency,
          label: block.delta.label ?? defaultLabel,
        })
      : null

  // Compare to previous period: the same measure over the window before the range wins over the
  // block's own comparison.
  if (compareOn(state) && !('literal' in block.value) && value !== null) {
    const before = previousPeriod(dashboard, block.datasetId, state)
    const previousValue = before ? kpiResult(dashboard, block, before.state).value : null
    if (before && previousValue !== null) {
      const compared = computeDelta(value, previousValue, {
        format: block.delta?.format ?? 'percent',
        good: block.delta?.good ?? 'up',
        valueFormat: block.format,
        currency: dashboard.currency,
        label: `vs ${formatDateRange(before.from, before.to)}`,
      })
      return { value, base: previousValue, delta: compared, trend, period, rowCount }
    }
  }

  return { value, base, delta, trend, period, rowCount }
}

export function computeDelta(
  value: number,
  base: number,
  options: {
    format: 'percent' | 'absolute'
    good: 'up' | 'down'
    valueFormat: ValueFormat
    currency?: string
    label: string
  },
): DeltaResult | null {
  let change: number
  if (options.format === 'percent') {
    if (base === 0) return null
    change = (value - base) / Math.abs(base)
  } else {
    change = value - base
  }
  const magnitude = (amount: number) =>
    options.format === 'percent'
      ? formatPercent(amount)
      : options.valueFormat === 'percent'
        ? `${trimNumber(amount * 100, 1)} pp`
        : formatValue(amount, options.valueFormat, options.currency)
  // A change too small to show ("+0 pp") is no change: the arrow must agree with the digits.
  if (magnitude(Math.abs(change)) === magnitude(0)) change = 0
  const direction = change > 0 ? 'up' : change < 0 ? 'down' : 'flat'
  const sentiment = direction === 'flat' ? 'neutral' : direction === options.good ? 'good' : 'bad'
  const sign = change > 0 ? '+' : change < 0 ? '−' : ''
  return { change, direction, sentiment, text: `${sign}${magnitude(Math.abs(change))}`, label: options.label }
}

/** The format a KPI's number uses when the column alone decides. */
export function defaultFormat(type: ColumnType | undefined): ValueFormat {
  if (type === 'currency') return 'currency'
  if (type === 'percent') return 'percent'
  return 'number'
}

// ---- Charts

export type ChartSeries = {
  key: string
  label: string
  /** The same measure over the period before (compare to previous period), drawn dashed. */
  previous?: boolean
  /** A fitted trendline (not data): drawn thin and dashed. */
  trend?: boolean
  /** Drawn against the right-hand value axis. */
  axis?: 'right'
}
export type ChartData = {
  /** One object per x: `x` holds the x value, `s0…sN` each series' value. */
  rows: Array<Record<string, string | number | null>>
  series: ChartSeries[]
  xType: ColumnType
}

export const MAX_SPLIT_SERIES = 12

export function chartData(dashboard: Dashboard, block: ChartBlock, state: FilterState): ChartData {
  const dataset = findDataset(dashboard, block.datasetId)
  const rows = datasetRows(dashboard, block.datasetId, stateForBlock(state, block.id))
  return chartFromRows(dataset, block, rows)
}

/** The label of the bucket `topN` folds the smaller categories into. */
export const OTHER_LABEL = 'Other'

/** A chart's points from given rows (the chart's own filters already applied). */
export function chartFromRows(dataset: Dataset | undefined, block: ChartBlock, rows: DataRow[]): ChartData {
  const xType = findColumn(dataset, block.x)?.type ?? 'string'
  const agg = block.agg ?? 'sum'

  if (block.type === 'scatter') {
    const first = block.series[0]
    const points = rows
      .filter((r) => typeof r[block.x] === 'number' && typeof r[first.key] === 'number')
      .map((r) => ({ x: r[block.x] as number, s0: r[first.key] as number }))
      .sort((a, b) => a.x - b.x)
    const series: ChartSeries[] = [{ key: 's0', label: first.label }]
    if (block.trendline && points.length > 1) {
      const fit = linearFit(points.map((p) => [p.x, p.s0] as const))
      if (fit) {
        points.forEach((p) => Object.assign(p, { t0: fit.slope * p.x + fit.intercept }))
        series.push({ key: 't0', label: `${first.label} trend`, trend: true })
      }
    }
    return { rows: points, series, xType }
  }

  // Dates and numbers read left to right in order; categories keep the dataset's order.
  let groups = groupRows(rows, block.x, xType !== 'string')
  let other: DataRow[] | null = null
  // Top N: the N largest categories by the first series (all series summed when split).
  if (block.topN !== undefined && xType === 'string' && groups.length > block.topN) {
    const sizeKey = block.series[0].key
    const ranked = groups
      .map((g) => ({ g, size: aggregate(g.rows.map((r) => r[sizeKey] ?? null), agg) ?? -Infinity }))
      .sort((a, b) => b.size - a.size)
    const keep = new Set(ranked.slice(0, block.topN).map((r) => r.g))
    other = groups.filter((g) => !keep.has(g)).flatMap((g) => g.rows)
    groups = groups.filter((g) => keep.has(g))
  }
  const buckets: Array<{ key: string | number; rows: DataRow[]; other?: boolean }> = [...groups]
  if (other && other.length) buckets.push({ key: OTHER_LABEL, rows: other, other: true })

  let series: ChartSeries[]
  let out: Array<Record<string, string | number | null>>
  if (block.split) {
    const splitKey = block.split
    const values = groupRows(rows, splitKey, false)
      .slice(0, MAX_SPLIT_SERIES)
      .map((g) => String(g.key))
    series = values.map((v, i) => ({ key: `s${i}`, label: v }))
    const column = block.series[0].key
    out = buckets.map((g) => {
      const row: Record<string, string | number | null> = { x: g.key }
      if (g.other) row[OTHER_FLAG] = 1
      values.forEach((v, i) => {
        row[`s${i}`] = aggregate(
          g.rows.filter((r) => String(r[splitKey]) === v).map((r) => r[column] ?? null),
          agg,
        )
      })
      return row
    })
  } else {
    const single = block.type === 'pie' || block.type === 'donut' || block.type === 'funnel'
    series = (single ? block.series.slice(0, 1) : block.series).map((s, i) => ({
      key: `s${i}`,
      label: s.label,
      ...(s.axis === 'right' ? { axis: 'right' as const } : {}),
    }))
    const columns = block.series.map((s) => s.key)
    out = buckets.map((g) => {
      const row: Record<string, string | number | null> = { x: g.key }
      if (g.other) row[OTHER_FLAG] = 1
      series.forEach((s, i) => {
        row[s.key] = aggregate(
          g.rows.map((r) => r[columns[i]] ?? null),
          agg,
        )
      })
      return row
    })
  }

  // Sorting: "Other" always stays last.
  if (block.sort) {
    const tail = out.filter((r) => r[OTHER_FLAG])
    const head = out.filter((r) => !r[OTHER_FLAG])
    const size = (r: Record<string, string | number | null>) =>
      series.reduce((sum, s) => sum + (typeof r[s.key] === 'number' ? (r[s.key] as number) : 0), 0)
    if (block.sort === 'x') head.sort((a, b) => compareCells(a.x as Cell, b.x as Cell))
    else head.sort((a, b) => (block.sort === 'value-asc' ? size(a) - size(b) : size(b) - size(a)))
    out = [...head, ...tail]
  }

  // Trendline: a least-squares line through the first series, over the points in order.
  if (block.trendline && !block.split && out.length > 1) {
    const pairs = out.flatMap((r, i) => (typeof r.s0 === 'number' ? [[i, r.s0] as const] : []))
    const fit = linearFit(pairs)
    if (fit) {
      out = out.map((r, i) => ({ ...r, t0: fit.slope * i + fit.intercept }))
      series = [...series, { key: 't0', label: `${series[0].label} trend`, trend: true }]
    }
  }

  return { rows: out, series, xType }
}

/** Marks the "Other" row of a top-N chart: it is not one category, so it never filters or drills. */
export const OTHER_FLAG = '__other'
export const isOtherRow = (row: Record<string, unknown>) => Boolean(row[OTHER_FLAG])

/** Least squares over (x, y) pairs; null with fewer than two distinct x. */
export function linearFit(points: ReadonlyArray<readonly [number, number]>): { slope: number; intercept: number } | null {
  const n = points.length
  if (n < 2) return null
  let sx = 0
  let sy = 0
  for (const [x, y] of points) {
    sx += x
    sy += y
  }
  const mx = sx / n
  const my = sy / n
  let num = 0
  let den = 0
  for (const [x, y] of points) {
    num += (x - mx) * (y - my)
    den += (x - mx) * (x - mx)
  }
  if (den === 0) return null
  const slope = num / den
  return { slope, intercept: my - slope * mx }
}

/**
 * Small multiples: the chart drawn once per value of its `facet` column (at most MAX_FACETS,
 * in the order values first appear; dates and numbers ascending), each from that value's rows.
 */
export function chartFacets(
  dashboard: Dashboard,
  block: ChartBlock,
  state: FilterState,
): Array<{ key: string | number; label: string; data: ChartData }> {
  if (!block.facet) return []
  const dataset = findDataset(dashboard, block.datasetId)
  const rows = datasetRows(dashboard, block.datasetId, stateForBlock(state, block.id))
  const type = findColumn(dataset, block.facet)?.type ?? 'string'
  return groupRows(rows, block.facet, type !== 'string')
    .slice(0, MAX_FACETS)
    .map((g) => ({ key: g.key, label: formatKey(g.key, type), data: chartFromRows(dataset, block, g.rows) }))
}

/** A log axis cannot show zero or less: the chart then draws linear and says why. */
export function logScaleUsable(data: ChartData): boolean {
  return data.rows.every((row) =>
    data.series.every((s) => {
      const v = row[s.key]
      return typeof v !== 'number' || v > 0
    }),
  )
}

/** Chart types that can draw the period before beside the period shown. */
const COMPARABLE: ReadonlySet<ChartBlock['type']> = new Set(['bar', 'line', 'area', 'combo'])

export type ChartComparison = { data: ChartData; compare: { from: string; to: string } | null }

/**
 * A chart's data with "Compare to previous period" applied: when the toggle is on and a date
 * range narrows the chart's dataset, the first series measured over the window just before is
 * added as a `previous` series. An x on the range's own date column is shifted forward by the
 * range's length (so Nov 9 lines up with Oct 26 for a 14-day range); any other x (categories)
 * is matched as is. Unsplit bar, line, area and combo charts only; nothing lines up is no
 * comparison.
 */
export function chartComparison(dashboard: Dashboard, block: ChartBlock, state: FilterState): ChartComparison {
  const data = chartData(dashboard, block, state)
  if (!compareOn(state) || block.split || !COMPARABLE.has(block.type) || data.rows.length === 0) {
    return { data, compare: null }
  }
  const before = previousPeriod(dashboard, block.datasetId, stateForBlock(state, block.id))
  if (!before) return { data, compare: null }
  const previous = chartData(dashboard, block, before.state)
  const shift = block.x === before.column
  const byKey = new Map<string, number | null>()
  for (const row of previous.rows) {
    const key = shift && typeof row.x === 'string' ? addDays(row.x, before.days) : String(row.x)
    byKey.set(key, typeof row.s0 === 'number' ? row.s0 : null)
  }
  let matched = 0
  const rows = data.rows.map((row) => {
    const value = byKey.get(String(row.x)) ?? null
    if (value !== null) matched++
    return { ...row, p0: value }
  })
  if (matched === 0) return { data, compare: null }
  const label = `${data.series[0].label}, ${formatDateRange(before.from, before.to)}`
  return {
    data: { ...data, rows, series: [...data.series, { key: 'p0', label, previous: true }] },
    compare: { from: before.from, to: before.to },
  }
}

// ---- Conditional colour

export type Tone = 'good' | 'bad' | 'warn' | 'info'
type Rule = { op: '>' | '>=' | '<' | '<=' | '=' | '!='; value: number | string; tone: Tone; label?: string }

/** Does a value meet a rule? Numbers compare as numbers, text case-sensitively and only by = / !=. */
export function ruleMatches(rule: Rule, value: Cell | undefined): boolean {
  if (value === null || value === undefined) return false
  if (typeof rule.value === 'string' || typeof value === 'string') {
    const same = String(value) === String(rule.value)
    return rule.op === '=' ? same : rule.op === '!=' ? !same : false
  }
  switch (rule.op) {
    case '>':
      return value > rule.value
    case '>=':
      return value >= rule.value
    case '<':
      return value < rule.value
    case '<=':
      return value <= rule.value
    case '=':
      return value === rule.value
    case '!=':
      return value !== rule.value
  }
}

/** The first rule a value meets (rules read in order, like a spreadsheet's), or null. */
export function firstRule<R extends Rule>(rules: readonly R[] | undefined, value: Cell | undefined): R | null {
  for (const rule of rules ?? []) if (ruleMatches(rule, value)) return rule
  return null
}

export const TONE_WORD: Record<Tone, string> = { good: 'good', bad: 'needs attention', warn: 'watch', info: 'note' }

/** How a KPI's rule reads aloud: its own label, else the tone in words. */
export const ruleWord = (rule: Rule) => rule.label?.trim() || TONE_WORD[rule.tone]

/** How a combo chart draws each series: as given, else bars first and lines after. */
export function comboKinds(block: ChartBlock): Array<'bar' | 'line'> {
  return block.series.map((s, i) => s.as ?? (i === 0 ? 'bar' : 'line'))
}

export type FunnelStep = {
  key: string | number
  value: number
  /** Share of the first step (1 for the first), or null when the first is 0. */
  ofFirst: number | null
  /** Share of the step before (null for the first, or when that one is 0). */
  ofPrevious: number | null
}

/** A funnel's steps in order, from the chart's first series. Steps with no value are left out. */
export function funnelSteps(data: ChartData): FunnelStep[] {
  const points = data.rows
    .map((row) => ({ key: row.x as string | number, value: row.s0 }))
    .filter((p): p is { key: string | number; value: number } => typeof p.value === 'number')
  const first = points[0]?.value ?? 0
  return points.map((p, i) => {
    const previous = i > 0 ? points[i - 1].value : null
    return {
      key: p.key,
      value: p.value,
      ofFirst: first > 0 ? p.value / first : null,
      ofPrevious: previous !== null && previous > 0 ? p.value / previous : null,
    }
  })
}

/**
 * Where an annotation sits among a chart's x values: on its own value when shown, else (dates and
 * numbers, which read in order) on the first x after it. Null when it falls past the end or the
 * axis is categories.
 */
export function snapToX(keys: Array<string | number>, x: string | number, ordered: boolean): string | number | null {
  const exact = keys.find((k) => String(k) === String(x))
  if (exact !== undefined) return exact
  if (!ordered) return null
  const after = keys.find((k) => compareCells(k, x) > 0)
  return after ?? null
}

/**
 * A heatmap scale over a column's cells: each number maps to 0 (lowest) … 1 (highest); a column
 * of one value maps to 0.5. Anything not a number maps to null.
 */
export function heatScale(cells: Array<Cell | undefined>): (cell: Cell | undefined) => number | null {
  const values = cells.filter((c): c is number => typeof c === 'number' && Number.isFinite(c))
  if (values.length === 0) return () => null
  const min = Math.min(...values)
  const max = Math.max(...values)
  return (cell) => {
    if (typeof cell !== 'number' || !Number.isFinite(cell)) return null
    return max === min ? 0.5 : (cell - min) / (max - min)
  }
}

/** An axis tick: always short ("$80K", "45K", "12%"). */
export function formatAxis(value: number, format: ValueFormat, currency = 'USD'): string {
  if (format === 'percent') return formatPercent(value)
  if (format === 'currency' && Math.abs(value) >= 1000) {
    return numberFormat(`axis-${currency}`, {
      style: 'currency',
      currency,
      notation: 'compact',
      maximumFractionDigits: 1,
    }).format(value)
  }
  return formatValue(value, format === 'currency' ? 'currency' : 'compact', currency)
}

/** The format a chart's values take: from its first series column. */
export function chartValueFormat(dashboard: Dashboard, block: ChartBlock): ValueFormat {
  if (block.valueFormat) return block.valueFormat
  const type = findColumn(findDataset(dashboard, block.datasetId), block.series[0].key)?.type
  if (type === 'currency') return 'currency'
  if (type === 'percent') return 'percent'
  return 'compact'
}

/**
 * The format of one drawn series (`s0`…): the chart's valueFormat when set, else that series'
 * own column (a right-axis "Orders" beside "Revenue" reads as a count, not dollars).
 */
export function seriesValueFormat(dashboard: Dashboard, block: ChartBlock, seriesKey: string): ValueFormat {
  if (block.valueFormat || block.split || !/^s\d+$/.test(seriesKey)) return chartValueFormat(dashboard, block)
  const series = block.series[Number(seriesKey.slice(1))]
  const type = series ? findColumn(findDataset(dashboard, block.datasetId), series.key)?.type : undefined
  if (type === 'currency') return 'currency'
  if (type === 'percent') return 'percent'
  return 'compact'
}

// ---- Tables

export function sortRows(rows: DataRow[], column: string, dir: 'asc' | 'desc'): DataRow[] {
  const sign = dir === 'asc' ? 1 : -1
  return rows
    .map((row, index) => ({ row, index }))
    .sort((a, b) => {
      const x = a.row[column]
      const y = b.row[column]
      const xEmpty = x === null || x === undefined
      const yEmpty = y === null || y === undefined
      // Empty cells sink, whichever way the column is sorted.
      if (xEmpty || yEmpty) return xEmpty === yEmpty ? a.index - b.index : xEmpty ? 1 : -1
      const order = compareCells(x, y)
      return order === 0 ? a.index - b.index : order * sign
    })
    .map(({ row }) => row)
}

export function paginate<T>(rows: T[], page: number, pageSize: number): { rows: T[]; page: number; pages: number } {
  const pages = Math.max(1, Math.ceil(rows.length / pageSize))
  const clamped = Math.min(Math.max(0, page), pages - 1)
  return { rows: rows.slice(clamped * pageSize, (clamped + 1) * pageSize), page: clamped, pages }
}

export function tableColumns(dashboard: Dashboard, block: TableBlock): DataColumn[] {
  const dataset = findDataset(dashboard, block.datasetId)
  if (!dataset) return []
  if (!block.columns) return dataset.columns
  return block.columns
    .map((key) => findColumn(dataset, key))
    .filter((c): c is DataColumn => c !== undefined)
}

// ---- Formatting

const cache = new Map<string, Intl.NumberFormat>()
function numberFormat(key: string, options: Intl.NumberFormatOptions): Intl.NumberFormat {
  let format = cache.get(key)
  if (!format) {
    format = new Intl.NumberFormat(LOCALE, options)
    cache.set(key, format)
  }
  return format
}

function trimNumber(value: number, digits: number): string {
  return numberFormat(`n${digits}`, { maximumFractionDigits: digits }).format(value)
}

/** 0.1234 → "12.3%". */
export function formatPercent(fraction: number): string {
  return `${trimNumber(fraction * 100, 1)}%`
}

export function formatValue(value: number | null | undefined, format: ValueFormat, currency = 'USD', percentDigits?: number): string {
  if (value === null || value === undefined || !Number.isFinite(value)) return EMPTY
  switch (format) {
    case 'integer':
      return numberFormat('int', { maximumFractionDigits: 0 }).format(value)
    case 'compact':
      if (Math.abs(value) < 1000) return trimNumber(value, Math.abs(value) < 10 ? 2 : 1)
      return numberFormat('compact', { notation: 'compact', maximumFractionDigits: 1 }).format(value)
    case 'currency': {
      const abs = Math.abs(value)
      if (abs >= 100_000) {
        return numberFormat(`cc-${currency}`, {
          style: 'currency',
          currency,
          notation: 'compact',
          maximumFractionDigits: 1,
        }).format(value)
      }
      const digits = abs >= 1000 || Number.isInteger(value) ? 0 : 2
      return numberFormat(`c${digits}-${currency}`, {
        style: 'currency',
        currency,
        minimumFractionDigits: digits,
        maximumFractionDigits: digits,
      }).format(value)
    }
    case 'percent':
      return percentDigits === undefined ? formatPercent(value) : `${trimNumber(value * 100, percentDigits)}%`
    case 'number':
      return trimNumber(value, Math.abs(value) >= 100 ? 0 : 2)
  }
}

const dateFormats = {
  short: new Intl.DateTimeFormat(LOCALE, { timeZone: 'UTC', month: 'short', day: 'numeric' }),
  long: new Intl.DateTimeFormat(LOCALE, { timeZone: 'UTC', month: 'short', day: 'numeric', year: 'numeric' }),
  // Date and time apart: ICU versions disagree on the joiner ("," or " at") when combined.
  stampDate: new Intl.DateTimeFormat(LOCALE, {
    timeZone: 'America/Los_Angeles',
    month: 'short',
    day: 'numeric',
    year: 'numeric',
  }),
  stampTime: new Intl.DateTimeFormat(LOCALE, {
    timeZone: 'America/Los_Angeles',
    hour: 'numeric',
    minute: '2-digit',
  }),
}

function utcDate(iso: string): Date | null {
  const match = /^(\d{4})-(\d{2})-(\d{2})$/.exec(iso)
  if (!match) return null
  const date = new Date(Date.UTC(Number(match[1]), Number(match[2]) - 1, Number(match[3])))
  return Number.isNaN(date.getTime()) ? null : date
}

/** "2026-11-09" → "Nov 9, 2026" (or "Nov 9" when `short`). Anything else is returned as-is. */
export function formatDate(iso: string, short = false): string {
  const date = utcDate(iso)
  if (!date) return iso
  return (short ? dateFormats.short : dateFormats.long).format(date)
}

/** "Oct 5 – Nov 9, 2026"; the year once when both ends share it. */
export function formatDateRange(from: string, to: string): string {
  if (from === to) return formatDate(from)
  const a = utcDate(from)
  const b = utcDate(to)
  if (!a || !b) return `${from} – ${to}`
  if (a.getUTCFullYear() === b.getUTCFullYear()) return `${formatDate(from, true)} – ${formatDate(to)}`
  return `${formatDate(from)} – ${formatDate(to)}`
}

/** The as-of stamp: a date reads as a calendar day, a date-time in Pacific time. */
export function formatAsOf(asOf: string): string {
  if (/^\d{4}-\d{2}-\d{2}$/.test(asOf)) return formatDate(asOf)
  const date = new Date(asOf)
  if (Number.isNaN(date.getTime())) return asOf
  return `${dateFormats.stampDate.format(date)}, ${dateFormats.stampTime.format(date)} PT`
}

/** A group or axis key: dates read as dates, numbers formatted. */
export function formatKey(key: string | number, type: ColumnType, short = false): string {
  if (type === 'date' && typeof key === 'string') return formatDate(key, short)
  if (typeof key === 'number') return formatCell(key, type)
  return key
}

export function formatCell(value: Cell | undefined, type: ColumnType, currency?: string): string {
  if (value === null || value === undefined) return EMPTY
  if (typeof value === 'string') return type === 'date' ? formatDate(value) : value
  if (type === 'currency') {
    return numberFormat(`cell-${currency ?? 'USD'}`, {
      style: 'currency',
      currency: currency ?? 'USD',
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    }).format(value)
  }
  if (type === 'percent') return formatPercent(value)
  return trimNumber(value, Number.isInteger(value) ? 0 : 2)
}

// ---- Provenance

/**
 * The time window a block's numbers cover. A date-range filter on its dataset wins (the range
 * chosen, its open ends filled from the data); then the source's own `window`; then the span
 * of the dataset's first date column in the rows shown.
 */
export function blockWindow(dashboard: Dashboard, block: DataBlock, state: FilterState): string | null {
  const datasetId = blockDatasetId(block)
  const dataset = findDataset(dashboard, datasetId)
  for (const filter of dashboard.filters ?? []) {
    const choice = state[filter.id]
    if (filter.type !== 'date-range' || !dataset || !filter.datasetIds.includes(dataset.id)) continue
    if (!choice || choice.kind !== 'date-range' || !isFilterActive(choice)) continue
    const bounds = dateBounds(dashboard, filter)
    const from = choice.from ?? bounds?.min
    const to = choice.to ?? bounds?.max
    if (from && to) return formatDateRange(from, to)
  }
  const source = findSource(dashboard, blockSourceId(dashboard, block))
  if (source?.window) return source.window
  const dateColumn = dataset?.columns.find((c) => c.type === 'date')
  if (!dataset || !dateColumn) return null
  const dates = filterRows(dataset, dashboard.filters, state)
    .map((r) => r[dateColumn.key])
    .filter((v): v is string => typeof v === 'string')
    .sort()
  if (dates.length === 0) return null
  return formatDateRange(dates[0], dates[dates.length - 1])
}

const AGG_WORD: Record<Agg, string> = {
  sum: 'Sum',
  avg: 'Average',
  min: 'Lowest',
  max: 'Highest',
  last: 'Latest',
  count: 'Count',
}

function describeMeasure(dataset: Dataset | undefined, m: Measure): string {
  const column = findColumn(dataset, m.column)
  return `${AGG_WORD[m.agg].toLowerCase()} of ${column?.label ?? m.column}`
}

/** How a KPI's number is worked out, in one sentence: "Sum of Active users over 3 rows …". */
export function explainKpi(dashboard: Dashboard, block: KpiBlock, state: FilterState): string {
  const dataset = findDataset(dashboard, block.datasetId)
  const result = kpiResult(dashboard, block, state)
  const v = block.value
  let how: string
  if ('literal' in v) how = 'A fixed number, as reported by the source'
  else if ('ratio' in v) {
    how = `${capitalise(describeMeasure(dataset, v.ratio.numerator))} divided by ${describeMeasure(dataset, v.ratio.denominator)}`
  } else how = capitalise(describeMeasure(dataset, v))
  const parts = [how]
  if (!('literal' in v)) {
    const sparkType = block.sparkline ? findColumn(dataset, block.sparkline)?.type : undefined
    const where =
      result.period !== null && block.sparkline
        ? ` for ${findColumn(dataset, block.sparkline)?.label ?? block.sparkline} ${formatKey(result.period, sparkType ?? 'string')}`
        : ''
    parts.push(`over ${result.rowCount} ${result.rowCount === 1 ? 'row' : 'rows'}${where}`)
  }
  const filters = activeFilterSummary(dashboard, block.datasetId, state)
  if (filters) parts.push(`(${filters})`)
  return `${parts.join(' ')}.`
}

/** "Platform: Web; Weeks: Oct 5 – Nov 9, 2026", for the filters that touch a dataset. */
export function activeFilterSummary(dashboard: Dashboard, datasetId: string | undefined, state: FilterState): string {
  if (!datasetId) return ''
  const parts: string[] = []
  for (const filter of dashboard.filters ?? []) {
    const choice = state[filter.id]
    if (!filter.datasetIds.includes(datasetId) || !isFilterActive(choice)) continue
    if (choice.kind === 'select') parts.push(`${filter.label}: ${choice.value}`)
    else if (choice.kind === 'multi') parts.push(`${filter.label}: ${choice.values.join(', ')}`)
    else if (choice.kind === 'date-range') {
      const bounds = dateBounds(dashboard, filter)
      const from = choice.from ?? bounds?.min
      const to = choice.to ?? bounds?.max
      if (from && to) parts.push(`${filter.label}: ${formatDateRange(from, to)}`)
    }
  }
  const cross = crossFilter(state)
  const dataset = findDataset(dashboard, datasetId)
  if (cross && dataset?.columns.some((c) => c.key === cross.column)) {
    const type = findColumn(dataset, cross.column)?.type ?? 'string'
    parts.push(`${cross.label}: ${formatKey(cross.value, type)}`)
  }
  return parts.join('; ')
}

const capitalise = (text: string) => text.charAt(0).toUpperCase() + text.slice(1)

export const SUMMARY_LIMIT = 1600

/** A short plain-text account of a block as shown, for "Ask Buffy". Never more than ~800 chars. */
export function describeBlock(dashboard: Dashboard, block: DataBlock, state: FilterState): string {
  const source = findSource(dashboard, blockSourceId(dashboard, block))
  const window = blockWindow(dashboard, block, state)
  const filters = activeFilterSummary(dashboard, blockDatasetId(block), state)
  const provenance = [
    source ? `Source: ${source.label}` : '',
    window ? `Window: ${window}` : '',
    filters ? `Filters: ${filters}` : '',
  ]
    .filter(Boolean)
    .join('. ')

  let body: string
  switch (block.kind) {
    case 'kpi': {
      const r = kpiResult(dashboard, block, state)
      body = `${block.label}: ${formatValue(r.value, block.format, dashboard.currency, block.percentDigits)}`
      if (r.delta) body += ` (${r.delta.text} ${r.delta.label}, ${r.delta.direction})`
      const rule = firstRule(block.rules, r.value)
      if (rule) body += `. Flagged: ${ruleWord(rule)}`
      if (r.trend.length > 1) {
        body += `. Trend: ${r.trend
          .map((p) => formatValue(p.value, block.format, dashboard.currency, block.percentDigits))
          .join(', ')}`
      }
      body += `. ${explainKpi(dashboard, block, state)}`
      break
    }
    case 'chart': {
      body = describeChart(dashboard, block, state)
      break
    }
    case 'table': {
      const rows = datasetRows(dashboard, block.datasetId, state)
      const columns = tableColumns(dashboard, block)
      body = `Table "${block.title}", ${rows.length} rows; columns ${columns.map((c) => c.label).join(', ')}.`
      const sorted = block.sort ? sortRows(rows, block.sort.column, block.sort.dir) : rows
      const sample = sorted
        .slice(0, 3)
        .map((r) => columns.map((c) => formatCell(r[c.key], c.type, dashboard.currency)).join(' | '))
      if (sample.length) body += ` First rows: ${sample.join('; ')}.`
      break
    }
    case 'text':
      body = block.markdown.replace(/\s+/g, ' ').trim()
      break
    case 'section':
      body = `Section "${block.title}"${block.description ? `: ${block.description}` : ''}.`
      break
  }
  // The body gives way first: where the numbers came from always survives the cap.
  const tail = provenance ? ` ${provenance}.` : ''
  const room = SUMMARY_LIMIT - tail.length
  const clipped = body.length > room ? `${body.slice(0, Math.max(0, room - 1))}…` : body
  return `${clipped}${tail}`
}

/**
 * "Explain this chart": what it plots and how rows combine, each series' total, range and
 * change from first to last point, the reference lines and annotations, the comparison when
 * one is drawn, and the last points.
 */
export function describeChart(dashboard: Dashboard, block: ChartBlock, state: FilterState): string {
  const { data, compare } = chartComparison(dashboard, block, state)
  const format = chartValueFormat(dashboard, block)
  const dataset = findDataset(dashboard, block.datasetId)
  const xLabel = findColumn(dataset, block.x)?.label ?? block.x
  const show = (value: number | null) => formatValue(value, format, dashboard.currency)
  const parts: string[] = []
  const plotted = data.series.filter((s) => !s.previous && !s.trend)
  parts.push(
    `${capitalise(block.type.replace('-', ' '))} chart "${block.title}", ${plotted.map((s) => s.label).join(', ')} by ${xLabel}, ${data.rows.length} points.`,
  )
  if (block.split) {
    parts.push(`One series per ${findColumn(dataset, block.split)?.label ?? block.split}.`)
  }
  if (block.type !== 'scatter') {
    parts.push(`Rows sharing a ${xLabel} are combined by ${AGG_WORD[block.agg ?? 'sum'].toLowerCase()}.`)
    const ordered = data.xType !== 'string'
    for (const s of plotted.slice(0, 4)) {
      const values = data.rows.map((r) => r[s.key]).filter((v): v is number => typeof v === 'number')
      if (values.length === 0) continue
      const min = Math.min(...values)
      const max = Math.max(...values)
      const lowAt = data.rows.find((r) => r[s.key] === min)?.x as string | number
      const highAt = data.rows.find((r) => r[s.key] === max)?.x as string | number
      const total = values.reduce((a, b) => a + b, 0)
      // A total of rates or averages means nothing: those read as their average.
      const additive = (block.agg === undefined || block.agg === 'sum' || block.agg === 'count') && format !== 'percent'
      const overall = additive ? `total ${show(total)}` : `average ${show(total / values.length)}`
      let line = `${s.label}: ${overall}, high ${show(max)} (${formatKey(highAt, data.xType, true)}), low ${show(min)} (${formatKey(lowAt, data.xType, true)})`
      if (ordered && values.length > 1 && values[0] !== 0) {
        line += `, ${formatPercent((values[values.length - 1] - values[0]) / Math.abs(values[0]))} first to last`
      }
      parts.push(`${line}.`)
    }
  }
  if (block.references?.length) {
    parts.push(`Target lines: ${block.references.map((r) => `${r.label ? `${r.label} ` : ''}${show(r.value)}`).join(', ')}.`)
  }
  if (block.annotations?.length) {
    parts.push(`Marked: ${block.annotations.map((a) => `${a.label} (${formatKey(a.x, data.xType, true)})`).join(', ')}.`)
  }
  if (compare) parts.push(`Dashed: the same measure over ${formatDateRange(compare.from, compare.to)}.`)
  const points = data.rows.slice(-8).map((row) => {
    const x = formatKey(row.x as string | number, data.xType, true)
    const ys = data.series.map((s) => show(row[s.key] as number | null))
    return `${x}: ${ys.join(' / ')}`
  })
  if (points.length) parts.push(`${data.rows.length > 8 ? 'Last 8: ' : ''}${points.join('; ')}.`)
  if (block.note) parts.push(`Note: ${block.note}`)
  return parts.join(' ')
}

export function blockTitle(block: DataBlock): string {
  switch (block.kind) {
    case 'kpi':
      return block.label
    case 'text':
      return block.markdown.replace(/[#*_`>]/g, '').trim().split('\n')[0].slice(0, 80)
    default:
      return block.title
  }
}

// ---- Freshness (agent-refreshed data: nothing on the page fetches)

const HOUR_MS = 3_600_000

export type Freshness = {
  /** When the rows were pulled: the dataset's own stamp, else the page's asOf. Null when unknown. */
  fetchedAt: string | null
  /** Whether the stamp is the dataset's own (true) or the page's asOf (false). */
  own: boolean
  ageHours: number | null
  staleAfterHours: number | null
  /** Older than its recipe allows. Never stale without a threshold or a stamp. */
  stale: boolean
  /** "Nov 14, 2026, 3:00 PM PT" (or a calendar day). */
  label: string
  /** "3 days ago", "just now"; empty when unknown. */
  age: string
}

/** Milliseconds since the epoch for an ISO date (start of that UTC day) or date-time; null if unreadable. */
export function stampTime(iso: string): number | null {
  const t = Date.parse(/^\d{4}-\d{2}-\d{2}$/.test(iso) ? `${iso}T00:00:00Z` : iso)
  return Number.isNaN(t) ? null : t
}

/** "just now", "40 minutes ago", "5 hours ago", "3 days ago". */
export function formatAge(hours: number): string {
  if (hours < 1 / 60) return 'just now'
  if (hours < 1) {
    const m = Math.round(hours * 60)
    return `${m} minute${m === 1 ? '' : 's'} ago`
  }
  if (hours < 48) {
    const h = Math.round(hours)
    return `${h} hour${h === 1 ? '' : 's'} ago`
  }
  const d = Math.round(hours / 24)
  return `${d} days ago`
}

export function datasetFreshness(dashboard: Dashboard, dataset: Dataset, now: number): Freshness {
  const own = dataset.provenance?.fetchedAt
  const fetchedAt = own ?? dashboard.asOf ?? null
  const staleAfterHours = dataset.refresh?.staleAfterHours ?? null
  const at = fetchedAt === null ? null : stampTime(fetchedAt)
  // A date-only stamp means some time that day: age it from the day's end, never from its start.
  const effective = at === null || fetchedAt === null ? null : /T/.test(fetchedAt) ? at : at + 24 * HOUR_MS - 1
  const ageHours = effective === null ? null : Math.max(0, (now - effective) / HOUR_MS)
  return {
    fetchedAt,
    own: own !== undefined,
    ageHours,
    staleAfterHours,
    stale: ageHours !== null && staleAfterHours !== null && ageHours > staleAfterHours,
    label: fetchedAt === null ? '' : formatAsOf(fetchedAt),
    age: ageHours === null ? '' : formatAge(ageHours),
  }
}

/** The freshness behind a block's numbers: its dataset's, or the page's asOf for a literal KPI. */
export function blockFreshness(dashboard: Dashboard, block: DataBlock, now: number): Freshness | null {
  const dataset = findDataset(dashboard, blockDatasetId(block))
  if (dataset) return datasetFreshness(dashboard, dataset, now)
  if (block.kind !== 'kpi' || !dashboard.asOf) return null
  const at = stampTime(dashboard.asOf)
  const ageHours = at === null ? null : Math.max(0, (now - at) / HOUR_MS)
  return {
    fetchedAt: dashboard.asOf,
    own: false,
    ageHours,
    staleAfterHours: null,
    stale: false,
    label: formatAsOf(dashboard.asOf),
    age: ageHours === null ? '' : formatAge(ageHours),
  }
}

/** Datasets older than their recipe allows, in page order. */
export function staleDatasets(dashboard: Dashboard, now: number): string[] {
  return dashboard.datasets.filter((d) => datasetFreshness(dashboard, d, now).stale).map((d) => d.id)
}

/** The page's own stamp: the oldest dataset stamp when datasets carry them, else asOf. */
export function pageFreshness(dashboard: Dashboard, now: number): { label: string; stale: number; oldest: Freshness | null } {
  let oldest: Freshness | null = null
  let stale = 0
  for (const dataset of dashboard.datasets) {
    const f = datasetFreshness(dashboard, dataset, now)
    if (f.stale) stale++
    if (f.fetchedAt === null) continue
    if (!oldest || (stampTime(f.fetchedAt) ?? 0) < (stampTime(oldest.fetchedAt ?? '') ?? 0)) oldest = f
  }
  const label = oldest?.label ?? (dashboard.asOf ? formatAsOf(dashboard.asOf) : '')
  return { label, stale, oldest }
}

/**
 * A refresh recipe as typed in a form: blank fields dropped, null when there is no recipe. The
 * typed `how` replaces the legacy free-text file and command (a form never writes those again).
 */
export function cleanRefresh(input: { recipe: string; how?: RecipeHow | null; staleAfterHours?: number | null }): DatasetRefresh | null {
  const recipe = input.recipe.trim().slice(0, 2000)
  if (!recipe) return null
  const out: DatasetRefresh = { recipe }
  if (input.how) out.how = input.how
  const hours = input.staleAfterHours
  if (hours !== null && hours !== undefined && Number.isFinite(hours) && hours >= 1) out.staleAfterHours = Math.min(Math.round(hours), 24 * 366)
  return out
}

// ---- Tables: formats, totals, search, TSV

/** How a table cell reads: the block's format for that column when it sets one, else by type. */
export function formatTableCell(block: TableBlock, column: DataColumn, value: Cell | undefined, currency?: string): string {
  const format = block.formats?.[column.key]
  if (format && typeof value === 'number') return formatValue(value, format, currency)
  return formatCell(value, column.type, currency)
}

export type TableTotal = { value: number | null; agg: 'sum' | 'avg' }

/**
 * A totals row: sums for number and currency columns, averages for percent (a sum of rates means
 * nothing). Columns in `skip` (calculated per row, like a ratio) get no total: summing a per-row
 * formula is rarely the formula over the total.
 */
export function tableTotals(columns: DataColumn[], rows: DataRow[], skip: ReadonlySet<string> = new Set()): Map<string, TableTotal> {
  const out = new Map<string, TableTotal>()
  for (const column of columns) {
    if (!isNumericType(column.type) || skip.has(column.key)) continue
    const agg = column.type === 'percent' ? 'avg' : 'sum'
    out.set(column.key, { value: aggregate(rows.map((r) => r[column.key] ?? null), agg), agg })
  }
  return out
}

/** Rows whose shown text contains every word of the query (case-insensitive). */
export function searchRows(rows: DataRow[], columns: DataColumn[], query: string, currency?: string): DataRow[] {
  const words = query.trim().toLowerCase().split(/\s+/).filter(Boolean)
  if (words.length === 0) return rows
  return rows.filter((row) => {
    const text = columns
      .map((c) => {
        const v = row[c.key]
        return v === null || v === undefined ? '' : `${String(v)} ${formatCell(v, c.type, currency)}`
      })
      .join(' ')
      .toLowerCase()
    return words.every((w) => text.includes(w))
  })
}

/**
 * One TSV cell: tabs and line breaks become spaces; text a spreadsheet would run as a formula
 * gets a leading `'`. A number's shown text ("-2", "-$5.00") is never guarded: it is a value.
 */
export function tsvCell(cell: string | { text: string; numeric: true }): string {
  const text = typeof cell === 'string' ? cell : cell.text
  const flat = text.replace(/[\t\r\n]+/g, ' ')
  // Text only (export-safe.ts): a formatted number such as -5 stays a value.
  return typeof cell === 'string' ? formulaSafeCell(flat) : flat
}

/** A table as shown (formatted cells), tab-separated, for pasting into a spreadsheet. */
export function toTsv(header: string[], body: Array<Array<string | { text: string; numeric: true }>>): string {
  return [header, ...body].map((line) => line.map(tsvCell).join('\t')).join('\n')
}

// ---- Explaining a single number

/** How one chart point is worked out: "Sum of Active users over 3 rows where Week is Nov 9 …". */
export function explainChartPoint(dashboard: Dashboard, block: ChartBlock, state: FilterState, x: string | number): string {
  const dataset = findDataset(dashboard, block.datasetId)
  const own = stateForBlock(state, block.id)
  const rows = datasetRows(dashboard, block.datasetId, own)
  const xCol = findColumn(dataset, block.x)
  const matching = rows.filter((r) => r[block.x] !== null && r[block.x] !== undefined && String(r[block.x]) === String(x))
  const agg = AGG_WORD[block.agg ?? 'sum'].toLowerCase()
  const what = block.split
    ? `${findColumn(dataset, block.series[0].key)?.label ?? block.series[0].key}, one series per ${findColumn(dataset, block.split)?.label ?? block.split}`
    : block.series.map((s) => findColumn(dataset, s.key)?.label ?? s.key).join(', ')
  const calcs = block.series
    .map((s) => dataset?.calculated?.find((c) => c.key === s.key))
    .filter((c): c is NonNullable<typeof c> => c !== undefined)
    .map((c) => `${c.label} = ${c.expr}`)
  const filters = activeFilterSummary(dashboard, block.datasetId, own)
  return [
    `${capitalise(agg)} of ${what} over ${matching.length} ${matching.length === 1 ? 'row' : 'rows'} where ${xCol?.label ?? block.x} is ${formatKey(x, xCol?.type ?? 'string')}`,
    calcs.length ? `; calculated per row as ${calcs.join('; ')}` : '',
    filters ? ` (${filters})` : '',
    '.',
  ].join('')
}

/** Where one table cell comes from: a stored value, or its calculation. */
export function explainCell(dashboard: Dashboard, datasetId: string, columnKey: string): string {
  const dataset = findDataset(dashboard, datasetId)
  const calc = dataset?.calculated?.find((c) => c.key === columnKey)
  const source = findSource(dashboard, dataset?.sourceId)
  if (calc) return `${calc.label} is calculated for each row: ${calc.expr}.`
  const column = findColumn(dataset, columnKey)
  return `${column?.label ?? columnKey} is a stored value in ${dataset?.title ?? datasetId}${source ? `, from ${source.label}` : ''}.`
}

// ---- The agent's readout: computed values, so it can check its own output

/** Things worth fixing on a page as computed, in plain words (the agent reads these first). */
export function dashboardChecks(dashboard: Dashboard, now: number): string[] {
  const out: string[] = []
  const state = defaultFilterState(dashboard)
  const doubts = new Map(datasetDoubts(dashboard).map((d) => [d.datasetId, d]))
  for (const dataset of dashboard.datasets) {
    const f = datasetFreshness(dashboard, dataset, now)
    if (f.stale) out.push(`dataset ${dataset.id} is stale: fetched ${f.age}, its recipe allows ${f.staleAfterHours} hours`)
    const doubt = doubts.get(dataset.id)
    if (doubt?.edited) out.push(`dataset ${dataset.id} was edited by hand (${doubt.edited.by}, ${doubt.edited.at}) and no longer matches its source`)
    if (doubt?.failed) out.push(`dataset ${dataset.id}: the last refresh failed (${doubt.failed.slice(0, 200)})`)
    if (doubt?.voidCertificate) out.push(`dataset ${dataset.id} changed since the owner certified it`)
    for (const problem of doubt?.checks ?? []) out.push(`dataset ${dataset.id} ${problem}`)
    if (dataset.rows.length === 0 && dashboard.blocks.some((b) => blockDatasetId(b) === dataset.id)) {
      out.push(`dataset ${dataset.id} has no rows but blocks read it`)
    }
    if (dataset.rows.length > 0) {
      for (const calc of dataset.calculated ?? []) {
        const empty = dataset.rows.filter((row) => row[calc.key] === null || row[calc.key] === undefined).length
        if (empty === dataset.rows.length) {
          out.push(`calculated column ${dataset.id}.${calc.key} is empty on every row (a missing input or a division by zero?)`)
        } else if (empty > 0) {
          // A per-row null is a missing input or a division by zero there: worth knowing, not always wrong.
          out.push(`calculated column ${dataset.id}.${calc.key} is empty on ${empty} of ${dataset.rows.length} rows`)
        }
      }
    }
  }
  for (const block of dashboard.blocks) {
    if (block.kind === 'kpi') {
      if (kpiResult(dashboard, block, state).value === null) out.push(`kpi ${block.id} shows no value`)
    } else if (block.kind === 'chart') {
      const data = chartData(dashboard, block, state)
      if (data.rows.length === 0) out.push(`chart ${block.id} has no points`)
      else if (block.scale === 'log' && !logScaleUsable(data)) out.push(`chart ${block.id} asks for a log scale but has values of 0 or less; it draws linear`)
      else if (data.rows.every((r) => data.series.every((s) => r[s.key] === null))) out.push(`chart ${block.id} has points but every value is empty`)
    } else if (block.kind === 'table') {
      if (datasetRows(dashboard, block.datasetId, state).length === 0) out.push(`table ${block.id} has no rows`)
    }
  }
  return out
}

/** A compact per-series account: points, total or average, low, high, first and last. */
function seriesLine(data: ChartData, key: string, label: string, show: (v: number | null) => string, additive: boolean): string {
  const values = data.rows.map((r) => r[key]).filter((v): v is number => typeof v === 'number')
  if (values.length === 0) return `${label}: no values`
  const total = values.reduce((a, b) => a + b, 0)
  const x = (row: Record<string, unknown> | undefined) =>
    row ? formatKey(row.x as string | number, data.xType, true) : ''
  const first = data.rows.find((r) => typeof r[key] === 'number')
  const last = [...data.rows].reverse().find((r) => typeof r[key] === 'number')
  return `${label}: ${additive ? `total ${show(total)}` : `avg ${show(total / values.length)}`}, low ${show(Math.min(...values))}, high ${show(Math.max(...values))}, first ${show(values[0])} (${x(first)}), last ${show(values[values.length - 1])} (${x(last)})`
}

export const READOUT_LIMIT = 6000

/**
 * What the page computes, as the agent's `read_artifact` sees it: checks first, then each
 * dataset's freshness, each KPI's value and change, and each chart's series. Default filters
 * and parameters; calculated columns computed. Capped at `limit` characters.
 */
export function dashboardReadout(stored: Dashboard, now: number, limit = READOUT_LIMIT): string {
  // Calculated columns are computed here exactly as the page computes them.
  const dashboard = materializeDashboard(stored)
  const state = defaultFilterState(dashboard)
  const lines: string[] = []
  const checks = dashboardChecks(dashboard, now)
  lines.push(checks.length ? `Checks: ${checks.join('; ')}.` : 'Checks: nothing to flag.')
  const fresh = dashboard.datasets.map((d) => {
    const f = datasetFreshness(dashboard, d, now)
    const recipe = d.refresh ? `, ${d.refresh.how ? `${d.refresh.how.kind} ` : ''}recipe${d.refresh.staleAfterHours ? ` (stale after ${d.refresh.staleAfterHours}h)` : ''}` : ', no recipe'
    const change = d.lastChange
      ? `, last change ${d.lastChange.at}: +${d.lastChange.added} -${d.lastChange.removed} ~${d.lastChange.changed} =${d.lastChange.unchanged}${d.lastChange.keyed ? '' : ' (unkeyed)'}`
      : ''
    const run = d.lastRun ? `, last run ${d.lastRun.status} ${d.lastRun.at}` : ''
    const mark = certification(d) === 'valid' ? ', certified' : certification(d) === 'void' ? ', certificate void' : ''
    const edited = d.edited ? `, EDITED by ${d.edited.by}` : ''
    return `${d.id} ${f.fetchedAt ? `fetched ${f.fetchedAt}${f.own ? '' : ' (page asOf)'}${f.stale ? ' STALE' : ''}` : 'no fetch time'}${recipe}${edited}${run}${change}${mark}`
  })
  if (fresh.length) lines.push(`Freshness: ${fresh.join('; ')}.`)
  for (const block of dashboard.blocks) {
    if (block.kind === 'kpi') {
      const r = kpiResult(dashboard, block, state)
      const rule = firstRule(block.rules, r.value)
      lines.push(
        `kpi ${block.id} = ${formatValue(r.value, block.format, dashboard.currency, block.percentDigits)}${r.value === null ? '' : ` (raw ${Number(r.value.toPrecision(12))})`}${r.delta ? `, ${r.delta.text} ${r.delta.label}` : ''}${rule ? `, flagged ${ruleWord(rule)}` : ''}${r.period !== null ? `, period ${r.period}` : ''}, ${r.rowCount} rows`,
      )
    } else if (block.kind === 'chart') {
      const data = chartData(dashboard, block, state)
      const series = data.series
        .filter((s) => !s.trend)
        .slice(0, 4)
        .map((s) => {
          const format = s.previous ? chartValueFormat(dashboard, block) : seriesValueFormat(dashboard, block, s.key)
          const show = (v: number | null) => formatValue(v, format, dashboard.currency)
          const additive = (block.agg === undefined || block.agg === 'sum' || block.agg === 'count') && format !== 'percent'
          return seriesLine(data, s.key, s.label, show, additive)
        })
      const facets = block.facet ? `, ${chartFacets(dashboard, block, state).length} small multiples` : ''
      lines.push(`chart ${block.id} (${block.type}) ${data.rows.length} points${facets}: ${series.join(' | ')}`)
    } else if (block.kind === 'table') {
      lines.push(`table ${block.id}: ${datasetRows(dashboard, block.datasetId, state).length} rows`)
    }
  }
  const text = lines.join('\n')
  return text.length > limit ? `${text.slice(0, limit - 1)}…` : text
}

 * Freebuff Data: the content model of a data artifact (dashboards, metrics pages, KPI trackers,
 * scorecards, data stories), version 1.
 *
 * ===== For the agent writing one =====
 *
 * A data artifact is ONE JSON object. It is declarative: no HTML, no scripts, no formulas as
 * code. Every number on the page is computed by Freebuff from the rows you put in `datasets`,
 * and every tile and chart names the source it came from. Unknown keys are rejected everywhere.
 *
 * {
 *   "version": 1,
 *   "title": "Weekly product metrics",
 *   "subtitle": "Freebuff, all platforms",           // optional, one line
 *   "asOf": "2026-11-14",                           // optional: ISO date or date-time; when the data was pulled
 *   "layout": "grid",                               // "grid" (dashboard) or "story" (one column, text between charts)
 *   "currency": "USD",                              // optional ISO 4217 code for currency values (default USD)
 *   "sources": [                                    // where numbers come from. At least one.

 *       "window": "Sep 21 – Nov 14, 2026 (PT)" }   // detail and window optional

 *   "filters": [ ... optional, see below ... ]

 *    LATEST group's value, a sparkline of every group, and a delta against the group before it.

  const name = base.replace(/\.exe$/i, '')
  if (/\{\{/.test(program)) return 'the program cannot be a {{blank}}'
  if (SHELLS.has(base.toLowerCase()) || SHELLS.has(name.toLowerCase())) return `"${base}" is a shell; give the program and its arguments instead`
  if (WRAPPERS.has(name)) return `"${name}" runs another command; give that command directly`
  if (Object.hasOwn(INLINE_CODE, name) || /^python\d/.test(name)) {
    if (name === 'awk') return '"awk" runs a program given inline; put it in a workspace file'
    const flags = (Object.hasOwn(INLINE_CODE, name) ? INLINE_CODE[name] : undefined) ?? INLINE_CODE.python!
    const bad = argv.slice(1).find((arg) => flags.some((flag) => arg === flag || arg.startsWith(`${flag}=`)))
    if (bad) return `"${name} ${bad}" runs code given inline; put it in a workspace file`
  }
  const odd = argv.find((arg) => SHELL_ONLY.test(arg))
  if (odd !== undefined) return 'an argument holds shell syntax or control characters; each argument is plain text'
  return null
}

/** Why a path cannot be a recipe's workspace file, or null: relative, inside the workspace. */
export function workspacePathProblem(path: string): string | null {
  if (/^[\\/~]/.test(path) || /^[A-Za-z]:/.test(path)) return 'a path inside the workspace, like queries/weekly.sql (not absolute)'
  if (path.split(/[\\/]/).some((part) => part === '..')) return 'a path inside the workspace (no "..")'
  if (/[\u0000-\u001f\u007f]/.test(path)) return 'no control characters'
  if (/\{\{/.test(path)) return 'a file path cannot hold a {{blank}}'
  return null
}

const NUMERIC: ReadonlySet<ColumnType> = new Set(['number', 'currency', 'percent'])
export const isNumericType = (type: ColumnType) => NUMERIC.has(type)

/**
 * The cross-references zod cannot see: unique ids, every id pointing at something that exists,
 * columns of the right type, cells matching their column. Each issue is `[path, message]`.
 */
function referenceIssues(d: Dashboard): Array<{ path: Array<string | number>; message: string }> {
  const issues: Array<{ path: Array<string | number>; message: string }> = []
  const add = (path: Array<string | number>, message: string) => issues.push({ path, message })

  const unique = (items: Array<{ id: string }>, what: string, key: string) => {
    const seen = new Set<string>()
    items.forEach((item, i) => {
      if (seen.has(item.id)) add([key, i, 'id'], `duplicate ${what} id "${item.id}"`)
      seen.add(item.id)
    })
  }
  unique(d.sources, 'source', 'sources')
  unique(d.datasets, 'dataset', 'datasets')
  unique(d.blocks, 'block', 'blocks')
  unique(d.filters ?? [], 'filter', 'filters')
  unique(d.params ?? [], 'parameter', 'params')

  const paramIds = new Set((d.params ?? []).map((p) => p.id))
  const blankIds = new Set([...paramIds, ...(d.filters ?? []).map((f) => f.id)])
  ;(d.params ?? []).forEach((p, i) => {
    if (p.min !== undefined && p.max !== undefined && p.min > p.max) add(['params', i, 'min'], 'min is above max')
    if (p.min !== undefined && p.default < p.min) add(['params', i, 'default'], 'the default is below min')
    if (p.max !== undefined && p.default > p.max) add(['params', i, 'default'], 'the default is above max')
  })

  const sources = new Set(d.sources.map((s) => s.id))
  const datasets = new Map(d.datasets.map((ds) => [ds.id, ds]))

  d.datasets.forEach((ds, i) => {
    if (!sources.has(ds.sourceId)) add(['datasets', i, 'sourceId'], `unknown source "${ds.sourceId}"`)
    const cols = new Map<string, ColumnType>()
    ds.columns.forEach((c, j) => {
      if (cols.has(c.key)) add(['datasets', i, 'columns', j, 'key'], `duplicate column "${c.key}"`)
      cols.set(c.key, c.type)
    })
    for (let r = 0; r < ds.rows.length; r++) {
      const row = ds.rows[r]
      for (const [key, value] of Object.entries(row)) {
        const type = cols.get(key)
        if (!type) {
          add(['datasets', i, 'rows', r, key], `"${key}" is not a column of dataset "${ds.id}"`)
          continue
        }
        if (value === null) continue
        if (isNumericType(type) && typeof value !== 'number') {
          add(['datasets', i, 'rows', r, key], `"${key}" is a ${type} column; expected a number`)
        } else if (type === 'date' && (typeof value !== 'string' || !ISO_DATE.test(value))) {
          add(['datasets', i, 'rows', r, key], `"${key}" is a date column; expected "YYYY-MM-DD"`)
        } else if (type === 'string' && typeof value !== 'string') {
          add(['datasets', i, 'rows', r, key], `"${key}" is a string column; expected a string`)
        }
      }
      // One bad dataset should not bury every other issue.
      if (issues.length > 50) return
    }
    ds.key?.forEach((key, j) => {
      if (!cols.has(key)) add(['datasets', i, 'key', j], `"${key}" is not a stored column of dataset "${ds.id}"`)
    })
    ds.expect?.required?.forEach((key, j) => {
      if (!cols.has(key)) add(['datasets', i, 'expect', 'required', j], `"${key}" is not a stored column of dataset "${ds.id}"`)
    })
    const how = ds.refresh?.how
    if (how) {
      const texts = how.kind === 'ask' ? [how.question] : how.kind === 'command' ? how.argv : how.kind === 'integration' ? integrationBlankTexts(how) : []
      for (const text of texts) {
        for (const match of text.matchAll(RECIPE_BLANK)) {
          if (!blankIds.has(match[1]!)) add(['datasets', i, 'refresh', 'how'], `{{${match[1]}}} names no filter or parameter`)
        }
      }
    }
    const calcKeys = new Set<string>()
    ;(ds.calculated ?? []).forEach((c, j) => {
      if (calcKeys.has(c.key)) add(['datasets', i, 'calculated', j, 'key'], `duplicate calculated column "${c.key}"`)
      calcKeys.add(c.key)
    })
    for (const issue of calculatedIssues(ds, paramIds)) {
      add(['datasets', i, 'calculated', issue.index, issue.field], issue.message)
    }
  })

  // Calculated columns read like stored ones everywhere a block or filter names a column.
  const columnOf = (dsId: string, key: string) => {
    const ds = datasets.get(dsId)
    return ds?.columns.find((c) => c.key === key) ?? ds?.calculated?.find((c) => c.key === key)
  }
  const checkRule = (path: Array<string | number>, rule: { op: string; value: number | string }) => {
    if (typeof rule.value === 'string' && rule.op !== '=' && rule.op !== '!=') {
      add([...path, 'value'], `"${rule.op}" compares numbers; use a number, or "=" / "!=" for text`)
    }
  }

  const checkMeasure = (
    path: Array<string | number>,
    dsId: string,
    m: Measure,
  ) => {
    const col = columnOf(dsId, m.column)
    if (!col) add([...path, 'column'], `unknown column "${m.column}" in dataset "${dsId}"`)
    else if (m.agg !== 'count' && m.agg !== 'last' && !isNumericType(col.type)) {
      add([...path, 'agg'], `"${m.agg}" needs a numeric column; "${m.column}" is ${col.type}`)
    }
  }

  d.blocks.forEach((b, i) => {
    const at = (...rest: Array<string | number>) => ['blocks', i, ...rest]
    if (b.kind === 'kpi') {
      if (!sources.has(b.sourceId)) add(at('sourceId'), `unknown source "${b.sourceId}"`)
      const needsDataset = !('literal' in b.value)
      if (b.datasetId === undefined) {
        if (needsDataset) add(at('datasetId'), 'a column or ratio value needs a datasetId')
        if (b.sparkline) add(at('sparkline'), 'a sparkline needs a datasetId')
      } else {
        const ds = datasets.get(b.datasetId)
        if (!ds) add(at('datasetId'), `unknown dataset "${b.datasetId}"`)
        else {
          if (ds.sourceId !== b.sourceId) {
            add(at('sourceId'), `must be "${ds.sourceId}", the source of dataset "${ds.id}"`)
          }
          if ('column' in b.value) checkMeasure(at('value'), ds.id, b.value)
          if ('ratio' in b.value) {
            checkMeasure(at('value', 'ratio', 'numerator'), ds.id, b.value.ratio.numerator)
            checkMeasure(at('value', 'ratio', 'denominator'), ds.id, b.value.ratio.denominator)
          }
          if (b.sparkline && !columnOf(ds.id, b.sparkline)) {
            add(at('sparkline'), `unknown column "${b.sparkline}" in dataset "${ds.id}"`)
          }
        }
      }
      if (b.delta?.vs === 'previous' && !b.sparkline) {
        add(at('delta', 'vs'), '"previous" compares against the period before; it needs a sparkline column')
      }
      if ('literal' in b.value && b.delta?.vs === 'previous') {
        add(at('delta', 'vs'), 'a literal value has no previous period; use a number')
      }
      if (b.display === 'trend' && !b.sparkline) {
        add(at('display'), '"trend" draws the sparkline column large; it needs a sparkline')
      }
      b.rules?.forEach((rule, j) => checkRule(at('rules', j), rule))
    } else if (b.kind === 'chart') {
      if (!sources.has(b.sourceId)) add(at('sourceId'), `unknown source "${b.sourceId}"`)
      const ds = datasets.get(b.datasetId)
      if (!ds) {
        add(at('datasetId'), `unknown dataset "${b.datasetId}"`)
        return
      }
      if (ds.sourceId !== b.sourceId) {
        add(at('sourceId'), `must be "${ds.sourceId}", the source of dataset "${ds.id}"`)
      }
      const x = columnOf(ds.id, b.x)
      if (!x) add(at('x'), `unknown column "${b.x}" in dataset "${ds.id}"`)
      b.series.forEach((s, j) => {
        const col = columnOf(ds.id, s.key)
        if (!col) add(at('series', j, 'key'), `unknown column "${s.key}" in dataset "${ds.id}"`)
        else if (!isNumericType(col.type)) add(at('series', j, 'key'), `series "${s.key}" must be numeric`)
      })
      if (b.split) {
        if (!columnOf(ds.id, b.split)) add(at('split'), `unknown column "${b.split}" in dataset "${ds.id}"`)
        if (b.series.length !== 1) add(at('series'), 'a split chart takes exactly one series')
      }
      if (b.type === 'scatter' && x && !isNumericType(x.type)) {
        add(at('x'), 'a scatter chart needs a numeric x column')
      }
      if ((b.type === 'pie' || b.type === 'donut' || b.type === 'funnel') && b.split) {
        add(at('split'), `a ${b.type} chart cannot be split`)
      }
      if (b.facet) {
        if (!columnOf(ds.id, b.facet)) add(at('facet'), `unknown column "${b.facet}" in dataset "${ds.id}"`)
        else if (b.facet === b.x || b.facet === b.split) add(at('facet'), 'facet by a column other than x and split')
        if (b.type === 'funnel') add(at('facet'), 'a funnel cannot be drawn as small multiples')
      }
      if (b.stack && !['bar', 'area', 'stacked-bar', 'horizontal-bar'].includes(b.type)) {
        add(at('stack'), 'stack applies to bar, horizontal-bar, stacked-bar and area charts')
      }
      if (b.scale === 'log' && (b.stack === 'percent' || ['pie', 'donut', 'funnel'].includes(b.type))) {
        add(at('scale'), 'a log scale needs a plain value axis (not a 100% stack, pie, donut or funnel)')
      }
      if (b.topN !== undefined && (b.type === 'scatter' || (x && x.type !== 'string'))) {
        add(at('topN'), 'topN keeps the largest categories; it needs a text x column (not dates or numbers, not scatter)')
      }
      if (b.trendline && (b.split || !['bar', 'line', 'area', 'combo', 'scatter'].includes(b.type))) {
        add(at('trendline'), 'a trendline goes on an unsplit bar, line, area, combo or scatter chart')
      }
      b.series.forEach((s, j) => {
        if (s.axis === 'right' && (b.split || !['bar', 'line', 'area', 'combo'].includes(b.type))) {
          add(at('series', j, 'axis'), 'a right axis is for unsplit bar, line, area and combo charts')
        }
      })
    } else if (b.kind === 'table') {
      const ds = datasets.get(b.datasetId)
      if (!ds) {
        add(at('datasetId'), `unknown dataset "${b.datasetId}"`)
        return
      }
      if (b.sourceId !== undefined && b.sourceId !== ds.sourceId) {
        add(at('sourceId'), `must be "${ds.sourceId}", the source of dataset "${ds.id}"`)
      }
      b.columns?.forEach((key, j) => {
        if (!columnOf(ds.id, key)) add(at('columns', j), `unknown column "${key}" in dataset "${ds.id}"`)
      })
      if (b.sort && !columnOf(ds.id, b.sort.column)) {
        add(at('sort', 'column'), `unknown column "${b.sort.column}" in dataset "${ds.id}"`)
      }
      b.heatmap?.forEach((key, j) => {
        const col = columnOf(ds.id, key)
        if (!col) add(at('heatmap', j), `unknown column "${key}" in dataset "${ds.id}"`)
        else if (!isNumericType(col.type)) add(at('heatmap', j), `heatmap column "${key}" must be numeric`)
      })
      b.rules?.forEach((rule, j) => {
        if (!columnOf(ds.id, rule.column)) add(at('rules', j, 'column'), `unknown column "${rule.column}" in dataset "${ds.id}"`)
        checkRule(at('rules', j), rule)
      })
      for (const key of Object.keys(b.formats ?? {})) {
        const col = columnOf(ds.id, key)
        if (!col) add(at('formats', key), `unknown column "${key}" in dataset "${ds.id}"`)
        else if (!isNumericType(col.type)) add(at('formats', key), `"${key}" is ${col.type}; formats apply to number columns`)
      }
    }
  })

  if (d.tabs && !d.blocks.some((b) => b.kind === 'section')) {
    add(['tabs'], 'tabs are made from section blocks; add a section block, or drop "tabs"')
  }

  ;(d.filters ?? []).forEach((f, i) => {
    f.datasetIds.forEach((dsId, j) => {
      if (!datasets.has(dsId)) {
        add(['filters', i, 'datasetIds', j], `unknown dataset "${dsId}"`)
        return
      }
      const col = columnOf(dsId, f.column)
      if (!col) add(['filters', i, 'column'], `unknown column "${f.column}" in dataset "${dsId}"`)
      else if (f.type === 'date-range' && col.type !== 'date') {
        add(['filters', i, 'type'], `a date-range filter needs a date column; "${f.column}" is ${col.type}`)
      }
    })
    if (f.multiple && f.type !== 'select') add(['filters', i, 'multiple'], 'only a select filter takes several values')
  })

  return issues
}

export type DashboardParse =
  | { ok: true; dashboard: Dashboard }
  | { ok: false; issues: string[] }

/** Validate an untrusted dashboard body. Never throws; issues read as `path: message`. */
export function parseDashboard(input: unknown): DashboardParse {
  const result = dashboardSchema.safeParse(input)
  if (result.success) return { ok: true, dashboard: result.data }
  return {
    ok: false,
    issues: result.error.issues.map((issue) => {
      const path = issue.path.map(String).join('.')
      return path ? `${path}: ${issue.message}` : issue.message
    }),
  }
}

/** The landmark a frame's layer name asks for (nav, header, footer, aside), else null. */
export function landmarkTag(name: string | undefined): 'nav' | 'header' | 'footer' | 'aside' | null {
  if (!name) return null
  const value = name.toLowerCase()
  if (/\b(nav|navbar|navigation|menu bar|tab bar)\b/.test(value)) return 'nav'
  if (/\bfooter\b/.test(value)) return 'footer'
  if (/\bheader\b/.test(value)) return 'header'

/** Whether a node (or anything drawn inside it) is itself interactive: a button, input or link. */
function hasInteractive(node: DesignNode, mains: ReadonlyMap<string, FrameNode>, depth = 0): boolean {
  if (depth > 40) return false
  if (node.type === 'button' || node.type === 'input') return true
  if (node.type !== 'frame') return false
  const shown = node.instanceOf !== undefined ? resolveInstance(node, mains) : node
  return shown.children.some((child) => !child.hidden && ((child.type === 'frame' && child.linkTo !== undefined) || hasInteractive(child, mains, depth + 1)))
}

/** Whether anything drawn inside a node is text a screen reader would read (text, a label). */
function hasText(node: DesignNode, mains: ReadonlyMap<string, FrameNode>, depth = 0): boolean {
  if (depth > 40) return false
  if (node.type === 'text') return node.text.trim().length > 0
  if (node.type === 'button') return true
  if (node.type === 'image') return node.alt.trim().length > 0
  if (node.type !== 'frame') return false
  const shown = node.instanceOf !== undefined ? resolveInstance(node, mains) : node
  return shown.children.some((child) => !child.hidden && hasText(child, mains, depth + 1))
}

  if (value === undefined) return undefined
  if (value === 'transparent' || HEX.test(value)) return value
  const hex = ctx.tokens?.colors[value] ?? DEFAULT_COLORS[value]
  if (!hex) return undefined
  return ctx.useVars ? `var(--color-${value})` : hex
}

const fontStackOf = (tokens: DesignTokens | undefined, role: 'heading' | 'body' | 'mono') =>
  role === 'mono' ? FONT_STACKS.mono.stack : FONT_STACKS[tokens?.fonts?.[role] ?? 'system'].stack
/** A font role: var(--font-heading) etc. in HTML, the stack itself for Tailwind. */
const font = (ctx: Ctx, role: 'heading' | 'body' | 'mono') => (ctx.useVars ? `var(--font-${role})` : fontStackOf(ctx.tokens, role))

const pxv = (value: number) => `${Math.round(value * 100) / 100}px`

function spacingCss(spacing: Padding | undefined): string | undefined {
  if (spacing === undefined) return undefined
  if (typeof spacing === 'number') return pxv(spacing)
  return spacing.map(pxv).join(' ')
}

function fillInsideMargin(margin: Padding | undefined, axis: 'width' | 'height', base = '100%'): string {
  if (margin === undefined) return base
  const [top, right, bottom = top, left = right] = typeof margin === 'number' ? [margin, margin] : margin
  const total = axis === 'width' ? left + right : top + bottom
  return total === 0 ? base : `calc(${base} - ${pxv(total)})`
}

type Parent = 'root' | 'stack-v' | 'stack-h' | 'grid' | 'absolute'

type Sized = {
  w?: Size; h?: Size; grow?: boolean; x?: number; y?: number
  opacity?: number; absolute?: boolean; anchor?: string; aspect?: number; blur?: number
  margin?: Padding
}

/** How a node sizes and places itself in its parent: the renderer's rules, as CSS. */
function sizeCss(node: Sized, parent: Parent, defaults: { w?: Size; h?: Size } = {}): Decl[] {
  const out: Decl[] = []
  const w = node.w ?? defaults.w ?? 'hug'
  const h = node.h ?? defaults.h ?? 'hug'
  if (node.absolute && parent !== 'absolute' && parent !== 'root') {
    const anchor = node.anchor ?? 'top-left'
    out.push(['position', 'absolute'])
    out.push([anchor.startsWith('bottom') ? 'bottom' : 'top', pxv(node.y ?? 0)])
    out.push([anchor.endsWith('right') ? 'right' : 'left', pxv(node.x ?? 0)])
    if (typeof w === 'number') out.push(['width', pxv(w)])
    if (typeof h === 'number') out.push(['height', pxv(h)])
  } else if (parent === 'root') {
    out.push(['width', typeof w === 'number' ? pxv(w) : fillInsideMargin(node.margin, 'width')])
    if (typeof h === 'number') out.push(['height', pxv(h)])
    else out.push(['min-height', fillInsideMargin(node.margin, 'height')])
  } else if (parent === 'absolute') {
    out.push(['position', 'absolute'], ['left', pxv(node.x ?? 0)], ['top', pxv(node.y ?? 0)])
    if (typeof w === 'number') out.push(['width', pxv(w)])
    else if (w === 'fill') out.push(['right', '0'])
    if (typeof h === 'number') out.push(['height', pxv(h)])
    else if (h === 'fill') out.push(['bottom', '0'])
  } else if (parent === 'grid') {
    if (typeof w === 'number') out.push(['width', pxv(w)])
    if (typeof h === 'number') out.push(['height', pxv(h)])
    else if (h === 'fill') out.push(['height', fillInsideMargin(node.margin, 'height')])
  } else {
    const row = parent === 'stack-h'
    const main = row ? w : h
    const cross = row ? h : w
    if (main === 'fill' || node.grow) out.push(['flex', '1 1 0%'], [row ? 'min-width' : 'min-height', '0'])
    else if (typeof main === 'number') out.push([row ? 'width' : 'height', pxv(main)], ['flex-shrink', '0'])
    if (cross === 'fill') out.push(['align-self', 'stretch'])
    else if (typeof cross === 'number') out.push([row ? 'height' : 'width', pxv(cross)])
  }
  const margin = spacingCss(node.margin)
  if (margin !== undefined) out.push(['margin', margin])
  if (node.opacity !== undefined && node.opacity < 1) out.push(['opacity', String(node.opacity)])
  if (node.aspect) out.push(['aspect-ratio', String(Math.round(node.aspect * 10000) / 10000)])
  if (node.blur) out.push(['filter', `blur(${node.blur}px)`])
  return out
}

const gradientCss = (g: { from: string; to: string; angle?: number; kind?: 'linear' | 'radial' }, ctx: Ctx) =>
  g.kind === 'radial'
    ? `radial-gradient(circle at center, ${color(g.from, ctx)}, ${color(g.to, ctx)})`
    : `linear-gradient(${g.angle ?? 180}deg, ${color(g.from, ctx)}, ${color(g.to, ctx)})`

function nameOf(node: DesignNode): string {
  const base = (node.name ?? node.id.split('~').pop() ?? node.type)
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, '-')
    .replace(/^-+|-+$/g, '')
  return base && /^[a-z]/.test(base) ? base : `${node.type}-${base || 'node'}`
}

function frameEl(raw: FrameNode, parent: Parent, ctx: Ctx, depth: number): El {
  const node = resolveNode(raw, { styles: ctx.styles })
  const shown = node.instanceOf !== undefined ? resolveNode(resolveInstance(node, ctx.mains), { styles: ctx.styles }) : node
  const css: Decl[] = [...sizeCss(shown, parent, parent === 'root' ? { w: 'fill', h: 'fill' } : {})]
  if (parent !== 'absolute' && !(shown.absolute && parent !== 'root')) {
    // Children that are positioned (absolute layout or pinned) need a containing block.
    if (shown.layout === 'absolute' || shown.children.some((child) => child.absolute)) css.push(['position', 'relative'])
  }
  if (shown.layout === 'stack-v' || shown.layout === 'stack-h') {
    css.push(['display', 'flex'], ['flex-direction', shown.layout === 'stack-h' ? 'row' : 'column'])
    const align = shown.align ?? (shown.layout === 'stack-h' ? 'center' : 'start')
    if (align !== 'stretch') css.push(['align-items', ALIGN[align]])
    if (shown.justify && shown.justify !== 'start') css.push(['justify-content', JUSTIFY[shown.justify]])
    if (shown.wrap) css.push(['flex-wrap', 'wrap'])
    if (shown.gap) css.push(['gap', pxv(shown.gap)])
    if (shown.wrap && shown.rowGap !== undefined) css.push(['row-gap', pxv(shown.rowGap)])
  } else if (shown.layout === 'grid') {
    css.push(['display', 'grid'], ['grid-template-columns', `repeat(${shown.columns ?? 2}, minmax(0, 1fr))`])
    if (shown.gap) css.push(['gap', pxv(shown.gap)])
    if (shown.align && shown.align !== 'stretch') css.push(['align-items', ALIGN[shown.align]])
  }
  // A multi-column grid stacks to one column on a narrow screen (the guide's phone rule).
  const small: Decl[] | undefined = mergeSmall(
    shown.layout === 'grid' && (shown.columns ?? 2) > 1 ? [['grid-template-columns', 'minmax(0, 1fr)']] : undefined,
    narrowDecls(raw, ctx),
  )
  if (shown.minW !== undefined) css.push(['min-width', pxv(shown.minW)])
  if (shown.maxW !== undefined) css.push(['max-width', pxv(shown.maxW)])
  if (shown.minH !== undefined) css.push(['min-height', pxv(shown.minH)])
  if (shown.maxH !== undefined) css.push(['max-height', pxv(shown.maxH)])
  const padding = spacingCss(shown.padding)
  if (padding) css.push(['padding', padding])
  const fill = color(shown.fill, ctx)
  if (fill) css.push(['background-color', fill])
  const layers: string[] = []
  if (shown.fillImage) layers.push(`url("${cssString(shown.fillImage.src)}")`)
  if (shown.gradient) layers.push(gradientCss(shown.gradient, ctx))
  if (layers.length) css.push(['background-image', layers.join(', ')])
  if (shown.fillImage) css.push(['background-size', shown.fillImage.fit ?? 'cover'], ['background-position', 'center'])
  if (shown.stroke) {
    const border = `${pxv(shown.strokeWidth ?? 1)} ${shown.strokePattern ?? 'solid'} ${color(shown.stroke, ctx)}`
    if (!shown.strokeSides) css.push(['border', border])
    else for (const side of shown.strokeSides) css.push([`border-${side}`, border])
  }
  if (shown.radius) css.push(['border-radius', pxv(shown.radius)])
  if (shown.shadow && shown.shadow !== 'none') css.push(['box-shadow', SHADOW[shown.shadow]])
  if (shown.clip) css.push(['overflow', 'hidden'])
  // A link cannot hold another control (a button inside an <a> is invalid HTML), so a linked
  // frame with buttons or inputs inside stays a div and says where it would go.
  const link = shown.linkTo !== undefined && !hasInteractive(shown, ctx.mains) ? shown.linkTo : undefined
  const landmark = ctx.inLandmark ? null : landmarkTag(shown.name)
  const tag = link ? 'a' : landmark ?? (depth === 1 && parent === 'stack-v' ? 'section' : 'div')
  const attrs: Array<[string, string]> = []
  if (link) {
    attrs.push(['href', `#${link}`])
    // A link drawn with icons or shapes only still needs a name.
    if (!hasText(shown, ctx.mains)) attrs.push(['aria-label', shown.name ?? `Go to ${ctx.boardTitles?.get(link) ?? link}`])
  } else if (tag === 'section' && shown.name) attrs.push(['aria-label', shown.name])
  const notes = [
    node.instanceOf !== undefined ? `instance of ${shown.name ?? node.instanceOf}` : null,
    shown.linkTo !== undefined && !link ? `links to #${shown.linkTo} (holds controls, so not an <a>)` : null,
  ].filter(Boolean)
  return {
    tag,
    name: nameOf(node),
    attrs,
    css,
    small,
    children: (() => {
      const outer = ctx.inLandmark
      if (landmark) ctx.inLandmark = true
      const out = shown.children.filter((child) => !child.hidden).map((child) => nodeEl(child, shown.layout, ctx, depth + 1))
      ctx.inLandmark = outer
      return out
    })(),
    comment: notes.length ? notes.join('; ') : undefined,
  }
}

function nodeEl(raw: DesignNode, parent: Parent, ctx: Ctx, depth: number): El {
  const el = nodeElInner(resolveNode(raw, { styles: ctx.styles }), parent, ctx, depth)
  if (raw.type === 'frame') return el
  const small = mergeSmall(el.small, narrowDecls(raw, ctx))
  return small ? { ...el, small } : el
}

/** A node's override at the board's narrow breakpoint, as the CSS that replaces its own there. */
function narrowDecls(node: DesignNode, ctx: Ctx): Decl[] | undefined {
  const override: ResponsiveOverride | undefined = ctx.narrow ? node.responsive?.[ctx.narrow] : undefined
  if (!override) return undefined
  const out: Decl[] = []
  if (override.hidden === true) out.push(['display', 'none'])
  if (node.type === 'frame') {
    if (override.layout === 'stack-v' || override.layout === 'stack-h') out.push(['flex-direction', override.layout === 'stack-h' ? 'row' : 'column'])
    if (override.gap !== undefined) out.push(['gap', pxv(override.gap)])
    const padding = spacingCss(override.padding)
    if (padding) out.push(['padding', padding])
    if (override.columns !== undefined) out.push(['grid-template-columns', override.columns === 1 ? 'minmax(0, 1fr)' : `repeat(${override.columns}, minmax(0, 1fr))`])
  }
  if (node.type === 'text' && (override.fontSize !== undefined || override.style !== undefined)) {
    out.push(['font-size', pxv(override.fontSize ?? TEXT_SCALE[override.style ?? node.style].size)])
  }
  if (node.type === 'text' && override.weight !== undefined) out.push(['font-weight', String(override.weight)])
  if (override.w !== undefined) out.push(['width', typeof override.w === 'number' ? pxv(override.w) : override.w === 'fill' ? fillInsideMargin(node.margin, 'width') : 'auto'])
  return out.length ? out : undefined
}

function mergeSmall(a: Decl[] | undefined, b: Decl[] | undefined): Decl[] | undefined {
  if (!a?.length) return b
  if (!b?.length) return a
  const props = new Set(b.map(([prop]) => prop))
  return [...a.filter(([prop]) => !props.has(prop)), ...b]
}

function nodeElInner(node: DesignNode, parent: Parent, ctx: Ctx, depth: number): El {
  switch (node.type) {
    case 'frame':
      return frameEl(node, parent, ctx, depth)
    case 'text': {
      const scale = TEXT_SCALE[node.style]
      // A fixed width in a column is a measure (a cap on line length): never wider than its
      // parent, so it shrinks on a narrow screen and is exactly the design at full width.
      const sized = sizeCss(node, parent).flatMap((decl): Decl[] =>
        decl[0] === 'width' && parent === 'stack-v' && typeof node.w === 'number' ? [decl, ['max-width', '100%']] : [decl],
      )
      if (node.margin === undefined) sized.push(['margin', '0'])
      const css: Decl[] = [...sized, ['font-family', font(ctx, scale.font)], ['font-size', pxv(node.fontSize ?? scale.size)]]
      css.push(['line-height', String(node.lineHeight ?? scale.line)], ['font-weight', String(node.weight ?? scale.weight)])
      const tracking = node.letterSpacing === undefined ? scale.tracking : scale.tracking === 'normal' ? pxv(node.letterSpacing) : `calc(${scale.tracking} + ${pxv(node.letterSpacing)})`
      if (tracking !== 'normal') css.push(['letter-spacing', tracking])
      const c = color(node.color, ctx)
      if (c) css.push(['color', c])
      if (node.align && node.align !== 'start') css.push(['text-align', node.align === 'end' ? 'right' : 'center'])
      if (node.decoration) css.push(['text-decoration', node.decoration])
      if (node.text.includes('\n')) css.push(['white-space', 'pre-wrap'])
      if (node.maxLines) css.push(['display', '-webkit-box'], ['-webkit-line-clamp', String(node.maxLines)], ['-webkit-box-orient', 'vertical'], ['overflow', 'hidden'])
      let tag = scale.tag
      if (tag === 'h1') {
        if (ctx.h1) tag = 'h2'
        else ctx.h1 = true
      }
      return { tag, name: nameOf(node), attrs: [], css, children: [node.text] }
    }
    case 'button': {
      const size = BUTTON_SIZE[node.size ?? 'md']
      const palette =
        node.variant === 'primary'
          ? { bg: color('primary', ctx), fg: color('on-primary', ctx), border: 'transparent' }
          : node.variant === 'secondary'
            ? { bg: color('background', ctx), fg: color('text', ctx), border: color('border', ctx) }
            : { bg: 'transparent', fg: color('text', ctx), border: 'transparent' }
      const css: Decl[] = [
        ...sizeCss(node, parent),
        ['display', 'inline-flex'], ['align-items', 'center'], ['justify-content', 'center'], ['gap', '8px'],
        ['height', pxv(size.height)], ['padding', `0 ${pxv(size.pad)}`], ['border-radius', pxv(ctx.tokens?.radius ?? 8)],
        ['border', `1px solid ${palette.border}`], ['background-color', palette.bg ?? 'transparent'], ['color', palette.fg ?? 'inherit'],
        ['font-family', font(ctx, 'body')], ['font-size', pxv(size.font)], ['font-weight', '600'], ['white-space', 'nowrap'],
      ]
      const icon: El | null = node.icon ? iconEl(node.icon, size.icon, undefined, ctx) : null
      const trailing = node.icon ? /^(arrow|chevron)-/.test(node.icon) : false
      const children: Array<El | string> = [node.label]
      if (icon) trailing ? children.push(icon) : children.unshift(icon)
      return {
        tag: node.linkTo ? 'a' : 'button',
        name: nameOf(node),
        attrs: node.linkTo ? [['href', `#${node.linkTo}`]] : [['type', 'button']],
        css,
        children,
      }
    }
    case 'image': {
      const defaults = node.placeholder ? PLACEHOLDER_SIZE[node.placeholder] : { w: 320, h: 200, radius: 0 }
      const css: Decl[] = [...sizeCss(node, parent, { w: defaults.w, h: defaults.h }), ['display', 'block'], ['flex-shrink', '0']]
      const radius = node.radius ?? defaults.radius
      if (radius) css.push(['border-radius', pxv(radius)])
      if (node.src) {
        css.push(['object-fit', node.fit ?? 'cover'])
        if (node.focus) css.push(['object-position', `${Math.round(node.focus[0] * 100)}% ${Math.round(node.focus[1] * 100)}%`])
        return { tag: 'img', name: nameOf(node), attrs: [['src', node.src], ['alt', node.alt]], css, children: [] }
      }
      css.push(['background-color', color('surface', ctx) ?? '#f1f5f9'])
      const alt = node.alt.trim()
      return {
        tag: 'div',
        name: nameOf(node),
        // No alt means decorative: hidden from screen readers rather than an image with no name.
        attrs: alt ? [['role', 'img'], ['aria-label', alt]] : [['aria-hidden', 'true']],
        css,
        children: [],
        comment: `${node.placeholder} placeholder: replace with a real image`,
      }
    }
    case 'icon':
      return iconEl(node.icon, node.size ?? 20, node, ctx, parent)
    case 'input':
      return inputEl(node, parent, ctx)
    case 'divider': {
      const vertical = parent === 'stack-h'
      const css: Decl[] = [
        ...sizeCss(node, parent),
        ['flex-shrink', '0'],
        ...(vertical ? ([['width', '1px'], ['align-self', 'stretch']] as Decl[]) : ([['height', '1px'], ['width', fillInsideMargin(node.margin, 'width')]] as Decl[])),
        ['background-color', color(node.color ?? 'border', ctx) ?? '#e2e8f0'],
      ]
      if (!vertical) {
        css.push(['border', '0'])
        if (node.margin === undefined) css.push(['margin', '0'])
      }
      return { tag: vertical ? 'div' : 'hr', name: nameOf(node), attrs: [], css, children: [] }
    }
    case 'shape': {
      if (isVectorShape(node)) return vectorEl(node, parent, ctx)
      const css: Decl[] = [...sizeCss(node, parent)]
      if (node.kind === 'line') {
        const height = typeof node.h === 'number' ? Math.max(node.h, 1) : 2
        const stroke = color(node.stroke ?? node.fill ?? 'border', ctx) ?? '#e2e8f0'
        const patterned = node.strokePattern !== undefined && node.strokePattern !== 'solid'
        css.push(['height', pxv(height)])
        if (node.strokeWidth !== undefined || patterned) css.push(['border-top', `${pxv(node.strokeWidth ?? height)} ${node.strokePattern ?? 'solid'} ${stroke}`])
        else css.push(['background-color', stroke])
      } else {
        const fill = color(node.fill, ctx)
        if (fill) css.push(['background-color', fill])
        if (node.gradient) css.push(['background-image', gradientCss(node.gradient, ctx)])
        if (node.stroke) css.push(['border', `${pxv(node.strokeWidth ?? 1.5)} ${node.strokePattern ?? 'solid'} ${color(node.stroke, ctx)}`])
        if (node.kind === 'ellipse') css.push(['border-radius', '50%'])
        else if (node.radius) css.push(['border-radius', pxv(node.radius)])
      }
      return { tag: 'div', name: nameOf(node), attrs: [['aria-hidden', 'true']], css, children: [] }
    }
  }
}

/** A vector shape as an inline SVG: one path, filled when closed, stroked otherwise. */
function vectorEl(node: ShapeNode, parent: Parent, ctx: Ctx): El {
  const w = typeof node.w === 'number' ? node.w : 100
  const h = typeof node.h === 'number' ? node.h : 100
  const geometry = shapeGeometry(node, w, h)
  const stroke = color(node.stroke, ctx) ?? (geometry.closed ? undefined : color('text', ctx))
  const fill = geometry.closed ? color(node.fill, ctx) ?? (stroke ? undefined : color('primary', ctx)) : undefined
  const path: Decl[] = [['fill', fill ?? 'none']]
  if (stroke) {
    const width = node.strokeWidth ?? 2
    path.push(['stroke', stroke], ['stroke-width', pxv(width)], ['stroke-linecap', 'round'], ['stroke-linejoin', 'round'])
    const dashes = strokeDashArray(node.strokePattern, width)
    if (dashes) path.push(['stroke-dasharray', dashes], ['vector-effect', 'non-scaling-stroke'])
  }
  return {
    tag: 'svg',
    name: nameOf(node),
    attrs: [['viewBox', `0 0 ${w} ${h}`], ['preserveAspectRatio', 'none'], ['aria-hidden', 'true']],
    css: [...sizeCss(node, parent), ['display', 'block'], ['overflow', 'visible']],
    children: [{ tag: 'path', name: `${nameOf(node)}-path`, attrs: [['d', geometry.d]], css: path, children: [] }],
  }
}

function iconEl(icon: IconName, size: number, node: DesignNode | undefined, ctx: Ctx, parent: Parent = 'stack-h'): El {
  const css: Decl[] = node ? sizeCss(node, parent) : []
  css.push(['width', pxv(size)], ['height', pxv(size)], ['flex-shrink', '0'])
  const c = node?.type === 'icon' ? color(node.color, ctx) : undefined
  if (c) css.push(['color', c])
  return { tag: 'span', name: node ? nameOf(node) : `icon-${icon}`, attrs: [['aria-hidden', 'true']], css, children: [], icon: { name: LUCIDE[icon], size } }
}

function inputEl(node: Extract<DesignNode, { type: 'input' }>, parent: Parent, ctx: Ctx): El {
  const text = color('text', ctx) ?? '#0f172a'
  const border = color('border', ctx) ?? '#e2e8f0'
  const base: Decl[] = [...sizeCss(node, parent), ['font-family', font(ctx, 'body')], ['font-size', '14px'], ['color', text]]
  const field: Decl[] = [
    ['height', '40px'], ['padding', '0 12px'], ['border', `1px solid ${border}`], ['border-radius', pxv(ctx.tokens?.radius ?? 8)],
    ['background-color', color('background', ctx) ?? '#ffffff'], ['font', 'inherit'], ['color', 'inherit'],
  ]
  const label = (children: Array<El | string>, css: Decl[]): El => ({ tag: 'label', name: nameOf(node), attrs: [], css: [...base, ...css], children })
  // With no visible label the control still needs an accessible name.
  const named: Array<[string, string]> = node.label?.trim()
    ? []
    : [['aria-label', node.placeholder?.trim() || node.name || (node.kind === 'search' ? 'Search' : node.kind === 'toggle' ? 'Toggle' : node.kind === 'checkbox' ? 'Checkbox' : 'Text field')]]
  if (node.kind === 'checkbox') {
    return label(
      [{ tag: 'input', name: `${nameOf(node)}-box`, attrs: [['type', 'checkbox'], ...named, ...(node.checked ? ([['defaultChecked', '']] as Array<[string, string]>) : [])], css: [['width', '16px'], ['height', '16px']], children: [] }, ...(node.label ? [node.label] : [])],
      [['display', 'inline-flex'], ['align-items', 'center'], ['gap', '10px']],
    )
  }
  if (node.kind === 'toggle') {
    return label(
      [
        ...(node.label ? [node.label] : []),
        { tag: 'input', name: `${nameOf(node)}-switch`, attrs: [['type', 'checkbox'], ['role', 'switch'], ...named, ...(node.checked ? ([['defaultChecked', '']] as Array<[string, string]>) : [])], css: [['width', '36px'], ['height', '20px']], children: [] },
      ],
      [['display', 'flex'], ['align-items', 'center'], ['justify-content', 'space-between'], ['gap', '12px']],
    )
  }
  const control: El =
    node.kind === 'select'
      ? { tag: 'select', name: `${nameOf(node)}-control`, attrs: [...named], css: field, children: [{ tag: 'option', name: '', attrs: [], css: [], children: [node.value ?? node.placeholder ?? 'Choose…'] }] }
      : {
          tag: 'input',
          name: `${nameOf(node)}-control`,
          attrs: [
            ['type', node.kind === 'search' ? 'search' : 'text'],
            ...named,
            ...(node.placeholder ? ([['placeholder', node.placeholder]] as Array<[string, string]>) : []),
            ...(node.value ? ([['defaultValue', node.value]] as Array<[string, string]>) : []),
          ],
          css: field,
          children: [],
        }
  return label(
    [...(node.label ? [{ tag: 'span', name: `${nameOf(node)}-label`, attrs: [], css: [['font-weight', '500']] as Decl[], children: [node.label] } as El] : []), control],
    [['display', 'flex'], ['flex-direction', 'column'], ['gap', '6px']],
  )
}

function boardEl(design: Design, stored: DesignBoard, useVars: boolean, inspectRoot = false): El {
  const mains = collectMains(design.boards)
  // Round 5: repeats, bindings and showIf drawn as the variables stand; a dark board's theme colours.
  const board = hasLogic(design) ? materializeBoardLogic(stored, logicScope(design, undefined, mains)) : stored
  const narrow = [...(board.breakpoints ?? [])].filter((entry) => entry.w < CODEGEN_BREAKPOINT).sort((a, b) => a.w - b.w)[0]
  const ctx: Ctx = {
    tokens: themedTokens(design.tokens, board.theme),
    mains,
    useVars,
    boardTitles: new Map(design.boards.map((entry) => [entry.id, entry.title])),
    styles: design.styles,
    narrow: breakpointOf(board, narrow?.id)?.id ?? null,
  }
  const root = frameEl(board.root, 'root', ctx, 0)
  // The board's width is the page's widest, not a fixed size: it narrows with the screen.
  const css: Decl[] = [

    ['font-family', font(ctx, 'body')],
    ['font-size', '16px'],

  ]

    // centring and min-height should not move with that frame.
    const content: El = {

  }
}

/** Tailwind's spacing scale: px values with a named step. */
const SPACING: Record<number, string> = { 0: '0', 1: 'px', 2: '0.5', 4: '1', 6: '1.5', 8: '2', 10: '2.5', 12: '3', 14: '3.5', 16: '4', 20: '5', 24: '6', 28: '7', 32: '8', 36: '9', 40: '10', 44: '11', 48: '12', 56: '14', 64: '16', 80: '20', 96: '24', 112: '28', 128: '32', 144: '36', 160: '40', 176: '44', 192: '48', 208: '52', 224: '56', 240: '60', 256: '64', 288: '72', 320: '80', 384: '96' }

/** `p-6`, `gap-[18px]`: a px length on the spacing scale, or an arbitrary value. */
function space(prefix: string, value: string): string {
  if (value === '0') return `${prefix}-0`
  const match = /^(-?\d+(?:\.\d+)?)px$/.exec(value)
  if (match) {
    const n = Number(match[1])
    if (n === 0) return `${prefix}-0`
    const step = SPACING[Math.abs(n)]
    if (step) return `${n < 0 ? '-' : ''}${prefix}-${step}`
  }
  if (value === '100%') return `${prefix}-full`
  return `${prefix}-[${arbitrary(value)}]`
}

function paddingClasses(value: string): string[] {
  const parts = value.split(' ')
  const [t, r, b, l] = parts.length === 1 ? [parts[0], parts[0], parts[0], parts[0]] : parts.length === 2 ? [parts[0], parts[1], parts[0], parts[1]] : parts
  if (t === r && r === b && b === l) return [space('p', t)]
  if (t === b && r === l) return [space('py', t), space('px', r)]
  return [space('pt', t), space('pr', r), space('pb', b), space('pl', l)]
}

/** CSS declarations as Tailwind classes; anything without a utility becomes `[prop:value]`. */
export function tailwindClasses(css: Decl[]): string[] {
  const out: string[] = []
  for (const [prop, value] of css) {
    switch (prop) {
      case 'display':
        out.push(value === 'flex' ? 'flex' : value === 'grid' ? 'grid' : value === 'inline-flex' ? 'inline-flex' : value === 'block' ? 'block' : `[display:${value}]`)
        break
      case 'flex-direction':
        out.push(value === 'column' ? 'flex-col' : 'flex-row')
        break
      case 'align-items':
        out.push({ 'flex-start': 'items-start', center: 'items-center', 'flex-end': 'items-end', stretch: 'items-stretch' }[value] ?? `items-[${value}]`)
        break
      case 'justify-content':
        out.push({ 'flex-start': 'justify-start', center: 'justify-center', 'flex-end': 'justify-end', 'space-between': 'justify-between', 'space-around': 'justify-around', 'space-evenly': 'justify-evenly' }[value] ?? `justify-[${value}]`)
        break
      case 'align-self':
        out.push(value === 'stretch' ? 'self-stretch' : `self-[${value}]`)
        break
      case 'flex-wrap':
        out.push('flex-wrap')
        break
      case 'flex':
        out.push(value === '1 1 0%' ? 'flex-1' : `flex-[${arbitrary(value)}]`)
        break
      case 'flex-shrink':
        out.push(value === '0' ? 'shrink-0' : 'shrink')
        break
      case 'gap':
        out.push(space('gap', value))
        break
      case 'padding':
        out.push(...paddingClasses(value))
        break
      case 'margin':
        out.push(value === '0' ? 'm-0' : value === '0 auto' ? 'mx-auto' : `m-[${arbitrary(value)}]`)
        break
      case 'max-width':
        out.push(value === '100%' ? 'max-w-full' : `max-w-[${arbitrary(value)}]`)
        break
      case 'max-height':
        out.push(`max-h-[${arbitrary(value)}]`)
        break
      case 'width':
        out.push(space('w', value))
        break
      case 'height':
        out.push(space('h', value))
        break
      case 'min-height':
        out.push(value === '100%' ? 'min-h-full' : value === '0' ? 'min-h-0' : `min-h-[${arbitrary(value)}]`)
        break
      case 'min-width':
        out.push(value === '0' ? 'min-w-0' : `min-w-[${arbitrary(value)}]`)
        break
      case 'position':
        out.push(value)
        break
      case 'top':
      case 'right':
      case 'bottom':
      case 'left':
        out.push(space(prop, value))
        break
      case 'grid-template-columns': {
        const match = /^repeat\((\d+), minmax\(0, 1fr\)\)$/.exec(value)
        out.push(match ? `grid-cols-${match[1]}` : value === 'minmax(0, 1fr)' ? 'grid-cols-1' : `grid-cols-[${arbitrary(value)}]`)
        break
      }
      case 'background-color':
        out.push(value === 'transparent' ? 'bg-transparent' : `bg-[${arbitrary(value)}]`)
        break
      case 'background-image':
        out.push(`bg-[${arbitrary(value.replace(/"/g, "'"))}]`)
        break
      case 'background-size':
        out.push(value === 'cover' ? 'bg-cover' : value === 'contain' ? 'bg-contain' : `[background-size:${arbitrary(value)}]`)
        break
      case 'background-position':
        out.push(value === 'center' ? 'bg-center' : `[background-position:${arbitrary(value)}]`)
        break
      case 'color':
        out.push(value === 'inherit' ? 'text-inherit' : `text-[${arbitrary(value)}]`)
        break
      case 'border-radius':
        out.push(RADIUS[value] ?? `rounded-[${arbitrary(value)}]`)
        break
      case 'box-shadow':
        out.push(`shadow-[${arbitrary(value)}]`)
        break
      case 'overflow':
        out.push(`overflow-${value}`)
        break
      case 'opacity':
        out.push(`opacity-[${value}]`)
        break
      case 'font-size':
        out.push(FONT_SIZE[value] ?? `text-[${value}]`)
        break
      case 'font-weight':
        out.push(WEIGHT[value] ?? `font-[${value}]`)
        break
      case 'line-height':
        out.push(`leading-[${value}]`)
        break
      case 'letter-spacing':
        out.push(`tracking-[${arbitrary(value)}]`)
        break
      case 'text-align':
        out.push(`text-${value}`)
        break
      case 'text-decoration':
        out.push(value === 'underline' ? 'underline' : 'line-through')
        break
      case 'white-space':
        out.push(value === 'nowrap' ? 'whitespace-nowrap' : 'whitespace-pre-wrap')
        break
      case 'object-fit':
        out.push(`object-${value}`)
        break
      case 'aspect-ratio':
        out.push(value === '1' ? 'aspect-square' : `aspect-[${value}]`)
        break
      case 'filter': {
        const match = /^blur\((\d+)px\)$/.exec(value)
        out.push(match ? `blur-[${match[1]}px]` : `[filter:${arbitrary(value)}]`)
        break
      }
      case 'font-family':
        // The body font is set once on the board; others are arbitrary families.
        out.push(`font-[${arbitrary(value.replace(/'/g, ''))}]`)
        break
      default:
        out.push(`[${prop}:${arbitrary(value)}]`)
    }
  }
  return [...new Set(out)]
}

/**
 * Mobile-first classes: a property `small` overrides gets the small value as its base class and
 * the design's value behind `md:` (`grid-cols-1 md:grid-cols-3`); the rest as `tailwindClasses`.
 */
export function responsiveClasses(css: Decl[], small: Decl[] | undefined): string[] {
  if (!small?.length) return tailwindClasses(css)
  const narrow = new Map(small)
  const out: string[] = []
  for (const decl of css) {
    const value = narrow.get(decl[0])
    if (value === undefined) out.push(...tailwindClasses([decl]))
    else out.push(...tailwindClasses([[decl[0], value]]), ...tailwindClasses([decl]).map((cls) => `md:${cls}`))
  }
  // A narrow-only property (a node hidden on phones) applies below md only.
  const own = new Set(css.map(([prop]) => prop))
  for (const decl of small) if (!own.has(decl[0])) out.push(...tailwindClasses([decl]).map((cls) => `max-md:${cls}`))
  return [...new Set(out)]
}

function applyOverride(node: DesignNode, override: InstanceOverride | undefined): DesignNode {
  if (!override) return node
  if (node.type === 'text' && override.text !== undefined) return { ...node, text: override.text }
  if (node.type === 'button' && override.label !== undefined) return { ...node, label: override.label }
  if (node.type === 'image') {

    } else if (override.placeholder !== undefined) {

      delete next.src

  instance: FrameNode,
  mains: ReadonlyMap<string, FrameNode>,
  depth = 0,
): FrameNode {
  const main = instance.instanceOf !== undefined ? mains.get(instance.instanceOf) : undefined
  const own = {
    id: instance.id,
    name: instance.name ?? main?.name,
    hidden: instance.hidden,
    x: instance.x,
    y: instance.y,
  }
  if (!main || depth >= MAX_INSTANCE_DEPTH) {
    return { ...instance, ...own, children: [] }
  }
  const prefix = (node: DesignNode, base: string, overrides: FrameNode['overrides']): DesignNode => {
    const id = `${base}${INSTANCE_SEP}${node.id}`
    const shown = applyOverride(node, overrides?.[node.id])
    if (shown.type !== 'frame') return { ...shown, id }
    if (shown.instanceOf !== undefined) {
      const inner = resolveInstance({ ...shown, id }, mains, depth + 1)
      return inner
    }
    return { ...shown, id, component: undefined, children: shown.children.map((child) => prefix(child, base, overrides)) }
  }
  const { component: _main, ...style } = main
  const resolved: FrameNode = {
    ...style,
    ...own,
    w: instance.w ?? main.w,
    h: instance.h ?? main.h,
    margin: instance.margin ?? main.margin,
    grow: instance.grow ?? main.grow,
    minW: instance.minW ?? main.minW,
    maxW: instance.maxW ?? main.maxW,
    minH: instance.minH ?? main.minH,
    maxH: instance.maxH ?? main.maxH,
    linkTo: instance.linkTo ?? main.linkTo,
    // The transition goes with whichever link is drawn.
    transition: instance.linkTo !== undefined ? instance.transition : main.transition,
    instanceOf: main.id,
    overrides: undefined,
    children: main.children.map((child) => prefix(child, instance.id, instance.overrides)),
  }
  const record = resolved as Record<string, unknown>
  for (const key of Object.keys(record)) if (record[key] === undefined) delete record[key]
  return resolved
}

/** The nodes of a main an instance can override, in reading order. */
export function overrideTargets(main: FrameNode): Array<{ id: string; node: DesignNode }> {
  const out: Array<{ id: string; node: DesignNode }> = []
  for (const { node } of walkNodes(main)) {
    if (node === main) continue
    if (node.type === 'text' || node.type === 'button' || node.type === 'image') out.push({ id: node.id, node })
  }
  return out
}

/** Sets (or with `undefined` fields, clears) one override on an instance. */
export function setOverride(
  design: Design,
  boardId: string,
  instanceId: string,
  mainNodeId: string,
  patch: InstanceOverride | null,
): Design {
  return updateBoard(design, boardId, (board) => {
    const hit = findNode(board.root, instanceId)
    if (!hit || !isInstance(hit.node)) return board
    const overrides = { ...(hit.node.overrides ?? {}) }
    const merged: InstanceOverride = patch === null ? {} : { ...(overrides[mainNodeId] ?? {}), ...patch }
    for (const key of Object.keys(merged) as Array<keyof InstanceOverride>) if (merged[key] === undefined) delete merged[key]
    if (merged.src !== undefined && patch?.placeholder !== undefined) delete merged.src
    if (merged.placeholder !== undefined && patch?.src !== undefined) delete merged.placeholder
    if (Object.keys(merged).length === 0) delete overrides[mainNodeId]
    else overrides[mainNodeId] = merged
    const next: FrameNode = { ...hit.node, overrides: Object.keys(overrides).length ? overrides : undefined }
    if (next.overrides === undefined) delete next.overrides
    return { ...board, root: swap(board.root, instanceId, next) as FrameNode }
  })
}

function swap(root: DesignNode, nodeId: string, node: DesignNode): DesignNode {
  if (root.id === nodeId) return node
  if (root.type !== 'frame') return root
  let changed = false
  const children = root.children.map((child) => {
    const next = swap(child, nodeId, node)
    if (next !== child) changed = true
    return next
  })
  return changed ? { ...root, children } : root
}

/** Marks a frame as a main component (or unmarks it, detaching its instances first). */
export function setComponent(design: Design, boardId: string, frameId: string, on: boolean): Design {
  const board = getBoard(design, boardId)
  const hit = board ? findNode(board.root, frameId) : null
  if (!hit || hit.node.type !== 'frame' || hit.node.instanceOf !== undefined) return design
  if (Boolean(hit.node.component) === on) return design
  const base = on ? design : detachInstancesOf(design, [frameId])
  return updateBoard(base, boardId, (current) => {
    const found = findNode(current.root, frameId)
    if (!found || found.node.type !== 'frame') return current
    const next: FrameNode = { ...found.node }
    if (on) next.component = true
    else delete next.component
    return { ...current, root: swap(current.root, frameId, next) as FrameNode }
  })
}

/** A new instance node of `mainId` (not yet placed). */
export function instanceNode(design: Design, mainId: string, options: { id?: string; overrides?: FrameNode['overrides'] } = {}): FrameNode | null {
  const main = collectMains(design.boards).get(mainId)
  if (!main) return null

/** Places an instance of `mainId` in frame `parentId`; returns its id. */
export function insertInstance(
  design: Design,
  boardId: string,
  parentId: string,
  mainId: string,
  index?: number,

/**
 * The instance turned into plain nodes: the main's subtree with overrides baked in and fresh
 * ids. One level: an instance inside the main stays an instance.
 */
function detachedCopy(instance: FrameNode, main: FrameNode, taken: Set<string>): FrameNode {
  const copy = (node: DesignNode): DesignNode => {
    const shown = applyOverride(node, instance.overrides?.[node.id])
    const id = freshId(taken, shown.id.replace(/-\d+$/, ''))
    if (shown.type !== 'frame') return { ...shown, id }
    const { component: _c, ...rest } = shown
    if (shown.instanceOf !== undefined) return { ...rest, id, children: [] }
    return { ...rest, id, children: shown.children.map(copy) }
  }
  const { component: _c, instanceOf: _i, overrides: _o, ...style } = main
  const out: FrameNode = {
    ...style,
    id: instance.id,
    children: main.children.map(copy),
  }
  for (const key of ['name', 'hidden', 'x', 'y', 'w', 'h', 'grow', 'linkTo', 'minW', 'maxW', 'minH', 'maxH'] as const) {
    const value = instance[key] ?? main[key]
    if (value !== undefined) (out as Record<string, unknown>)[key] = value
    else delete (out as Record<string, unknown>)[key]
  }
  // The transition goes with whichever link the copy keeps.
  const transition = instance.linkTo !== undefined ? instance.transition : main.transition
  if (transition !== undefined) out.transition = transition
  else delete out.transition
  return out
}

/** Detaches one instance (it keeps its id); the design is unchanged when it is not one. */
export function detachInstance(design: Design, boardId: string, instanceId: string): Design {
  const board = getBoard(design, boardId)
  const hit = board ? findNode(board.root, instanceId) : null
  if (!hit || !isInstance(hit.node)) return design
  const main = collectMains(design.boards).get(hit.node.instanceOf)
  const taken = allIds(design)
  const plain: FrameNode = main
    ? detachedCopy(hit.node, main, taken)
    : (({ instanceOf: _i, overrides: _o, ...rest }) => rest)(hit.node)
  return updateBoard(design, boardId, (current) => ({ ...current, root: swap(current.root, instanceId, plain) as FrameNode }))
}

/** Detaches every instance of the given mains (before those mains are deleted). */
export function detachInstancesOf(design: Design, mainIds: string[]): Design {
  if (mainIds.length === 0) return design
  const doomed = new Set(mainIds)
  let next = design
  for (let pass = 0; pass < 8; pass += 1) {
    const found: Array<{ boardId: string; nodeId: string }> = []
    for (const board of next.boards) {
      for (const { node } of walkNodes(board.root)) {
        if (isInstance(node) && doomed.has(node.instanceOf)) found.push({ boardId: board.id, nodeId: node.id })
      }
    }
    if (found.length === 0) break
    for (const entry of found) next = detachInstance(next, entry.boardId, entry.nodeId)
  }
  return next
}

/** removeNode that first detaches instances of any main component inside the removed subtree. */
export function removeNodeDetaching(design: Design, boardId: string, nodeId: string): Design {
  const board = getBoard(design, boardId)
  const hit = board ? findNode(board.root, nodeId) : null
  if (!hit?.parent) return design
  return removeNode(detachInstancesOf(design, mainsInSubtree(hit.node)), boardId, nodeId)
}

/** removeBoard that first detaches instances of the board's main components elsewhere. */
export function removeBoardDetaching(design: Design, boardId: string): Design {
  const board = getBoard(design, boardId)
  if (!board) return design
  return removeBoard(detachInstancesOf(design, mainsInSubtree(board.root)), boardId)
}

/** The node as drawn at `breakpointId`: its override's keys (those its type honours) win. */
export function applyResponsive<T extends DesignNode>(node: T, breakpointId: string | null | undefined): T {

  const override = node.responsive[breakpointId]
  if (!override) return node
  const out: Record<string, unknown> = { ...node }
  for (const key of KEYS_BY_TYPE[node.type]) {
    const value = override[key]
    if (value !== undefined) out[key] = value
  }
  return out as T

/** The node with its shared style's fields applied (the style wins for the fields it sets). */
export function applyStyles<T extends DesignNode>(node: T, styles: DesignStyles | undefined): T {
  if (node.type === 'text' && node.textStyle !== undefined) {
    const style = styles?.text?.find((entry) => entry.id === node.textStyle)
    if (!style) return node
    const out: Record<string, unknown> = { ...node, style: style.style }
    for (const key of ['fontSize', 'weight', 'lineHeight', 'letterSpacing', 'color'] as const) {
      if (style[key] !== undefined) out[key] = style[key]
    }
    return out as T
  }

export type Size2 = { w: number; h: number }

const round = (value: number) => Math.round(value * 100) / 100

function constrainAxis(
  pos: number,
  len: number | undefined,
  kind: string | undefined,
  from: number,
  to: number,
): { pos: number; len: number | undefined } {
  const delta = to - from
  switch (kind) {
    case 'right':
    case 'bottom':
      return { pos: round(pos + delta), len }
    case 'leftRight':
    case 'topBottom':
      return len === undefined ? { pos, len } : { pos, len: round(Math.max(1, len + delta)) }
    case 'center':
      return { pos: round(pos + delta / 2), len }
    case 'scale': {
      if (from <= 0) return { pos, len }
      const ratio = to / from
      return { pos: round(pos * ratio), len: len === undefined ? undefined : round(Math.max(1, len * ratio)) }
    }
    default:
      return { pos, len }
  }
}

/**
 * Where an absolute child goes when its parent goes from `from` to `to`, by its constraints
 * (left/top, the default, never move). Only numeric sizes stretch or scale.
 */
export function constrainNode<T extends DesignNode>(node: T, from: Size2, to: Size2, constraints: Constraints | undefined = node.constraints): T {
  if (!constraints || (from.w === to.w && from.h === to.h)) return node
  const record = node as Record<string, unknown>
  const w = typeof record.w === 'number' ? record.w : undefined
  const h = typeof record.h === 'number' ? record.h : undefined
  const hx = constrainAxis(node.x ?? 0, w, constraints.h, from.w, to.w)
  const vy = constrainAxis(node.y ?? 0, h, constraints.v, from.h, to.h)
  const out: Record<string, unknown> = { ...node }
  if (hx.pos !== (node.x ?? 0)) out.x = hx.pos
  if (vy.pos !== (node.y ?? 0)) out.y = vy.pos
  if (hx.len !== undefined && hx.len !== w) out.w = hx.len
  if (vy.len !== undefined && vy.len !== h) out.h = vy.len
  return out as T
}

/** An absolute frame's children moved by their constraints for a resize of the frame. */
export function constrainChildren(frame: FrameNode, from: Size2, to: Size2): FrameNode {
  if (frame.layout !== 'absolute' || (from.w === to.w && from.h === to.h)) return frame
  let changed = false
  const children = frame.children.map((child) => {
    const next = constrainNode(child, from, to)
    if (next !== child) changed = true
    return next
  })

/**
 * Applies a w/h patch to a node and, when it is an absolute frame with a numeric size before and
 * after, moves its children by their constraints (the editor's resize and the inspector's W/H).
 */
export function resizeNodeConstrained(design: Design, boardId: string, nodeId: string, patch: Record<string, unknown>): Design {
  return updateBoard(design, boardId, (board) => {
    const hit = findNode(board.root, nodeId)
    if (!hit) return board
    const before = hit.node
    let after = { ...before, ...patch } as DesignNode
    for (const [key, value] of Object.entries(patch)) if (value === undefined) delete (after as Record<string, unknown>)[key]
    if (after.type === 'frame' && before.type === 'frame') {
      const from = numericSize(before)
      const to = numericSize(after)
      if (from && to) after = constrainChildren(after, from, to)
    }
    return { ...board, root: swapIn(board.root, nodeId, after) as FrameNode }
  })
}

function materializeNode(node: DesignNode, ctx: ResolveContext, parent: { from: Size2; to: Size2 } | null, self?: { from: Size2; to: Size2 }): DesignNode {
  const styled = applyStyles(node, ctx.styles)
  let out: DesignNode = applyResponsive(styled, ctx.breakpoint)
  if (parent && out.constraints) out = constrainNode(out, parent.from, parent.to)
  const plain = { ...out } as Record<string, unknown>
  delete plain.responsive
  delete plain.textStyle
  delete plain.fillStyle
  out = plain as unknown as DesignNode
  if (out.type !== 'frame') return out
  const sizes = self ?? (() => {
    const from = numericSize(styled)
    const to = numericSize(out)
    return from && to ? { from, to } : null
  })()
  const forChildren = out.layout === 'absolute' && sizes ? sizes : null
  return { ...out, children: out.children.map((child) => materializeNode(child, ctx, forChildren)) }
}

/**
 * A board as drawn at `breakpointId` (its size, overrides, styles and constraints resolved, with
 * the `responsive`/`textStyle`/`fillStyle` keys dropped). A board that does not declare the
 * breakpoint draws at its base size. For drawing and checking only: never stored.

  const w = previewWidth ?? breakpoint?.w ?? board.w
  const h = breakpoint?.h ?? board.h
  const ctx: ResolveContext = { styles, breakpoint: breakpoint?.id ?? null }
  const root = materializeNode(board.root, ctx, null, { from: { w: board.w, h: board.h }, to: { w, h } }) as FrameNode

export function materializeDesign(

// ---- editing at a breakpoint

/**
 * Splits a node patch for the active breakpoint: keys the type honours there go to the override,
 * the rest (text, colours, names) to the base node. With no breakpoint everything is base.
 */
export function routePatch(
  type: NodeType,
  patch: Record<string, unknown>,
  breakpointId: string | null | undefined,
): { base: Record<string, unknown>; override: Record<string, unknown> } {
  if (!breakpointId) return { base: patch, override: {} }
  const keys = new Set<string>(KEYS_BY_TYPE[type])
  const base: Record<string, unknown> = {}
  const override: Record<string, unknown> = {}
  for (const [key, value] of Object.entries(patch)) (keys.has(key) ? override : base)[key] = value
  return { base, override }
}

/** The node with `override` merged into `responsive[breakpointId]` (undefined deletes a key). */
export function withOverride<T extends DesignNode>(node: T, breakpointId: string, override: Record<string, unknown>): T {
  const current: Record<string, unknown> = { ...(node.responsive?.[breakpointId] ?? {}) }
  for (const [key, value] of Object.entries(override)) {
    if (value === undefined) delete current[key]
    else current[key] = value
  }
  const responsive: Record<string, ResponsiveOverride> = { ...(node.responsive ?? {}) }
  if (Object.keys(current).length === 0) delete responsive[breakpointId]
  else responsive[breakpointId] = current as ResponsiveOverride
  const out: Record<string, unknown> = { ...node }
  if (Object.keys(responsive).length === 0) delete out.responsive
  else out.responsive = responsive
  return out as T
}

/**
 * The editor's one write path for a node patch: base fields on the node, breakpoint fields into
 * its override when a breakpoint (one its board declares) is active. Size changes on an absolute
 * frame at the base move its constrained children.
 */
export function patchNodeAt(
  design: Design,
  boardId: string,
  nodeId: string,
  patch: Record<string, unknown>,
  breakpointId: string | null | undefined,
): Design {
  const board = design.boards.find((entry) => entry.id === boardId)
  const hit = board ? findNode(board.root, nodeId) : null
  if (!board || !hit) return design
  const active = breakpointOf(board, breakpointId)?.id ?? null
  const { base, override } = routePatch(hit.node.type, patch, active)
  let next = design
  if (Object.keys(base).length > 0) next = resizeNodeConstrained(next, boardId, nodeId, base)
  if (active && Object.keys(override).length > 0) {
    next = updateBoard(next, boardId, (current) => {
      const found = findNode(current.root, nodeId)
      if (!found) return current
      return { ...current, root: swapIn(current.root, nodeId, withOverride(found.node, active, override)) as FrameNode }
    })
  }
  return next
}

/** Drops a node's override at one breakpoint (or one key of it). */
export function clearOverride(design: Design, boardId: string, nodeId: string, breakpointId: string, key?: keyof ResponsiveOverride): Design {
  return updateBoard(design, boardId, (board) => {
    const hit = findNode(board.root, nodeId)
    if (!hit?.node.responsive?.[breakpointId]) return board
    const patch: Record<string, unknown> = key ? { [key]: undefined } : Object.fromEntries(Object.keys(hit.node.responsive[breakpointId]).map((k) => [k, undefined]))
    return { ...board, root: swapIn(board.root, nodeId, withOverride(hit.node, breakpointId, patch)) as FrameNode }
  })
}

export function strokeDashArray(pattern: StrokePattern | undefined, width: number): string | undefined {

  if (pattern === 'dashed') return `${3 * width} ${2 * width}`
  if (pattern === 'dotted') return `0.01 ${2 * width}`

/** Regular polygon points inscribed in the box (first vertex at the top). */
export function polygonPoints(sides: number, w: number, h: number): Array<[number, number]> {
  const n = Math.max(3, Math.min(24, Math.round(sides)))
  const out: Array<[number, number]> = []
  for (let i = 0; i < n; i += 1) {
    const angle = -Math.PI / 2 + (i * 2 * Math.PI) / n
    out.push([r(w / 2 + (w / 2) * Math.cos(angle)), r(h / 2 + (h / 2) * Math.sin(angle))])
  }
  return out

/** Star points (outer and inner alternating, first point at the top). */
export function starPoints(points: number, innerRatio: number, w: number, h: number): Array<[number, number]> {
  const n = Math.max(3, Math.min(24, Math.round(points)))
  const inner = Math.max(0.1, Math.min(0.95, innerRatio))
  const out: Array<[number, number]> = []
  for (let i = 0; i < n * 2; i += 1) {
    const angle = -Math.PI / 2 + (i * Math.PI) / n
    const k = i % 2 === 0 ? 1 : inner
    out.push([r(w / 2 + (w / 2) * k * Math.cos(angle)), r(h / 2 + (h / 2) * k * Math.sin(angle))])
  }
  return out
}

const pathOf = (points: Array<[number, number]>, closed: boolean) =>
  points.map(([x, y], i) => `${i === 0 ? 'M' : 'L'}${r(x)} ${r(y)}`).join(' ') + (closed ? ' Z' : '')

/**
 * A path whose segment i (point i to i + 1; with `closed`, the last back to the first) is a cubic
 * curve when `curves[i]` gives its two control points (shares of the box), else a line.
 */
export function curvedPathOf(
  points: Array<[number, number]>,
  curves: ReadonlyArray<readonly [number, number, number, number] | null>,
  closed: boolean,
  w: number,
  h: number,
): string {
  if (points.length === 0) return ''
  const parts = [`M${r(points[0][0])} ${r(points[0][1])}`]
  const segments = closed ? points.length : points.length - 1
  for (let i = 0; i < segments; i += 1) {
    const [x, y] = points[(i + 1) % points.length]
    const curve = curves[i]
    parts.push(curve ? `C${r(curve[0] * w)} ${r(curve[1] * h)} ${r(curve[2] * w)} ${r(curve[3] * h)} ${r(x)} ${r(y)}` : `L${r(x)} ${r(y)}`)
  }
  return parts.join(' ') + (closed ? ' Z' : '')
}

/**
 * The SVG path of a vector shape in its w×h box, plus whether it is filled (closed) or only
 * stroked. An arrow runs left to right through the box's middle with its head at the end.
 */
export function shapeGeometry(node: ShapeNode, w: number, h: number): { d: string; closed: boolean } {
  switch (node.kind) {
    case 'polygon':
      return { d: pathOf(polygonPoints(node.sides ?? 6, w, h), true), closed: true }
    case 'star':
      return { d: pathOf(starPoints(node.sides ?? 5, node.innerRatio ?? 0.5, w, h), true), closed: true }
    case 'path': {
      const points = (node.vertices ?? [[0, 0], [1, 1]]).map(([x, y]) => [x * w, y * h] as [number, number])
      if (!node.curves?.some(Boolean)) return { d: pathOf(points, node.closed === true), closed: node.closed === true }
      return { d: curvedPathOf(points, node.curves, node.closed === true, w, h), closed: node.closed === true }
    }
    case 'arrow': {
      const stroke = node.strokeWidth ?? 2
      const head = Math.min(w / 2, Math.max(8, stroke * 4))
      const mid = h / 2
      return {
        d: `M0 ${r(mid)} L${r(w)} ${r(mid)} M${r(w - head)} ${r(mid - head / 2)} L${r(w)} ${r(mid)} L${r(w - head)} ${r(mid + head / 2)}`,
        closed: false,
      }
    }
    default:
      return { d: `M0 0 L${r(w)} 0 L${r(w)} ${r(h)} L0 ${r(h)} Z`, closed: true }
  }
}

/**
 * A pen path from board px points (and per-segment cubic control points, also board px): its box
 * at (x, y), and the points and controls as shares of that box, which is how a path shape stores
 * them. The editor's pen and the agent's addPath both build paths with this.
 */
export function pathShapeFromPoints(
  points: Array<[number, number]>,
  curves: Array<[number, number, number, number] | null> | undefined,
  taken: ReadonlySet<string>,
  id: string,
): ShapeNode {
  const minX = Math.min(...points.map(([x]) => x))
  const minY = Math.min(...points.map(([, y]) => y))
  const w = Math.max(1, Math.max(...points.map(([x]) => x)) - minX)
  const h = Math.max(1, Math.max(...points.map(([, y]) => y)) - minY)
  const share = (value: number, from: number, size: number) => Math.round(((value - from) / size) * 1000) / 1000
  const clampShare = (value: number) => Math.min(2, Math.max(-1, value))
  let nodeId = id
  for (let n = 2; taken.has(nodeId); n += 1) nodeId = `${id}-${n}`
  return {
    type: 'shape',
    id: nodeId,
    kind: 'path',
    x: Math.round(minX),
    y: Math.round(minY),
    w: Math.round(w),
    h: Math.round(h),
    vertices: points.map(([x, y]) => [share(x, minX, w), share(y, minY, h)] as [number, number]),
    ...(curves?.some(Boolean)
      ? {
          curves: curves.map((curve) =>
            curve ? ([clampShare(share(curve[0], minX, w)), clampShare(share(curve[1], minY, h)), clampShare(share(curve[2], minX, w)), clampShare(share(curve[3], minY, h))] as [number, number, number, number]) : null,
          ),
        }
      : {}),
  }
}
