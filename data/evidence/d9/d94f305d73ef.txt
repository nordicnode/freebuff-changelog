/**
 * True when `prompt` is the compact command AND received `compactPrompt` from
 * the map above. run-agent-step's history-replacing branch must use this
 * rather than matching the prompt itself: a trigger that matches prompts the
 * map does not (e.g. `/Compact` under a lowercased comparison) replaces the
 * whole history with an ordinary answer instead of a summary.
 */
export function isCompactCommandPrompt(prompt: string | undefined): boolean {
  return (
    prompt !== undefined &&
    prompt in additionalSystemPrompts &&
    additionalSystemPrompts[prompt as keyof typeof additionalSystemPrompts] ===
      compactPrompt
  )
}

Note: terminal commands run in Git Bash on Windows, not cmd.exe or PowerShell.
- Use POSIX syntax — \`ls\`, \`mv\`, \`rm\`, \`cp\`, \`grep\` — never \`dir\`, \`move\`, \`del\`, \`copy\` or \`findstr\`. Write paths with forward slashes: bash treats \`\\\` as an escape, so \`C:\\Users\\me\` becomes \`C:Usersme\`.
- When a Windows-only task needs PowerShell, write the script to a \`.ps1\` file and run \`powershell.exe -NoProfile -ExecutionPolicy Bypass -File script.ps1\`. A short command with no quotes inside it can go in single quotes: \`powershell.exe -NoProfile -Command 'Get-Process node'\`. Never put PowerShell in bash double quotes: bash replaces \`$_\`, \`$env:NAME\` and backticks before PowerShell sees them.
- Git Bash turns an argument that starts with \`/\` into a Windows path, so native switches such as \`/nologo\` break. Run such a command as \`MSYS2_ARG_CONV_EXCL='*' tool /switch\`, or from a \`.cmd\` or \`.ps1\` file.
- Write a script longer than a few lines (Python, SQL, PowerShell) to a file and run the file, not through a heredoc or a \`-c\` one-liner.
`.trim()

import { runCommand, type RunCommand } from './git'

export type ProcessRow = { pid: number; ppid: number; command: string }

/** Every process with its parent and command line. Read-only; bounded by the command's time limit. */
export async function processTable(platform: string, run: RunCommand = runCommand): Promise<ProcessRow[]> {
  if (platform === 'win32') {
    const result = await run('powershell.exe', ['-NoProfile', '-NonInteractive', '-Command',
      'Get-CimInstance Win32_Process | ForEach-Object { "$($_.ProcessId)`t$($_.ParentProcessId)`t$($_.CommandLine)" }'], { timeoutMs: 8_000 })
    return result.stdout.split(/\r?\n/).flatMap(line => {
      const [pid, ppid, ...command] = line.split('\t')
      return pid && ppid ? [{ pid: Number(pid), ppid: Number(ppid), command: command.join('\t') }] : []
    })
  }
  const result = await run('ps', ['-axww', '-o', 'pid=,ppid=,command='], { timeoutMs: 4_000 })
  return result.stdout.split('\n').flatMap(line => {
    const match = /^\s*(\d+)\s+(\d+)\s+(.*)$/.exec(line)
    return match ? [{ pid: Number(match[1]), ppid: Number(match[2]), command: match[3]! }] : []
  })
}
