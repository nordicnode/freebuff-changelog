import { createRequire } from 'module'

import { Argument, Command } from 'commander'

import { IS_FREEBUFF, type AgentMode } from './utils/constants'
import { getCliEnv } from './utils/env'

const require = createRequire(import.meta.url)

export type ParsedArgs = {
  initialPrompt: string | null
  command?: string
  agent?: string
  clearLogs: boolean
  continue: boolean
  continueId?: string | null
  cwd?: string
  initialMode?: AgentMode
  /** Load repository `.agents` content without the trust prompt (CI opt-in). */
  trustAgents: boolean
}

export function loadPackageVersion(): string {
  const env = getCliEnv()
  if (env.CODEBUFF_CLI_VERSION) {
    return env.CODEBUFF_CLI_VERSION
  }

  try {
    const pkg = require('../package.json') as { version?: string }
    if (pkg.version) {
      return pkg.version
    }
  } catch {
    // Continue to dev fallback
  }

  return 'dev'
}

export function parseArgs({
  argv = process.argv,
  isFreebuff = IS_FREEBUFF,
  version = loadPackageVersion(),
}: {
  argv?: string[]
  isFreebuff?: boolean
  version?: string
} = {}): ParsedArgs {
  const program = new Command()

  if (isFreebuff) {
    // Freebuff: simplified CLI - no prompt args, no agent override, no clear-logs
    program
      .name('freebuff')
      .description('Freebuff - Free AI coding assistant')
      .version(version, '-v, --version', 'Print the CLI version')
      .option(
        '--continue [conversation-id]',
        'Continue from a previous conversation (optionally specify a conversation id)',
      )
      .option(
        '--cwd <directory>',
        'Set the working directory (default: current directory)',
      )
      .option(
        '--trust-agents',
        "Load this repository's .agents files and mcp.json without asking (for CI)",
      )
      .addArgument(
        new Argument('[command]', 'Command to run: login, or doctor to check this computer for common problems').choices(['login', 'doctor']),
      )
      .helpOption('-h, --help', 'Show this help message')
  } else {
    // Codebuff: full CLI with all options
    program
      .name('codebuff')
      .description('Codebuff CLI - AI-powered coding assistant')
      .version(version, '-v, --version', 'Print the CLI version')
      .option(
        '--agent <agent-id>',
        'Run a specific agent id (skips loading local .agents overrides)',
      )
      .option(
        '--clear-logs',
        'Remove any existing CLI log files before starting',
      )
      .option(
        '--continue [conversation-id]',
        'Continue from a previous conversation (optionally specify a conversation id)',
      )
      .option(
        '--cwd <directory>',
        'Set the working directory (default: current directory)',
      )
      .option(
        '--trust-agents',
        "Load this repository's .agents files and mcp.json without asking (for CI)",
      )
      .option('--lite', 'Start in LITE mode')
      .option('--free', 'Start in LITE mode (deprecated alias)')
      .option('--max', 'Start in MAX mode')
      .option('--plan', 'Start in PLAN mode')
      .addHelpText(
        'after',
        '\nCommands:\n  login                          Log in to your account\n  publish                        Publish agents to the registry\n  doctor                         Check this computer for common problems',
      )
      .helpOption('-h, --help', 'Show this help message')
      .argument('[prompt...]', 'Initial prompt to send to the agent')
      .allowExcessArguments(true)
  }

  program.parse(argv)

  const options = program.opts()
  const args = program.args

  const continueFlag = options.continue

  // Determine initial mode from flags (last flag wins if multiple specified)
  // Freebuff always uses LITE mode
  let initialMode: AgentMode | undefined
  if (isFreebuff) {
    initialMode = 'LITE'
  } else {
    if (options.free || options.lite) initialMode = 'LITE'
    if (options.max) initialMode = 'MAX'
    if (options.plan) initialMode = 'PLAN'
  }

  return {
    initialPrompt: !isFreebuff && args.length > 0 ? args.join(' ') : null,
    command: args[0],
    agent: options.agent,
    clearLogs: options.clearLogs || false,
    continue: Boolean(continueFlag),
    continueId:
      typeof continueFlag === 'string' && continueFlag.trim().length > 0
        ? continueFlag.trim()
        : null,
    cwd: options.cwd,
    initialMode,
    trustAgents: Boolean(options.trustAgents),
  }
}

function createQueryClient(): QueryClient {
  return new QueryClient({
    defaultOptions: {
      queries: {
        staleTime: 5 * 60 * 1000, // 5 minutes - auth tokens don't change frequently
        gcTime: 10 * 60 * 1000, // 10 minutes - keep cached data a bit longer
        retry: false, // Don't retry failed auth queries automatically
        refetchOnWindowFocus: false, // CLI doesn't have window focus
        refetchOnReconnect: true, // Refetch when network reconnects
        refetchOnMount: false, // Don't refetch on every mount
      },
      mutations: {
        retry: 1, // Retry mutations once on failure
      },
    },
  })

async function main(): Promise<void> {
  // CI gate: `<binary> --smoke-tree-sitter` proves the embedded wasm boots
  // through Parser.init end-to-end. Has to live BEFORE commander.parse() —
  // an earlier attempt put this in a pre-init module with top-level await,
  // and on Windows that didn't actually pause module evaluation (commander
  // still ran first and rejected the unknown flag).
  if (process.argv.includes('--smoke-tree-sitter')) {
    const wasmBinary = (
      globalThis as { __CODEBUFF_TREE_SITTER_WASM_BINARY__?: Uint8Array }
    ).__CODEBUFF_TREE_SITTER_WASM_BINARY__
    const wasmPath = (
      globalThis as { __CODEBUFF_TREE_SITTER_WASM_PATH__?: string }
    ).__CODEBUFF_TREE_SITTER_WASM_PATH__

    // Diagnostic dump so CI logs (and bug reports) show exactly what
    // the runtime saw when smoke fails. process.execPath, the
    // siblingPath we expect, and what's actually in that directory.
    const fs = await import('fs')
    const path = await import('path')
    const execDir = path.dirname(process.execPath)
    const siblingPath = path.join(execDir, 'tree-sitter.wasm')
    let dirListing: string[] = []
    try {
      dirListing = fs.readdirSync(execDir)
    } catch (err) {
      dirListing = [
        `<readdir failed: ${err instanceof Error ? err.message : err}>`,
      ]
    }
    console.error(
      `[smoke diag] execPath=${process.execPath}\n` +
        `[smoke diag] execDir=${execDir}\n` +
        `[smoke diag] siblingPath=${siblingPath}\n` +
        `[smoke diag] siblingExists=${fs.existsSync(siblingPath)}\n` +
        `[smoke diag] dir contents (${dirListing.length}): ${dirListing.slice(0, 30).join(', ')}\n` +
        `[smoke diag] globalThis wasmPath=${wasmPath ?? '<unset>'}\n` +
        `[smoke diag] globalThis wasmBinary bytes=${wasmBinary?.byteLength ?? 0}\n`,
    )

    try {
      const { Parser } = await import('web-tree-sitter')
      // Pick the best wasm source available, falling back to the
      // sibling-of-execPath lookup if pre-init couldn't reach it. By
      // main() time process.execPath has stabilized to the disk path
      // even on Windows, where it was the bunfs path during pre-init.
      let effectiveBinary = wasmBinary
      let effectivePath = wasmPath
      if (!effectiveBinary && !effectivePath && fs.existsSync(siblingPath)) {
        effectivePath = siblingPath
        effectiveBinary = new Uint8Array(fs.readFileSync(siblingPath))
      }

      if (effectiveBinary) {
        await Parser.init({ wasmBinary: effectiveBinary })
        // Marker grepped by cli/scripts/smoke-binary.ts — keep this exact text.
        console.log(
          `tree-sitter smoke ok (wasmBinary, ${effectiveBinary.byteLength} bytes)`,
        )
      } else if (effectivePath) {
        await Parser.init({
          locateFile: (name: string) =>
            name === 'tree-sitter.wasm' ? effectivePath! : name,
        })
        console.log(`tree-sitter smoke ok (locateFile, path=${effectivePath})`)
      } else {
        console.error(
          'tree-sitter smoke FAIL: no wasm available — pre-init published ' +
            'nothing and the sibling-of-execPath fallback also missed. See ' +
            'the diag above for paths.',
        )
        process.exit(1)
      }
      process.exit(0)
    } catch (err) {
      console.error('tree-sitter smoke FAIL:', err)
      process.exit(1)
    }
  }

  // CI gate: `<binary> --smoke-api-url` prints the backend URL the SDK resolved
  // and exits, so cli/scripts/smoke-binary.ts can prove dotenv files in the CWD
  // cannot steer it (see --no-compile-autoload-dotenv in build-binary.ts). Must
  // precede commander.parse(), which rejects unknown flags.
  if (process.argv.includes('--smoke-api-url')) {
    // Marker parsed by cli/scripts/smoke-binary.ts — keep this exact prefix.
    console.log(`api-url smoke: ${getWebsiteUrl()}`)
    process.exit(0)
  }

  // Native-Windows release gate. The external harness starts the packaged
  // binary inside winpty so this exercises a real console, OpenTUI renderer,
  // console-free broker, Git Bash child, and SDK process lifecycle.
  // Keep this before commander.parse(), which intentionally knows nothing
  // about internal smoke-only flags.

  )
  const endOfOptionsIndex = process.argv.indexOf('--')

    terminalBrokerSmokeIndex !== -1 &&

        configDir: getConfigDir(),
        apiUrl: getWebsiteUrl(),

    command,
    agent,
    clearLogs,
    continue: continueChat,
    continueId,
    cwd,
    initialMode,
    trustAgents,

  const isPublishCommand = command === 'publish'

    'started',

    }

  // next --continue launch still has a prior row for the health dashboard's
  // rapid-resume and interruption joins.

  }

  // When --agent is provided, skip local .agents to avoid overrides.
  // Repository-scoped .agents directories (cwd and its parent) execute code
  // on import and spawn mcp.json servers, so they pass the trust gate first;
  // ~/.agents is the user's own and is always loaded.
  if (isPublishCommand || !hasAgentOverride) {

      candidateDirs: getDefaultAgentDirs(),
      homeAgentsDir: path.join(os.homedir(), '.agents'),
      interactive: Boolean(process.stdin.isTTY && process.stdout.isTTY),

        console.log(yellow(message))
      },
    })

        '[agents] Untrusted repository .agents directories were not loaded',
      )
    }

    const agentIds = process.argv.slice(publishIndex + 1)
    const result = await handlePublish(agentIds)

      for (const agent of result.agents) {
        logger.info(
          cyan(
            `  - ${agent.displayName} (${result.publisherId}/${agent.id}@${agent.version})`,
          ),
        )
      }
      process.exit(0)
    } else {

      if (result.error) logger.error(red(`Error: ${result.error}`))
      if (result.details) logger.error(red(result.details))
      if (result.hint) logger.warn(yellow(`Hint: ${result.hint}`))
      process.exit(1)
    }
  }

    clearLogFile()

  // the full conversation to log.jsonl. Deferred to keep the stat sweep over
  // chat directories off the startup path.

 * Checks only the CLI needs: how it is installed, what the shell runs when you type its name,

const plural = (n: number, one: string, many = `${one}s`) => `${n} ${n === 1 ? one : many}`
const mb = (bytes: number) => `${Math.max(1, Math.round(bytes / 1024 ** 2))} MB`

export function compareVersions(a: string, b: string) {
  const parts = (v: string) => v.split(/[.+-]/).map(n => Number.parseInt(n, 10)).map(n => (Number.isFinite(n) ? n : 0))
  const x = parts(a)
  const y = parts(b)
  for (let i = 0; i < Math.max(x.length, y.length); i++) {
    const d = (x[i] ?? 0) - (y[i] ?? 0)
    if (d) return Math.sign(d)
  }
  return 0
}

export function installFindings(input: {
  version: string
  latest: string | null
  launcherVersion: string | null
  packageName: string
}): DoctorFinding[] {
  const out: DoctorFinding[] = []
  if (input.version === 'dev') return out
  if (input.latest && compareVersions(input.latest, input.version) > 0)
    out.push(finding('install.outdated', 'warning', 'install', `${input.packageName} ${input.latest} is available (you have ${input.version})`,

    out.push(finding('install.current', 'ok', 'install', `${input.packageName} is up to date`, `Version ${input.version}.`))

    out.push(finding('install.old-launcher', 'info', 'install', 'The npm launcher is older than the app it starts',

export async function binaryFindings(input: { configDir: string; packageName: string; platform: string; redact: DoctorRedactor; fs?: Fs; run?: RunCommand }): Promise<DoctorFinding[]> {
  const files = input.fs ?? fs
  const binary = path.join(input.configDir, input.platform === 'win32' ? `${input.packageName}.exe` : input.packageName)
  let stats: fs.Stats
  try { stats = files.statSync(binary) } catch { return [] } // run from source or another install method
  if (!stats.size)
    return [finding('install.binary-empty', 'error', 'install', 'The downloaded app is empty',
      `${input.redact.path(binary)} has no content, usually from a download that was cut off.`,
      ['Delete that file and start the CLI again; the launcher downloads it again.'])]
  if (input.platform !== 'win32' && !(stats.mode & 0o111))
    return [finding('install.binary-not-executable', 'error', 'install', 'The downloaded app cannot be run',
      `${input.redact.path(binary)} is not marked as executable.`, [`Run \`chmod +x ${input.redact.path(binary)}\`, or delete it and start the CLI again.`])]
  if (input.platform === 'darwin' && input.run) {
    const quarantine = await input.run('xattr', ['-p', 'com.apple.quarantine', binary], { timeoutMs: 3_000 })
    if (quarantine.code === 0 && quarantine.stdout.trim())

        'Gatekeeper may refuse to run a file marked as downloaded from the internet.',
        [`Run \`xattr -d com.apple.quarantine ${input.redact.path(binary)}\`.`])]
  }
  return []
}

/** Every file the shell could run for `name`, in PATH order. */
export function commandsOnPath(name: string, env: Record<string, string | undefined>, platform: string, files: Fs = fs) {
  const separator = platform === 'win32' ? ';' : ':'
  const dirs = (env.PATH ?? env.Path ?? '').split(separator).filter(Boolean)
  const extensions = platform === 'win32' ? ['.exe', '.cmd', '.bat', '.ps1', ''] : ['']
  const found: string[] = []
  const seen = new Set<string>()
  const normalized = (dir: string) => (platform === 'win32' ? dir.toLowerCase() : dir).replace(/[\\/]+$/, '')
  let repeated = 0
  for (const dir of dirs) {
    const key = normalized(dir)
    if (seen.has(key)) { repeated++; continue }
    seen.add(key)
    for (const ext of extensions) {
      const candidate = path.join(dir, name + ext)
      try { if (files.statSync(candidate).isFile()) { found.push(candidate); break } } catch { /* not here */ }
    }
  }
  return { found, repeated }
}

  if (dirs.length > 1)

const PID_DIR = /^\.download-temp-(\d+)$/

/** Superseded engine copies, interrupted downloads, and download folders whose process is gone. */
export function leftoverFindings(input: { configDir: string; now: number; isAlive: (pid: number) => boolean; fs?: Fs }): DoctorFinding[] {
  const files = input.fs ?? fs
  let names: string[]
  try { names = files.readdirSync(input.configDir) } catch { return [] }
  let oldCopies = 0
  let partial = 0
  let bytes = 0
  let stale = 0
  for (const name of names) {
    const full = path.join(input.configDir, name)
    let stats: fs.Stats
    try { stats = files.statSync(full) } catch { continue }
    if (/\.old\.\d+(?:\.\d+)?$/.test(name)) { oldCopies++; bytes += stats.size }
    else if (/\.(?:part|crdownload)$/.test(name) && input.now - stats.mtimeMs > 60 * 60_000) { partial++; bytes += stats.size }
    else {
      const pid = PID_DIR.exec(name)?.[1]
      if (pid && stats.isDirectory() && !input.isAlive(Number(pid))) stale++
    }
  }
  const out: DoctorFinding[] = []
  if (oldCopies || partial)
    out.push(finding('install.leftover-copies', 'info', 'install', `Old app copies and unfinished downloads use ${mb(bytes)}`,
      `${[oldCopies ? plural(oldCopies, 'replaced copy', 'replaced copies') : '', partial ? plural(partial, 'unfinished download') : ''].filter(Boolean).join(' and ')} are left in the CLI's folder from earlier updates. On Windows a copy still in use cannot be deleted until it exits.`,
      ['Quit every running copy of the CLI, then delete the files ending in `.old.<number>` or `.part` in its config folder.']))
  if (stale)
    out.push(finding('install.stale-download', 'info', 'install', `${plural(stale, 'download folder')} from a process that is gone`,
      'An update download was interrupted and its folder was left behind. The next update starts over.', ['Delete the `.download-temp-<number>` folders in the CLI\'s config folder.']))
  return out
}

/** The credentials file's shape and permissions. The token itself is never read into a finding. */
export function authFindings(input: { configDir: string; platform: string; redact: DoctorRedactor; fs?: Fs; command: string }): DoctorFinding[] {
  const files = input.fs ?? fs
  const file = path.join(input.configDir, 'credentials.json')
  let text: string
  try { text = files.readFileSync(file, 'utf8') as string } catch (error) {
    if ((error as NodeJS.ErrnoException).code === 'ENOENT')
      return [finding('auth.signed-out', 'info', 'auth', 'Not signed in', 'There is no saved sign-in on this computer.', [`Run \`${input.command} login\`.`])]
    return [finding('auth.unreadable', 'error', 'auth', 'The saved sign-in cannot be read',
      `${input.redact.path(file)} exists but could not be opened.`, ['Check that your user owns the file, or delete it and sign in again.'])]
  }
  let profiles: unknown[]
  try {
    const parsed = JSON.parse(text) as unknown
    profiles = parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? Object.values(parsed) : []
  } catch {
    return [finding('auth.corrupt', 'error', 'auth', 'The saved sign-in is damaged',
      `${input.redact.path(file)} is not valid JSON, so every request runs signed out.`, [`Delete it and run \`${input.command} login\`.`])]
  }
  const usable = profiles.some(p => p && typeof p === 'object' &&
    (typeof (p as { authToken?: unknown }).authToken === 'string' || (p as { tokenStore?: unknown }).tokenStore === 'keychain'))
  const out: DoctorFinding[] = usable
    ? [finding('auth.ok', 'ok', 'auth', 'A sign-in is saved', 'The saved sign-in file reads cleanly.')]
    : [finding('auth.empty', 'warning', 'auth', 'The saved sign-in has no account in it', 'The file exists but holds no usable sign-in.', [`Run \`${input.command} login\`.`])]
  if (input.platform !== 'win32') {
    try {
      if (files.statSync(file).mode & 0o077)
        out.push(finding('auth.permissions', 'warning', 'auth', 'Other users on this computer can read your sign-in',
          `${input.redact.path(file)} is readable beyond your own account.`, [`Run \`chmod 600 ${input.redact.path(file)}\`.`]))
    } catch { /* checked above */ }
  }
  return out
}

/** Size of local history, newest log tail, and the last fatal crash the CLI recorded. */
export function historyFindings(input: { configDir: string; now: number; fs?: Fs; maxEntries?: number }): { findings: DoctorFinding[]; newestLog: string | null } {
  const files = input.fs ?? fs
  const projects = path.join(input.configDir, 'projects')
  let bytes = 0
  let unreadable = 0
  let newest: { file: string; mtime: number } | null = null
  let seen = 0
  const stack = [projects]
  while (stack.length && seen < (input.maxEntries ?? 20_000)) {
    const dir = stack.pop()!
    let names: string[]
    try { names = files.readdirSync(dir) } catch { if (dir !== projects) unreadable++; continue }
    for (const name of names) {
      seen++
      const full = path.join(dir, name)
      let stats: fs.Stats
      try { stats = files.statSync(full) } catch { unreadable++; continue }
      if (stats.isDirectory()) stack.push(full)
      else {
        bytes += stats.size
        if (name === 'log.jsonl' && (!newest || stats.mtimeMs > newest.mtime)) newest = { file: full, mtime: stats.mtimeMs }
      }
    }
  }
  const out: DoctorFinding[] = []
  if (unreadable)
    out.push(finding('history.unreadable', 'warning', 'state', `${plural(unreadable, 'history file')} cannot be read`,
      'Some saved chats cannot be opened, so `--continue` may not find them.', ['Check that your user owns the CLI\'s config folder.']))
  if (bytes > 2 * 1024 ** 3)
    out.push(finding('history.large', 'info', 'state', `Local history uses ${mb(bytes)}`,
      'Old chats are kept until you remove them.', ['Delete project folders you no longer need under the CLI\'s `projects` folder.']))
  try {
    const crash = JSON.parse(files.readFileSync(path.join(input.configDir, 'last-fatal-crash.json'), 'utf8') as string) as { errorName?: unknown; crashedAt?: unknown; crashedVersion?: unknown }
    const at = typeof crash.crashedAt === 'string' ? Date.parse(crash.crashedAt) : NaN
    if (!Number.isFinite(at) || input.now - at < 7 * 86_400_000)
      out.push(finding('crash.recent', 'warning', 'state', 'The CLI crashed recently',
        `The last run stopped on a fatal error${typeof crash.errorName === 'string' ? ` (${crash.errorName.slice(0, 60)})` : ''}${typeof crash.crashedVersion === 'string' ? ` in version ${crash.crashedVersion.slice(0, 20)}` : ''}.`,
        ['Update to the current version.', 'If it happens again, send this report to support.']))
  } catch { /* no crash recorded */ }
  return { findings: out, newestLog: newest?.file ?? null }
}

/** The npm launcher runs on Node. Its version, and how many Node installs compete on PATH. */
export async function nodeFindings(input: { run: RunCommand; env: Record<string, string | undefined>; platform: string; fs?: Fs }): Promise<DoctorFinding[]> {

  }

    out.push(finding('node.several', 'info', 'install', `${plural(nodes.length, 'Node install')} on your PATH`,

  if (registry && !/registry\.npmjs\.org/.test(registry))

      'npm_config_registry points at a registry other than npmjs.org. If it mirrors slowly, installs and updates lag behind.', []))
  return out
}

export async function latestPublishedVersion(packageName: string, fetcher: (url: string, init?: RequestInit) => Promise<Response> = fetch) {

/** Every process with its parent and command line. Read-only; bounded by the command's time limit. */
export async function processTable(platform: string, run: RunCommand = runCommand): Promise<ProcessRow[]> {
  if (platform === 'win32') {
    const result = await run('powershell.exe', ['-NoProfile', '-NonInteractive', '-Command',
      'Get-CimInstance Win32_Process | ForEach-Object { "$($_.ProcessId)`t$($_.ParentProcessId)`t$($_.CommandLine)" }'], { timeoutMs: 8_000 })
    return result.stdout.split(/\r?\n/).flatMap(line => {
      const [pid, ppid, ...command] = line.split('\t')
      return pid && ppid ? [{ pid: Number(pid), ppid: Number(ppid), command: command.join('\t') }] : []
    })
  }
  const result = await run('ps', ['-axww', '-o', 'pid=,ppid=,command='], { timeoutMs: 4_000 })
  return result.stdout.split('\n').flatMap(line => {
    const match = /^\s*(\d+)\s+(\d+)\s+(.*)$/.exec(line)
    return match ? [{ pid: Number(match[1]), ppid: Number(match[2]), command: match[3]! }] : []
  })
}

+export type ProcessRow = { pid: number; ppid: number; command: string }
+
+/** Every process with its parent and command line. Read-only; bounded by the command's time limit. */
+export async function processTable(platform: string, run: RunCommand = runCommand): Promise<ProcessRow[]> {
+  if (platform === 'win32') {
+    const result = await run('powershell.exe', ['-NoProfile', '-NonInteractive', '-Command',
+      'Get-CimInstance Win32_Process | ForEach-Object { "$($_.ProcessId)`t$($_.ParentProcessId)`t$($_.CommandLine)" }'], { timeoutMs: 8_000 })
+    return result.stdout.split(/\r?\n/).flatMap(line => {
+      const [pid, ppid, ...command] = line.split('\t')
+      return pid && ppid ? [{ pid: Number(pid), ppid: Number(ppid), command: command.join('\t') }] : []
+    })
+  }
+  const result = await run('ps', ['-axww', '-o', 'pid=,ppid=,command='], { timeoutMs: 4_000 })
+  return result.stdout.split('\n').flatMap(line => {
+    const match = /^\s*(\d+)\s+(\d+)\s+(.*)$/.exec(line)
+    return match ? [{ pid: Number(match[1]), ppid: Number(match[2]), command: match[3]! }] : []
+  })
