/** When the CLI says its plan's limit resets, as it printed it (the computer's own clock). */
function limitReset(message: string): string | null {
  const at =
    /\btry again at ([^()\n]{1,60}?)\.?\s*$/i.exec(message) ??
    /\bresets (?:at )?([^()\n·∙|]{1,40}?(?: \([^)\n]{1,40}\))?)\.?\s*(?:$|[·∙|])/i.exec(message)
  return at?.[1]?.trim() || null
}

/**
 * The user-facing message for a native turn's journal `error`, or null when it
 * is not one we can name. Desktop builds before these messages existed wrote the
 * SDK's own text for every abort ("Claude Code process aborted by user").
 *
 * The CLI runs on the user's computer with their own sign-in, so most failures
 * are theirs to fix there: signed out, plan limit used up, a model their plan
 * cannot use. Only fixed classes are named; the CLI's raw text never is (it can
 * quote the request).
 */
export function deviceHarnessFailureMessage(
  message: string,
  harness?: DeviceHarness,
): string | null {
  if (
    message === DEVICE_HARNESS_TIME_LIMIT_MESSAGE ||
    message === DEVICE_HARNESS_STOPPED_MESSAGE
  )
    return message
  if (/process aborted by user/i.test(message))
    return DEVICE_HARNESS_STOPPED_MESSAGE
  if (message.length <= 1000 && DESKTOP_WORDED.some((pattern) => pattern.test(message)))
    return message
  if (!harness) return null
  const label = HARNESS_LABEL[harness]
  const account = HARNESS_ACCOUNT[harness]
  if (
    /\b(hit your (usage )?limit|usage limit reached|(5-hour|weekly|session) limit reached)\b/i.test(
      message,
    )
  ) {
    const reset = limitReset(message)
    return `Your ${account} plan’s ${label} usage limit is used up${reset ? ` until ${reset}` : ''}. Wait for it to reset, or ${PICK_FREEBUFF}.`
  }
  const unsupported =
    /The '([\w.:/-]{1,80})' model is not supported when using Codex with a ChatGPT account/.exec(
      message,
    )
  if (unsupported)
    return `Your ChatGPT plan can’t use ${unsupported[1]} in Codex. Choose another model from the model picker.`
  if (/spend cap set by the owner of your workspace/i.test(message))
    return `Your ${account} workspace’s spend cap for ${label} is used up. Ask a workspace owner to raise it, or ${PICK_FREEBUFF}.`
  if (/\bworkspace is out of credits\b/i.test(message))
    return `Your ${account} workspace is out of ${label} credits. Add credits there, or ${PICK_FREEBUFF}.`
  if (/^Quota exceeded\. Check your plan and billing details|credit balance is too low/i.test(message))
    return `The account ${label} signs in with on your computer is out of quota or credit. Check its plan and billing, or ${PICK_FREEBUFF}.`
  if (new RegExp(`^Could not install ${label}\\b`).test(message))
    return `Freebuff couldn’t install ${label} on your computer. Check that computer’s internet connection, then send your message again.`
  return null
}

export function createCloudHistoryClient(
  options: {
    baseUrl?: string
    getToken?: () => string | null | Promise<string | null>
    fetch?: (url: string, init?: RequestInit) => Promise<Response>
  } = {},
) {
  const base = (options.baseUrl ?? '/api/v1/cloud/history').replace(/\/$/, '')
  async function get<T>(
    path: string,
    page: CloudHistoryOptions = {},
    projectId?: string,
    update?: CloudHistoryChatUpdate | CloudHistoryProjectUpdate | CloudHistoryChatFork | CloudHistoryChatRollback | CloudHistoryAttentionAck,
    search?: string,
    method: 'GET' | 'PATCH' | 'POST' = update ? 'PATCH' : 'GET',
  ): Promise<T> {
    const query = new URLSearchParams()
    if (search) query.set('search', search)
    if (page.cursor != null) query.set('cursor', page.cursor)
    if (page.limit !== undefined) query.set('limit', String(page.limit))
    if (projectId !== undefined) query.set('projectId', projectId)
    const token = await options.getToken?.()
    const res = await (options.fetch ?? fetch)(
      `${base}/${path}${query.size ? `?${query}` : ''}`,
      {
        credentials: 'same-origin',
        cache: 'no-store',
        signal: page.signal,
        method,
        body: update ? JSON.stringify(update) : undefined,
        headers: update
          ? {
              'Content-Type': 'application/json',
              ...(token ? bearer(token) : {}),
            }
          : token
            ? bearer(token)
            : undefined,
      },
    )
    if (!res.ok) {
      const body = await res.json().catch(() => null)
      throw new CloudHistoryError(
        res.status,
        body?.error ?? 'Cloud history unavailable',
      )
    }
    return res.json() as Promise<T>
  }
  const id = encodeURIComponent
  return {
    listProjects: (page?: CloudHistoryOptions) =>
      get<CloudHistoryPage<CloudHistoryProject>>('projects', page),
    getProject: (projectId: string, signal?: AbortSignal) =>
      get<CloudHistoryProject>(`projects/${id(projectId)}`, { signal }),
    updateProject: (
      projectId: string,
      update: CloudHistoryProjectUpdate,
      signal?: AbortSignal,
    ) =>
      get<CloudHistoryProject>(
        `projects/${id(projectId)}`,
        { signal },
        undefined,
        update,
      ),
    listChats: (page: CloudHistoryOptions & { projectId?: string; search?: string } = {}) =>
      get<CloudHistoryPage<CloudHistoryChat>>('chats', page, page.projectId, undefined, page.search),
    getChat: (chatId: string, signal?: AbortSignal) =>
      get<CloudHistoryChat>(`chats/${id(chatId)}`, { signal }),
    forkChat: (chatId: string, fork: CloudHistoryChatFork, signal?: AbortSignal) =>
      get<CloudHistoryChat>(`chats/${id(chatId)}/fork`, { signal }, undefined, fork, undefined, 'POST'),
    openSideChat: (chatId: string, side: CloudHistorySideChat, signal?: AbortSignal) =>
      get<CloudHistoryChat>(`chats/${id(chatId)}/side`, { signal }, undefined, side, undefined, 'POST'),
    rollbackChat: (chatId: string, rollback: CloudHistoryChatRollback, signal?: AbortSignal) =>
      get<CloudHistoryChatRolledBack>(`chats/${id(chatId)}/rollback`, { signal }, undefined, rollback, undefined, 'POST'),
    getAttachment: (chatId: string, signal?: AbortSignal) =>
      get<CloudHistoryAttachment>(`chats/${id(chatId)}/attachment`, { signal }),
    updateChat: (
      chatId: string,
      update: CloudHistoryChatUpdate,
      signal?: AbortSignal,
    ) =>
      get<CloudHistoryChat>(
        `chats/${id(chatId)}`,
        { signal },
        undefined,
        update,
      ),
    acknowledgeAttention: (
      chatId: string,
      throughRevision: number,
      signal?: AbortSignal,
    ) =>
      get<CloudHistoryChat>(
        `chats/${id(chatId)}/attention`,
        { signal },
        undefined,
        { throughRevision },
      ),
    listRuns: (chatId: string, page?: CloudHistoryOptions) =>
      get<CloudHistoryPage<CloudHistoryRun>>(`chats/${id(chatId)}/runs`, page),
    getRun: (runId: string, signal?: AbortSignal) =>
      get<CloudHistoryRun>(`runs/${id(runId)}`, { signal }),
    listEvents: (runId: string, page?: CloudHistoryOptions) =>
      get<CloudHistoryEvents>(`runs/${id(runId)}/events`, page),
    listTraces: (runId: string, page?: CloudHistoryOptions) =>
      get<CloudHistoryEvents>(`runs/${id(runId)}/traces`, page),
    /** Whether this account may use Freebuff Projects (admins and an allowlist). */
    getProjectsAccess: (signal?: AbortSignal) =>
      get<{ allowed: boolean }>('projects-access', { signal }),
  }
}
