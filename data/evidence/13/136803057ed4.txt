function getStringProperty(
  properties: AnalyticsProperties,
  key: string,

  const value = properties?.[key]
  return typeof value === 'string' && value.trim() ? value : undefined
}

function getPropertyUserId(
  properties: AnalyticsProperties,
): string | undefined {
  const direct =
    getStringProperty(properties, 'userId') ??
    getStringProperty(properties, 'user_id') ??
    getStringProperty(properties, 'distinct_id')
  if (direct) {
    return direct

  const user = properties?.user

/**
 * Deterministic sampling for high-volume logs/events: hashes a stable key
 * (typically a userId) so the same key is always in or out of the sample and
 * per-key funnels stay coherent. Callers should record the rate on sampled
 * rows (e.g. `sampleRate: 0.02`) so counts can be re-inflated in queries.
 */
export function shouldSampleByKey(key: string, rate: number): boolean {
  if (rate >= 1) return true
  if (rate <= 0) return false
  return hashString(key) / 0xffffffff < rate
}

function hashString(input: string): number {
  let hash = 2166136261
  for (let i = 0; i < input.length; i++) {
    hash ^= input.charCodeAt(i)
    hash = Math.imul(hash, 16777619)
  }
  return hash >>> 0
}

function getSamplingKey(params: {
  event: AnalyticsEvent
  distinctId?: string
  properties?: AnalyticsProperties
}): string {
  return (
    params.distinctId ??
    getPropertyUserId(params.properties) ??
    getStringProperty(params.properties, 'clientSessionId') ??
    getStringProperty(params.properties, 'userInputId') ??
    params.event
  )
}

export function shouldTrackAnalyticsEvent(params: {
  event: AnalyticsEvent
  distinctId?: string
  properties?: AnalyticsProperties
}): boolean {
  if (isFullTelemetryEnabled(params)) {
    return true
  }

  const rate = getEventSampleRate(params.event, params.properties)
  if (rate >= 1) {
    return true
  }
  if (rate <= 0) {
    return false
  }

  const bucket =
    hashString(`${params.event}:${getSamplingKey(params)}`) / 0xffffffff
  return bucket < rate
}

function valueKind(value: unknown): string {
  if (Array.isArray(value)) {
    return 'array'
  }
  if (value === null) {
    return 'null'
  }
  return typeof value
}

export function summarizeAnalyticsValue(
  value: unknown,
): Record<string, unknown> {
  if (value === null || value === undefined) {
    return { kind: valueKind(value) }
  }

  if (typeof value === 'string') {
    return { kind: 'string', length: value.length }
  }

  if (Array.isArray(value)) {
    return { kind: 'array', length: value.length }
  }

  if (typeof value === 'object') {
    const keys = Object.keys(value as Record<string, unknown>)
    return {
      kind: 'object',
      keyCount: keys.length,
      keys: keys.slice(0, 25),
    }
  }

  return { kind: valueKind(value) }
}

/** Public, noncredential receipt syntax shared by delivery and Desktop. HMAC verification is server-only. */
export const DESKTOP_RECEIPT_PLATFORMS = [
  'mac-arm64',
  'mac-intel',
  'windows',
  'windows-arm64',
  'windows-baseline',
  'linux',
  'linux-arm64',
  'linux-appimage',
  'linux-arm64-appimage',
] as const
export type DesktopReceiptPlatform = (typeof DESKTOP_RECEIPT_PLATFORMS)[number]
export const DESKTOP_RECEIPT_MAX_AGE_SECONDS = 30 * 24 * 60 * 60
const UUID =
  '[0-9a-f]{8}-[0-9a-f]{4}-[1-8][0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}'
const TOKEN = new RegExp(
  `^v1\\.(${UUID})\\.([0-9a-z]{6,10})\\.(${DESKTOP_RECEIPT_PLATFORMS.join('|')})\\.([A-Za-z0-9_-]{43})$`,
)

export function parseDesktopAcquisitionReceipt(value: unknown) {
  if (typeof value !== 'string' || value.length > 180) return null
  const match = TOKEN.exec(value)
  if (!match || match[0] !== value) return null
  const issued_at = Number.parseInt(match[2]!, 36)
  if (
    !Number.isSafeInteger(issued_at) ||
    issued_at <= 0 ||
    issued_at.toString(36) !== match[2]
  )
    return null
  return {
    acquisition_id: match[1]!,
    issued_at,
    platform: match[3]! as DesktopReceiptPlatform,
    token: value,
  }
}

export function extractDesktopAcquisitionReceipt(filename: unknown) {
  if (typeof filename !== 'string') return null
  const name = filename.split(/[\\/]/).pop() ?? ''
  const marker = name.lastIndexOf('--fb_')
  if (marker < 0) return null
  const token = name
    .slice(marker + 5)
    .replace(/(?: \([0-9]+\))?\.(?:exe|app|zip|AppImage|deb)$/, '')
  return parseDesktopAcquisitionReceipt(token)
}

import { gzipSync, gunzipSync } from 'node:zlib'

import { parseDesktopAcquisitionReceipt } from './desktop-acquisition-receipt'

const AR_MAGIC = Buffer.from('!<arch>\n')
const CONTROL_LIMIT = 2 * 1024 * 1024
const CONTROL_TAR_LIMIT = 8 * 1024 * 1024
export const DESKTOP_DEB_RECEIPT_MARKER =
  '# FREEBUFF_INSTALL_RECEIPT_PLACEHOLDER'

/** Reads only as much of a release stream as the current archive member needs. */
class ArchiveReader {
  private readonly iterator: AsyncIterator<Uint8Array>
  private chunk: Uint8Array = new Uint8Array(0)
  private offset = 0

  constructor(source: AsyncIterable<Uint8Array>) {
    this.iterator = source[Symbol.asyncIterator]()
  }

  async read(size: number): Promise<Buffer> {
    const output = Buffer.alloc(size)
    let written = 0
    while (written < size) {
      if (this.offset === this.chunk.length) {
        const next = await this.iterator.next()
        if (next.done) throw new Error('Truncated Debian installer')
        this.chunk = next.value
        this.offset = 0
        if (!this.chunk.length) continue
      }
      const count = Math.min(size - written, this.chunk.length - this.offset)
      output.set(this.chunk.subarray(this.offset, this.offset + count), written)
      this.offset += count
      written += count
    }
    return output
  }

  async *rest(): AsyncGenerator<Uint8Array> {
    if (this.offset < this.chunk.length) yield this.chunk.subarray(this.offset)
    for (;;) {
      const next = await this.iterator.next()
      if (next.done) return
      if (next.value.length) yield next.value
    }
  }

  async cancel(): Promise<void> {
    await this.iterator.return?.()
  }
}

function arMember(header: Buffer): { name: string; size: number } {
  if (header.length !== 60 || header.toString('ascii', 58) !== '`\n')
    throw new Error('Invalid Debian archive member')
  const name = header.toString('ascii', 0, 16).trim().replace(/\/$/, '')
  const sizeText = header.toString('ascii', 48, 58).trim()
  if (!/^\d+$/.test(sizeText)) throw new Error('Invalid Debian archive size')
  const size = Number(sizeText)
  if (!Number.isSafeInteger(size))
    throw new Error('Invalid Debian archive size')
  return { name, size }
}

function tarName(header: Buffer): string {
  const name = header.toString('utf8', 0, 100).replace(/\0.*$/s, '')
  const prefix = header.toString('utf8', 345, 500).replace(/\0.*$/s, '')
  return (prefix ? `${prefix}/${name}` : name).replace(/^\.\//, '')
}

function tarSize(header: Buffer): number {
  const sizeText = header.toString('ascii', 124, 136).replace(/\0/g, '').trim()
  if (!/^[0-7]+$/.test(sizeText))
    throw new Error('Unsupported Debian control tar size')
  const size = Number.parseInt(sizeText, 8)
  if (!Number.isSafeInteger(size) || size > CONTROL_TAR_LIMIT)
    throw new Error('Debian control tar member is too large')
  return size
}

function resizedTarHeader(original: Buffer, size: number): Buffer {
  const header = Buffer.from(original)
  header.fill(0, 124, 136)
  header.write(size.toString(8).padStart(11, '0'), 124, 11, 'ascii')
  header.fill(32, 148, 156)
  const checksum = header.reduce((total, byte) => total + byte, 0)
  header.write(`${checksum.toString(8).padStart(6, '0')}\0 `, 148, 8, 'ascii')
  return header
}

function receiptInstallScript(receipt: string): string {
  // The caller verified the signature; the shared parser also limits this literal to shell-safe
  // ASCII. This file is outside the application and is not replaced by ordinary auto-updates.
  return `# Persist only the original acquisition, never replace it during an upgrade.\nif [ ! -e /var/lib/freebuff/install-receipt.txt ] && install -d -m 0755 /var/lib/freebuff; then\n    ( set -C; printf '%s\\n' '${receipt}' > /var/lib/freebuff/install-receipt.txt ) 2>/dev/null || true\n    chmod 0644 /var/lib/freebuff/install-receipt.txt 2>/dev/null || true\nfi`
}

function personalizeControl(control: Buffer, receipt: string): Buffer {
  const archive = gunzipSync(control, { maxOutputLength: CONTROL_TAR_LIMIT })
  if (archive.length > CONTROL_TAR_LIMIT)
    throw new Error('Debian control tar is too large')
  const parts: Buffer[] = []
  let offset = 0
  let rewritten = false
  while (offset + 512 <= archive.length) {
    const header = archive.subarray(offset, offset + 512)
    if (header.every((byte) => byte === 0)) {
      parts.push(archive.subarray(offset))
      offset = archive.length
      break
    }
    const size = tarSize(header)
    const end = offset + 512 + Math.ceil(size / 512) * 512
    if (end > archive.length) throw new Error('Truncated Debian control tar')
    if (tarName(header) === 'postinst') {
      if (rewritten || (header[156] !== 0 && header[156] !== 48))
        throw new Error('Invalid Debian post-install hook')
      const original = archive
        .subarray(offset + 512, offset + 512 + size)
        .toString('utf8')
      if (original.split(DESKTOP_DEB_RECEIPT_MARKER).length !== 2)
        throw new Error(
          'Debian installer does not support acquisition receipts',
        )
      const updated = Buffer.from(
        original.replace(
          DESKTOP_DEB_RECEIPT_MARKER,
          receiptInstallScript(receipt),
        ),
      )
      parts.push(resizedTarHeader(header, updated.length), updated)
      if (updated.length % 512)
        parts.push(Buffer.alloc(512 - (updated.length % 512)))
      rewritten = true
    } else {
      parts.push(archive.subarray(offset, end))
    }
    offset = end
  }
  if (!rewritten || offset !== archive.length)
    throw new Error('Invalid Debian control tar')
  return gzipSync(Buffer.concat(parts))
}

/**
 * Personalize a website download without buffering its application payload. Only the small
 * control.tar.gz install hook changes; the data archive and every byte after it pass through.
 * The returned stream is ready only after the receipt hook has been checked, so unsupported
 * releases can fail before the HTTP response starts. The caller must verify the receipt HMAC.
 */
export async function prepareDesktopDebReceipt(
  source: AsyncIterable<Uint8Array>,
  receipt: string,
): Promise<AsyncIterable<Uint8Array>> {
  const parsed = parseDesktopAcquisitionReceipt(receipt)
  if (
    !parsed ||
    (parsed.platform !== 'linux' && parsed.platform !== 'linux-arm64')
  )
    throw new Error('Invalid Debian acquisition receipt')
  const reader = new ArchiveReader(source)
  try {
    if (!(await reader.read(8)).equals(AR_MAGIC))
      throw new Error('Invalid Debian installer')
    const binaryHeader = await reader.read(60)
    const binary = arMember(binaryHeader)
    if (binary.name !== 'debian-binary' || binary.size !== 4)
      throw new Error('Unsupported Debian installer')
    const binaryBody = await reader.read(4)
    if (binaryBody.toString('ascii') !== '2.0\n')
      throw new Error('Unsupported Debian version')
    const controlHeader = await reader.read(60)
    const control = arMember(controlHeader)
    if (control.name !== 'control.tar.gz' || control.size > CONTROL_LIMIT)
      throw new Error('Unsupported Debian control archive')
    const updated = personalizeControl(await reader.read(control.size), receipt)
    if (control.size % 2) await reader.read(1)
    const header = Buffer.from(controlHeader)
    header.write(String(updated.length).padEnd(10, ' '), 48, 10, 'ascii')
    return (async function* () {
      try {
        yield AR_MAGIC
        yield binaryHeader
        yield binaryBody
        yield header
        yield updated
        if (updated.length % 2) yield Buffer.from('\n')
        yield* reader.rest()
      } finally {
        await reader.cancel()
      }
    })()
  } catch (error) {
    await reader.cancel()
    throw error
  }
}

        data.readUInt32LE(at + 4) !== entry.compressed ||
        data.readUInt32LE(at + 8) !== entry.uncompressed

    header.writeUInt16LE(entry.name.length, 26)

  const footer = Buffer.from(data.subarray(end))
  footer.writeUInt32LE(outputSize - directoryOffset, 12)
  footer.writeUInt32LE(directoryOffset, 16)
  return Buffer.concat([...chunks, footer])

+export function parseDesktopAcquisitionReceipt(value: unknown) {
+  if (typeof value !== 'string' || value.length > 180) return null
+  const match = TOKEN.exec(value)
+  if (!match || match[0] !== value) return null
+  const issued_at = Number.parseInt(match[2]!, 36)
+  if (
+    !Number.isSafeInteger(issued_at) ||
+    issued_at <= 0 ||
+    issued_at.toString(36) !== match[2]
+  )
+    return null
+  return {
+    acquisition_id: match[1]!,
+    issued_at,
+    platform: match[3]! as DesktopReceiptPlatform,
+    token: value,
+  }

+import { parseDesktopAcquisitionReceipt } from './desktop-acquisition-receipt'

+const AR_MAGIC = Buffer.from('!<arch>\n')
+const CONTROL_LIMIT = 2 * 1024 * 1024
+const CONTROL_TAR_LIMIT = 8 * 1024 * 1024
+export const DESKTOP_DEB_RECEIPT_MARKER =
+  '# FREEBUFF_INSTALL_RECEIPT_PLACEHOLDER'
+
+/** Reads only as much of a release stream as the current archive member needs. */
+class ArchiveReader {
+  private readonly iterator: AsyncIterator<Uint8Array>
+  private chunk: Uint8Array = new Uint8Array(0)
+  private offset = 0
+
+  constructor(source: AsyncIterable<Uint8Array>) {
+    this.iterator = source[Symbol.asyncIterator]()
+  }
+
+  async read(size: number): Promise<Buffer> {
+    const output = Buffer.alloc(size)
+    let written = 0
+    while (written < size) {
+      if (this.offset === this.chunk.length) {
+        const next = await this.iterator.next()
+        if (next.done) throw new Error('Truncated Debian installer')
+        this.chunk = next.value
+        this.offset = 0
+        if (!this.chunk.length) continue
+      }
+      const count = Math.min(size - written, this.chunk.length - this.offset)
+      output.set(this.chunk.subarray(this.offset, this.offset + count), written)
+      this.offset += count
+      written += count
+    }
+    return output
+  }
+
+  async *rest(): AsyncGenerator<Uint8Array> {
+    if (this.offset < this.chunk.length) yield this.chunk.subarray(this.offset)
+    for (;;) {
+      const next = await this.iterator.next()
+      if (next.done) return
+      if (next.value.length) yield next.value
+    }
+  }
+
+  async cancel(): Promise<void> {
+    await this.iterator.return?.()
+  }
+}
+
+function arMember(header: Buffer): { name: string; size: number } {
+  if (header.length !== 60 || header.toString('ascii', 58) !== '`\n')
+    throw new Error('Invalid Debian archive member')
+  const name = header.toString('ascii', 0, 16).trim().replace(/\/$/, '')
+  const sizeText = header.toString('ascii', 48, 58).trim()
+  if (!/^\d+$/.test(sizeText)) throw new Error('Invalid Debian archive size')
+  const size = Number(sizeText)
+  if (!Number.isSafeInteger(size))
+    throw new Error('Invalid Debian archive size')
+  return { name, size }
+}
+
+function tarName(header: Buffer): string {
+  const name = header.toString('utf8', 0, 100).replace(/\0.*$/s, '')
+  const prefix = header.toString('utf8', 345, 500).replace(/\0.*$/s, '')
+  return (prefix ? `${prefix}/${name}` : name).replace(/^\.\//, '')
+}
+
+function tarSize(header: Buffer): number {
+  const sizeText = header.toString('ascii', 124, 136).replace(/\0/g, '').trim()
+  if (!/^[0-7]+$/.test(sizeText))
+    throw new Error('Unsupported Debian control tar size')
+  const size = Number.parseInt(sizeText, 8)
+  if (!Number.isSafeInteger(size) || size > CONTROL_TAR_LIMIT)
+    throw new Error('Debian control tar member is too large')
+  return size
+}
+
+function resizedTarHeader(original: Buffer, size: number): Buffer {
+  const header = Buffer.from(original)
+  header.fill(0, 124, 136)
+  header.write(size.toString(8).padStart(11, '0'), 124, 11, 'ascii')
+  header.fill(32, 148, 156)
+  const checksum = header.reduce((total, byte) => total + byte, 0)
+  header.write(`${checksum.toString(8).padStart(6, '0')}\0 `, 148, 8, 'ascii')
+  return header
+}

+function receiptInstallScript(receipt: string): string {
+  // The caller verified the signature; the shared parser also limits this literal to shell-safe
+  // ASCII. This file is outside the application and is not replaced by ordinary auto-updates.
+  return `# Persist only the original acquisition, never replace it during an upgrade.\nif [ ! -e /var/lib/freebuff/install-receipt.txt ] && install -d -m 0755 /var/lib/freebuff; then\n    ( set -C; printf '%s\\n' '${receipt}' > /var/lib/freebuff/install-receipt.txt ) 2>/dev/null || true\n    chmod 0644 /var/lib/freebuff/install-receipt.txt 2>/dev/null || true\nfi`
+}

+function personalizeControl(control: Buffer, receipt: string): Buffer {
+  const archive = gunzipSync(control, { maxOutputLength: CONTROL_TAR_LIMIT })
+  if (archive.length > CONTROL_TAR_LIMIT)
+    throw new Error('Debian control tar is too large')
+  const parts: Buffer[] = []
+  let offset = 0
+  let rewritten = false
+  while (offset + 512 <= archive.length) {
+    const header = archive.subarray(offset, offset + 512)
+    if (header.every((byte) => byte === 0)) {
+      parts.push(archive.subarray(offset))
+      offset = archive.length
+      break
+    }
+    const size = tarSize(header)
+    const end = offset + 512 + Math.ceil(size / 512) * 512
+    if (end > archive.length) throw new Error('Truncated Debian control tar')
+    if (tarName(header) === 'postinst') {
+      if (rewritten || (header[156] !== 0 && header[156] !== 48))
+        throw new Error('Invalid Debian post-install hook')
+      const original = archive
+        .subarray(offset + 512, offset + 512 + size)
+        .toString('utf8')
+      if (original.split(DESKTOP_DEB_RECEIPT_MARKER).length !== 2)
+        throw new Error(
+          'Debian installer does not support acquisition receipts',
+        )
+      const updated = Buffer.from(
+        original.replace(
+          DESKTOP_DEB_RECEIPT_MARKER,
+          receiptInstallScript(receipt),
+        ),
+      )
+      parts.push(resizedTarHeader(header, updated.length), updated)
+      if (updated.length % 512)
+        parts.push(Buffer.alloc(512 - (updated.length % 512)))
+      rewritten = true
+    } else {
+      parts.push(archive.subarray(offset, end))
+    }
+    offset = end
+  }
+  if (!rewritten || offset !== archive.length)
+    throw new Error('Invalid Debian control tar')
+  return gzipSync(Buffer.concat(parts))
+}

+/**
+ * Personalize a website download without buffering its application payload. Only the small
+ * control.tar.gz install hook changes; the data archive and every byte after it pass through.
+ * The returned stream is ready only after the receipt hook has been checked, so unsupported
+ * releases can fail before the HTTP response starts. The caller must verify the receipt HMAC.
+ */
+export async function prepareDesktopDebReceipt(
+  source: AsyncIterable<Uint8Array>,
+  receipt: string,
+): Promise<AsyncIterable<Uint8Array>> {
+  const parsed = parseDesktopAcquisitionReceipt(receipt)
+  if (
+    !parsed ||
+    (parsed.platform !== 'linux' && parsed.platform !== 'linux-arm64')
+  )
+    throw new Error('Invalid Debian acquisition receipt')
+  const reader = new ArchiveReader(source)
+  try {
+    if (!(await reader.read(8)).equals(AR_MAGIC))
+      throw new Error('Invalid Debian installer')
+    const binaryHeader = await reader.read(60)
+    const binary = arMember(binaryHeader)
+    if (binary.name !== 'debian-binary' || binary.size !== 4)
+      throw new Error('Unsupported Debian installer')
+    const binaryBody = await reader.read(4)
+    if (binaryBody.toString('ascii') !== '2.0\n')
+      throw new Error('Unsupported Debian version')
+    const controlHeader = await reader.read(60)
+    const control = arMember(controlHeader)
+    if (control.name !== 'control.tar.gz' || control.size > CONTROL_LIMIT)
+      throw new Error('Unsupported Debian control archive')
+    const updated = personalizeControl(await reader.read(control.size), receipt)
+    if (control.size % 2) await reader.read(1)
+    const header = Buffer.from(controlHeader)
+    header.write(String(updated.length).padEnd(10, ' '), 48, 10, 'ascii')
+    return (async function* () {
+      try {
+        yield AR_MAGIC
+        yield binaryHeader
+        yield binaryBody
+        yield header
+        yield updated
+        if (updated.length % 2) yield Buffer.from('\n')
+        yield* reader.rest()
+      } finally {
+        await reader.cancel()
+      }
+    })()
+  } catch (error) {
+    await reader.cancel()
+    throw error
+  }
+}

+      data.readUInt16LE(entry.offset + 4) >= 45 ||
+      data.readUInt16LE(entry.offset + 6) !== entry.flags ||
+      data.readUInt16LE(entry.offset + 8) !== entry.method
+    )
+      throw Error('Invalid local ZIP entry')
+    const n = data.readUInt16LE(entry.offset + 26),
+      x = data.readUInt16LE(entry.offset + 28)
+    if (
+      entry.offset + 30 + n + x > next ||
+      !data
+        .subarray(entry.offset + 30, entry.offset + 30 + n)
+        .equals(entry.oldName)
+    )
+      throw Error('Inconsistent ZIP entry')
+    extraFields(data.subarray(entry.offset + 30 + n, entry.offset + 30 + n + x))
+    const localCrc = data.readUInt32LE(entry.offset + 14),
+      localCompressed = data.readUInt32LE(entry.offset + 18),
+      localUncompressed = data.readUInt32LE(entry.offset + 22)
+    const descriptor = Boolean(entry.flags & 8)
+    if (
+      (localCrc !== entry.crc && !(descriptor && localCrc === 0)) ||
+      (localCompressed !== entry.compressed &&
+        !(descriptor && localCompressed === 0)) ||
+      (localUncompressed !== entry.uncompressed &&
+        !(descriptor && localUncompressed === 0))
+    )
+      throw Error('Inconsistent ZIP sizes')
+    const payloadEnd = entry.offset + 30 + n + x + entry.compressed
+    const descriptorSize = next - payloadEnd
+    if (!descriptor && descriptorSize !== 0)
+      throw Error('Inconsistent ZIP payload')
+    if (descriptor) {
+      if (
+        ![12, 16].includes(descriptorSize) ||
+        payloadEnd < entry.offset ||
+        payloadEnd + descriptorSize > next
+      )
+        throw Error('Invalid ZIP descriptor')
+      const at = descriptorSize === 16 ? payloadEnd + 4 : payloadEnd
+      if (
+        (descriptorSize === 16 &&
+          data.readUInt32LE(payloadEnd) !== 0x08074b50) ||
+        data.readUInt32LE(at) !== entry.crc ||
+        data.readUInt32LE(at + 4) !== entry.compressed ||
+        data.readUInt32LE(at + 8) !== entry.uncompressed
+      )
+        throw Error('Inconsistent ZIP descriptor')
+    }
+    const header = Buffer.from(data.subarray(entry.offset, entry.offset + 30))
+    header.writeUInt16LE(entry.name.length, 26)
+    offsets.set(entry.offset, outputSize)
+    const rest = data.subarray(entry.offset + 30 + n, next)
+    chunks.push(header, entry.name, rest)
+    outputSize += header.length + entry.name.length + rest.length
+  }
+  const directoryOffset = outputSize
+  for (const entry of entries) {
+    const n = entry.central.readUInt16LE(28),
+      header = Buffer.from(entry.central.subarray(0, 46)),
+      rest = entry.central.subarray(46 + n)
+    header.writeUInt16LE(entry.name.length, 28)
+    header.writeUInt32LE(offsets.get(entry.offset)!, 42)
+    chunks.push(header, entry.name, rest)
+    outputSize += header.length + entry.name.length + rest.length
+  }
+  const footer = Buffer.from(data.subarray(end))
+  footer.writeUInt32LE(outputSize - directoryOffset, 12)
+  footer.writeUInt32LE(directoryOffset, 16)
+  return Buffer.concat([...chunks, footer])
+}
