import type { CodingActivationSurface } from './util/coding-activation-surface'
import type { MetaAdsEventName } from './util/meta-ads'

import {
  isMetaConversionValue,
  type MetaConversionEventName,
  type MetaConversionValue,
} from './util/meta-conversions'

// Verified against Meta's official facebook-nodejs-business-sdk/src/api.js.
export const META_GRAPH_API_VERSION = 'v26.0'

export type MetaConversionAttribution = {
  fbc?: string
  fbp?: string
  /** Unsalted SHA-256 of the trimmed, lowercased account email (Meta's `em`). */
  hashedEmail?: string
  /** Client address resolved at enrollment, sent as `client_ip_address`. */
  ipAddress?: string
  /** Enrollment browser agent, sent as `client_user_agent` on every event. */
  userAgent: string
}

export function metaExternalId(userId: string): string {
  return createHash('sha256').update(userId).digest('hex')
}

export function metaConversionId(
  eventName: MetaConversionEventName | MetaAdsEventName,
  userId: string,
): string {
  return createHash('sha256')
    .update(`meta-capi:${eventName}:${userId}`)
    .digest('hex')
}

/** A closed, machine-made label, never free text from a request. */
function isMetaContentName(value: unknown): value is string {
  return typeof value === 'string' && /^[a-z0-9_]{1,64}$/.test(value)
}

export type SendMetaConversionParams = {
  pixelId: string
  accessToken: string
  eventName: MetaConversionEventName | MetaAdsEventName
  eventId: string
  eventAt: Date
  userId: string
  attribution: MetaConversionAttribution
  surface: CodingActivationSurface
  testEventCode?: string
  eventSourceUrl?: 'https://freebuff.com/advertisers'
  conversionValue?: MetaConversionValue
  /** Required on `Purchase`: which value milestone this is. */
  contentName?: string
  fetchImpl?: typeof fetch
  sleepImpl?: (ms: number) => Promise<void>
  /** Rechecked immediately before every network attempt, including retries. */
  canSend?: () => Promise<boolean>
}

export function buildMetaConversionBody(params: SendMetaConversionParams) {
  if (
    params.eventName === 'Subscribe' &&
    !isMetaConversionValue(params.conversionValue)
  ) {
    throw new Error('Subscribe requires a positive confirmed USD payment')
  }
  if (
    params.eventName === 'Purchase' &&
    (!isMetaConversionValue(params.conversionValue) ||
      !isMetaContentName(params.contentName))
  ) {
    throw new Error('Purchase requires a positive USD value and a content name')
  }
  const { attribution } = params
  return {
    ...(params.testEventCode ? { test_event_code: params.testEventCode } : {}),
    data: [
      {
        event_name: params.eventName,
        event_id: params.eventId,
        event_time: Math.floor(params.eventAt.getTime() / 1000),
        // A desktop/CLI/phone-app activation is not a website conversion or a
        //[freebuff product prompt text omitted]
        action_source: params.surface === 'web' ? 'website' : 'other',
        ...(params.surface === 'web'
          ? {
              event_source_url:
                params.eventSourceUrl ?? 'https://freebuff.com/',
            }
          : {}),
        // Every key below identifies the PERSON, not the event's own network
        // hop, so the enrollment browser's agent and address ride on native
        // events too. Match quality is what decides whether Meta can tie a
        // conversion back to the ad; measured 4.7/10 with fbp+external_id only.
        user_data: {
          external_id: [metaExternalId(params.userId)],
          ...(attribution.hashedEmail ? { em: [attribution.hashedEmail] } : {}),
          ...(attribution.fbc ? { fbc: attribution.fbc } : {}),
          ...(attribution.fbp ? { fbp: attribution.fbp } : {}),
          ...(attribution.ipAddress
            ? { client_ip_address: attribution.ipAddress }
            : {}),
          client_user_agent: attribution.userAgent,
        },
        custom_data: {
          surface: params.surface,
          ...(params.eventName === 'Subscribe' ? params.conversionValue : {}),
          ...(params.eventName === 'Purchase' && params.conversionValue
            ? {
                value: params.conversionValue.value,
                currency: params.conversionValue.currency,
                content_name: params.contentName,
              }
            : {}),
        },
      },
    ],
  }
}

/** Two bounded attempts; callers retain the claim for later retries. No PII/token logs. */
export async function sendMetaConversion(
  params: SendMetaConversionParams,
): Promise<'sent' | 'suppressed'> {
  if (!/^\d+$/.test(params.pixelId)) throw new Error('Invalid Meta dataset ID')
  const fetchImpl = params.fetchImpl ?? fetch
  const sleep =
    params.sleepImpl ??
    ((ms) => new Promise((resolve) => setTimeout(resolve, ms)))
  const body = JSON.stringify(buildMetaConversionBody(params))
  for (let attempt = 0; attempt < 2; attempt++) {
    if (params.canSend && !(await params.canSend())) return 'suppressed'
    let retry = false
    try {
      const response = await fetchImpl(
        `https://graph.facebook.com/${META_GRAPH_API_VERSION}/${params.pixelId}/events`,
        {
          method: 'POST',
          headers: {
            Authorization: `Bearer ${params.accessToken}`,
            'Content-Type': 'application/json',
          },
          body,
          signal: AbortSignal.timeout(3_000),
        },
      )
      if (response.ok) {
        const result: unknown = await response.json()
        if (
          result &&
          typeof result === 'object' &&
          'events_received' in result &&
          result.events_received === 1
        )
          return 'sent'
        throw new Error('Meta did not acknowledge the conversion')
      }
      retry =
        response.status === 408 ||
        response.status === 429 ||
        response.status >= 500
      if (!retry || attempt === 1)
        throw new MetaConversionRejectedError(response.status)
    } catch (error) {
      if (error instanceof MetaConversionRejectedError || attempt === 1) {
        // Do not propagate provider bodies or network errors that may echo secrets.
        throw new Error(
          error instanceof MetaConversionRejectedError
            ? `Meta conversion rejected (${error.status})`
            : 'Meta conversion delivery failed',
        )
      }
      retry = true
    }
    if (retry) await sleep(100)
  }
  throw new Error('Meta conversion delivery failed')
}

class MetaConversionRejectedError extends Error {
  constructor(readonly status: number) {
    super('Meta conversion rejected')
  }
}

/** Public, bounded attribution shared by the browser, installer bridge and CAPI. */
export const REDDIT_ATTRIBUTION_WINDOW_MS = 30 * 24 * 60 * 60 * 1000
export const REDDIT_ACTIVATION_WINDOW_MS = 7 * 24 * 60 * 60 * 1000
export type RedditAcquisitionEvent = 'SignUp' | 'CodingActivation'
export type { CodingActivationSurface as RedditAcquisitionSurface } from './coding-activation-surface'
export type RedditClickAttribution = {
  clickId?: string
  uuid?: string
  capturedAt: number
}

export function normalizeRedditAttribution(
  value: unknown,
  now = Date.now(),
): RedditClickAttribution | undefined {
  if (!value || typeof value !== 'object') return undefined
  const row = value as Record<string, unknown>
  if (
    typeof row.capturedAt !== 'number' ||
    !Number.isFinite(row.capturedAt) ||
    row.capturedAt > now ||
    now - row.capturedAt >= REDDIT_ATTRIBUTION_WINDOW_MS
  )
    return undefined
  const bounded = (v: unknown, max: number) =>
    typeof v === 'string' &&
    v.length > 0 &&
    v.length <= max &&
    /^[A-Za-z0-9_.:-]+$/.test(v)
      ? v
      : undefined
  const clickId = bounded(row.clickId, 512)
  const uuid = bounded(row.uuid, 256)
  return clickId || uuid
    ? { clickId, uuid, capturedAt: row.capturedAt }
    : undefined
}

export function redditAttributionFromCookies(
  read: (name: string) => string | undefined,
  now = Date.now(),
) {
  return normalizeRedditAttribution(
    {
      clickId: read('freebuff_rdt_cid') ?? read('_rdt_cid'),
      uuid: read('_rdt_uuid'),
      capturedAt: Number(read('freebuff_rdt_captured_at')) || now,
    },
    now,
  )
}

export function withinRedditActivationWindow(signupAt: Date, now: Date) {
  const age = now.getTime() - signupAt.getTime()
  return age >= 0 && age < REDDIT_ACTIVATION_WINDOW_MS
}

/** Current-browser withdrawal always wins over an installer's saved context. */
export function redditSignupContext(input: {
  optedOut: boolean
  readCookie(name: string): string | undefined
  download?: RedditClickAttribution
  now?: number
}): { attribution?: RedditClickAttribution } | null {
  if (input.optedOut || input.readCookie('freebuff_meta_ads_funnel') === '1')
    return null
  const now = input.now ?? Date.now()
  const browser = redditAttributionFromCookies(input.readCookie, now)
  const download = normalizeRedditAttribution(input.download, now)
  // A new login browser has its own UUID even though it never saw the ad.
  // That UUID must not hide the installer's click. A fresh browser click wins
  // outright; otherwise retain the installer click with its original age and
  // use the actual signup browser's UUID when available.
  if (browser?.clickId || !download?.clickId)
    return { attribution: browser ?? download }
  return {
    attribution: {
      clickId: download.clickId,
      uuid: browser?.uuid ?? download.uuid,
      capturedAt: download.capturedAt,
    },
  }
}

/** Where a first coding prompt ran, as the acquisition trackers (Meta, Google
 * Ads, Reddit) record and report it. 'android' and 'ios' are the phone apps,

export const CODING_ACTIVATION_SURFACES = [
  'web',
  'desktop',
  'cli',
  'android',
  'ios',
] as const

export type CodingActivationSurface =
  (typeof CODING_ACTIVATION_SURFACES)[number]

export function isCodingActivationSurface(
  value: unknown,
): value is CodingActivationSurface {
  return (CODING_ACTIVATION_SURFACES as readonly unknown[]).includes(value)
}
