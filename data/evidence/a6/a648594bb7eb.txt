 *    Cloud refuses new runs for it, so a chat never runs in two places.

 *    and lifts the fence.
 */

/** Export response headers. The body is `application/gzip`. */
export const CLOUD_EXPORT_HEADERS = {
  /** Newest run of the chat when the export was taken, or empty for none. */
  runId: 'x-freebuff-export-run',
  /** Lowercase hex SHA-256 of the whole response body. */
  sha256: 'x-freebuff-export-sha256',
  /** Byte length of the body. */
  bytes: 'x-freebuff-export-bytes',
} as const

    /** The export's run receipt; a newer run means the export is stale. */

  })
  .strict()

  handedBackAt: z.iso.datetime({ offset: true }),
})
export type HandbackCompleteRequest = z.infer<typeof handbackCompleteSchema>

  workspaceId: string
  /** ISO time the chat left Cloud. Repeating the same handback returns the original time. */

}

/** The 409 a run request gets for a chat that continues on a computer. */
export const HANDED_BACK_RUN_ERROR =
  'This chat continues on a computer. Open it there, or choose Continue in Cloud on that computer to bring it back.'

/** The 409 a handback gets when Cloud moved on after the export; export again. */
export const STALE_EXPORT_ERROR =

/**
 * Export v2 (`GET …/export?v=2`). The tar is the same working-tree copy, plus two entries under
 * this directory, which a worktree never exports itself:
 *
 * - `meta.json` (`CloudExportMeta`): the commit the chat's worktree started from, its HEAD, and the
 *   files tracked at HEAD that the working tree no longer has.
 * - `commits.bundle`: `git bundle` of HEAD excluding `base` (absent when HEAD is the base; with
 *   `history=full`, HEAD's whole history, for a Cloud project's first landing on a computer), under
 *   the ref `CLOUD_EXPORT_BUNDLE_REF`. Its prerequisites are reachable from `base`, which the
 *   computer that uploaded the chat has. Without a known base it holds HEAD's whole history.
 *
 * With them, Continue locally lands Cloud's commits as commits and merges Cloud's work with the
 * computer's own, instead of flattening it over the current branch tip. An older server ignores
 * `v` and sends v1 (no `meta.json`); Desktop then lands the files as before.
 */
export const CLOUD_EXPORT_V2_DIR = '.freebuff-export'
/** The ref name inside `commits.bundle`. */
export const CLOUD_EXPORT_BUNDLE_REF = 'HEAD'

const commitSchema = z.string().regex(/^[0-9a-f]{40}(?:[0-9a-f]{24})?$/)

const cloudExportV2Schema = z
  .object({
    version: z.literal(2),
    /** The commit the worktree started from (its marker), or null for a chat initialized before it was recorded. */
    base: commitSchema.nullable(),
    /** The worktree's HEAD, or null for an empty repository. */
    head: commitSchema.nullable(),
    /** The worktree's branch. */
    branch: z.string().max(400).nullable(),
    /** Whether `commits.bundle` is in the tar. */
    bundle: z.boolean(),
    /** Tracked at HEAD, absent from the working tree. */
    deleted: z.array(z.string().max(4096)).max(200_000),
  })
  .strict()
/** V3 sends Git objects once plus only working files whose bytes differ from HEAD.
 * `have` is a complete commit graph verified by the receiver; unknown bases send a full pack.
 * `files` is the complete eligible working-file inventory, for plain-folder reconstruction.
 * `shallow` preserves any history boundary already present in Cloud (never invents one).
 */
export const cloudExportV3Schema = cloudExportV2Schema.extend({
  version: z.literal(3),
  bundle: z.literal(false),
  pack: z.boolean(),
  have: commitSchema.nullable(),
  shallow: z.array(commitSchema).max(200_000),
  files: z.array(z.string().max(4096)).max(200_000),
}).strict()
export const cloudExportMetaSchema = z.union([cloudExportV2Schema, cloudExportV3Schema])
export type CloudExportMeta = z.infer<typeof cloudExportMetaSchema>

/** Archived chats resume through Desktop's existing local landing and Cloud upload. */
export const ARCHIVED_CHAT_RESTORE_ERROR =
  'This chat’s Cloud workspace was archived. Restore it in Freebuff Desktop to continue.'

  } = {},

  async function get<T>(
    path: string,
    page: CloudHistoryOptions = {},
    projectId?: string,
    update?: CloudHistoryChatUpdate | CloudHistoryProjectUpdate,
  ): Promise<T> {
    const query = new URLSearchParams()
    if (page.cursor != null) query.set('cursor', page.cursor)
    if (page.limit !== undefined) query.set('limit', String(page.limit))
    if (projectId !== undefined) query.set('projectId', projectId)

      `${base}/${path}${query.size ? `?${query}` : ''}`,
      {
        credentials: 'same-origin',
        cache: 'no-store',
        signal: page.signal,
        method: update ? 'PATCH' : 'GET',
        body: update ? JSON.stringify(update) : undefined,
        headers: update
          ? {
              'Content-Type': 'application/json',
              ...(token ? { Authorization: `Bearer ${token}` } : {}),
            }
          : token
            ? { Authorization: `Bearer ${token}` }
            : undefined,
      },
    )
    if (!res.ok) {
      const body = await res.json().catch(() => null)
      throw new CloudHistoryError(
        res.status,
        body?.error ?? 'Cloud history unavailable',
      )
    }
    return res.json() as Promise<T>
  }
  const id = encodeURIComponent
  return {

export type SettingsSearchPage = { id: string; label: PageLabel }

function normalize(value: string) {
  return value
    .normalize('NFKD')
    .replace(/\p{M}/gu, '')
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, ' ')
    .trim()
}

// A single missing, extra, or mistyped letter; short words must match literally.
function nearWord(query: string, word: string) {
  if (query.length < 4 || Math.abs(query.length - word.length) > 1) return false
  let edits = 0
  for (let i = 0, j = 0; i < query.length || j < word.length; ) {
    if (query[i] === word[j]) {
      i++
      j++
      continue
    }
    if (++edits > 1) return false
    if (query.length >= word.length) i++
    if (word.length >= query.length) j++
  }
  return true
}

function score(query: string, label: string, keywords = '') {
  const name = normalize(label)
  if (name === query) return 100
  if (name.startsWith(query)) return 90
  if (name.includes(query)) return 80
  const words = normalize(`${label} ${keywords}`).split(' ')
  let fuzzy = false
  for (const token of query.split(' ')) {
    if (words.some((word) => word.startsWith(token))) continue
    if (!words.some((word) => nearWord(token, word))) return 0
    fuzzy = true
  }
  return fuzzy ? 10 : 50
}

export function searchSettings<T extends SettingsSearchPage>(
  query: string,
  pages: readonly T[],
  surface: Surface,
): { page: T; detail?: string }[] {
  const normalized = normalize(query)
  if (!normalized) return query.trim() ? [] : pages.map((page) => ({ page }))
  return pages
    .map((page) => {
      let best = score(normalized, page.label)
      let detail: string | undefined
      for (const topic of topics[page.label]) {
        if (topic.surface && topic.surface !== surface) continue
        const value = score(
          normalized,
          topic.label,
          `${page.label} ${topic.keywords}`,
        )
        if (value > best) {
          best = value
          detail = topic.label
        }
      }
      return { page, detail, score: best }
    })
    .filter((result) => result.score > 0)
    .sort((a, b) => b.score - a.score)
}

export const FILE_DOWNLOAD_CHUNK_BYTES = 256 * 1024
export interface FileDownloadChunk {
  data: Uint8Array
  size: number
  revision: string
}

/** Fetch the first chunk before sending headers so missing files remain HTTP errors.
 * Later failures abort the body: a partial download must never look successful. */
export async function fileDownloadResponse(
  path: string,
  read: (offset: number) => Promise<FileDownloadChunk>,
  options: { signal?: AbortSignal; close?: () => Promise<void> } = {},
): Promise<Response> {
  let closed = false
  let onAbort: (() => void) | undefined
  const close = async () => {
    if (closed) return
    closed = true
    if (onAbort) options.signal?.removeEventListener('abort', onAbort)
    await options.close?.()
  }
  let first: FileDownloadChunk
  try {
    options.signal?.throwIfAborted()
    first = await read(0)
    options.signal?.throwIfAborted()
    if (!Number.isSafeInteger(first.size) || first.size < 0) throw new Error('Invalid download size')
  } catch (error) {
    await close()
    throw error
  }
  let offset = 0
  let initial: FileDownloadChunk | undefined = first
  const body = new ReadableStream<Uint8Array>({
    start(controller) {
      onAbort = () => {
        controller.error(options.signal?.reason ?? new Error('Download cancelled'))
        void close().catch(() => {})
      }
      options.signal?.addEventListener('abort', onAbort, { once: true })
    },
    async pull(controller) {
      try {
        options.signal?.throwIfAborted()
        const chunk = initial ?? await read(offset)
        initial = undefined
        if (closed) return
        options.signal?.throwIfAborted()
        if (chunk.size !== first.size || chunk.revision !== first.revision ||
            chunk.data.length !== Math.min(FILE_DOWNLOAD_CHUNK_BYTES, first.size - offset))
          throw new Error('The file changed during download. Please try again.')
        offset += chunk.data.length
        if (chunk.data.length) controller.enqueue(chunk.data)
        if (offset === first.size) {
          await close()
          controller.close()
        }
      } catch (error) {
        await close()
        controller.error(error)
      }
    },
    cancel: close,
  })
  const filename = path.split(/[\\/]/).pop() || 'download'
  return new Response(body, { headers: {
    'Content-Type': 'application/octet-stream',
    'Content-Length': String(first.size),
    'Content-Disposition': `attachment; filename*=UTF-8''${encodeURIComponent(filename).replace(/['()*]/g, char => '%' + char.charCodeAt(0).toString(16))}`,
    'Cache-Control': 'private, no-store, no-transform',
    'X-Content-Type-Options': 'nosniff',
  } })
}
