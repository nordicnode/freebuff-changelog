import { z } from 'zod/v4'
import { bundleSite, readSiteSecret } from './local-files'
import { SITE_ARGS } from './tools'

import { SitesClientError } from './errors'
export { SitesClientError } from './errors'
// The SAME objects the tools publish as their input schemas, so a constraint
// cannot be enforced here without the model being shown it. See `SITE_ARGS`.
const uuid = SITE_ARGS.siteId
const environment = SITE_ARGS.environment
const key = SITE_ARGS.requestId
const name = SITE_ARGS.bindingName
const target = z.object({ siteId: uuid, environment })
const invalid = (message: string) =>
  new SitesClientError(400, 'invalid_site_request', message)

/**
 * Parse ONE argument, keeping its name on the issue.
 *
 * `schema.parse(args.name)` produces an issue with an empty path, so a refusal
 * could only ever name the failing CHECK and not the argument — the half of
 * this that a model cannot work around.
 */
function field<T>(schema: z.ZodType<T>, argName: string, value: unknown): T {
  return z.object({ [argName]: schema }).parse({ [argName]: value })[
    argName
  ] as T
}

/**
 * Name the arguments a Zod check rejected, and nothing else.
 *
 * The opaque fallback below exists because a filesystem or validator error may
 * carry file contents or a secret value. A Zod issue cannot: this reports only
 * the argument's path and the kind of check that failed, never the value that
 * failed it and never the issue's own message (`invalid_value` carries the
 * received literal). Without this an argument the model can fix reads as
 * "check arguments and workspace files", so it retries the same call.
 */
function rejectedArguments(error: z.ZodError): string {
  const named = [
    ...new Set(
      error.issues.map((issue) => {
        const path = issue.path.map(String).join('.')
        return path ? `${path} (${issue.code})` : issue.code
      }),
    ),
  ]
  return named.length > 0
    ? `Sites rejected these arguments: ${named.join(', ')}. Read the tool's input schema and send values that match it.`
    : 'Could not prepare the Sites request. Check arguments and workspace files.'
}

  async domainAction(siteId: string, env: string, raw: unknown, requestId: string) {
    const input = z.discriminatedUnion('action', [
      z.strictObject({ action: z.literal('connect'), hostname: z.string().min(1).max(253) }),
      z.strictObject({ action: z.enum(['verify', 'primary', 'remove']), domainId: uuid }),
    ]).safeParse(raw)
    if (!uuid.safeParse(siteId).success || !environment.safeParse(env).success || !key.safeParse(requestId).success || !input.success) throw invalid('Invalid domain request.')

  }
  async setPublished(
    siteId: string,
    env: string,
    published: boolean,
    requestId: string,
    signal?: AbortSignal,
  ) {
    try {
      uuid.parse(siteId)
      environment.parse(env)
      key.parse(requestId)
      z.boolean().parse(published)
    } catch {
      throw invalid('Use a valid site, environment and request id.')
    }
    return this.request(
      `/${siteId}/environments/${env}/publication`,
      published ? 'PUT' : 'DELETE',
      undefined,
      requestId,
      signal,
    )
  }
  async deleteSite(siteId: string) {
    if (!uuid.safeParse(siteId).success) throw invalid('Use a valid site id.')
    return this.request(`/${siteId}`, 'DELETE')
  }
  async writeSecret(
    siteId: string,
    env: string,
    binding: string,
    method: 'PUT' | 'DELETE',
    raw: unknown,
    requestId: string,
  ) {
    try {
      uuid.parse(siteId)
      environment.parse(env)
      name.parse(binding)
      key.parse(requestId)
      const input =
        method === 'PUT'
          ? z
              .strictObject({
                value: z
                  .string()
                  .min(1)
                  .max(5120)
                  .refine((value) => Buffer.byteLength(value, 'utf8') <= 5120),
              })
              .parse(raw)
          : undefined
      return await this.request(
        `/${siteId}/environments/${env}/secrets/${binding}`,
        method,
        input,
        requestId,
      )
    } catch (error) {
      if (error instanceof SitesClientError) throw error
      throw invalid(
        'Use an uppercase variable name and a non-empty value of at most 5 KB. Platform binding names are reserved.',
      )
    }
  }
  async read(parts: string[], query = new URLSearchParams()) {
    if (parts.length === 0) return this.request('')
    uuid.parse(parts[0])
    if (parts.length === 1) return this.request('/' + parts[0])
    if (parts[1] !== 'environments') throw invalid('Unknown Sites path.')
    environment.parse(parts[2])
    const suffix = parts.slice(3)
    if (
      !(
        suffix.length === 0 ||
        (suffix.length === 1 &&
          ['operations', 'analytics', 'domains'].includes(suffix[0])) ||
        (suffix.length === 2 &&
          suffix[0] === 'auth' &&
          ['analytics', 'users'].includes(suffix[1])) ||
        (suffix.length === 2 &&
          suffix[0] === 'operations' &&
          uuid.safeParse(suffix[1]).success) ||
        (suffix.length === 3 &&
          suffix[0] === 'storage' &&
          name.safeParse(suffix[1]).success &&
          ['tables', 'rows', 'objects'].includes(suffix[2]))
      )
    )
      throw invalid('Unknown Sites path.')
    const search = new URLSearchParams()
    for (const field of ['table', 'offset', 'cursor', 'q', 'days']) {
      const value = query.get(field)
      if (value !== null && value.length <= 2000) search.set(field, value)
    }
    return this.request(
      '/' +
        parts.map(encodeURIComponent).join('/') +
        (search.size ? '?' + search : ''),
    )
  }

    tool: string,
    raw: unknown,
    cwd: string,
    mode: string,
    signal?: AbortSignal,
    context?: { originProjectId: string; originProjectType: 'local' | 'cloud' },
  ) {
    const run = () => this.runTool(tool, raw, cwd, mode, signal)
    return context ? this.context.run(context, run) : run()
  }
  private async runTool(
    tool: string,
    raw: unknown,
    cwd: string,
    mode: string,
    signal?: AbortSignal,
  ) {
    try {
      if (mode === 'plan' && !['sites_list', 'sites_inspect'].includes(tool))
        throw invalid('Sites mutations are unavailable in plan mode.')
      if (tool === 'sites_list')
        return await this.request('', 'GET', undefined, undefined, signal)
      if (tool === 'sites_create') {
        const args = z
          .object({ siteId: uuid, name: z.string().min(1).max(80) })
          .strict()
          .parse(raw)
        return await this.request(
          '',
          'POST',
          { id: args.siteId, name: args.name },
          undefined,
          signal,
        )
      }
      const args = target.passthrough().parse(raw)
      const path = `/${args.siteId}/environments/${args.environment}`
      if (tool === 'sites_set_published') {
        const input = target
          .extend({ published: z.boolean(), requestId: key })
          .strict()
          .parse(raw)
        return await this.setPublished(
          input.siteId,
          input.environment,
          input.published,
          input.requestId,
          signal,
        )
      }
      if (tool === 'sites_inspect') {
        const [environment, activity] = await Promise.all([
          this.request(path, 'GET', undefined, undefined, signal),
          this.request(
            path + '/operations',
            'GET',
            undefined,
            undefined,
            signal,
          ),
        ])
        return { environment, activity }
      }
      if (tool === 'sites_enable_auth') {
        const input = target
          .extend({ localOrigin: z.string().url().optional() })
          .strict()
          .parse(raw)
        return await this.request(
          path + '/auth',
          'POST',
          { localOrigin: input.localOrigin },
          undefined,
          signal,
        )
      }
      const requestId = field(key, 'requestId', args.requestId)
      if (tool === 'sites_deploy') {
        const manifestPath = field(
          SITE_ARGS.workspacePath,
          'manifestPath',
          args.manifestPath,
        )
        const bundle = await (this.options.files?.bundle ?? bundleSite)(
          cwd,
          manifestPath,
        )
        return await this.request(
          path + '/deployments',
          'POST',
          bundle,
          requestId,
          signal,
        )
      }
      if (tool === 'sites_add_storage') {
        const kind = field(z.enum(['d1', 'r2']), 'kind', args.kind)
        return await this.request(
          path + '/' + kind,
          'POST',
          { name: field(name, 'name', args.name) },
          requestId,
          signal,
        )
      }
      if (tool === 'sites_set_secret') {
        const binding = field(name, 'name', args.name)
        const value = await (this.options.files?.secret ?? readSiteSecret)(
          cwd,
          field(SITE_ARGS.workspacePath, 'valueFile', args.valueFile),
        )
        return await this.request(
          path + '/secrets/' + binding,
          'PUT',
          { value },
          requestId,
          signal,
        )
      }
      throw invalid('Unknown Sites tool.')
    } catch (error) {
      // People read the connection message; an agent also needs to know an
      // unanswered mutation may have landed and is only safe to repeat by key.
      if (error instanceof SitesClientError)
        return {
          error: error.code,
          message:
            error.code === 'sites_connection_failed' &&
            IDEMPOTENT_TOOLS.has(tool)
              ? `${error.message} The operation may still be running: retry with the same requestId.`
              : error.message,
        }
      // Filesystem and validator errors must never echo file contents / secret values.
      return {
        error: 'invalid_site_request',
        message:
          error instanceof z.ZodError
            ? rejectedArguments(error)
            : 'Could not prepare the Sites request. Check arguments and workspace files.',
      }
    }
  }
}

export type SiteToolSpec = {

/** Worker bindings the platform owns; `sites_enable_auth` and the asset
 *  handler inject them, so a project may not claim one. */
export const SITE_RESERVED_BINDINGS = [
  'ASSETS',
  'FREEBUFF_CLIENT_ID',
  'FREEBUFF_ISSUER',
  'APP_ORIGIN',
] as const

/**
 * The Sites tool arguments, defined ONCE.
 *
 * `SitesClient` validates every call against these, and `SITE_TOOL_SPECS`
 * below publishes the same objects as the tools' input schemas, so the
 * constraints the model is shown are exactly the ones it is held to. They used
 * to be two sets — a bare `z.string()` in the spec and a uuid, a binding regex
 * or a length bound in the client — and a model that sent a slug for `siteId`,
 * a lowercase binding name or an over-long site name was refused for a reason
 * no schema had mentioned. Keep them in one place.
