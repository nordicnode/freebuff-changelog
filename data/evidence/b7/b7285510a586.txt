/** Capability detection is presentation only; native IPC checks its sender/account. */
export function hostedWorkspaceHost(
  value: unknown,
): HostedWorkspaceHost | null {
  if (!value || typeof value !== 'object') return null
  const host = value as Partial<HostedWorkspaceHost>
  return host.protocolVersion === HOSTED_WORKSPACE_PROTOCOL_VERSION &&
    typeof host.platform === 'string' &&
    typeof host.appVersion === 'string' &&
    ['status', 'openLocalProject', 'disconnect', 'returnToDesktop'].every(
      (name) => typeof host[name as keyof HostedWorkspaceHost] === 'function',
    )
    ? (host as HostedWorkspaceHost)
    : null
}

// ------------------------------------------------------------------ clock

/** Past this, Desktop skips turn-start probes whose 25 s deadline already "passed" (PR #9473). */
export const CLOCK_SKEW_WARN_MS = 20_000
/** Past this, commands (5 min deadline), sign-in and certificate checks break too. */
export const CLOCK_SKEW_ERROR_MS = 5 * 60_000

export type ClockReading = { skewMs: number; rttMs: number } | { error: string }

/**
 * Compares this computer's clock with the server's `Date` header. The header has one-second
 * precision, so the server time is taken as the middle of that second and the local time as the
 * middle of the round trip; the error is under a second plus half the round trip.
 */
export async function readClockSkew(url: string, options: { fetch?: Fetch; now?: () => number; timeoutMs?: number } = {}): Promise<ClockReading> {
  const fetcher = options.fetch ?? fetch
  const now = options.now ?? Date.now
  const started = now()
  try {
    const response = await fetcher(url, { method: 'HEAD', redirect: 'manual', signal: AbortSignal.timeout(options.timeoutMs ?? 6_000) })
    const ended = now()
    await response.body?.cancel().catch(() => {})
    const header = response.headers.get('date')
    const server = header ? Date.parse(header) : NaN
    if (!Number.isFinite(server)) return { error: 'no-date' }
    return { skewMs: Math.round((started + ended) / 2 - (server + 500)), rttMs: ended - started }
  } catch (error) {
    return { error: classifyNetworkError(error) }
  }
}

export function classifyNetworkError(error: unknown): NetworkErrorKind {
  const e = error as { code?: unknown; name?: unknown; message?: unknown; cause?: { code?: unknown; message?: unknown } } | undefined
  const text = [e?.code, e?.name, e?.message, e?.cause?.code, e?.cause?.message].filter(v => typeof v === 'string').join(' ')

// Issuers that only appear when something on this computer or network re-signs HTTPS traffic.
const INTERCEPTORS = /Avast|AVG|Kaspersky|ESET|Bitdefender|Norton|Symantec|McAfee|Sophos|Trend ?Micro|Webroot|F-Secure|Malwarebytes|Zscaler|Netskope|Fortinet|FortiGate|Palo Alto|Cisco Umbrella|Blue ?Coat|Forcepoint|Barracuda|Check ?Point|Cloudflare Gateway|mitmproxy|Charles Proxy|Fiddler|DO_NOT_TRUST|Proxy/i

export type TlsInspection =
  | { authorized: boolean; authorizationError: string | null; issuers: string[] }
  | { error: NetworkErrorKind }

export function inspectTls(host: string, options: { connect?: typeof tlsConnect; timeoutMs?: number } = {}): Promise<TlsInspection> {
  return new Promise(resolve => {
    let socket: TLSSocket | undefined
    const timer = setTimeout(() => { socket?.destroy(); resolve({ error: 'timeout' }) }, options.timeoutMs ?? 6_000)
    try {
      // rejectUnauthorized: false so a rejected chain can still be described; nothing is sent.
      const opened: TLSSocket = (options.connect ?? tlsConnect)({ host, port: 443, servername: host, rejectUnauthorized: false }, () => {
        clearTimeout(timer)
        const issuers: string[] = []
        try {
          let cert: PeerCertificate | undefined = opened.getPeerCertificate(true)
          const seen = new Set<string>()
          while (cert && Object.keys(cert).length && issuers.length < 6) {
            const name = [cert.issuer?.O, cert.issuer?.CN].flat().filter(Boolean).join(' / ')
            if (name) issuers.push(name)
            const next: PeerCertificate | undefined = (cert as PeerCertificate & { issuerCertificate?: PeerCertificate }).issuerCertificate
            if (!next || seen.has(cert.fingerprint256) || next.fingerprint256 === cert.fingerprint256) break
            seen.add(cert.fingerprint256)
            cert = next
          }
        } catch { /* a runtime without chain access still reports authorization */ }
        const authorizationError = opened.authorizationError ? String(opened.authorizationError) : null
        resolve({ authorized: opened.authorized, authorizationError, issuers })
        opened.end()
      })
      socket = opened
      opened.on('error', error => { clearTimeout(timer); resolve({ error: classifyNetworkError(error) }) })
    } catch (error) {
      clearTimeout(timer)
      resolve({ error: classifyNetworkError(error) })
    }
  })
}

export function tlsFinding(host: string, inspection: TlsInspection): DoctorFinding {
  if ('error' in inspection)

  const interceptor = inspection.issuers.find(name => INTERCEPTORS.test(name))
  if (!inspection.authorized || interceptor) {
    const by = interceptor ? ` The certificate was issued by "${interceptor}", not by a public authority.` : ''
    return finding('tls.intercepted', inspection.authorized ? 'warning' : 'error', 'network',

}

const PROXY_VARS = ['HTTPS_PROXY', 'https_proxy', 'HTTP_PROXY', 'http_proxy', 'ALL_PROXY', 'all_proxy']

export function configuredProxies(env: Record<string, string | undefined>) {
  const seen = new Map<string, { variable: string; host: string; port: number }>()
  for (const variable of PROXY_VARS) {
    const value = env[variable]
    if (!value) continue
    try {
      const url = new URL(value.includes('://') ? value : `http://${value}`)
      const port = Number(url.port) || (url.protocol === 'https:' ? 443 : 80)
      // Only host and port: a proxy URL can carry a user name and password.
      const key = `${url.hostname}:${port}`
      if (!seen.has(key)) seen.set(key, { variable, host: url.hostname, port })
    } catch {
      seen.set(`invalid:${variable}`, { variable, host: '', port: 0 })
    }
  }
  return [...seen.values()]
}

export function tcpReachable(host: string, port: number, timeoutMs = 3_000, connect = netConnect): Promise<boolean> {
  return new Promise(resolve => {
    const socket = connect({ host, port })
    const done = (ok: boolean) => { clearTimeout(timer); socket.destroy(); resolve(ok) }
    const timer = setTimeout(() => done(false), timeoutMs)
    socket.once('connect', () => done(true))
    socket.once('error', () => done(false))
  })
}

/** Counts files under `roots` whose full path passes Windows' 260-character limit. Bounded. */
export async function longPathFinding(roots: string[], platform = process.platform, maxEntries = 20_000,
  list: (dir: string) => Promise<Array<{ name: string; isDirectory(): boolean }>> | Array<{ name: string; isDirectory(): boolean }> = dir => readdir(dir, { withFileTypes: true })): Promise<DoctorFinding | null> {

  while (stack.length && seen < maxEntries) {
    const dir = stack.pop()!
    let entries: Array<{ name: string; isDirectory(): boolean }>
    try { entries = await list(dir) } catch { continue }

      seen++

      if (full.length >= 260) over++
      if (entry.isDirectory() && entry.name !== 'node_modules' && entry.name !== '.git') stack.push(full)
    }
  }

  return finding('system.long-paths', 'warning', 'system', `${over} file path${over === 1 ? ' is' : 's are'} longer than Windows allows by default`,
    'Paths over 260 characters fail in many Windows tools unless long paths are enabled.',
    ['Enable long paths: run `git config --global core.longpaths true`, and turn on "Enable Win32 long paths" in Group Policy (or the LongPathsEnabled registry value).', 'Keep projects in a short folder such as C:\\code.'],

/**
 * Report redaction. By default a report names no full path, no user name and no email: a path
 * keeps only its last folder (`~/…/my-repo`), which is what support needs to follow along. The
 * user can opt in to full paths for their own debugging. Secrets are removed either way.
 */

export type DoctorRedactor = {
  readonly includePaths: boolean
  /** One path, shortened unless the user opted in. */
  path(value: string): string
  /** Free text (an error message): paths, the user name, emails and secrets taken out. */
  text(value: string): string
}

const EMAIL = /[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g
// Bearer tokens, `token=`/`key=` pairs and long opaque keys. Applied whatever the opt-in says.
const SECRETS: Array<[RegExp, string]> = [
  [/\b(Bearer|Basic)\s+[A-Za-z0-9._~+/=-]{8,}/gi, '$1 <secret>'],
  [/\b((?:api[_-]?key|token|secret|password|authorization|cookie)["']?\s*[:=]\s*["']?)[^\s"'&,;]+/gi, '$1<secret>'],
  [/\b(?:sk|pk|rk|fb|cb|ghp|gho|ghu|github_pat|xox[abp])[-_][A-Za-z0-9_-]{16,}\b/g, '<secret>'],
]
// An absolute path: `C:\…`, `\\server\…` or `/…`, not the `//` of a URL or a relative `a/b`.
const ABSOLUTE_PATH = /(?<![\w:/.\\-])(?:[A-Za-z]:[\\/]|\\\\|\/)(?:[^\s"'`<>|*?:\\/]+[\\/])*[^\s"'`<>|*?:\\/]*/g

const escape = (text: string) => text.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')

export function createRedactor(options: {
  home: string
  username?: string
  includePaths?: boolean
}): DoctorRedactor {
  const includePaths = options.includePaths === true
  const home = options.home.replace(/[\\/]+$/, '')
  const homePattern = home
    ? new RegExp(escape(home).replace(/\\\\|\//g, '[\\\\/]'), 'gi')
    : null
  const user = options.username && options.username.length >= 2 ? options.username : undefined

  function shorten(value: string) {
    const parts = value.split(/[\\/]+/).filter(Boolean)
    const last = parts.at(-1) ?? ''
    const underHome = home !== '' && value.toLowerCase().startsWith(home.toLowerCase())
    const prefix = underHome ? '~' : ''
    if (!last || (underHome && value.length <= home.length + 1)) return underHome ? '~' : '…'
    const name = user ? last.replace(new RegExp(escape(user), 'gi'), '<user>') : last
    return `${prefix}/…/${name}`
  }

  function secrets(value: string) {
    return SECRETS.reduce((text, [pattern, replacement]) => text.replace(pattern, replacement), value)
  }

  return {
    includePaths,
    path(value) {
      return includePaths ? value : shorten(value)
    },
    text(value) {
      let text = secrets(value)
      if (includePaths) return text
      text = text.replace(ABSOLUTE_PATH, match => (match.length > 1 ? shorten(match) : match))
      if (homePattern) text = text.replace(homePattern, '~')
      text = text.replace(EMAIL, '<email>')
      if (user) text = text.replace(new RegExp(`\\b${escape(user)}\\b`, 'gi'), '<user>')
      return text
    },
  }
}

import type { DoctorFinding, DoctorReport, DoctorSeverity } from './types'

export function doctorCounts(report: Pick<DoctorReport, 'findings'>) {
  const counts: Record<DoctorSeverity, number> = { error: 0, warning: 0, info: 0, ok: 0 }
  for (const item of report.findings) counts[item.severity]++

}

export function doctorSummary(report: Pick<DoctorReport, 'findings'>) {

  if (!error && !warning) return 'No problems found.'

  const shown = report.findings.filter(item => item.severity !== 'ok')
  for (const item of shown) {
    lines.push(`### ${LABEL[item.severity]}: ${item.title}`, '', `\`${item.id}\` (${item.area})`, '', item.detail, '')
    if (item.steps.length) lines.push(...steps(item, ''), '')

+export const CLOCK_SKEW_WARN_MS = 20_000
+/** Past this, commands (5 min deadline), sign-in and certificate checks break too. */
+export const CLOCK_SKEW_ERROR_MS = 5 * 60_000

+export type ClockReading = { skewMs: number; rttMs: number } | { error: string }
+
+/**
+ * Compares this computer's clock with the server's `Date` header. The header has one-second
+ * precision, so the server time is taken as the middle of that second and the local time as the
+ * middle of the round trip; the error is under a second plus half the round trip.
+ */
+export async function readClockSkew(url: string, options: { fetch?: Fetch; now?: () => number; timeoutMs?: number } = {}): Promise<ClockReading> {
+  const fetcher = options.fetch ?? fetch
+  const now = options.now ?? Date.now
+  const started = now()
+  try {
+    const response = await fetcher(url, { method: 'HEAD', redirect: 'manual', signal: AbortSignal.timeout(options.timeoutMs ?? 6_000) })
+    const ended = now()
+    await response.body?.cancel().catch(() => {})
+    const header = response.headers.get('date')
+    const server = header ? Date.parse(header) : NaN
+    if (!Number.isFinite(server)) return { error: 'no-date' }
+    return { skewMs: Math.round((started + ended) / 2 - (server + 500)), rttMs: ended - started }
+  } catch (error) {
+    return { error: classifyNetworkError(error) }
+  }
+}

+export type TlsInspection =
+  | { authorized: boolean; authorizationError: string | null; issuers: string[] }
+  | { error: NetworkErrorKind }

+export function inspectTls(host: string, options: { connect?: typeof tlsConnect; timeoutMs?: number } = {}): Promise<TlsInspection> {
+  return new Promise(resolve => {
+    let socket: TLSSocket | undefined
+    const timer = setTimeout(() => { socket?.destroy(); resolve({ error: 'timeout' }) }, options.timeoutMs ?? 6_000)
+    try {
+      // rejectUnauthorized: false so a rejected chain can still be described; nothing is sent.
+      const opened: TLSSocket = (options.connect ?? tlsConnect)({ host, port: 443, servername: host, rejectUnauthorized: false }, () => {
+        clearTimeout(timer)
+        const issuers: string[] = []
+        try {
+          let cert: PeerCertificate | undefined = opened.getPeerCertificate(true)
+          const seen = new Set<string>()
+          while (cert && Object.keys(cert).length && issuers.length < 6) {
+            const name = [cert.issuer?.O, cert.issuer?.CN].flat().filter(Boolean).join(' / ')
+            if (name) issuers.push(name)
+            const next: PeerCertificate | undefined = (cert as PeerCertificate & { issuerCertificate?: PeerCertificate }).issuerCertificate
+            if (!next || seen.has(cert.fingerprint256) || next.fingerprint256 === cert.fingerprint256) break
+            seen.add(cert.fingerprint256)
+            cert = next
+          }
+        } catch { /* a runtime without chain access still reports authorization */ }
+        const authorizationError = opened.authorizationError ? String(opened.authorizationError) : null
+        resolve({ authorized: opened.authorized, authorizationError, issuers })
+        opened.end()
+      })
+      socket = opened
+      opened.on('error', error => { clearTimeout(timer); resolve({ error: classifyNetworkError(error) }) })
+    } catch (error) {
+      clearTimeout(timer)
+      resolve({ error: classifyNetworkError(error) })
+    }
+  })

+/**
+ * Report redaction. By default a report names no full path, no user name and no email: a path
+ * keeps only its last folder (`~/…/my-repo`), which is what support needs to follow along. The
+ * user can opt in to full paths for their own debugging. Secrets are removed either way.
+ */
+
+export type DoctorRedactor = {
+  readonly includePaths: boolean
+  /** One path, shortened unless the user opted in. */
+  path(value: string): string
+  /** Free text (an error message): paths, the user name, emails and secrets taken out. */
+  text(value: string): string
+}
+
+const EMAIL = /[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g
+// Bearer tokens, `token=`/`key=` pairs and long opaque keys. Applied whatever the opt-in says.
+const SECRETS: Array<[RegExp, string]> = [
+  [/\b(Bearer|Basic)\s+[A-Za-z0-9._~+/=-]{8,}/gi, '$1 <secret>'],
+  [/\b((?:api[_-]?key|token|secret|password|authorization|cookie)["']?\s*[:=]\s*["']?)[^\s"'&,;]+/gi, '$1<secret>'],
+  [/\b(?:sk|pk|rk|fb|cb|ghp|gho|ghu|github_pat|xox[abp])[-_][A-Za-z0-9_-]{16,}\b/g, '<secret>'],
+]
+// An absolute path: `C:\…`, `\\server\…` or `/…`, not the `//` of a URL or a relative `a/b`.
+const ABSOLUTE_PATH = /(?<![\w:/.\\-])(?:[A-Za-z]:[\\/]|\\\\|\/)(?:[^\s"'`<>|*?:\\/]+[\\/])*[^\s"'`<>|*?:\\/]*/g
+
+const escape = (text: string) => text.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
+
+export function createRedactor(options: {
+  home: string
+  username?: string
+  includePaths?: boolean
+}): DoctorRedactor {
+  const includePaths = options.includePaths === true
+  const home = options.home.replace(/[\\/]+$/, '')
+  const homePattern = home
+    ? new RegExp(escape(home).replace(/\\\\|\//g, '[\\\\/]'), 'gi')
+    : null
+  const user = options.username && options.username.length >= 2 ? options.username : undefined
+
+  function shorten(value: string) {
+    const parts = value.split(/[\\/]+/).filter(Boolean)
+    const last = parts.at(-1) ?? ''
+    const underHome = home !== '' && value.toLowerCase().startsWith(home.toLowerCase())
+    const prefix = underHome ? '~' : ''
+    if (!last || (underHome && value.length <= home.length + 1)) return underHome ? '~' : '…'
+    const name = user ? last.replace(new RegExp(escape(user), 'gi'), '<user>') : last
+    return `${prefix}/…/${name}`
+  }
+
+  function secrets(value: string) {
+    return SECRETS.reduce((text, [pattern, replacement]) => text.replace(pattern, replacement), value)
+  }
+
+  return {
+    includePaths,
+    path(value) {
+      return includePaths ? value : shorten(value)
+    },
+    text(value) {
+      let text = secrets(value)
+      if (includePaths) return text
+      text = text.replace(ABSOLUTE_PATH, match => (match.length > 1 ? shorten(match) : match))
+      if (homePattern) text = text.replace(homePattern, '~')
+      text = text.replace(EMAIL, '<email>')
+      if (user) text = text.replace(new RegExp(`\\b${escape(user)}\\b`, 'gi'), '<user>')
+      return text
+    },
+  }
+}

+      `${report.options.deep ? ', deep scan' : ''}${report.options.includePaths ? ', full paths included' : ', paths redacted'}`,
+  ]

+  return item.steps.map((step, index) => `${bullet}${index + 1}. ${step}`)

+/** The "Copy report" text: Markdown for a support ticket or Discord. Already redacted by the scan. */
+export function doctorReportMarkdown(report: DoctorReport): string {
+  const lines = ['## Freebuff diagnostics', '', ...header(report).map(line => `- ${line}`), `- ${doctorSummary(report)}`, '']
+  const shown = report.findings.filter(item => item.severity !== 'ok')

+  const missing = report.checks.filter(check => check.outcome === 'timeout' || check.outcome === 'failed')

+import { compareFindings, finding, type DoctorArea, type DoctorCheckRun, type DoctorFinding } from './types'
+
+export type DoctorCheck<Context> = {
+  id: string
+  area: DoctorArea
+  /** Runs only when the user asks for a deep scan (a full database integrity check). */
+  deep?: boolean
+  /** Per-check bound; a slow check reports that it timed out and the scan carries on. */
+  timeoutMs?: number
+  run(context: Context, signal: AbortSignal): Promise<DoctorFinding[]> | DoctorFinding[]
+}
+
+const DEFAULT_TIMEOUT_MS = 8_000
+
+/**
+ * Runs every check concurrently, each under its own time limit. A check that throws or times out
+ * becomes one `info` finding naming the check, never the error text (which may hold a path), so
+ * one broken check can neither fail the scan nor leak into the report.
+ */
+export async function runDoctorChecks<Context>(
+  checks: ReadonlyArray<DoctorCheck<Context>>,
+  context: Context,
+  options: { deep?: boolean; signal?: AbortSignal; now?: () => number } = {},
+): Promise<{ findings: DoctorFinding[]; checks: DoctorCheckRun[] }> {
+  const now = options.now ?? (() => performance.now())
+  const runs = await Promise.all(checks.map(async (check): Promise<{ run: DoctorCheckRun; findings: DoctorFinding[] }> => {
+    if (check.deep && !options.deep) return { run: { id: check.id, ms: 0, outcome: 'skipped' }, findings: [] }
+    const started = now()
+    const controller = new AbortController()
+    const abort = () => controller.abort()
+    options.signal?.addEventListener('abort', abort, { once: true })
+    let timer: ReturnType<typeof setTimeout> | undefined
+    try {
+      const timeout = new Promise<'timeout'>(resolve => {
+        timer = setTimeout(() => resolve('timeout'), check.timeoutMs ?? DEFAULT_TIMEOUT_MS)
+      })
+      const result = await Promise.race([Promise.resolve().then(() => check.run(context, controller.signal)), timeout])
+      if (result === 'timeout') {
+        controller.abort()
+        return {
+          run: { id: check.id, ms: Math.round(now() - started), outcome: 'timeout' },
+          findings: [finding(`${check.id}.timeout`, 'info', check.area, `A check took too long (${check.id})`,
+            'This check did not finish in time, so its result is missing from this report. A slow disk or network can cause this.',
+            ['Run the scan again. If it keeps timing out, mention it when you contact support.'])],
+        }
+      }
+      return { run: { id: check.id, ms: Math.round(now() - started), outcome: 'done' }, findings: result }
+    } catch {
+      return {
+        run: { id: check.id, ms: Math.round(now() - started), outcome: 'failed' },
+        findings: [finding(`${check.id}.failed`, 'info', check.area, `A check could not run (${check.id})`,
+          'This check stopped with an error, so its result is missing from this report.', [])],
+      }
+    } finally {
+      clearTimeout(timer)
+      options.signal?.removeEventListener('abort', abort)
+    }
+  }))
+  return {
+    findings: runs.flatMap(run => run.findings).sort(compareFindings),
+    checks: runs.map(run => run.run),
+  }
+}
