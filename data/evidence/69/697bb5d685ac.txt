```
import z from 'zod/v4'

import {
  $getNativeToolCallExampleString,
  jsonToolResultSchema,
  repairInput,
} from '../utils'

import type { $ToolParams } from '../../constants'

const toolName = 'web_search'
const endsAgentStep = true
const keywordQuery = z.string().trim().min(1).max(200)
const searchSchema = z
  .object({
    query: keywordQuery.describe(
      '[freebuff product prompt text omitted]',
    ),
    objective: z
      .string()
      .trim()
      .min(1)
      .max(5000)
      .optional()
      .describe(
        '[freebuff product prompt text omitted]',
      ),
    additional_queries: z
      .array(keywordQuery)
      .max(2)
      .optional()
      .describe(
        'Up to two alternate keyword queries for the same objective. Submit them together in this search.',
      ),
    live: z
      .boolean()
      .optional()
      .describe(
        'Request fresh pages only when the answer is time-sensitive. Slower; pages fetched within the last 10 minutes may be reused.',
      ),
  })
  .describe('Search the web for relevant source excerpts and URLs.')
/** More than two `additional_queries` (~230 calls/day, Nov 2026): search the first two. */
function keepTwoAdditionalQueries(input: Record<string, unknown>) {
  if (Array.isArray(input.additional_queries))
    input.additional_queries = input.additional_queries.slice(0, 2)
  return input
}
const inputSchema = repairInput(keepTwoAdditionalQueries, searchSchema)
const description = `
Start web research with web_search. Give a short keyword query and describe the information you need in objective. For multiple angles on the same question, add related queries in the same call. Keep exact identifiers, numbers, and names in the relevant language.

Read the returned excerpts first. Answer with their source URLs when they provide the evidence you need. Use read_url for a known URL or when an excerpt lacks necessary detail. Indexed content is fine for most searches. Set live only if recently changed information matters.

Example:
${$getNativeToolCallExampleString({
  toolName,
  inputSchema,
  input: {
    query: '@tanstack/react-query staleTime',
    objective:
      'Find the documented default staleTime in TanStack Query v5 and what it means for background refetches.',
  },
  endsAgentStep,
})}
`.trim()

  toolName,
  endsAgentStep,
  description,
  inputSchema,

      z.object({

      }),

        errorMessage: z.string(),

} satisfies $ToolParams
```

```
import z from 'zod/v4'

import { updateFileResultSchema } from './str-replace'
import {
  $getNativeToolCallExampleString,
  jsonToolResultSchema,
  repairInput,
} from '../utils'

import type { $ToolParams } from '../../constants'

const toolName = 'write_file'
const endsAgentStep = false
const fileSchema = z
  .object({
    path: z
      .string()
      .min(1, 'Path cannot be empty')
      .describe(
        `Path to the file, relative to the **project root**. Local hosts (CLI, Desktop, SDK) also accept an absolute path, which is not limited to the project.`,
      ),
    instructions: z
      .string()
      .optional()
      .describe('What the change is intended to do in only one sentence.'),
    content: z.string().describe(`Complete file content to write to the file.`),
  })
  .describe(`Create or overwrite a file with the given content.`)
/** `contents` or `_content` for `content` (~115-270 calls/day, Nov 2026). */
function contentAlias(input: Record<string, unknown>) {
  if (input.content === undefined) {
    const alias = input.contents ?? input._content
    if (typeof alias === 'string') input.content = alias
  }
  return input
}
const inputSchema = repairInput(contentAlias, fileSchema)
const description = `
Create or replace a file with the given content.

Format the \`content\` parameter with the entire content of the file.

#### Additional Info

[freebuff product prompt text omitted]

Examples:

Example 1 - Simple file creation:
${$getNativeToolCallExampleString({
  toolName,
  inputSchema,
  input: {
    path: 'new-file.ts',
    instructions: 'Prints Hello, world',
    content: 'console.log("Hello, world!");',
  },
  endsAgentStep,
})}

Example 2 - Overwriting a file:
${$getNativeToolCallExampleString({
  toolName,
  inputSchema,
  input: {
    path: 'foo.ts',
    instructions: 'Update foo function',
    content: `function foo() {
  doSomethingNew();
}
  
function bar() {
  doSomethingOld();
}
`,
  },
  endsAgentStep,
})}
`.trim()

export const writeFileParams = {
  toolName,
  endsAgentStep,
  description,
  inputSchema,
  outputSchema: jsonToolResultSchema(updateFileResultSchema),
} satisfies $ToolParams
```
