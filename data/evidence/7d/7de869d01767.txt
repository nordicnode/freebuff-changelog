export function getFreebuffModelPresets({
  chatMode = false,
  ...params
}: {
  accessTier?: FreebuffAccessTier | null
  isSubscriber?: boolean
  balance?: number
  pricing?: Pick<FreebuffFreebucksInfo, 'prices'> | null
  models?: readonly FreebuffPresetModel[]
  /** The catalog's recommended row for this viewer (a catalog key). With
   *  `models`, used only while it is listed and available. */
  recommendedModelId?: string | null
  /** The Chat-mode slider's stops (workspace-mode.ts). */
  chatMode?: boolean
} = {}): readonly FreebuffModelPreset[] {
  const presets = tierPresets(params)
  if (!chatMode || !flashSlowOffered(params.models)) return presets
  return presets.map((preset) =>
    preset.id === 'balanced'
      ? {
          id: 'balanced',
          label: preset.label,
          modelId: FREEBUFF_DEEPSEEK_FLASH_SLOW_CATALOG_KEY,
          reasoningEffort: null,
        }
      : preset,
  )
}

/** DeepSeek V4.1 Flash Slow is listed and open in `models`; with no rows to
 *  check against (the CLI, a caller before its catalog), its key is taken as
 *  is. */

  return (
    !models ||
    models.some(
      (row) =>
        row.id === FREEBUFF_DEEPSEEK_FLASH_SLOW_CATALOG_KEY &&
        row.available !== false,
    )
  )
}

function tierPresets({
  accessTier,
  isSubscriber = false,
  balance = 0,
  pricing,
  models,
  recommendedModelId,
}: {
  accessTier?: FreebuffAccessTier | null
  isSubscriber?: boolean
  balance?: number
  pricing?: Pick<FreebuffFreebucksInfo, 'prices'> | null
  models?: readonly FreebuffPresetModel[]
  recommendedModelId?: string | null
}): readonly FreebuffModelPreset[] {
  const only = models?.length === 1 ? models[0] : undefined
  if (only?.efforts && only.efforts.length > 1)
    return thinkingLevelPresets(only, only.efforts)

  const limited = accessTier === 'limited'
  const premium = isSubscriber || balance >= 100
  const rowFor = (id: string) =>
    models?.find((row) => freebuffModelIdMatches(row.compiledId ?? row.id, id))
  const preset = (
    id: FreebuffModelPreset['id'],
    label: string,
    modelId: string,
    reasoningEffort: ReasoningEffort | null,
  ): FreebuffModelPreset => {
    // If a catalog withdraws a recommendation, keep the control usable with
    // the closest listed fallback. A listed but locked row still goes through
    // the ordinary paywall; recommendations never grant model access.
    const row =
      rowFor(modelId) ??
      (modelId !== FREEBUFF_MIMO_V25_MODEL_ID
        ? rowFor(FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID)
        : undefined) ??
      rowFor(FREEBUFF_MIMO_V25_MODEL_ID)
    const resolvedId = row?.compiledId ?? row?.id ?? modelId
    return {
      id,
      label,
      modelId: row?.id ?? modelId,
      reasoningEffort: freebuffModelIdMatches(resolvedId, modelId)
        ? reasoningEffort
        : freebuffModelIdMatches(resolvedId, FREEBUFF_MIMO_V25_MODEL_ID)
          ? null
          : 'high',
    }
  }

  if (!limited)
    return [
      preset('efficient', 'Fast', FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID, 'low'),
      preset('balanced', 'Balanced', FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID, 'high'),
      preset('powerful', 'Powerful', FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID, 'max'),
    ]

  // DeepSeek V4.1 Flash Slow where it is listed and open (with no rows to check
  // against, as in the CLI, its key is taken as is), else Flash at low effort.
  const flashSlow = (
    id: FreebuffModelPreset['id'],
    label: string,
  ): FreebuffModelPreset =>
    flashSlowOffered(models)
      ? {
          id,
          label,
          modelId: FREEBUFF_DEEPSEEK_FLASH_SLOW_CATALOG_KEY,
          reasoningEffort: null,
        }
      : preset(id, label, FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID, 'low')

  // Only a live, open, explicitly zero-priced row qualifies. Display names
  // identify catalog-only candidates without publishing their upstream IDs.
  const free = [/^glyph\b/i, /^solar mini\b/i, /^solar pro\b/i]
    .map((name) =>
      models?.find(
        (row) =>
          row.available !== false &&
          name.test(row.displayName) &&
          (row.price ?? pricing?.prices[row.id]) === 0,
      ),
    )
    .find((row) => row !== undefined)
  // The base of this ladder: the server's pick for the viewer when it is one
  // they can start, else DeepSeek V4.1 Flash Slow.
  const recommended: Pick<FreebuffPresetModel, 'id' | 'defaultEffort'> | undefined = !recommendedModelId
    ? undefined
    : models
      ? models.find(
          (row) => row.id === recommendedModelId && row.available !== false,
        )
      : { id: recommendedModelId }
  const base = (
    id: FreebuffModelPreset['id'],
    label: string,
  ): FreebuffModelPreset =>
    recommended
      ? {
          id,
          label,
          modelId: recommended.id,
          reasoningEffort: recommended.defaultEffort ?? null,
        }
      : flashSlow(id, label)
  return [
    free
      ? {
          id: 'efficient',
          label: 'Efficient',
          modelId: free.id,
          reasoningEffort: free.defaultEffort ?? free.efforts?.[0] ?? null,
        }
      : base('efficient', 'Efficient'),
    base('balanced', 'Balanced'),
    premium
      ? preset(
          'powerful',
          'Powerful',
          FREEBUFF_MUSE_SPARK_13_CONTRIBUTOR_MODEL_ID,
          FREEBUFF_MUSE_SPARK_REASONING_EFFORT,
        )
      : preset('powerful', 'Powerful', FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID, 'high'),
  ]
}

/**
 * One model's thinking levels as the stops: its lowest rung (Fast), its
 * default (Default, the Balanced stop new chats start on) and its highest
 * (Max). "Fast" rather than "Low": a quick answer is a reason to choose it,
 * where "low" sounds like less. DeepSeek V4.1 Flash's ladder is exactly
 * low / high / max.
 */
const THINKING_LEVEL_LABELS: Record<FreebuffModelPreset['id'], string> = {
  efficient: 'Fast',
  balanced: 'Default',
  powerful: 'Max',
}

/** What each of the one-model picker's thinking levels does: its stop's hover
 *  text (product, 2026-11-11). */
export const FREEBUFF_THINKING_LEVEL_HINTS: Record<
  FreebuffModelPreset['id'],
  string
> = {
  efficient: 'Fast: light reasoning, quickest replies. Good for small edits and questions.',
  balanced: 'Default: thinks the task through before answering. Best for most work.',
  powerful: 'Max: the most reasoning, slowest replies. For hard bugs and large changes.',
}

/** The three levels in one line, for the one-model picker's info icon. */
export const FREEBUFF_THINKING_LEVELS_SUMMARY =
  'Fast answers quickly with light reasoning. Default thinks the task through and suits most work. Max reasons the longest, for hard problems.'

function thinkingLevelPresets(
  row: FreebuffPresetModel,
  efforts: readonly ReasoningEffort[],
): readonly FreebuffModelPreset[] {
  const balanced =
    row.defaultEffort && efforts.includes(row.defaultEffort)
      ? row.defaultEffort
      : efforts[efforts.length - 1]!
  const stops = [
    ...new Set([efforts[0]!, balanced, efforts[efforts.length - 1]!]),
  ]
  return stops.map((effort): FreebuffModelPreset => {
    const id =
      effort === balanced
        ? 'balanced'
        : efforts.indexOf(effort) < efforts.indexOf(balanced)
          ? 'efficient'
          : 'powerful'
    return {
      id,
      label: THINKING_LEVEL_LABELS[id],
      modelId: row.id,
      reasoningEffort: effort,
      thinkingLevel: true,
    }
  })
}

/** A manual effort override is custom too: never hide it behind a category. */
export function matchingFreebuffModelPreset(
  presets: readonly FreebuffModelPreset[],
  modelId: string | null | undefined,
  effort: ReasoningEffort | null | undefined,
): FreebuffModelPreset | undefined {
  // If no free promotion is available, the base row fills both lower stops. Prefer
  // Balanced, the default, instead of making a new chat appear downgraded.
  const matches = (preset: FreebuffModelPreset) =>
    !!modelId &&
    freebuffModelIdMatches(modelId, preset.modelId) &&
    (effort ?? null) === preset.reasoningEffort
  return (
    presets.find((preset) => preset.id === 'balanced' && matches(preset)) ??
    presets.find(matches)
  )
}

 * `showAllModels`). On, the picker lists the whole catalog. Off, it lists one
 * model, DeepSeek V4.1 Flash, and a one-model catalog is drawn as that
 * model's thinking levels (Fast / Default / Max, `getFreebuffModelPresets`).
 * Only what is listed changes: prices and admission never read this setting.
 */

/** What a one-model picker says its model is, on the info icon by the Model
 *  heading: with Show all models off here, and for a new account's one row
 *  (`FREEBUFF_ONE_ROW`). */
export const FREEBUFF_ONE_MODEL_TAGLINE =
  'Performs at the level of Claude Opus 5 & Fable 5 on benchmarks'

/**
 * Where the switch stands for this viewer: undefined when the catalog does
 * not offer it (no switch, and every model is listed), else the viewer's own
 * `choice`, or the catalog's default until they make one.
 */
export function showAllModelsSwitch(
  catalog: Pick<FreebuffModelCatalog, 'showAllModels'> | null | undefined,
  choice: boolean | undefined,
): boolean | undefined {
  const option = catalog?.showAllModels
  return option ? (choice ?? option.default) : undefined
}

/**
 * `catalog` as the picker lists it. With the switch off, only DeepSeek V4.1
 * Flash's row, under `FREEBUFF_ONE_MODEL_TAGLINE`, which is also the
 * recommended and fallback row; the whole
 * catalog when that row is missing or locked, so the picker is never left
 * empty or holding only a row nobody can start.
 */
export function pickerFreebuffCatalog(
  catalog: FreebuffModelCatalog,
  choice: boolean | undefined,
): FreebuffModelCatalog {
  if (showAllModelsSwitch(catalog, choice) !== false) return catalog
  const row = catalog.rows.find(
    (candidate) =>
      compiledFreebuffModelIdOfRow(candidate) ===
      FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID,
  )
  if (!row || row.access !== 'open') return catalog
  return {
    ...catalog,
    // Clients redraw when the version changes.
    version: `${catalog.version}.one`,
    rows: [{ ...row, tagline: FREEBUFF_ONE_MODEL_TAGLINE }],
    recommendedKey: row.key,
    fallbackKey: row.key,
  }
}
