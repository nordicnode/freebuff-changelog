```
/**
 * Keeps mouse reports that arrive split across stdin reads out of the input box.
 *
 * With mouse tracking on, the terminal sends each move or wheel tick as an SGR
 * report, `ESC [ < 35 ; 76 ; 40 M`. OpenTUI's stdin parser waits a short time
 * for the rest of an unfinished sequence, then flushes it. A report split
 * right after `ESC [` (a slow ssh or relay hop, a busy machine) loses its
 * introducer to that flush as an opaque `ESC [` response, and the rest,
 * `<35;76;40M`, is then parsed as typed characters. OpenTUI recovers a report
 * split after a lone ESC or after `ESC [ <`, but not this one.
 *
 * The guard watches the renderer's input sequences: after a flushed `ESC [`,
 * keys that spell an SGR report body are swallowed. Nothing a person types
 * produces a lone flushed `ESC [` followed by `<`, so typing is untouched.
 */
```

function findLineStart(text: string, cursor: number): number {
  let pos = clamp(cursor, 0, text.length)
  while (pos > 0 && text[pos - 1] !== '\n') {
    pos--
  }
  return pos

function findPreviousWordBoundary(text: string, cursor: number): number {
  let pos = clamp(cursor, 0, text.length)

  // Skip whitespace backwards
  while (pos > 0 && /\s/.test(text[pos - 1])) {
    pos--
  }

  // Skip word characters backwards
  while (pos > 0 && !/\s/.test(text[pos - 1])) {
    pos--
  }

  return pos
}

function findNextWordBoundary(text: string, cursor: number): number {
  let pos = clamp(cursor, 0, text.length)

  // Skip non-whitespace forwards
  while (pos < text.length && !/\s/.test(text[pos])) {
    pos++
  }

  // Skip whitespace forwards
  while (pos < text.length && /\s/.test(text[pos])) {
    pos++
  }

  return pos
}

/**
 * Check if a key event represents printable character input (not a special key).
 * Uses a positive heuristic based on key.name length rather than a brittle deny-list.
 *
 * The key insight is that OpenTUI's parser assigns descriptive multi-character names
 * to special keys (like 'backspace', 'up', 'f1') while regular printable characters
 * either have no name (multi-byte input like Chinese) or a single-character name.
 */
function isPrintableCharacterKey(key: KeyEvent): boolean {
  const name = key.name

  // No name = likely multi-byte input (Chinese, Japanese, Korean, etc.) - treat as printable
  if (!name) return true

  // Single character name = regular ASCII printable (a, b, 1, $, etc.)
  if (name.length === 1) return true

  // Special case: space key has name 'space' but is printable
  if (name === 'space') return true

  // Multi-char name = special key (up, f1, backspace, etc.)
  return false
}

function getPrintableKeySequence(key: KeyEvent): string | null {
  if (!key.sequence || key.sequence.length < 1) return null
  if (key.ctrl || key.meta || key.option) return null

  const keypadValue = getKeypadPrintableSequence(key)
  if (keypadValue !== null) return keypadValue

  if (!CONTROL_CHAR_REGEX.test(key.sequence) && isPrintableCharacterKey(key)) {
    return key.sequence
  }

  return null
}

// Helper to convert render position (in tab-expanded string) to original text position
function renderPositionToOriginal(text: string, renderPos: number): number {
  let originalPos = 0
  let currentRenderPos = 0

  while (originalPos < text.length && currentRenderPos < renderPos) {
    if (text[originalPos] === '\t') {
      currentRenderPos += TAB_WIDTH
    } else {
      currentRenderPos += 1
    }
    originalPos++
  }

  return Math.min(originalPos, text.length)
}

        if (handleSelectionDeletion()) return true
        const newValue =
          value.slice(0, wordStart) + value.slice(cursorPosition)

          text: newValue,
          cursorPosition: wordStart,
          lastEditDueToNav: false,

        return true

      if (key.name === 'delete' && key.meta && !isAltLikeModifier) {
        preventKeyDefault(key)

        let newValue = originalValue
        let nextCursor = cursorPosition

            cursorPosition === lineStart &&
            value[cursorPosition - 1] === '\n'
          ) {
            newValue =
              value.slice(0, cursorPosition - 1) + value.slice(cursorPosition)
            nextCursor = cursorPosition - 1
          } else {
            newValue = value.slice(0, lineStart) + value.slice(cursorPosition)
            nextCursor = lineStart
          }

        if (newValue === originalValue && cursorPosition > 0) {
          newValue =
            value.slice(0, cursorPosition - 1) + value.slice(cursorPosition)
          nextCursor = cursorPosition - 1
        }

        if (newValue !== originalValue) {
          onChange({
            text: newValue,
            cursorPosition: nextCursor,
            lastEditDueToNav: false,
          })
        }
        return true

      if (key.name === 'delete' && isAltLikeModifier) {
        preventKeyDefault(key)
        if (handleSelectionDeletion()) return true
        const newValue = value.slice(0, cursorPosition) + value.slice(wordEnd)
        onChange({
          text: newValue,
          cursorPosition,
          lastEditDueToNav: false,
        })
        return true
      }

      if (key.ctrl && lowerKeyName === 'k' && !key.meta && !key.option) {
        preventKeyDefault(key)
        if (handleSelectionDeletion()) return true
        const newValue = value.slice(0, cursorPosition) + value.slice(lineEnd)
        onChange({ text: newValue, cursorPosition, lastEditDueToNav: false })
        return true

        preventKeyDefault(key)
        if (handleSelectionDeletion()) return true

            value.slice(0, cursorPosition - 1) + value.slice(cursorPosition)

            text: newValue,
            cursorPosition: cursorPosition - 1,
            lastEditDueToNav: false,

        return true

        if (cursorPosition < value.length) {

            value.slice(0, cursorPosition) + value.slice(cursorPosition + 1)
          onChange({
            text: newValue,
            cursorPosition,
            lastEditDueToNav: false,
          })

        return true

      if (key.name === 'backspace' && !key.ctrl && !key.meta && !key.option) {
        preventKeyDefault(key)
        if (handleSelectionDeletion()) return true
        if (cursorPosition > 0) {
          const newValue =

            cursorPosition: cursorPosition - 1,
            lastEditDueToNav: false,

        : logicalLineStart
      const visualLineEnd = lineStarts[visualLineIndex + 1] !== undefined

      scrollX={false}
      stickyScroll={true}
      stickyStart="bottom"

        wrapperOptions: {
          paddingLeft: 1,
          paddingRight: 1,
          border: false,
        },
        contentOptions: {
          justifyContent: 'flex-start',

            {shouldHighlight ? (

                : ''
              : afterCursor}

          </>
        ) : (
          <>
            {displayValueForRendering}
            {layoutMetrics.gutterEnabled ? '\n' : ''}
          </>
        )}
      </text>
    </scrollbox>
  )
})

import type { CliRenderer } from '@opentui/core'

/**
 * Keeps mouse reports that arrive split across stdin reads out of the input box.
 *
 * With mouse tracking on, the terminal sends each move or wheel tick as an SGR
 * report, `ESC [ < 35 ; 76 ; 40 M`. OpenTUI's stdin parser waits a short time
 * for the rest of an unfinished sequence, then flushes it. A report split
 * right after `ESC [` (a slow ssh or relay hop, a busy machine) loses its
 * introducer to that flush as an opaque `ESC [` response, and the rest,
 * `<35;76;40M`, is then parsed as typed characters. OpenTUI recovers a report
 * split after a lone ESC or after `ESC [ <`, but not this one.
 *
 * The guard watches the renderer's input sequences: after a flushed `ESC [`,
 * keys that spell an SGR report body are swallowed. Nothing a person types
 * produces a lone flushed `ESC [` followed by `<`, so typing is untouched.
 */

/** The introducer OpenTUI flushes on its own when a report is split after it. */
const FLUSHED_CSI = '\x1b['
/** How long after the flush the rest of the report may still arrive. */
const REST_OF_REPORT_WINDOW_MS = 2_000
/**
 * Waiting this long for the rest of a split sequence before OpenTUI flushes it
 * (its own default is 20ms). Terminals (Ghostty, iTerm2, VS Code) also split
 * bracketed paste across reads when files are dropped, and a flush corrupts
 * paste detection. Still unnoticeable on a lone Escape press.
 */
export const STDIN_PARSER_TIMEOUT_MS = 100

type GuardRenderer = Pick<
  CliRenderer,
  'prependInputHandler' | 'removeInputHandler'
>

/**
 * Progress through `<` digits `;` digits `;` digits (`M` | `m`).
 * `field` counts the semicolons seen; `digits` whether this field has one.
 */
type ReportBody = { field: number; digits: boolean }

export class SplitMouseReportGuard {
  private armedAt: number | null = null
  private body: ReportBody | null = null

  constructor(private readonly now: () => number = Date.now) {}

  /** True when `sequence` is part of a split mouse report and must not be typed. */
  readonly handleInput = (sequence: string): boolean => {
    if (sequence === FLUSHED_CSI) {
      this.armedAt = this.now()
      this.body = null
      return false
    }
    if (this.body) return this.continueBody(sequence)
    if (this.armedAt === null) return false
    const armedAt = this.armedAt
    this.armedAt = null
    if (sequence !== '<' || this.now() - armedAt > REST_OF_REPORT_WINDOW_MS) {
      return false
    }
    this.body = { field: 0, digits: false }
    return true
  }

  private continueBody(sequence: string): boolean {
    const body = this.body!
    if (sequence.length === 1 && sequence >= '0' && sequence <= '9') {
      body.digits = true
      return true
    }
    if (sequence === ';' && body.digits && body.field < 2) {
      this.body = { field: body.field + 1, digits: false }
      return true
    }
    this.body = null
    // The final byte ends the report; anything else is not a report, so it is typed.
    return (sequence === 'M' || sequence === 'm') && body.digits && body.field === 2
  }
}

/** OpenTUI's stdin parser, whose flush timeout is not a renderer option. */
type RendererWithStdinParser = { stdinParser?: { timeoutMs?: unknown } | null }

/**
 * Give split sequences longer to arrive whole before the parser flushes them,
 * and swallow the remains of a split mouse report it flushed anyway.
 * Returns the uninstaller.
 */
export function installSplitMouseReportGuard(
  renderer: GuardRenderer,
): () => void {
  const parser = (renderer as unknown as RendererWithStdinParser).stdinParser
  if (parser && typeof parser.timeoutMs === 'number') {
    parser.timeoutMs = Math.max(parser.timeoutMs, STDIN_PARSER_TIMEOUT_MS)
  }
  const guard = new SplitMouseReportGuard()
  renderer.prependInputHandler(guard.handleInput)
  return () => renderer.removeInputHandler(guard.handleInput)
}

        mice.push(event.raw)
      } else if (event.type === 'response') {
        handler?.(event.sequence)

      drain()
      now += elapsedMs
      parser.flushTimeout(now)
      drain()
    },
    get typed() {
      return typed
    },
    mice,
  }
}

    terminal,
    advance(ms: number) {
      now += ms
    },
  }
}

```diff
+import type { CliRenderer } from '@opentui/core'
+
+/**
+ * Keeps mouse reports that arrive split across stdin reads out of the input box.
+ *
+ * With mouse tracking on, the terminal sends each move or wheel tick as an SGR
+ * report, `ESC [ < 35 ; 76 ; 40 M`. OpenTUI's stdin parser waits a short time
+ * for the rest of an unfinished sequence, then flushes it. A report split
+ * right after `ESC [` (a slow ssh or relay hop, a busy machine) loses its
+ * introducer to that flush as an opaque `ESC [` response, and the rest,
+ * `<35;76;40M`, is then parsed as typed characters. OpenTUI recovers a report
+ * split after a lone ESC or after `ESC [ <`, but not this one.
+ *
+ * The guard watches the renderer's input sequences: after a flushed `ESC [`,
+ * keys that spell an SGR report body are swallowed. Nothing a person types
+ * produces a lone flushed `ESC [` followed by `<`, so typing is untouched.
+ */
+
+/** The introducer OpenTUI flushes on its own when a report is split after it. */
+const FLUSHED_CSI = '\x1b['
+/** How long after the flush the rest of the report may still arrive. */
+const REST_OF_REPORT_WINDOW_MS = 2_000
+/**
+ * Waiting this long for the rest of a split sequence before OpenTUI flushes it
+ * (its own default is 20ms). Terminals (Ghostty, iTerm2, VS Code) also split
+ * bracketed paste across reads when files are dropped, and a flush corrupts
+ * paste detection. Still unnoticeable on a lone Escape press.
+ */
+export const STDIN_PARSER_TIMEOUT_MS = 100
+
+type GuardRenderer = Pick<
+  CliRenderer,
+  'prependInputHandler' | 'removeInputHandler'
+>
+
+/**
+ * Progress through `<` digits `;` digits `;` digits (`M` | `m`).
+ * `field` counts the semicolons seen; `digits` whether this field has one.
+ */
+type ReportBody = { field: number; digits: boolean }
+
+export class SplitMouseReportGuard {
+  private armedAt: number | null = null
+  private body: ReportBody | null = null
+
+  constructor(private readonly now: () => number = Date.now) {}
+
+  /** True when `sequence` is part of a split mouse report and must not be typed. */
+  readonly handleInput = (sequence: string): boolean => {
+    if (sequence === FLUSHED_CSI) {
+      this.armedAt = this.now()
+      this.body = null
+      return false
+    }
+    if (this.body) return this.continueBody(sequence)
+    if (this.armedAt === null) return false
+    const armedAt = this.armedAt
+    this.armedAt = null
+    if (sequence !== '<' || this.now() - armedAt > REST_OF_REPORT_WINDOW_MS) {
+      return false
+    }
+    this.body = { field: 0, digits: false }
+    return true
+  }
+
+  private continueBody(sequence: string): boolean {
+    const body = this.body!
+    if (sequence.length === 1 && sequence >= '0' && sequence <= '9') {
+      body.digits = true
+      return true
+    }
+    if (sequence === ';' && body.digits && body.field < 2) {
+      this.body = { field: body.field + 1, digits: false }
+      return true
+    }
+    this.body = null
+    // The final byte ends the report; anything else is not a report, so it is typed.
+    return (sequence === 'M' || sequence === 'm') && body.digits && body.field === 2
+  }
+}
+
+/** OpenTUI's stdin parser, whose flush timeout is not a renderer option. */
+type RendererWithStdinParser = { stdinParser?: { timeoutMs?: unknown } | null }
+
+/**
+ * Give split sequences longer to arrive whole before the parser flushes them,
+ * and swallow the remains of a split mouse report it flushed anyway.
+ * Returns the uninstaller.
+ */
+export function installSplitMouseReportGuard(
+  renderer: GuardRenderer,
+): () => void {
+  const parser = (renderer as unknown as RendererWithStdinParser).stdinParser
+  if (parser && typeof parser.timeoutMs === 'number') {
+    parser.timeoutMs = Math.max(parser.timeoutMs, STDIN_PARSER_TIMEOUT_MS)
+  }
+  const guard = new SplitMouseReportGuard()
+  renderer.prependInputHandler(guard.handleInput)
+  return () => renderer.removeInputHandler(guard.handleInput)
+}
```
