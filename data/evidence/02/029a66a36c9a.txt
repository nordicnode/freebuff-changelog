/** Stable product messages, also used when Desktop imports Cloud history. */
export const CLOUD_RESOURCE_MESSAGE =
  'Your cloud environment has reached its resource limit. Open this chat in the Desktop app to continue locally.'
export const CLOUD_RESOURCE_PENDING =
  'Your cloud environment has reached its resource limit. Your files are being saved for recovery. Open this chat in the Desktop app to continue locally.'
export const CLOUD_RESOURCE_MOVE_REASON =
  'The cloud environment is running low on resources. Continue the same task locally, preserving completed work and checking any interrupted command before retrying it.'
export const CLOUD_RESOURCE_EVENT = 'resource_limit'
export type CloudResourceKind =
  | 'disk'
  | 'memory'
  | 'cpu'
  | 'processes'
  | 'files'
  | 'time'
export const isCloudResourceMessage = (message: string | null | undefined) =>
  message === CLOUD_RESOURCE_MESSAGE || message === CLOUD_RESOURCE_PENDING

/**
 * Limits a running process hits, not the project itself: a watcher holding too
 * many files open, a burst of processes or threads, a memory spike. The chat's
 * saved copy restores without the processes that tripped it, so the chat may
 * run in Cloud again. A full disk (the files themselves are the problem) and
 * sustained CPU (the shape of mining) still keep a chat out of Cloud.
 * Web-only users had no way back from these at all (Lescence, Brendan, 11-04).
 */
export const CLOUD_RESTARTABLE_RESOURCES: readonly CloudResourceKind[] = ['files', 'processes', 'memory']
export const cloudResourceRestarts = (kind: string | null | undefined) =>
  (CLOUD_RESTARTABLE_RESOURCES as readonly string[]).includes(kind ?? '')
/** Restarts after a restartable trip, per chat per rolling day; past it the chat waits or goes to Desktop. */
export const CLOUD_RESOURCE_RESTARTS_PER_DAY = 3
/** The failed run's error after a restartable trip, for clients that cannot move the chat locally. */
export const CLOUD_RESOURCE_RESTART_MESSAGE =
  'Cloud stopped this turn at a resource limit and saved this chat’s files. Send a message to continue it in Cloud from those files; whatever was running (dev servers, watchers) is not started again.'
/** A send while the restartable trip's copy is still being saved. */
export const CLOUD_RESOURCE_SAVING_MESSAGE =
  'Cloud is still saving this chat’s files after its resource limit. Send again in a few minutes to continue it in Cloud.'
export const CLOUD_RESOURCE_RESTART_LIMIT_MESSAGE =
  `Cloud stopped this chat at a resource limit ${CLOUD_RESOURCE_RESTARTS_PER_DAY} times in the last day, so it can’t restart in Cloud until a day after the first stop. Its files are saved with this chat, and the Freebuff Desktop app can continue it now (open the chat there and choose Local).`

/** Which limit was hit, in plain words, and what usually takes that resource. */
const RESOURCE_EXPLANATIONS: Record<
  CloudResourceKind,
  { what: string; advice: string }
> = {
  disk: {
    what: "Cloud stopped this turn because the project's disk is almost full (under 2 GB free).",
    advice:
      'Build output and large folders such as node_modules usually take the space; remove what you no longer need before running this project in Cloud again.',
  },
  memory: {
    what: 'Cloud stopped this turn because the project used up its memory.',
    advice:
      'A dev server, build or test run that keeps growing is the usual cause; stop it, or split the task into smaller steps, before running this project in Cloud again.',
  },
  cpu: {
    what: 'Cloud stopped this turn because the project kept the CPU fully busy for over 30 seconds.',
    advice:
      'A runaway build or loop is the usual cause; stop it before running this project in Cloud again.',
  },
  processes: {
    what: 'Cloud stopped this turn because the project is running too many processes.',
    advice:
      'Dev servers and watchers left running are the usual cause; stop the ones you no longer need before running this project in Cloud again.',
  },
  files: {
    what: 'Cloud stopped this turn because a process has too many files open.',
    advice:
      'A file watcher over a huge folder is the usual cause; stop it before running this project in Cloud again.',
  },
  time: {
    what: 'Cloud stopped this turn because it passed the two-hour limit for one turn.',
    advice: 'Send the work as smaller steps.',
  },
}

/**
 * Where the files went, for a run whose client cannot move it to this computer
 * (web). Recovery takes the chat's files out of Cloud into a copy only Desktop
 * reads; a new Cloud chat starts from the project's source, not from them, and
 * deleting the chat purges the copy. "Your files are saved" alone sent web users
 * to a new chat that was empty (2026-10-28).
 */
export const CLOUD_RESOURCE_FILES_WHERE =
  'They stay with this chat, and the Freebuff Desktop app can continue from them on your computer (open the chat there and choose Local). A new chat starts from your project, not from these files, and deleting this chat deletes them.'

/** Where the files went after a restartable trip (`cloudResourceRestarts`). */
export const CLOUD_RESOURCE_RESTART_WHERE =
  'Send a message here to continue this chat in Cloud from them; whatever was running is not started again. The Freebuff Desktop app can also continue them on your computer (open the chat there and choose Local).'

/**
 * The transcript note written next to a resource-limit failure: which limit,
 * whether the files are already safe, what to clean up. CLOUD_RESOURCE_MESSAGE
 * itself stays the fixed sentence Desktop builds compare against
 * (`isCloudResourceMessage`), so the explanation travels as a text event.
 * `movesLocally` is false when the run's client cannot continue it locally.
 */
export function cloudResourceExplanation(
  kind: CloudResourceKind,
  recovered: boolean,
  movesLocally = true,
): string {
  const { what, advice } = RESOURCE_EXPLANATIONS[kind]
  const files = recovered
    ? 'Your files are saved.'
    : 'Your files are being saved for recovery; this can take a few minutes.'
  if (movesLocally) return `${what} ${files} ${advice}`
  return cloudResourceRestarts(kind)
    ? `${what} ${files} ${CLOUD_RESOURCE_RESTART_WHERE} ${advice}`
    : `${what} ${files} ${CLOUD_RESOURCE_FILES_WHERE} ${advice}`
}

A new chat starts from your project, not from these files, and deleting this chat deletes them.   expect((['files', 'processes', 'memory'] as const).every(cloudResourceRestarts)).toBe(true)
  expect((['disk', 'cpu', 'time', 'archive', 'local_handoff', undefined] as const).some(cloudResourceRestarts)).toBe(false)
  // Web: the dead end "open this chat in Desktop" was all a browser user heard (Lescence, Brendan).
  expect(cloudResourceExplanation('files', true, false)).toBe(

  )
  expect(cloudResourceExplanation('files', true, false)).not.toContain(CLOUD_RESOURCE_FILES_WHERE)
  // Desktop moves locally as before, and the run error is not the sentence it acts on.
  expect(cloudResourceExplanation('files', true, true)).not.toContain(CLOUD_RESOURCE_RESTART_WHERE)
  expect(isCloudResourceMessage(CLOUD_RESOURCE_RESTART_MESSAGE)).toBe(false)
})
