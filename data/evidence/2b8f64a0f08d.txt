 * Must stay in step with the `ad_engagement_status` pg enum.

 * typed against `AdEngagementStatus` silently could not represent the one
 * status that carries a consequence. The pg enum is the source of truth and

 * for.

  approved: 'Approved',
  rejected: 'Rejected',
  skipped: 'Skipped',

}
