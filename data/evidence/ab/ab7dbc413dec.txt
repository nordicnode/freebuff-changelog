/**
 * Compaction by structured summary, prepared in the background.
 *
 * The goals: a summary is ready before it is needed (low latency), it keeps
 * decisions, requirements and next steps (accurate), and a user who returns
 * after the prompt cache expired sends a small request (no cold start).
 *
 * 1. When a turn ends (or is cancelled or fails) with the context at
 *    FIRST_SUMMARY_TOKENS, or SUMMARY_GROWTH_TOKENS past the last summary,
 *    the runtime asks the agent's own model for a summary of the whole
 *    conversation. Mid-turn, after a step, it asks only once the context is
 *    in the band below the compaction threshold (`summaryBandStart`), the
 *    only summary the limit will use. The request is the agent's normal
 *    request plus one instruction, so the provider reads the history from its
 *    warm cache. It runs in the background. A new summary replaces the old
 *    one.
 * 2. The full history is still sent while the cache is warm.
 * 3. The summary is used only when the user returns after the model's idle
 *    threshold (`summaryIdleMs`), or when the context reaches its limit.
 *    Then the request is the summary, the messages after it, and the new
 *    message.
 *
 * The summary waits on the agent state (`compactionSummary`), outside the
 * history. A summary that finishes after its run ended reaches the next run
 * of the same conversation through an in-process store.
 */

/**
 * The idle gap after which the summary replaces the history. Five minutes is
 * the shortest prompt cache measured on our lanes (Anthropic via Respan,
 * prod 2026-10-26); Pareto keeps it 10-15 minutes, OpenAI about 30.
 */
export const SUMMARY_IDLE_MS = 5 * 60 * 1000

/**
 * DeepSeek V4 Flash, fast mode included, waits as long as its cache lasts.
 * Measured in prod 2026-11-02/03 on returns that kept the full history (no
 * summary was ready): on Pareto its cache was still warm (over half the
 * prompt cached) on 92% of them after 5-15 minutes, and on 89% after 15-30.
 * Half of its idle triggers fell in that 5-15 minute window, and each swap
 * there turned a warm prefix into a cold one (8.6% cached on the request
 * after a swap) while dropping detail the user then saw as "compacting
 * constantly". Fifteen
 * minutes is the figure its idle policy already uses
 * (common/src/constants/compaction-policy.ts).
 */
const MODEL_SUMMARY_IDLE_MS: Readonly<Record<string, number>> = {
  [FREEBUFF_DEEPSEEK_V4_FLASH_MODEL_ID]:
    DEEPSEEK_FLASH_COMPACTION_POLICY.cacheExpiryMs,
  [FREEBUFF_DEEPSEEK_V4_FLASH_FAST_MODEL_ID]:
    DEEPSEEK_FLASH_COMPACTION_POLICY.cacheExpiryMs,
}

/** The idle gap after which `model`'s summary replaces the history. */
export function summaryIdleMs(model: string): number {
  return MODEL_SUMMARY_IDLE_MS[model] ?? SUMMARY_IDLE_MS
}

export const FIRST_SUMMARY_TOKENS = 40_000
export const SUMMARY_GROWTH_TOKENS = 30_000
/** Small windows use shares of the window instead of the fixed sizes. */
const FIRST_SUMMARY_SHARE = 0.2
const SUMMARY_GROWTH_SHARE = 0.15
const SUMMARY_TARGET_TOKENS = 2_000
const SUMMARY_TARGET_SHARE = 0.1
const SUMMARY_MAX_OUTPUT_TOKENS = 16_384
const MIN_SUMMARY_CHARS = 200
/** Of the format's eight sections, how many mark a reply with no English one. */
const MIN_TRANSLATED_SECTIONS = 6

export const CONVERSATION_SUMMARY_TAG = 'CONVERSATION_SUMMARY'

const SUMMARY_NOTE =
  'Historical memory only: it replaces the earlier conversation. Messages after it are real messages. The latest user request can appear both in the memory and after it: the memory then records work already done on it, so continue that work, do not start over. If no user message follows, resume the latest request in the memory. Use real tool calls when further action is needed or give the final response when the work is complete.'

export function summaryThresholds(maxContextLength: number): {
  first: number
  growth: number
} {
  return {
    first: Math.min(
      FIRST_SUMMARY_TOKENS,
      Math.floor(maxContextLength * FIRST_SUMMARY_SHARE),
    ),
    growth: Math.min(
      SUMMARY_GROWTH_TOKENS,
      Math.floor(maxContextLength * SUMMARY_GROWTH_SHARE),
    ),
  }
}

/** How long a summary is asked to be: small windows get a shorter one. */
export function summaryTargetTokens(maxContextLength: number): number {
  return Math.min(
    SUMMARY_TARGET_TOKENS,
    Math.floor(maxContextLength * SUMMARY_TARGET_SHARE),
  )
}

/**
 * Whether a manual compact is worth a summary request. A chat under twice the
 * summary's own target gains little from swapping in a summary of about that
 * size, and a summary already in the history is not counted: pressing Compact
 * twice in a row is billed once. The caller's scaffolding is not conversation.
 */
export function manualCompactionIsWorthwhile(params: {
  messages: Message[]
  maxContextLength: number
}): boolean {
  const conversation = params.messages.filter(
    (message) =>
      !isEphemeral(message) &&
      !message.tags?.includes('INSTRUCTIONS_PROMPT') &&
      !message.tags?.includes(CONVERSATION_SUMMARY_TAG),
  )
  return (
    conversation.length >= 2 &&
    countTokensMessages(conversation) >=
      2 * summaryTargetTokens(params.maxContextLength)
  )
}

/** Whether the context grew enough since `latest` to need a new summary. */
export function summaryIsDue(params: {
  contextTokens: number
  maxContextLength: number
  /** The newest summary made or in progress, as its context size. */
  latestContextTokens: number | undefined
}): boolean {
  const { first, growth } = summaryThresholds(params.maxContextLength)
  if (params.latestContextTokens === undefined)
    return params.contextTokens >= first
  return params.contextTokens - params.latestContextTokens >= growth
}

/**
 * Where the near-limit band starts: one growth step below the compaction
 * threshold (270k on DeepSeek V4 Flash's 375k budget, 290k on 400k).
 */
export function summaryBandStart(maxContextLength: number): number {
  return (
    compactionThreshold(maxContextLength) -
    summaryThresholds(maxContextLength).growth
  )
}

/**
 * Whether a step inside a turn should start a summary. Only one in the band
 * is ever used at the limit; a mid-turn summary below it is replaced by the
 * turn-end one or a later one before anything reads it. Measured 2026-11-10:
 * 416k mid-turn summaries a day against ~107k applied. So: one summary once
 * the context reaches the band, and none while the newest is already in it.
 */
export function midTurnSummaryIsDue(params: {
  contextTokens: number
  maxContextLength: number
  /** The newest summary made or in progress, as its context size. */
  latestContextTokens: number | undefined
}): boolean {
  const bandStart = summaryBandStart(params.maxContextLength)
  if (params.contextTokens < bandStart) return false
  return (
    params.latestContextTokens === undefined ||
    params.latestContextTokens < bandStart
  )
}

/**
 * Tags a prompt the host wrote to carry on work the user already asked for
 * (a Cloud turn resumed after a runner drain, a mission's next pass). It is
 * not the user returning: the gap before it is a queue, not a user who left.
 */
export const CONTINUATION_PROMPT_TAG = 'CONTINUATION_PROMPT'

/**
 * Whether the live prompt arrived after the conversation sat idle longer than
 * `idleMs`, measured from the later of the last turn end or checkpoint
 * (`lastActivityAt`) and the last assistant message. Never true after an
 * answer or for a continuation prompt.
 */
export function isIdleReturn(
  messages: Message[],
  idleMs = SUMMARY_IDLE_MS,
  now = Date.now(),
  lastActivityAt?: number,
): boolean {
  const promptIndex = messages.findLastIndex((m) =>
    m.tags?.includes('USER_PROMPT'),
  )
  if (promptIndex <= 0) return false
  if (messages.slice(promptIndex + 1).some((m) => m.role === 'assistant'))
    return false
  const prompt = messages[promptIndex]
  if (prompt.tags?.includes(CONTINUATION_PROMPT_TAG)) return false
  const sentAt = prompt.sentAt ?? now
  // The later of the two: a runtime that predates `lastActivityAt` keeps the
  // field but does not move it, so it can be older than the last reply.
  let last = lastActivityAt
  for (let i = promptIndex - 1; i >= 0; i--) {
    const message = messages[i]
    if (message.role !== 'assistant') continue
    if (message.sentAt !== undefined)
      last = Math.max(last ?? message.sentAt, message.sentAt)
    break
  }
  return last !== undefined && sentAt - last > idleMs
}

/** The history exactly as the agent's next request sends it. */

  return dropUnansweredToolCalls(expireMessages(messages, 'agentStep'))
}

function isEphemeral(message: Message): boolean {
  return (
    message.timeToLive !== undefined ||
    message.tags?.includes('STEP_PROMPT') === true
  )
}

/** A short, stable id for a message's content (FNV-1a). */
export function messageFingerprint(message: Message): string {
  const text =
    message.role +
    '\u0000' +
    (message.role === 'tool' ? message.toolCallId : '') +
    '\u0000' +
    JSON.stringify(message.content)
  let hash = 0x811c9dc5
  for (let i = 0; i < text.length; i++) {
    hash ^= text.charCodeAt(i)
    hash = Math.imul(hash, 0x01000193)
  }
  return `${(hash >>> 0).toString(16)}:${text.length}`
}

/**
 * Where the summary's covered part ends in `messages`, or -1 when the last
 * covered message is gone (an edit removed it). Searched from its recorded
 * position first, then from the end: dropped scaffolding can move it.
 */
function anchorIndex(
  messages: Message[],
  summary: ConversationSummary,
): number {
  const expected = summary.coveredMessages - 1
  if (
    messages[expected] &&
    messageFingerprint(messages[expected]) === summary.anchor
  )
    return expected
  for (let i = Math.min(expected, messages.length - 1); i >= 0; i--) {
    if (messageFingerprint(messages[i]) === summary.anchor) return i
  }
  return -1
}

export function summaryMatchesHistory(
  messages: Message[],
  summary: ConversationSummary,
): boolean {
  return anchorIndex(messages, summary) >= 0
}

export function buildConversationSummaryMessage(
  text: string,
  now = Date.now(),
): Message {
  return {
    role: 'user',
    tags: [CONVERSATION_SUMMARY_TAG],
    sentAt: now,
    content: [
      {
        type: 'text',
        text: `<conversation_summary>\n${SUMMARY_HEADER}\n\n<historical_memory>\n${text}\n</historical_memory>\n</conversation_summary>\n\n${SUMMARY_NOTE}`,
      },
    ],
  }
}

/**
 * The history with the summary in place of what it covers, and the summary
 * re-anchored on its own message (it now covers itself). Covered messages
 * that `keep` accepts (the current turn's own user messages) stay as real
 * messages right after it: a summary made at the limit, or during a turn,
 * covers the request it is answering, and that request must reach the model
 * word for word. Its size is unset until the provider reports the next
 * request's. Null when the summary no longer matches.
 */
export function applyConversationSummary(params: {
  messages: Message[]
  summary: ConversationSummary
  keep?: (message: Message) => boolean
  now?: number
}): { messages: Message[]; summary: ConversationSummary } | null {
  const index = anchorIndex(params.messages, params.summary)
  if (index < 0) return null
  const head = buildConversationSummaryMessage(params.summary.text, params.now)
  const { contextTokens: _, ...summary } = params.summary
  const kept = params.keep
    ? params.messages.slice(0, index + 1).filter(params.keep)
    : []
  return {
    messages: [head, ...kept, ...params.messages.slice(index + 1)],
    summary: {
      ...summary,
      coveredMessages: 1,
      anchor: messageFingerprint(head),
    },
  }
}

export type SummaryStream = (params: {
  messages: Message[]
  /** Absent: the agent's own limit. */
  maxOutputTokens?: number
  signal: AbortSignal
  extraCodebuffMetadata: Record<string, string>
  onFinishReason: (reason: string) => void
  onUsageReceived: (usage: ModelUsageData) => void
}) => ReturnType<PromptAiSdkStreamFn>

export class SummaryError extends Error {
  constructor(
    readonly kind:
      | 'empty'
      | 'unrecognized'
      | 'output_limit'
      | 'provider_error'
      | 'no_history',
    message: string,
  ) {
    super(message)
  }
}

/**
 * The summary in a model's reply, or '' when the reply is not one. It starts
 * at its first heading; a preamble before it is dropped. The format's headings
 * are in English, but a model can translate them with the rest (GLM 5.3 Flash
 * did on 12% of its summaries, prod 2026-10-30, and each one was thrown away),
 * so a reply with the format's shape counts without the name.
 */
export function summaryFromReply(reply: string): string {
  const text = reply.replace(/<think>[\s\S]*?<\/think>/g, '')
  const english = text.indexOf('## Objective')
  if (english >= 0) return text.slice(english).trim()
  const sections = [...text.matchAll(/^## /gm)]
  return sections.length >= MIN_TRANSLATED_SECTIONS
    ? text.slice(sections[0].index).trim()
    : ''
}

/**
 * Asks the model for the summary. `history` must be what the agent's next
 * request sends (`summaryRequestHistory`), or the cache misses.
 */
export async function generateConversationSummary(params: {
  summaryId?: string
  history: Message[]
  system: string
  /** The provider's count when the summary was requested, if known. */
  contextTokens?: number
  targetTokens: number
  maxOutputTokens?: number
  stream: SummaryStream
  signal: AbortSignal
  onUsage?: (usage: ModelUsageData) => void
}): Promise<{ summary: ConversationSummary; finishReason?: string }> {
  const { history } = params
  const summaryId = params.summaryId ?? crypto.randomUUID()
  const createdAt = Date.now()
  const anchor = history.findLastIndex((m) => !isEphemeral(m))
  if (anchor < 0)
    throw new SummaryError('no_history', 'No history to summarize')
  let finishReason: string | undefined
  const stream = params.stream({
    messages: [
      systemMessage(params.system),
      ...history,
      // Tagged so the server's prompt classifier skips it as product text.
      userMessage({
        content: withSystemInstructionTags(
          buildCompactionSummaryPrompt(params.targetTokens),
        ),
      }),
    ],
    maxOutputTokens:
      params.maxOutputTokens !== undefined && params.maxOutputTokens > 0
        ? Math.min(SUMMARY_MAX_OUTPUT_TOKENS, params.maxOutputTokens)
        : SUMMARY_MAX_OUTPUT_TOKENS,
    signal: params.signal,
    extraCodebuffMetadata: {
      [COMPACTION_SUMMARY_METADATA_KEY]: '1',
      compaction_summary_id: summaryId,
      compaction_request_id: summaryId,
      compaction_request_group: 'summary',
    },
    onFinishReason: (reason) => {
      finishReason = reason
    },
    onUsageReceived: (usage) => params.onUsage?.(usage),
  })
  let text = ''
  for (;;) {
    const next = await stream.next()
    if (next.done) {
      if (next.value.aborted) params.signal.throwIfAborted()
      break
    }
    const chunk = next.value
    if (chunk.type === 'error')
      throw new SummaryError('provider_error', chunk.message)
    if (chunk.type === 'text') text += chunk.text
  }
  params.signal.throwIfAborted()
  if (finishReason === 'length')
    throw new SummaryError('output_limit', 'The summary hit the output limit')
  const summaryText = summaryFromReply(text)
  if (summaryText.length < MIN_SUMMARY_CHARS)
    throw new SummaryError(
      text.trim().length < MIN_SUMMARY_CHARS ? 'empty' : 'unrecognized',
      'The model returned no summary',
    )
  return {
    finishReason,
    summary: {
      id: summaryId,
      text: summaryText,
      coveredMessages: anchor + 1,
      anchor: messageFingerprint(history[anchor]),
      ...(params.contextTokens === undefined
        ? {}
        : { contextTokens: params.contextTokens }),
      createdAt,
    },
  }
}

type Entry = {
  latest?: ConversationSummary
  pending?: {
    promise: Promise<ConversationSummary | null>
    contextTokens: number
  }
}

const MAX_CONVERSATIONS = 200
const conversations = new Map<string, Entry>()

function entry(key: string): Entry {
  let value = conversations.get(key)
  if (value) {
    conversations.delete(key)
  } else {
    value = {}
  }
  conversations.set(key, value)
  while (conversations.size > MAX_CONVERSATIONS) {
    const oldest = conversations.keys().next().value
    if (oldest === undefined) break
    conversations.delete(oldest)
  }
  return value
}

/** The newest of the stored summary and the one in `state`. */
function newer(
  a: ConversationSummary | undefined,
  b: ConversationSummary | undefined,
): ConversationSummary | undefined {
  if (!a) return b
  if (!b) return a
  return b.createdAt > a.createdAt ? b : a
}

/**
 * One conversation's summaries in this process: the newest finished one and
 * the one in progress. Keyed by conversation and agent, so a summary that
 * finishes after its run ended reaches the next turn.
 */
export function conversationSummaries(key: string) {
  return {
    /** The newest summary that still matches `messages`. */
    latest(
      fromState: ConversationSummary | undefined,
      messages: Message[],
    ): ConversationSummary | undefined {
      const summary = newer(fromState, entry(key).latest)
      return summary && summaryMatchesHistory(messages, summary)
        ? summary
        : undefined
    },
    pending() {
      return entry(key).pending
    },
    /** Records a summary the history now carries (after it was applied). */
    replace(summary: ConversationSummary | undefined) {
      entry(key).latest = summary
    },
    /** Starts a summary unless one is running. */
    start(
      contextTokens: number,
      run: () => Promise<ConversationSummary | null>,
    ): Promise<ConversationSummary | null> {
      const current = entry(key)
      if (current.pending) return current.pending.promise
      const promise = run()
        .then((summary) => {
          if (summary) {
            const target = entry(key)
            target.latest = newer(target.latest, summary)
          }
          return summary
        })
        .catch(() => null)
        .finally(() => {
          const target = conversations.get(key)
          if (target?.pending?.promise === promise) target.pending = undefined
        })
      current.pending = { promise, contextTokens }
      return promise
    },
  }
}

export function resetConversationSummariesForTest(): void {
  conversations.clear()
}

     * that refusal as a turn's request would. An in-band failure (a 5xx or a
     * cut stream, an empty reply) still returns null: trying again can work.
     */
    rethrowRefusals?: boolean
    telemetry?: Record<string, string>
  },
): Promise<ConversationSummary | null> {
  const {
    logger,
    runId,
    model,
    mode,
    midTurn,
    cancelSignal,
    rethrowRefusals,
    telemetry,
    ...summaryParams
  } = params
  const started = Date.now()
  const summaryId = params.summaryId ?? crypto.randomUUID()
  let usage: ModelUsageData | undefined
  const log = (data: Record<string, unknown>, message: string) => {
    try {
      logger.info(
        {
          axiomEvent: COMPACTION_SUMMARY_EVENT,
          ...telemetry,
          summary_id: summaryId,
          agent_run_id: runId,
          model,
          mode,
          mid_turn: midTurn,
          context_token_count: params.contextTokens,
          duration_ms: Date.now() - started,
          ...(usage
            ? {
                input_tokens: usage.inputTokens,
                cached_input_tokens: usage.cachedInputTokens,
                output_tokens: usage.outputTokens,
              }
            : {}),
          ...data,
        },
        message,
      )
    } catch {
      // Logging must never fail a summary.
    }
  }
  try {
    const { summary, finishReason } = await generateConversationSummary({
      ...summaryParams,
      summaryId,
      onUsage: (received) => {
        usage = received
        summaryParams.onUsage?.(received)
      },
    })
    log(
      {
        outcome: 'completed',
        finish_reason: finishReason,
        summary_tokens: countTokens(summary.text),
      },
      'Conversation summary completed',
    )
    return summary
  } catch (error) {
    log(
      {
        outcome: cancelSignal?.aborted ? 'cancelled' : 'failed',
        error_kind:
          error instanceof SummaryError
            ? error.kind
            : isAbortError(error) || summaryParams.signal.aborted
              ? 'timeout'
              : 'provider_error',
        // Not allowlisted for Axiom; local/debug logs only.
        error: getErrorObject(error),
      },
      'Conversation summary failed',
    )
    if (cancelSignal?.aborted) throw error
    if (
      rethrowRefusals &&
      !(error instanceof SummaryError) &&
      !isAbortError(error) &&
      !summaryParams.signal.aborted
    )
      throw error
    return null
  }
}

/** Waits for `promise`, or rejects when `signal` aborts first. */
export function untilAborted<T>(
  promise: Promise<T>,
  signal: AbortSignal,
): Promise<T> {
  signal.throwIfAborted()
  return new Promise<T>((resolve, reject) => {
    const onAbort = () => reject(new AbortError())
    signal.addEventListener('abort', onAbort, { once: true })
    promise.then(resolve, reject).finally(() => {
      signal.removeEventListener('abort', onAbort)
    })
  })
}

/**
 * The reasoning settings for a summary request: the agent's own, at low
 * effort. On a high-effort model a summary at the limit thought for minutes
 * (a 150k-token summary took 304 s on DeepSeek V4.1 Flash) and one spent its
 * whole output budget on reasoning, leaving no summary. Anthropic models keep
 * their settings: changing thinking parameters there invalidates the cached
 * messages, the summary's whole saving.
 */
export function summaryReasoningOptions(
  model: string,
  reasoning: OpenRouterReasoningOptions | undefined,
): OpenRouterReasoningOptions | undefined {
  if (!reasoning || !('effort' in reasoning)) return reasoning
  if (model.startsWith('anthropic/')) return reasoning
  if (['none', 'minimal', 'low'].includes(reasoning.effort)) return reasoning
  return { ...reasoning, effort: 'low' }
}

        ...usage,
        isRoot: !agentState.parentId,
        agentId: agentState.agentId,
      })
    },
    onUsageIncomplete: params.onAgentUsageIncomplete,
    onFinishReason: (reason) => {
      finishReason = reason
    },
    template: agentTemplate,
    onCostCalculated,
  })

  const {
    fullResponse: fullResponseAfterStream,
    hadToolCallError,
    messageId,
    toolCalls,
    toolResults: newToolResults,
  } = await processStream({
    ...params,
    agentContext,
    agentState,
    agentStepId,
    agentTemplate,
    fullResponse,
    messages: agentState.messageHistory,
    repoId,
    stream,
    onCostCalculated,
    stopStream: () => streamStop.abort(),
    streamFinishReason: () => finishReason,
  }).finally(() => {
    // Inline agents can replace history (set_messages). A receipt for the old
    // request is not a baseline for a replacement summary.
    if (
      contextUsage &&
      inputHistory.every((m, i) => agentState.messageHistory[i] === m)
    ) {
      recordContextUsage({
        agentState,
        model,
        usage: contextUsage,
        estimatedInputTokens,
        estimatedOutputTokens: countTokensMessages(
          agentState.messageHistory
            .slice(inputHistory.length)
            .filter((m) => m.role === 'assistant'),
        ),
      })
    }
    agentState.contextTokenCount = estimateContextTokens({
      agentState,
      model,
      systemPrompt: system,
      toolsForTokenCount: contextTools,
    })
  })

  toolResults.push(...newToolResults)

  fullResponse = fullResponseAfterStream

  // A successful no-op is not progress. GLM can keep emitting the same to-do
  // list (even while acknowledging the loop in prose) until the general step
  // budget is exhausted. Give it a chance to recover, then fail explicitly.
  if (
    !hadToolCallError &&
    toolCalls.length > 0 &&
    toolCalls.every((call) => call.toolName === 'write_todos')
  ) {
    const consecutive = trailingIdenticalTodoCalls(agentState.messageHistory)
    if (consecutive >= TODO_LOOP_RECOVERY_THRESHOLD) {
      const stopped = consecutive >= TODO_LOOP_STOP_THRESHOLD
      logger.warn(
        {
          metric: stopped ? 'todo_loop_stopped' : 'todo_loop_recovery',
          consecutive,
          model: agentTemplate.model,
          agentId: agentTemplate.id,
          fileEditToolsOffered: hasFileEditTool(agentTemplate.toolNames),
          userId,
          runId: agentState.runId,
        },
        stopped
          ? 'Stopping a turn stuck repeating an unchanged to-do list'
          : 'Model is repeating an unchanged to-do list; adding recovery guidance',
      )
      if (stopped) {
        throw new TodoLoopError()
      }
      agentState.messageHistory.push(
        userMessage({
          content: withSystemTags(
            todoLoopRecoveryMessage(agentTemplate.toolNames),
          ),
          tags: [TODO_LOOP_RECOVERY_TAG],
        }),
      )
    }
  }

  // Measured first, and deliberately before the guards below so that a turn
  // a guard stops still leaves this row behind. The loops users photograph
  // are a model retrying the SAME file, varying the edit a little or reading
  // the file in between; the identical-edit guard needs byte-identical
  // arguments and an unbroken run of edit-only steps, so most of them score
  // below its threshold. Over 2026-10-09..14 the only per-rejection signal
  // Axiom had (write_file's "same as the old content", one `logger.info`)
  // showed ~35% of same-file loops on guard-capable clients producing no guard
  // row. This event makes the real rate a division: `consecutive` is the loose
  // count, `strictConsecutive` and `guardEligible` are what the identical-edit
  // guard saw at the same moment. The same loose count drives the varied-edit
  // guard below.
  const nearMiss = toolCalls.some((call) => isFileEditLoopTool(call.toolName))
    ? trailingRejectedEditsToSameFiles(
        agentState.messageHistory,
        FILE_EDIT_LOOP_VARIED_STOP_THRESHOLD,
      )
    : null
  // Once per episode as it crosses each of the guards' thresholds, so a long
  // loop is a few rows rather than one per step.
  if (
    nearMiss &&
    (nearMiss.consecutive === FILE_EDIT_LOOP_RECOVERY_THRESHOLD ||
      nearMiss.consecutive === FILE_EDIT_LOOP_STOP_THRESHOLD)
  ) {
    logger.info(
      {
        axiomEvent: FILE_EDIT_LOOP_EVENT,
        metric:
          nearMiss.consecutive === FILE_EDIT_LOOP_STOP_THRESHOLD
            ? 'file_edit_loop_same_file_stop_depth'
            : 'file_edit_loop_same_file',
        consecutive: nearMiss.consecutive,
        strictConsecutive: trailingIdenticalRejectedEdits(
          agentState.messageHistory,
        ),
        guardEligible:
          !hadToolCallError &&
          toolCalls.every((call) => isFileEditLoopTool(call.toolName)),
        varied: nearMiss.varied,
        interleaved: nearMiss.interleaved,
        mixedStep: nearMiss.mixedStep,
        toolName: nearMiss.toolName,
        model: agentTemplate.model,
        agentId: agentTemplate.id,
        runId: agentState.runId,
      },
      'Trailing steps re-sent a rejected edit to the same file',
    )
  }
  let fileEditLoopGuided = false

  // The same shape with file edits: an identical write_file/str_replace is
  // answered with the same rejection every step, each result forces another
  // step, and the model re-sends the edit until the step budget is gone.
  if (
    !hadToolCallError &&
    toolCalls.length > 0 &&
    toolCalls.every((call) => isFileEditLoopTool(call.toolName))
  ) {
    const consecutive = trailingIdenticalRejectedEdits(
      agentState.messageHistory,
    )
    if (consecutive >= FILE_EDIT_LOOP_RECOVERY_THRESHOLD) {
      const stopped = consecutive >= FILE_EDIT_LOOP_STOP_THRESHOLD
      logger.warn(
        {
          metric: stopped
            ? 'file_edit_loop_stopped'
            : 'file_edit_loop_recovery',
          consecutive,
          model: agentTemplate.model,
          agentId: agentTemplate.id,
          userId,
          runId: agentState.runId,
        },
        stopped
          ? 'Stopping a turn stuck re-sending the same rejected file edit'
          : 'Model is re-sending the same rejected file edit; adding recovery guidance',
      )
      if (stopped) {
        throw new FileEditLoopError()
      }
      agentState.messageHistory.push(
        userMessage({
          content: withSystemTags(FILE_EDIT_LOOP_RECOVERY_MESSAGE),
          tags: [FILE_EDIT_LOOP_RECOVERY_TAG],
        }),
      )
      fileEditLoopGuided = true
    }
  }

  // The loop the identical-edit guard misses: rejected edits to the same files
  // that vary a little each time (a stale old string, then an old string equal
  // to the new one, then another), or with a read between them. No edit has
  // landed for the whole streak, so it is not progress however it varies.
  if (nearMiss && nearMiss.consecutive >= FILE_EDIT_LOOP_VARIED_STOP_THRESHOLD) {
    logger.warn(
      {
        metric: 'file_edit_loop_stopped',
        variedGuard: true,
        consecutive: nearMiss.consecutive,
        varied: nearMiss.varied,
        interleaved: nearMiss.interleaved,
        mixedStep: nearMiss.mixedStep,
        model: agentTemplate.model,
        agentId: agentTemplate.id,
        userId,
        runId: agentState.runId,
      },
      'Stopping a turn whose file edits to the same files were all rejected',
    )
    throw new FileEditLoopError()
  }
  if (
    nearMiss?.consecutive === FILE_EDIT_LOOP_STOP_THRESHOLD &&
    !fileEditLoopGuided
  ) {
    logger.warn(
      {
        metric: 'file_edit_loop_recovery',
        variedGuard: true,
        consecutive: nearMiss.consecutive,
        model: agentTemplate.model,
        agentId: agentTemplate.id,
        userId,
        runId: agentState.runId,
      },
      'File edits to the same files keep being rejected; adding recovery guidance',
    )
    agentState.messageHistory.push(
      userMessage({
        content: withSystemTags(FILE_EDIT_LOOP_VARIED_RECOVERY_MESSAGE),
        tags: [FILE_EDIT_LOOP_RECOVERY_TAG],
      }),
    )
  }

  agentState.messageHistory = expireMessages(
    agentState.messageHistory,
    'agentStep',
  )

  const hasNoToolResults =
    toolCalls.filter(
      (call) => !TOOLS_WHICH_WONT_FORCE_NEXT_STEP.includes(call.toolName),
    ).length === 0 &&
    toolResults.filter(
      (result) => !TOOLS_WHICH_WONT_FORCE_NEXT_STEP.includes(result.toolName),
    ).length === 0 &&
    !hadToolCallError // Tool call errors should also force another step so the agent can retry

  const hasTaskCompleted = toolCalls.some(
    (call) =>
      call.toolName === 'task_completed' || call.toolName === 'end_turn',
  )

  // If the response is only <think>...</think> scaffolding (including orphan
  // </think> closes that native-reasoning providers sometimes leak into
  // content), the model was just thinking and should continue rather than end.
  const isThinkOnly = hasNoToolResults && isThinkOnlyResponse(fullResponse)

  // If the agent has the task_completed tool, it must be called to end its turn.
  const requiresExplicitCompletion =
    agentTemplate.toolNames.includes('task_completed')

  let shouldEndTurn: boolean
  if (requiresExplicitCompletion) {
    // For models requiring explicit completion, only end turn when:
    // - task_completed is called, OR
    // - end_turn is called (backward compatibility)
    shouldEndTurn = !hadToolCallError && hasTaskCompleted
  } else {
    // For other models, also end turn when there are no tool calls
    // Exception: if the response is only <think> tags, continue the turn
    shouldEndTurn =
      !hadToolCallError &&
      (hasTaskCompleted || (hasNoToolResults && !isThinkOnly))
  }

  // The answer to a Gravity report nudge ends the turn once its reports run;
  // see util/gravity-report-nudge.ts.
  if (
    !shouldEndTurn &&
    !hadToolCallError &&
    endsOnGravityReport(toolCalls, agentState.messageHistory)
  ) {
    shouldEndTurn = true
  }

  // Ending on followup cards while the agent's own to-do list is part done
  // stops a task the user asked to have finished (CodebuffAI/freebuff#1434).
  // Bounded per user prompt; see util/followup-todo-nudge.ts. An explicit
  // task_completed/end_turn is left alone.
  if (
    shouldEndTurn &&
    !hasTaskCompleted &&
    toolCalls.some((call) => call.toolName === 'suggest_followups')
  ) {
    const decision = decideFollowupTodoNudge(agentState.messageHistory)
    if (decision) {
      logger.info(
        {
          metric: 'followups_with_open_todos',
          action: decision.action,
          reason: decision.action === 'end' ? decision.reason : undefined,
          openTodos: decision.openTodos.length,
          nudgesSoFar: decision.nudgesSoFar,
          model: agentTemplate.model,
          agentId: agentTemplate.id,
          userId,
          runId: agentState.runId,
        },
        decision.action === 'nudge'
          ? 'Turn ended on followups with unfinished to-dos; continuing'
          : 'Turn ended on followups with unfinished to-dos; nudge limit reached',
      )
      if (decision.action === 'nudge') {
        agentState.messageHistory.push(
          userMessage({
            content: withSystemTags(decision.message),
            tags: [FOLLOWUP_TODO_NUDGE_TAG],
          }),
        )
        shouldEndTurn = false
      }
    }
  }

  // Cards before the answer; see util/suggest-prompts-before-answer.ts. After
  // the to-do nudge on purpose: an unfinished list sends the agent back to
  // work, which beats asking it for an answer.
  if (
    shouldEndTurn &&
    !hasTaskCompleted &&
    endsOnSuggestPromptsBeforeAnswer(
      toolCalls,
      fullResponse,
      agentState.messageHistory,
    )
  ) {
    logger.info({ metric: 'suggest_prompts_before_answer' }, 'Continuing')
    agentState.messageHistory.push(
      userMessage({
        content: withSystemTags(SUGGEST_PROMPTS_BEFORE_ANSWER_MESSAGE),
        tags: [SUGGEST_PROMPTS_BEFORE_ANSWER_TAG],
      }),
    )
    shouldEndTurn = false
  }

  // A turn that integrated a gravity_index recommendation but never reported
  // it gets one step to do so, with the exact search_id and slug; see
  // util/gravity-report-nudge.ts. Once per user prompt. Runs even after
  // task_completed: the conversion is lost if the turn ends here.
  if (shouldEndTurn && agentTemplate.toolNames.includes('gravity_index')) {
    const nudge = decideGravityReportNudge(agentState.messageHistory)
    if (nudge) {
      logger.info(
        {
          axiomEvent: GRAVITY_REPORT_NUDGE_EVENT,
          pending: nudge.candidates.length,
          editsSinceRecommendation: nudge.editsSinceRecommendation,
          model: agentTemplate.model,
          agentId: agentTemplate.id,
          runId: agentState.runId,
        },
        'Turn ending with an unreported Gravity integration; reminding',
      )
      agentState.messageHistory.push(
        userMessage({
          content: withSystemTags(nudge.message),
          tags: [GRAVITY_REPORT_NUDGE_TAG],
        }),
      )
      shouldEndTurn = false
    }
  }

  // A to-do list the model never ticks off reads 0/N until the turn's last
  // step; see util/todo-progress-reminder.ts. Once per list, mid-turn only.
  if (!shouldEndTurn) {
    const reminder = decideTodoProgressReminder(agentState.messageHistory)
    if (reminder) {
      logger.info(
        {
          axiomEvent: TODO_PROGRESS_REMINDER_EVENT,
          done: reminder.done,
          total: reminder.total,
          callsSinceUpdate: reminder.callsSinceUpdate,
          model: agentTemplate.model,
          agentId: agentTemplate.id,
          runId: agentState.runId,
        },
        'To-do list not updated for several tool calls; reminding',
      )
      agentState.messageHistory.push(
        userMessage({
          content: withSystemTags(reminder.message),
          tags: [TODO_PROGRESS_REMINDER_TAG],
        }),
      )
    }
  }

  agentState = {
    ...agentState,
    stepsRemaining: agentState.stepsRemaining - 1,
    agentContext,
  }

  // Capture the assistant response and tool results added during this step
  params.traceWriter?.recordStep({
    agentId: agentState.agentId,
    agentType: String(agentType),
    runId: agentState.runId,
    userInputId,
    step: iterationNum,
    system,
    messages: agentState.messageHistory,
  })

  logger.debug(
    {
      iteration: iterationNum,
      agentId: agentState.agentId,
      model,
      prompt,
      shouldEndTurn,
      duration: Date.now() - startTime,
      fullResponse,
      // Summarize instead of logging the full message history: logging it
      // every step bloats log files quadratically over the course of a chat.
      messageCount: agentState.messageHistory.length,
      toolCalls,
      toolResults,
      stepCreditsUsed,
    },
    `End agent ${agentType} step ${iterationNum} (${userInputId}${prompt ? ` - Prompt: ${prompt.slice(0, 20)}` : ''})`,
  )

  return {
    agentState,
    fullResponse,
    shouldEndTurn,
    messageId,
    nResponses: undefined,
    usedWorkTools: toolCalls.some((call) =>
      ['run_terminal_command', 'write_file', 'str_replace'].includes(
        call.toolName,
      ),
    ),
    stepToolCalls: toolCalls,
    stepToolResults: toolResults,
  }
}

/**
 * Runs the agent loop.
 *
 * IMPORTANT: This function mutates `params.agentState` in place throughout the
 * run (not just at return time). Fields like `messageHistory`, `systemPrompt`,
 * `toolDefinitions`, `creditsUsed`, and `output` are updated as work progresses
 * so that callers holding a reference to the same object (e.g. the SDK's
 * `sessionState.mainAgentState`) see in-progress work immediately — which
 * matters when an error is thrown mid-run and the normal return path is
 * skipped.
 */
export async function loopAgentSteps(
  params: {
    addAgentStep: AddAgentStepFn
    agentState: AgentState
    agentType: string
    clearUserPromptMessagesAfterResponse?: boolean
    clientSessionId: string
    content?: Array<TextPart | ImagePart>
    costMode?: string
    fileContext: ProjectFileContext
    finishAgentRun: FinishAgentRunFn
    localAgentTemplates: Record<string, AgentTemplate>
    logger: Logger
    parentSystemPrompt?: string
    parentTools?: ToolSet
    prompt: string | undefined
    signal: AbortSignal
    /** Host-enforced wall-clock deadline in Unix milliseconds. */
    deadlineAt?: number
    /** Optional steering hook. Drained at each step boundary (after a step's LLM
     * call + tools complete, before the next one). Any returned messages are appended
     * to the message history as user prompts and keep the turn going, letting a
     * host "steer" a running agent without aborting or losing the current step. */
    drainSteeringMessages?: DrainSteeringMessages
    /** The prompt continues work the user already asked for (a drain resume,
     * a mission pass): it is tagged so it never counts as an idle return. */
    promptIsContinuation?: boolean
    spawnParams: Record<string, any> | undefined
    startAgentRun: StartAgentRunFn
    userId: string | undefined
    userInputId: string
    agentTemplate?: AgentTemplate
  } & ParamsExcluding<typeof additionalToolDefinitions, 'agentTemplate'> &
    ParamsExcluding<
      typeof runProgrammaticStep,
      | 'agentState'
      | 'onCostCalculated'
      | 'prompt'
      | 'runId'
      | 'stepNumber'
      | 'stepsComplete'
      | 'system'
      | 'template'
      | 'toolCallParams'
      | 'tools'
    > &
    ParamsExcluding<typeof getAgentTemplate, 'agentId'> &
    ParamsExcluding<
      typeof getAgentPrompt,
      | 'agentTemplate'
      | 'promptType'
      | 'agentTemplates'
      | 'additionalToolDefinitions'
    > &
    ParamsExcluding<
      typeof getMCPToolData,
      'toolNames' | 'mcpServers' | 'writeTo'
    > &
    ParamsExcluding<StartAgentRunFn, 'agentId' | 'ancestorRunIds'> &
    ParamsExcluding<
      FinishAgentRunFn,
      'runId' | 'status' | 'totalSteps' | 'directCredits' | 'totalCredits'
    > &
    ParamsExcluding<
      typeof runAgentStep,
      | 'additionalToolDefinitions'
      | 'agentState'
      | 'agentTemplate'
      | 'prompt'
      | 'runId'
      | 'spawnParams'
      | 'system'
      | 'tools'
    > &
    ParamsExcluding<
      AddAgentStepFn,
      | 'agentRunId'
      | 'stepNumber'
      | 'credits'
      | 'childRunIds'
      | 'messageId'
      | 'status'
      | 'startTime'
    >,
): Promise<{
  agentState: AgentState
  output: AgentOutput
}> {
  const loopStarted = Date.now()
  let agentTemplate = params.agentTemplate
  if (!agentTemplate) {
    agentTemplate =
      (await getAgentTemplate({
        ...params,
        agentId: params.agentType,
      })) ?? undefined
  }
  if (!agentTemplate) {
    throw new Error(`Agent template not found for type: ${params.agentType}`)
  }

  // The context pruner is a programmatic no-model-call agent spawned before
  // every main-agent step. Recording its runs in the ledger cost three awaited
  // web-API round trips (start/step/finish) per main-agent step, so it mints a
  // local run id and skips run tracking entirely. Matches bundled and
  // publisher-qualified ids ('context-pruner', 'codebuff/context-pruner@1.0.0')
  // by agent name, never by substring: an untracked id is refused by the web
  // API, so any model-calling agent whose id merely CONTAINED the name (e.g.
  // 'context-pruner-test-agent') failed its first LLM call with
  // "runId Not Found".
  if (parseAgentId(agentTemplate.id).agentId === 'context-pruner') {
    params = {
      ...params,
      startAgentRun: async () =>
        `${UNTRACKED_RUN_ID_PREFIX}${crypto.randomUUID()}`,
      addAgentStep: async () => null,
      finishAgentRun: async () => {},
    }
  }

  const {
    addAgentStep,
    agentState: initialAgentState,
    agentType,
    clearUserPromptMessagesAfterResponse = true,
    clientSessionId,
    content,
    fileContext,
    finishAgentRun,
    localAgentTemplates,
    logger,
    parentSystemPrompt,
    parentTools,
    prompt,
    signal,
    spawnParams,
    startAgentRun,
    userId,
    userInputId,
    clientEnv,
    ciEnv,
  } = params

  const offersProjectProfile = shouldOfferProjectProfileTool({
    costMode: params.costMode,
    agentId: agentTemplate.id,
    isRoot: !initialAgentState.parentId,
    toolNames: agentTemplate.toolNames,
    getProjectProfile: params.getProjectProfile,
    reportProjectProfile: params.reportProjectProfile,
  })
  if (offersProjectProfile) {
    agentTemplate = {
      ...agentTemplate,
      toolNames: [...agentTemplate.toolNames, PROJECT_PROFILE_TOOL_NAME],
    }
  }

  if (signal.aborted) {
    return {
      agentState: initialAgentState,
      output: {
        type: 'error',
        message: 'Run cancelled by user',
      },
    }
  }

  const runId = await startAgentRun({
    ...params,
    agentId: agentTemplate.id,
    ancestorRunIds: initialAgentState.ancestorRunIds,
  })
  if (!runId) {
    // registration ends on the run's signal; an abort landing there is a cancel, not a failure
    if (signal.aborted) {
      return {
        agentState: initialAgentState,
        output: {
          type: 'error',
          message: 'Run cancelled by user',
        },
      }
    }
    throw new Error('Failed to start agent run')
  }
  initialAgentState.runId = runId

  let cachedAdditionalToolDefinitions: CustomToolDefinitions | undefined
  // Use parent's tools for prompt caching when inheritParentSystemPrompt is true
  const useParentTools =
    agentTemplate.inheritParentSystemPrompt && parentTools !== undefined

  const manualCompaction = isCompactCommandPrompt(prompt)
  // Initialize message history with user prompt and instructions on first iteration
  const instructionsPrompt = await getAgentPrompt({
    ...params,
    agentTemplate,
    promptType: { type: 'instructionsPrompt' },
    agentTemplates: localAgentTemplates,
    useParentTools,
    additionalToolDefinitions: async () => {
      if (!cachedAdditionalToolDefinitions) {
        cachedAdditionalToolDefinitions = await additionalToolDefinitions({
          ...params,
          agentTemplate,
        })
      }
      return cachedAdditionalToolDefinitions
    },
  })

  // Build the initial message history with user prompt and instructions
  // Generate system prompt once, using parent's if inheritParentSystemPrompt is true
  let system: string
  if (manualCompaction && initialAgentState.systemPrompt) {
    system = initialAgentState.systemPrompt
  } else if (agentTemplate.inheritParentSystemPrompt && parentSystemPrompt) {
    system = parentSystemPrompt
  } else {
    const systemPrompt = await getAgentPrompt({
      ...params,
      agentTemplate,
      promptType: { type: 'systemPrompt' },
      agentTemplates: localAgentTemplates,
      additionalToolDefinitions: async () => {
        if (!cachedAdditionalToolDefinitions) {
          cachedAdditionalToolDefinitions = await additionalToolDefinitions({
            ...params,
            agentTemplate,
          })
        }
        return cachedAdditionalToolDefinitions
      },
    })
    system = systemPrompt ?? ''
  }

  // Build agent tools (agents as direct tool calls) for non-inherited tools
  const agentTools = useParentTools
    ? {}
    : await buildAgentToolSet({
        ...params,
        spawnableAgents: modelSpawnableAgents(agentTemplate),
        agentTemplates: localAgentTemplates,
      })

  const tools = useParentTools
    ? parentTools
    : await getToolSet({
        toolNames: agentTemplate.toolNames,
        windowedFileReads: agentTemplate.windowedFileReads === true,
        suppressCommitAttribution:
          agentTemplate.suppressCommitAttribution === true,
        additionalToolDefinitions: async () => {
          if (!cachedAdditionalToolDefinitions) {
            cachedAdditionalToolDefinitions = await additionalToolDefinitions({
              ...params,
              agentTemplate,
            })
          }
          return cachedAdditionalToolDefinitions
        },
        agentTools,
        skills: fileContext.skills ?? {},
      })

  const hasUserMessage =
    !manualCompaction &&
    Boolean(
      prompt ||
      (spawnParams && Object.keys(spawnParams).length > 0) ||
      (content && content.length > 0),
    )

  // Named so that a size refusal can take it back out of the history.
  const userPromptMessage: Message | undefined = hasUserMessage
    ? {
        // Actual user message!
        role: 'user',
        content: buildUserMessageContent(prompt, spawnParams, content),
        tags: params.promptIsContinuation
          ? ['USER_PROMPT', CONTINUATION_PROMPT_TAG]
          : ['USER_PROMPT'],
        sentAt: Date.now(),

        // James: Deprecate the below, only use tags, which are not prescriptive.
        keepDuringTruncation: true,
      }
    : undefined

  const initialMessages = buildArray<Message>(
    ...initialAgentState.messageHistory,

    hasUserMessage && [
      userPromptMessage,
      prompt &&
        prompt in additionalSystemPrompts &&
        userMessage(
          withSystemInstructionTags(
            additionalSystemPrompts[
              prompt as keyof typeof additionalSystemPrompts
            ],
          ),
        ),
      ,
    ],

    instructionsPrompt &&
      userMessage({
        content: instructionsPrompt,
        tags: ['INSTRUCTIONS_PROMPT'],

        // James: Deprecate the below, only use tags, which are not prescriptive.
        keepLastTags: ['INSTRUCTIONS_PROMPT'],
      }),
  )

  // Convert tools to a serializable format for context-pruner token counting
  // Tool definitions live in agent state (persisted, snapshotted, shipped over
  // the wire), so every inputSchema must be plain JSON Schema: Zod instances
  // are cyclic and break any JSON.stringify over the state.
  const toolDefinitions =
    manualCompaction && initialAgentState.toolDefinitions
      ? initialAgentState.toolDefinitions
      : mapValues(tools, (tool) => ({
          description:
            typeof tool.description === 'string' ? tool.description : undefined,
          inputSchema: toTokenCountInputSchema(tool.inputSchema) ?? {},
        }))

  const additionalToolDefinitionsWithCache = async () => {
    if (!cachedAdditionalToolDefinitions) {
      cachedAdditionalToolDefinitions = await additionalToolDefinitions({
        ...params,
        agentTemplate,
      })
    }
    return cachedAdditionalToolDefinitions
  }

  // Mutate initialAgentState so that in-progress work propagates back to the
  // caller's shared reference (e.g. SDK's sessionState.mainAgentState) even if
  // an error is thrown before we return.
  // What earlier turns left; a USER_PROMPT not in it is this turn's own.
  const earlierMessages = new Set(initialAgentState.messageHistory)
  initialAgentState.messageHistory = initialMessages
  initialAgentState.systemPrompt = system
  initialAgentState.toolDefinitions = toolDefinitions
  let currentAgentState: AgentState = initialAgentState
  if (
    !sameContextModel(
      currentAgentState.contextTokenBaseline?.model,
      agentTemplate.model,
    )
  ) {
    currentAgentState.contextTokenBaseline = undefined
  }

  // Estimate the wire schemas, not Zod's internal object graph.
  const toolsForTokenCount = toolsForContextEstimate(toolDefinitions)

  // Keep the persisted/displayed count aligned after cancellation and TTL edits.
  // Uses the latest receipt plus cheap deltas; never re-tokenizes the history.
  const recountContextTokensForTurnEnd = () => {
    currentAgentState.contextTokenCount = recountContextTokens({
      agentState: currentAgentState,
      model: agentTemplate.model,
      systemPrompt: system,
      toolsForTokenCount,
    })
  }

  let shouldEndTurn = false
  let hasRetriedOutputSchema = false
  let hasCheckedCompletion = false
  // Only the first optional check may fall back to a completed answer. Once
  // it produces text or invokes ANY tool, its findings/effects must be kept.
  let completionCheckpoint: Message[] | undefined
  let usedWorkTools = false
  let deadlineReminderLevel = 0
  let currentPrompt = prompt
  let currentParams = spawnParams
  let totalSteps = 0
  let llmStepNumber = 0
  // Root turns only. A subagent keeps its own step ceiling, and most are
  // read-only by design. A programmatic agent's work happens outside the steps
  // recorded here, so the count would miss it.
  const stopsOnNoProgress =
    !initialAgentState.parentId && !agentTemplate.handleSteps
  const turnProgress = createTurnProgress()
  let nResponses: string[] | undefined = undefined
  let compactionWindow: CompactionWindow | undefined
  let firstOutputLogged = false
  const idleMs = summaryIdleMs(agentTemplate.model)
  const idleAtStart = isIdleReturn(
    currentAgentState.messageHistory,
    idleMs,
    undefined,
    currentAgentState.lastActivityAt,
  )

  const compactionVersion = agentTemplate.compactContext
    ? SUMMARY_COMPACTION_VERSION
    : 'not-enabled'
  const inheritedMetadata = clearCompactionRequestMetadata(
    params.extraCodebuffMetadata,
  )
  const runMetadata = {
    ...inheritedMetadata,
    ...(!initialAgentState.parentId &&
    initialAgentState.contextTokenBaseline?.model === agentTemplate.model
      ? {
          compaction_root_start_context_tokens: String(
            initialAgentState.contextTokenBaseline.tokens,
          ),
        }
      : {}),
    compaction_version: compactionVersion,
    compaction_root_model: initialAgentState.parentId
      ? (inheritedMetadata.compaction_root_model ?? 'unknown')
      : agentTemplate.model,
    compaction_root_version: initialAgentState.parentId
      ? (inheritedMetadata.compaction_root_version ?? 'unknown')
      : compactionVersion,
    compaction_root_run_id: initialAgentState.parentId
      ? (inheritedMetadata.compaction_root_run_id ?? runId)
      : runId,
  }
  // Programmatic spawns also need the root cohort, even before an LLM step.
  params = { ...params, extraCodebuffMetadata: runMetadata }
  const telemetry = {
    compaction_version: compactionVersion,
    compaction_root_version: runMetadata.compaction_root_version,
    compaction_root_run_id: runMetadata.compaction_root_run_id,
    trace_session_id: params.extraCodebuffMetadata?.trace_session_id ?? runId,
    // Lets Axiom count resumed turns and confirm none compacts as idle.
    ...(params.promptIsContinuation ? { continuation_prompt: 'true' } : {}),
  }

  const compactionPolicy =
    typeof agentTemplate.compactContext === 'object'
      ? agentTemplate.compactContext
      : {}
  const maxContextLength =
    compactionPolicy.maxContextLength ??
    contextPrunerBudgetForModel(agentTemplate.model)
  // Keyed by conversation, so a summary that finishes after this run ended
  // reaches the next turn of the same thread.
  const summaries = conversationSummaries(
    `${params.extraCodebuffMetadata?.trace_session_id ?? runId}:${initialAgentState.agentId}`,
  )
  let idleChecked = false
  /** The context size the provider reported for this run's latest step:
   *  input plus output tokens. Summary decisions use only this. */
  let stepUsageTokens: number | undefined
  /** The last size any step of this run reported: a cancelled or failed
   *  step reports none, and its turn-end summary is decided on this. */
  let lastStepUsageTokens: number | undefined
  /** Mid-turn summaries start in the band below the compaction threshold. */
  const bandStart = summaryBandStart(maxContextLength)
  /** Set when compaction left the run over its hard budget. */
  let stoppedOverBudget = false
  // The server accepts a request only while its run is open, so the run
  // finishes after the summaries it started.
  const runSummaries: Promise<unknown>[] = []

  const summarize = (options: {
    summaryId?: string
    history: Message[]
    contextTokens: number | undefined
    mode: 'background' | 'foreground'
    midTurn: boolean
    /** Only a foreground summary knows why it was made. */
    trigger?: CompactionTrigger | 'manual'
  }) =>
    summarizeAndLog({
      summaryId: options.summaryId,
      history: options.history,
      system,
      contextTokens: options.contextTokens,
      targetTokens: summaryTargetTokens(maxContextLength),
      maxOutputTokens: compactionPolicy.maxOutputTokens,
      mode: options.mode,
      midTurn: options.midTurn,
      // No time limit: a summary the run waits for is its only way under the
      // budget, and stopping a slow one ended the run (base3 eval, 10-28).
      // A background summary is not cancelled with its run: the next turn
      // can still use it. A dead stream ends on the SDK's idle timeout.
      signal:
        options.mode === 'foreground' ? signal : new AbortController().signal,
      cancelSignal: options.mode === 'foreground' ? signal : undefined,
      // A manual compact's summary is the run's only request, so its refusal
      // (a session that ended, a policy refusal) is the run's outcome: "try
      // again" would only be refused again (prod 11-02..04).
      rethrowRefusals: options.trigger === 'manual',
      logger,
      runId,
      model: agentTemplate.model,
      telemetry: options.trigger
        ? { ...telemetry, trigger_reason: options.trigger }
        : telemetry,
      stream: (request) =>
        getAgentStreamFromTemplate({
          ...params,
          agentId: initialAgentState.parentId
            ? initialAgentState.agentId
            : undefined,
          template: {
            ...agentTemplate,
            reasoningOptions: summaryReasoningOptions(
              agentTemplate.model,
              agentTemplate.reasoningOptions,
            ),
          },
          runId,
          messages: request.messages,
          tools,
          includeCacheControl: supportsCacheControl(agentTemplate.model),
          maxOutputTokens: request.maxOutputTokens,
          signal: request.signal,
          extraCodebuffMetadata: {
            ...runMetadata,
            llm_step_number: String(llmStepNumber + 1),
            ...request.extraCodebuffMetadata,
            ...(options.trigger ? { compaction_trigger: options.trigger } : {}),
          },
          onFinishReason: request.onFinishReason,
          onUsageReceived: request.onUsageReceived,
          onCostCalculated: async () => {},
        }),
    })

  /** Starts a background summary when one is due: after a step, once the
   *  context is in the near-limit band; at a turn's end, when it grew enough
   *  since the last one. The cache is warm here: a step or a turn just ended. */
  const startSummaryIfDue = (
    midTurn: boolean,
    fixedTokenCount: number,
    contextTokens = stepUsageTokens,
  ) => {
    if (!agentTemplate.compactContext || summaries.pending()) return
    const history = currentAgentState.messageHistory
    // A window too small for a summary to clear room (a large fixed prompt
    // in a small BYOK window) waits for the hard budget instead.
    if (
      !automaticCompactionIsWorthwhile({
        messages: history,
        maxContextLength,
        thresholdTokens: compactionThreshold(maxContextLength),
        fixedTokenCount,
        maxOutputTokens: compactionPolicy.maxOutputTokens,
      })
    )
      return
    if (contextTokens === undefined) return
    const latest = summaries.latest(
      currentAgentState.compactionSummary,
      history,
    )
    // Swapped in and not sized by a response yet: growth has no start.
    if (latest && latest.contextTokens === undefined) return
    const due = midTurn ? midTurnSummaryIsDue : summaryIsDue
    if (
      !due({
        contextTokens,
        maxContextLength,
        latestContextTokens: latest?.contextTokens,
      })
    )
      return
    const requestHistory = summaryRequestHistory(history)
    const done = summaries.start(contextTokens, () =>
      summarize({
        history: requestHistory,
        contextTokens,
        mode: 'background',
        midTurn,
      }),
    )
    runSummaries.push(
      done.then((summary) => {
        if (!summary || initialAgentState.parentId) return
        try {
          params.onCompactionSummary?.(summary)
        } catch (error) {
          logger.warn(
            { error: getErrorObject(error) },
            'Saving the conversation summary failed',
          )
        }
      }),
    )
  }

  /**
   * A cancelled or failed turn still leaves the user a history to return to,
   * and the last step's request warmed its cache: summarize it as a finished
   * turn would be. No step size is reported for the step that stopped, so it
   * is decided on the last one this run had.
   */
  const startSummaryAfterTurnStopped = () => {
    if (!agentTemplate.compactContext || manualCompaction) return
    try {
      startSummaryIfDue(
        false,
        countTokens(system) + countTokensJson(toolsForTokenCount),
        lastStepUsageTokens,
      )
    } catch (error) {
      logger.warn(
        { error: getErrorObject(error) },
        'Starting the summary of a stopped turn failed',
      )
    }
  }

  /** The server accepts a request only while its run is open: the run
   *  finishes after its background work, without holding up the caller. */
  const finishAfter = async (
    background: Promise<unknown>[],
    finish: () => Promise<unknown>,
  ) => {
    if (background.length) void Promise.allSettled(background).then(finish)
    else await finish()
  }

  /**
   * Replaces the covered history with the newest summary. Waits for a summary
   * in progress. At the context limit, or on a manual request, with no usable
   * summary it makes one now (the cache is warm). After an idle gap it never
   * does: the cache is cold, and the full history costs the same.
   */
  const compactWithSummary = async (options: {
    trigger: CompactionTrigger | 'manual'
    thresholdTokens: number
    /** The provider's last count, when known. Reported, never decided on. */
    preTokens: number | undefined
  }): Promise<boolean> => {
    const { trigger, thresholdTokens, preTokens } = options
    const started = Date.now()
    const compactionId = crypto.randomUUID()
    let outcome: 'applied' | 'noop' | 'failed' | 'cancelled' = 'noop'
    let readySummary = false
    let waitedForPending = false
    let generatedForeground = false
    let summaryId: string | undefined
    let summaryContextTokens: number | undefined
    const isRoot = !initialAgentState.parentId
    const before = currentAgentState.messageHistory
    // This turn's own request (and steering) stays a real message.
    const keep = (message: Message) =>
      message.tags?.includes('USER_PROMPT') === true &&
      !earlierMessages.has(message)
    // A summary is worth using when it replaces messages. Whether the result
    // is small enough is not guessed: the next response's count says, and a
    // context still over the limit then gets a fresh summary.
    const usable = (
      applied: ReturnType<typeof applyConversationSummary>,
    ): applied is NonNullable<typeof applied> =>
      applied !== null && applied.messages.length < before.length
    let statusShown = false
    const showStatus = () => {
      if (statusShown || !isRoot) return
      statusShown = true
      params.onCompactionStart?.({ trigger })
    }
    try {
      let summary = summaries.latest(
        currentAgentState.compactionSummary,
        before,
      )
      let applied = summary
        ? applyConversationSummary({ messages: before, summary, keep })
        : null
      readySummary = usable(applied)
      // A ready summary is used at once, even when a newer one is in
      // progress: waiting for it cost ~1 min a swap in the base3 eval
      // (10-28). The newer one replaces it at the next swap, when it adds
      // something. The run waits for it with nothing ready, or at the limit
      // when only it is in the near-limit band: mid-turn summaries start
      // only there, so the ready one is from an earlier turn's end and would
      // drop everything since.
      const pending = summaries.pending()
      const pendingIsFresher =
        pending !== undefined &&
        (trigger === 'context_limit' ||
          trigger === 'context_limit_and_cache_expiry') &&
        pending.contextTokens >= bandStart &&
        (summary?.contextTokens === undefined ||
          summary.contextTokens < bandStart)
      if (pending && (!usable(applied) || pendingIsFresher)) {
        waitedForPending = true
        showStatus()
        const completed = await untilAborted(pending.promise, signal)
        if (!completed) outcome = 'failed'
        summary = summaries.latest(currentAgentState.compactionSummary, before)
        applied = summary
          ? applyConversationSummary({ messages: before, summary, keep })
          : null
      }
      // Never billed for a chat a summary would not shrink, however often the
      // user asks. The run ends with nothing changed: that is the no-op.
      const summaryWorthwhile =
        trigger === 'manual'
          ? manualCompactionIsWorthwhile({ messages: before, maxContextLength })
          : trigger !== 'cache_expiry'
      if (!usable(applied) && summaryWorthwhile) {
        generatedForeground = true
        summaryId = crypto.randomUUID()
        showStatus()
        const fresh = await summarize({
          summaryId,
          history: summaryRequestHistory(before),
          contextTokens: stepUsageTokens,
          mode: 'foreground',
          midTurn: false,
          trigger,
        })
        if (fresh) {
          summary = fresh
          applied = applyConversationSummary({
            messages: before,
            summary: fresh,
            keep,
          })
        } else outcome = 'failed'
      }
      if (!summary || !usable(applied)) return false
      summaryId = summary.id
      summaryContextTokens = summary.contextTokens

      if (compactionWindow)
        closeCompactionWindow({
          window: compactionWindow,
          messages: before,
          endedBy: 'compaction',
          nextTrigger: trigger,
          logger,
          runId,
          model: agentTemplate.model,
        })
      compactionWindow = openCompactionWindow({
        before,
        after: applied.messages,
        trigger,
        compactionId,
      })
      currentAgentState.messageHistory = applied.messages
      currentAgentState.compactionSummary = applied.summary
      summaries.replace(applied.summary)
      currentAgentState.pendingCompactionTelemetry = {
        id: compactionId,
        summaryId,
        trigger,
      }
      outcome = 'applied'
      currentAgentState.contextTokenBaseline = undefined
      // Not reported, and decides nothing: the next response's count does.
      currentAgentState.contextTokenCount = estimateContextTokens({
        agentState: currentAgentState,
        model: agentTemplate.model,
        systemPrompt: system,
        toolsForTokenCount,
      })
      try {
        logger.info(
          {
            axiomEvent: CONTEXT_COMPACTION_COMPLETED_EVENT,
            ...telemetry,
            compaction_id: compactionId,
            summary_id: summaryId,
            agent_run_id: runId,
            trigger_reason: trigger,
            context_token_count: preTokens,
            max_context_length: maxContextLength,
            message_count: before.length,
            mid_turn: before
              .slice(
                before.findLastIndex((m) => m.tags?.includes('USER_PROMPT')) +
                  1,
              )
              .some((m) => m.role === 'assistant'),
            summary_estimated_tokens: countTokens(summary.text),
            preserved_fresh_messages: applied.messages.length - 1,
          },
          'Context compaction completed',
        )
      } catch {
        // Telemetry must never block the compaction.
      }
      if (isRoot)
        params.onCompaction?.({
          trigger,
          thresholdTokens,
          summary: summary.text,
          preTokens,
          durationMs: Date.now() - started,
        })
      return true
    } catch (error) {
      outcome = signal.aborted || isAbortError(error) ? 'cancelled' : 'failed'
      throw error
    } finally {
      try {
        logger.info(
          {
            axiomEvent: COMPACTION_ATTEMPT_EVENT,
            ...telemetry,
            agent_run_id: runId,
            model: agentTemplate.model,
            compaction_id: compactionId,
            summary_id: summaryId,
            trigger_reason: trigger,
            outcome,
            duration_ms: Date.now() - started,
            ready_summary: readySummary,
            waited_for_pending: waitedForPending,
            generated_foreground: generatedForeground,
            summary_context_tokens: summaryContextTokens,
            context_token_count: preTokens,
            max_context_length: maxContextLength,
          },
          'Compaction attempt finished',
        )
      } catch {
        // Telemetry must never change the turn's outcome.
      }
      if (statusShown) params.onCompactionEnd?.()
    }
  }

  try {
    while (true) {
      totalSteps++
      if (signal.aborted) {
        throw new AbortError()
      }

      if (totalSteps === 1 && userPromptMessage) {
        assertUserContentSize([userPromptMessage], agentTemplate)
      }

      const startTime = new Date()

      const stepPrompt = await getAgentPrompt({
        ...params,
        agentTemplate,
        promptType: { type: 'stepPrompt' },
        fileContext,
        agentState: currentAgentState,
        agentTemplates: localAgentTemplates,
        logger,
        additionalToolDefinitions: additionalToolDefinitionsWithCache,
      })
      const messagesWithStepPrompt = buildArray(
        ...currentAgentState.messageHistory,
        stepPrompt &&
          userMessage({
            content: stepPrompt,
          }),
      )

      // Each completed response supplies the baseline. Only content added or
      // removed since then (tools, steering, step scaffolding) is estimated.
      currentAgentState.contextTokenCount = estimateContextTokens({
        agentState: {
          ...currentAgentState,
          messageHistory: messagesWithStepPrompt,
        },
        model: agentTemplate.model,
        systemPrompt: system,
        toolsForTokenCount,
      })

      if (
        (agentTemplate.compactContext && !shouldEndTurn) ||
        manualCompaction
      ) {
        const thresholdTokens = compactionThreshold(maxContextLength)
        const fixedTokenCount =
          countTokens(system) +
          countTokensJson(toolsForTokenCount) +
          (stepPrompt
            ? countTokensMessages([userMessage({ content: stepPrompt })])
            : 0)
        // A summary made before an edit to a message it covers is dropped.
        currentAgentState.compactionSummary = summaries.latest(
          currentAgentState.compactionSummary,
          currentAgentState.messageHistory,
        )
        const idle =
          !idleChecked &&
          isIdleReturn(
            currentAgentState.messageHistory,
            idleMs,
            undefined,
            currentAgentState.lastActivityAt,
          )
        idleChecked = true
        // The provider's count for the latest response (input + output),
        // from this run or the saved state. What was added since (tool
        // results, a new message) is not guessed: the limit leaves 20% of
        // the budget for it. No count yet (a new chat, right after a swap):
        // no limit decision until the next response.
        const baseline = currentAgentState.contextTokenBaseline
        const contextTokens =
          baseline && sameContextModel(baseline.model, agentTemplate.model)
            ? baseline.tokens
            : undefined
        const overLimit =
          contextTokens !== undefined &&
          contextTokens > thresholdTokens &&
          // The threshold is a convenience, the budget is not. When even a
          // perfect compaction would land back near the threshold (a small
          // BYOK window under a large fixed prompt and tool catalog), keep
          // working until the hard budget instead.
          (contextTokens > maxContextLength ||
            automaticCompactionIsWorthwhile({
              messages: currentAgentState.messageHistory,
              maxContextLength,
              thresholdTokens,
              fixedTokenCount,
              maxOutputTokens: compactionPolicy.maxOutputTokens,
            }))
        const trigger: CompactionTrigger | 'manual' | null = manualCompaction
          ? 'manual'
          : overLimit
            ? idle
              ? 'context_limit_and_cache_expiry'
              : 'context_limit'
            : idle
              ? 'cache_expiry'
              : null
        const compacted = trigger
          ? await compactWithSummary({
              trigger,
              thresholdTokens,
              preTokens: contextTokens,
            })
          : false
        // A manual request is a maintenance operation, never a coding turn.
        if (manualCompaction) break
        if (
          !compacted &&
          contextTokens !== undefined &&
          contextTokens > maxContextLength
        ) {
          stoppedOverBudget = true
          throw new Error(
            compactionOverBudgetMessage({
              maxContextLength,
              byok: runId.startsWith(BYOK_RUN_ID_PREFIX),
            }),
          )
        }
      }

      // 1. Run programmatic step first if it exists
      let n: number | undefined = undefined

      if (agentTemplate.handleSteps) {
        const programmaticResult = await runProgrammaticStep({
          ...params,

          agentState: currentAgentState,
          localAgentTemplates,
          nResponses,
          onCostCalculated: async (credits: number) => {
            currentAgentState.creditsUsed += credits
            currentAgentState.directCreditsUsed += credits
          },
          prompt: currentPrompt,
          runId,
          stepNumber: totalSteps,
          stepsComplete: shouldEndTurn,
          system,
          tools,
          template: agentTemplate,
          toolCallParams: currentParams,
        })
        const {
          agentState: programmaticAgentState,
          endTurn,
          stepNumber,
          generateN,
        } = programmaticResult
        n = generateN

        Object.assign(initialAgentState, programmaticAgentState)
        currentAgentState = initialAgentState
        totalSteps = stepNumber

        shouldEndTurn = endTurn
      }

      // Check if output is required but missing
      if (
        agentTemplate.outputSchema &&
        currentAgentState.output === undefined &&
        shouldEndTurn &&
        !hasRetriedOutputSchema
      ) {
        hasRetriedOutputSchema = true
        logger.warn(
          {
            agentType,
            agentId: currentAgentState.agentId,
            runId,
          },
          'Agent finished without setting required output, restarting loop',
        )

        // Add system message instructing to use set_output
        const outputSchemaMessage = withSystemTags(
          `[freebuff product prompt text omitted]`,
        )

        currentAgentState.messageHistory = [
          ...currentAgentState.messageHistory,
          userMessage({
            content: outputSchemaMessage,
            keepDuringTruncation: true,
          }),
        ]

        // Reset shouldEndTurn to continue the loop
        shouldEndTurn = false
      }

      // One opportunity per invocation, only after concrete work. Keep normal
      // step, spend, abort and deadline limits; never override a programmatic end.
      if (
        shouldEndTurn &&
        agentTemplate.completionCheck &&
        !agentTemplate.handleSteps &&
        usedWorkTools &&
        !hasCheckedCompletion &&
        currentAgentState.stepsRemaining > 0 &&
        (params.deadlineAt === undefined ||
          !Number.isFinite(params.deadlineAt) ||
          params.deadlineAt - Date.now() >= 120_000)
      ) {
        hasCheckedCompletion = true
        // Require an actual answer, not just reasoning or end_turn.
        if (
          agentTemplate.outputMode === 'last_message' &&
          typeof outputForParent(
            getAgentOutput(currentAgentState, agentTemplate),
          ) === 'string'
        ) {
          completionCheckpoint = [...currentAgentState.messageHistory]
        }
        currentAgentState.messageHistory.push(
          userMessage({
            content: withSystemTags(
              "Final completion check: compare the actual deliverables with the user request: exact paths, interfaces, formats, correctness, and any performance requirements. If a necessary check or repair is missing, use the available tools to finish it. Verify the final edited files; rerun affected checks after repairs, but do not repeat passing checks when the relevant files have not changed. Preserve the tested command's exit status when filtering output (for example, Bash pipefail or explicit status capture); a successful output filter is not a passing test. Fix causes rather than skipping tests, weakening assertions, swallowing errors, or adding type/lint suppressions just to pass. If changing a check is required by the requested behavior, explain why and verify that behavior. Respect permission boundaries, refusals, and requests to stop; if awaiting user input or genuinely blocked, report that and end. Otherwise give a concise final answer stating what was verified, which checks failed or could not run, and any remaining limitations.",
            ),
          }),
        )
        shouldEndTurn = false
      }

      // End turn if programmatic step ended turn, or if the previous runAgentStep ended turn
      if (shouldEndTurn) {
        break
      }

      if (
        params.deadlineAt !== undefined &&
        Number.isFinite(params.deadlineAt)
      ) {
        const seconds = Math.max(
          0,
          Math.ceil((params.deadlineAt - Date.now()) / 1000),
        )
        const level = seconds <= 60 ? 3 : seconds <= 300 ? 2 : 1
        if (level > deadlineReminderLevel) {
          deadlineReminderLevel = level
          currentAgentState.messageHistory.push(
            userMessage({
              content: withSystemTags(
                `Time budget: approximately ${seconds} seconds remain before the host deadline. Prioritize saving usable deliverables and checking the required contract. Avoid optional exploration or starting work that cannot finish in time.`,
              ),
            }),
          )
        }
      }

      const creditsBefore = currentAgentState.directCreditsUsed
      const childrenBefore = currentAgentState.childRunIds.length
      stepUsageTokens = undefined
      llmStepNumber++
      const appliedCompaction = currentAgentState.pendingCompactionTelemetry
      // Consume at dispatch, not on success: retries belong to this logical
      // first request too. A manual /compact carries it to the next turn.
      delete currentAgentState.pendingCompactionTelemetry
      const {
        agentState: newAgentState,
        shouldEndTurn: llmShouldEndTurn,
        usedWorkTools: stepUsedWorkTools,
        stepToolCalls,
        stepToolResults,
        messageId,
        nResponses: generatedResponses,
      } = await runAgentStep({
        ...params,

        onAgentUsageReceived: (usage) => {
          // Spawned agents report through the same callback.
          if (usage.agentId === initialAgentState.agentId) {
            stepUsageTokens = usage.inputTokens + usage.outputTokens

          // Conservatively refuse fallback even for a read-only tool or an
          // error event: either may have uncovered a failed verification.

              ? chunk.trim().length > 0
              : chunk.type !== 'reasoning_delta' && chunk.type !== 'start'

          }

          statusCode === 408 ||
          (statusCode !== undefined && statusCode >= 500 && statusCode < 600) ||
          ((statusCode === undefined || statusCode === 200) &&
            (errorCode === 'flex_unavailable' ||
              errorCode === 'resource_unavailable' ||
              isTransientNetworkError(error) ||
              isFetchIdleTimeoutError(error)))

        // provider outage. Leave quota/payment/access refusals to the caller.

          !transient ||
          signal.aborted ||
          isAbortError(error)
        ) {
          throw error
        }

          '\n\nThe additional completion check could not finish because the model request failed. The preceding result is preserved; no additional verification was completed.'

          ...completionCheckpoint,
          assistantMessage(notice),
        ]
        currentAgentState.stepsRemaining--
        logger.warn(
          {
            metric: 'completion_check_unavailable',
            agentId: currentAgentState.agentId,
            runId,
            statusCode,

        )

          agentState: currentAgentState,
          shouldEndTurn: true,
          usedWorkTools: false,
          stepToolCalls: undefined,
          stepToolResults: undefined,
          messageId: null,
          nResponses: undefined,
        }

          ...params,

          childRunIds: newAgentState.childRunIds.slice(childrenBefore),
          messageId,
          status: 'completed',
          startTime,
        })

        logger.error('No runId found for agent state after finishing agent run')
      }

      currentAgentState = initialAgentState

      // summary's growth is measured from. The swap may have been in an
      // earlier run (manual /compact ends its run right after it).
      if (
        stepUsageTokens !== undefined &&
        currentAgentState.compactionSummary &&
        currentAgentState.compactionSummary.contextTokens === undefined

          ...currentAgentState.compactionSummary,
          contextTokens: stepUsageTokens,
        }
        summaries.replace(currentAgentState.compactionSummary)

      // request just warmed the cache.

          countTokens(system) + countTokensJson(toolsForTokenCount),

      currentParams = undefined

      // state) and keep the turn going so the agent responds to them next step,

      // framed as sent mid-turn: bare, the model read it as the whole new
      // request and dropped the one it was working on.

        ...params,
        runId,
        status,
        totalSteps,
        directCredits: currentAgentState.directCreditsUsed,
        totalCredits: currentAgentState.creditsUsed,
        errorMessage,
      }),
    )

      throw error
    }

    return {
      agentState: currentAgentState,
      output: {
        type: 'error',
        message:
          hasServerMessage || isIdleTimeout || isNetworkError || isModelLoop
            ? errorMessage
            : 'Agent run error: ' + errorMessage,
        ...(statusCode !== undefined && { statusCode }),
        ...(apiErrorDetails.errorCode !== undefined && {
          error: apiErrorDetails.errorCode,
        }),
        ...(apiErrorDetails.countryCode !== undefined && {
          countryCode: apiErrorDetails.countryCode,

        ...(apiErrorDetails.countryBlockReason !== undefined && {

        ...(apiErrorDetails.ipPrivacySignals !== undefined && {
          ipPrivacySignals: apiErrorDetails.ipPrivacySignals,
        }),
      },
    }
  } finally {

    if (compactionWindow)
      closeCompactionWindow({
        window: compactionWindow,
        messages: currentAgentState.messageHistory,
        endedBy: 'run_end',
        logger,
        runId,
        model: agentTemplate.model,
      })
    // The endTurn path inside runProgrammaticStep handles normal completion,
    // but abort/error exits (e.g. chat SSE disconnects) would otherwise leak
    // the run's generator, STEP_ALL flag, and proposed file content forever.
    clearProgrammaticRunState(runId)
  }
}

 * The review of a finished turn: one request with the agent's own system
 * prompt, tools and history plus `TURN_REVIEW_PROMPT`, so the provider reads
 * the conversation from its warm cache. Its reply becomes one-click next
 * prompts and a verdict for the host to show; nothing joins the history.
 */

import {
  parseTurnReview,
  TURN_REVIEW_METADATA_KEY,
  TURN_REVIEW_PROMPT,
} from '@codebuff/common/constants/turn-review'
import { TURN_REVIEW_EVENT } from '@codebuff/common/util/axiom-only-log'
import { getErrorObject } from '@codebuff/common/util/error'
import { systemMessage, userMessage } from '@codebuff/common/util/messages'

import { withSystemInstructionTags } from './util/messages'

import type { SummaryStream } from './conversation-summary'
import type { TurnReview } from '@codebuff/common/constants/turn-review'
import type { ModelUsageData } from '@codebuff/common/types/contracts/llm'
import type { Logger } from '@codebuff/common/types/contracts/logger'

/**
 * A review may think as long as it needs: no output cap beyond the agent's own,
 * and this generous limit. A 4,000-token cap and 90 s limit failed 5.5% of
 * reviews (11-12), mostly a model's reasoning spending the whole budget.
 */
export const TURN_REVIEW_TIMEOUT_MS = 10 * 60_000

/** Asks for the review. `history` must be what the agent's next request sends. */
export async function reviewTurn(params: {
  history: Message[]
  system: string
  /** TURN_REVIEW_PROMPT, or `turnReviewPrompt` with the user's recent messages. */
  prompt?: string
  stream: SummaryStream
  logger: Logger
  runId: string
  model: string
  telemetry?: Record<string, string>
}): Promise<TurnReview | null> {
  const started = Date.now()
  let usage: ModelUsageData | undefined
  let finishReason: string | undefined
  let outcome = 'failed'
  let review: TurnReview | null = null
  try {
    const stream = params.stream({
      messages: [
        systemMessage(params.system),
        ...params.history,
        // Tagged so the server's prompt classifier skips it as product text.
        userMessage({
          content: withSystemInstructionTags(params.prompt ?? TURN_REVIEW_PROMPT),
        }),
      ],
      signal: AbortSignal.timeout(TURN_REVIEW_TIMEOUT_MS),
      extraCodebuffMetadata: { [TURN_REVIEW_METADATA_KEY]: '1' },
      onFinishReason: (reason) => {
        finishReason = reason
      },
      onUsageReceived: (received) => {
        usage = received
      },
    })
    let text = ''
    for (;;) {
      const next = await stream.next()
      if (next.done) break
      if (next.value.type === 'error') throw new Error(next.value.message)
      if (next.value.type === 'text') text += next.value.text
    }
    review = parseTurnReview(text)
    outcome = review ? 'completed' : 'unrecognized'
    return review
  } catch (error) {
    params.logger.debug?.({ error: getErrorObject(error) }, 'Turn review failed')
    return null
  } finally {
    try {
      params.logger.info(
        {
          axiomEvent: TURN_REVIEW_EVENT,
          ...params.telemetry,
          agent_run_id: params.runId,
          model: params.model,
          outcome,
          finish_reason: finishReason,
          verdict: review?.verdict ?? undefined,
          action_count: review?.actions.length ?? 0,
          duration_ms: Date.now() - started,
          ...(usage
            ? {
                input_tokens: usage.inputTokens,
                cached_input_tokens: usage.cachedInputTokens,
                output_tokens: usage.outputTokens,
              }
            : {}),
        },
        'Turn review finished',
      )
    } catch {
      // Logging must never fail a review.
    }
  }
}

  read_image?: import('@codebuff/common/types/contracts/client').RequestImageFileFn
  read_files?: (input: {
    filePaths: string[]
    /** Present only for `windowedFileReads` agents. An override that ignores
     *  it returns whole files, which is a cost regression rather than a
     *  correctness bug — but it is the whole point of the flag, so overrides
     *  on hosted surfaces window the content themselves (they hold the file
     *  before their own read budget truncates it). */
    fileWindows?: Record<string, FileReadWindow[]>
    /** Set only when the SDK reads a file to compute an edit (str_replace,
     *  write_file), never from a model's read_files call. The override must
     *  then return the whole file, skipping its read budget: an edit computed
     *  from a truncated read cannot match anything past the cut, and the patch
     *  it produces carries the truncation notice, so it fails to apply to the
     *  real file. An override that ignores it keeps that old behaviour. */
    fullContent?: true
  }) => Promise<Record<string, string | null>>
}

function isRunPauseError(error: unknown) {
  return (
    !!error &&
    typeof error === 'object' &&
    (('codebuffRunPaused' in error &&
      (error as { codebuffRunPaused?: unknown }).codebuffRunPaused === true) ||
      ('name' in error &&
        (error as { name?: unknown }).name === 'CodebuffRunPausedError'))
  )
}

/** Remove the resolved credential and conventional provider variables before a
 * terminal tool inherits the host environment. This does not disable normal
 * user tooling network access; it only prevents an agent from reading BYOK
 * credentials through a subprocess. */
function scrubByokCredentialEnv(
  env: Record<string, string> | undefined,
  credential: string,
): Record<string, string> {
  const result: Record<string, string> = {}
  for (const [key, value] of Object.entries(env ?? {})) {
    if (value === credential || /(?:openrouter|openai|anthropic).*api.*key/i.test(key)) continue
    result[key] = value
  }
  return result
}

/** How often onStateSnapshot fires while a run is in flight. */
const STATE_SNAPSHOT_INTERVAL_MS = 5_000

export const STATE_SNAPSHOT_INTERRUPTION_MESSAGE =
  'The session ended before this response completed. Partial progress has been preserved.'

/**
 * Copy a SessionState for a checkpoint / cancellation snapshot: the caller
 * appends an interruption message and must not disturb the live session.
 *
 * lodash cloneDeep of the whole session is expensive — ~230ms on an ~8 MB
 * session — and this runs on the CLI's render/input thread every ~5s at each
 * in-flight snapshot, a major source of long-session freezes.
 *
 * Only mainAgentState needs an independent copy: the mutations that would
 * otherwise bleed into an already-captured snapshot all live there —
 * messageHistory (.push), agentContext subgoals (update_subgoal). We copy it
 * with a JSON round-trip, which is ~50x faster than cloneDeep (~4ms). The
 * snapshot is only ever consumed as JSON — persisted to disk, and re-serialized
 * when fed back as previousRunState — so a JSON round-trip is byte-for-byte
 * parity with the prior cloneDeep→JSON.stringify path. It also can't choke on
 * the URL / Buffer / Uint8Array instances the message schema permits in
 * image/file content (structuredClone throws on URL and rewrites
 * Buffer→Uint8Array; both change the persisted bytes or fall back).
 *
 * fileContext is large, effectively read-only during a run, and already
 * persisted as-is, so we share it by reference rather than copy it.
 *
 * Falls back to cloneDeep only if JSON.stringify throws (circular refs /
 * BigInt — which AgentState isn't expected to contain, as it uses IDs rather
 * than object back-refs) so a snapshot — or the final error state, which shares
 * this path — can never fail to build.
 */
export function cloneSessionState(
  state: SessionState,
  logger?: Logger,
): SessionState {
  let mainAgentState: SessionState['mainAgentState']
  try {
    mainAgentState = JSON.parse(JSON.stringify(state.mainAgentState))
  } catch (error) {
    logger?.debug?.(
      { error: error instanceof Error ? error.message : String(error) },
      'JSON clone of mainAgentState failed; falling back to cloneDeep',
    )
    mainAgentState = cloneDeep(state.mainAgentState)
  }
  return { fileContext: state.fileContext, mainAgentState }
}

/**
 * The session state a cancelled or errored turn persists, and the state a
 * follow-up prompt resumes from.
 *
 * This is the LAST thing that edits the history — after the runtime's
 * end-of-turn recount, not before it. It drops the half-step an interrupted
 * turn can leave behind and appends the message explaining why the turn ended,
 * so `contextTokenCount` has to follow: a count taken before these edits
 * describes a history that was never stored, and the host shows that number to
 * the user before their next message.
 *
 * The adjustment is a difference rather than a recount because the system
 * prompt and tool schemas — the other half of the count — are not in scope
 * here; carrying the delta keeps them exactly.
 */
export function buildCancelledSessionState(params: {
  sessionState: SessionState
  /** The runtime replaced the shared messageHistory, i.e. it got far enough to
   *  record the user's prompt itself. */
  runtimeMadeProgress: boolean
  /** The user's prompt, re-added only when the runtime never recorded it. */
  promptMessage?: Message
  /** Why the turn ended. Appended as a system-tagged user message. */
  message: string
  logger?: Logger
}): SessionState {
  const { sessionState, runtimeMadeProgress, promptMessage, message, logger } =
    params

  const state = cloneSessionState(sessionState, logger)
  // Also used for in-flight checkpoints: a host that exits mid-response must
  // not count the time spent generating that response as user inactivity.
  state.mainAgentState.lastActivityAt = Date.now()
  const previousHistory = state.mainAgentState.messageHistory
  // A checkpoint can land after an assistant tool call is recorded but before
  // its result arrives. Drop that half-step at the persistence boundary so a
  // resumed run starts from structurally valid history.
  //
  // Copied unconditionally: dropUnansweredToolCalls returns its input when
  // there is nothing to drop, and the appends below must not reach the array
  // `previousHistory` names.
  const nextHistory = [...dropUnansweredToolCalls(previousHistory)]

  if (!runtimeMadeProgress && promptMessage) {
    nextHistory.push(promptMessage)
  }
  nextHistory.push({
    role: 'user' as const,
    content: [{ type: 'text' as const, text: withSystemTags(message) }],
  })

  state.mainAgentState.messageHistory = nextHistory
  state.mainAgentState.contextTokenCount = adjustContextTokenCountForHistoryEdit(
    {
      contextTokenCount: state.mainAgentState.contextTokenCount,
      previousHistory,
      nextHistory,
      scale: state.mainAgentState.contextTokenBaseline?.scale,
    },
  )
  return state
}

/** The text of a single-text-part user message, else undefined. */
function userMessageText(message: Message | undefined): string | undefined {
  if (!message || message.role !== 'user') return undefined
  const content = message.content as unknown
  if (typeof content === 'string') return content
  if (!Array.isArray(content) || content.length !== 1) return undefined
  const [part] = content as Array<{ type?: string; text?: unknown }>
  return part?.type === 'text' && typeof part.text === 'string'
    ? part.text
    : undefined
}

/**
 * The run state a host resumes from after the USER stopped the turn (Esc).
 *
 * A cancelled run state from this SDK ends in the message explaining why the
 * turn ended, and `output.message` is that same text: the periodic
 * `onStateSnapshot` says the session ended (true only when the host died
 * mid-turn), and an early abort says the run was cancelled. After Esc the model
 * must read that the user interrupted it, as the runtime itself writes on
 * abort. This swaps the SDK's trailing message for the runtime's, once: a
 * history that already ends in it (the runtime's own abort state) is returned
 * as-is, and so is any state that does not end in an SDK cancellation message
 * (a run that completed as Esc landed).
 *
 * Returns undefined when there is no session state to resume from.
 */
export function toUserInterruptedRunState(runState: RunState): RunState | undefined {
  const sessionState = runState.sessionState
  if (!sessionState) return undefined
  const previousHistory = sessionState.mainAgentState.messageHistory
  const interrupted = withSystemTags(USER_INTERRUPTED_MESSAGE)
  const last = userMessageText(previousHistory.at(-1))
  if (last === interrupted) return runState

  const appended =
    runState.output?.type === 'error' && runState.output.message
      ? withSystemTags(runState.output.message)
      : undefined
  if (!appended || last !== appended) return runState

  const nextHistory = previousHistory.slice(0, -1)
  if (userMessageText(nextHistory.at(-1)) !== interrupted) {
    nextHistory.push({
      role: 'user' as const,
      content: [{ type: 'text' as const, text: interrupted }],
    })
  }
  return {
    ...runState,
    sessionState: {
      ...sessionState,
      mainAgentState: {
        ...sessionState.mainAgentState,
        messageHistory: nextHistory,
        contextTokenCount: adjustContextTokenCountForHistoryEdit({
          contextTokenCount: sessionState.mainAgentState.contextTokenCount,
          previousHistory,
          nextHistory,
          scale: sessionState.mainAgentState.contextTokenBaseline?.scale,
        }),
      },
    },
    output: { type: 'error', message: USER_INTERRUPTED_MESSAGE },
  }
}

const createAbortError = (signal?: AbortSignal) => {
  if (signal?.reason instanceof Error) {
    return signal.reason
  }
  const error = new Error('Aborted')
  error.name = 'AbortError'
  return error
}

type RunExecutionOptions = RunOptions &
  CodebuffClientOptions & {
    apiKey: string
    fingerprintId: string
  }
type RunReturnType = RunState

export async function run(options: RunExecutionOptions): Promise<RunState> {
  const requestedInference = options.byok
    ? { source: 'byok' as const, connectionId: options.byok.id, revision: options.byok.revision, model: options.byok.model }
    : { source: 'codebuff' as const }
  const previousInference = options.previousRun?.inference
  if (options.previousRun && !previousInference && options.byok && !options.allowInferenceSourceChange) {
    return {
      sessionState: options.previousRun.sessionState,
      traceSessionId: options.previousRun.traceSessionId ??
        options.traceSessionId ??
        crypto.randomUUID(),
      inference: undefined,
      output: { type: 'error', message: 'This session has no inference source pin; start a new BYOK task.' },
    }
  }
  // Preserve source pinning by default. A host with an explicit user selection
  // can transfer history; provider failures never authorize a transfer.
  if (
    !options.allowInferenceSourceChange &&
    previousInference &&
    ((previousInference.source !== requestedInference.source &&
        !(previousInference.source === 'codebuff' && requestedInference.source === 'byok')) ||
      (previousInference.source === 'byok' &&
        requestedInference.source === 'byok' &&
        (previousInference.connectionId !== requestedInference.connectionId ||
          previousInference.revision !== requestedInference.revision ||
          previousInference.model !== requestedInference.model)))
  ) {
    return {
      sessionState: options.previousRun?.sessionState,
      traceSessionId: options.previousRun?.traceSessionId ??
        options.traceSessionId ??
        crypto.randomUUID(),
      inference: previousInference,
      output: { type: 'error', message: 'This session is bound to a different inference connection; start a new task.' },
    }
  }
  const { signal } = options

  if (signal?.aborted) {
    const abortError = createAbortError(signal)
    return {
      sessionState: options.previousRun?.sessionState,
      traceSessionId:
        options.previousRun?.traceSessionId ??
        options.traceSessionId ??
        crypto.randomUUID(),
      inference: requestedInference,
      output: {
        type: 'error',
        message: abortError.message,
      },
    }
  }

  const result = await runOnce({
    ...options,
    onStateSnapshot: options.onStateSnapshot
      ? (snapshot) =>
          options.onStateSnapshot?.({
            ...snapshot,
            inference: requestedInference,
          })
      : undefined,
  })
  return { ...result, inference: requestedInference }
}

async function runOnce({
  apiKey,
  fingerprintId,
  byok,
  trustedAgentPublishers,
  disableAgentRegistry,
  requestHeaders,

  cwd,
  skillsDir,
  skillsLoader,
  includeHomeSkills,
  projectFiles,
  projectIndex,
  skipSymbolIndex,
  knowledgeFiles,
  agentDefinitions,
  agentDirs,
  maxAgentSteps = MAX_AGENT_STEPS_DEFAULT,
  env,
  terminalCommandBroker,

  handleEvent,
  handleStreamChunk,

  fileFilter,
  overrideTools,
  customToolDefinitions,

  fsSource = () => require('fs').promises,
  spawnSource,
  logger,
  traceWriter,

  agent,
  prompt,
  userId: requestedUserId,
  apiKeyUserId,
  content,
  params,
  previousRun,
  extraToolResults,
  signal,
  deadlineAt,
  drainSteeringMessages,
  promptIsContinuation,
  costMode,
  extraCodebuffMetadata,
  onStateSnapshot,
  onUsage,
  onUsageIncomplete,
  onCompaction,
  onCompactionStart,
  onCompactionEnd,
  onCompactionSummary,
  onTurnReview,
  turnReviewPrompt,
  terminalShell,
  projectKey,
  traceSessionId: traceSessionIdOption,
  confineFileToolsToProject,
  refuseNonTextFileReads,
}: RunExecutionOptions): Promise<RunState> {
  const fsSourceValue = typeof fsSource === 'function' ? fsSource() : fsSource
  const fs = await fsSourceValue
  let spawn: CodebuffSpawn
  if (spawnSource) {
    const spawnSourceValue = await spawnSource
    spawn = spawnSourceValue as CodebuffSpawn
  } else {
    spawn = require('child_process').spawn as CodebuffSpawn
  }
  let activeCustomToolDefinitions = customToolDefinitions ?? []
  // Never hand a provider credential to an agent subprocess. The credential is
  // only closed over by the model implementation below.
  const toolEnv = byok ? scrubByokCredentialEnv(env, byok.apiKey) : env

  // Init session state
  let agentId
  if (typeof agent !== 'string') {
    const clonedDefs = agentDefinitions ? cloneDeep(agentDefinitions) : []
    agentDefinitions = [...clonedDefs, agent]
    agentId = agent.id
  } else {
    agentId = agent
  }
  let sessionState: SessionState
  if (previousRun?.sessionState) {
    // applyOverridesToSessionState handles deep cloning and applying any provided overrides
    sessionState = await applyOverridesToSessionState(
      cwd,
      previousRun.sessionState,
      {
        knowledgeFiles,
        agentDefinitions,
        customToolDefinitions,
        projectFiles,
        projectIndex,
        skipSymbolIndex,
        maxAgentSteps,
      },
    )
  } else {
    // No previous run, so create a fresh session state
    sessionState = await initialSessionState({
      cwd,
      skillsDir,
      skillsLoader,
      includeHomeSkills,
      knowledgeFiles,
      agentDefinitions,
      agentDirs,
      customToolDefinitions,
      projectFiles,
      projectIndex,
      skipSymbolIndex,
      maxAgentSteps,
      fs,
      spawn,
      logger,
    })
  }
  const traceSessionId =
    previousRun?.traceSessionId ?? traceSessionIdOption ?? crypto.randomUUID()
  applyTerminalShell(sessionState, terminalShell)

  if (byok) {
    // The selected connection is a run snapshot. Apply its model to every
    // local helper/subagent template so no helper silently consumes Freebuff
    // inference or switches to a different provider.
    const limits = byokModelLimits(byok)
    for (const template of Object.values(sessionState.fileContext.agentTemplates)) {
      if (template && typeof template === 'object') {
        const byokTemplate = template as {
          model?: string
          compactContext?: boolean | Record<string, unknown>
        }
        byokTemplate.model = byok.model
        byokTemplate.compactContext = {
          ...(typeof byokTemplate.compactContext === 'object'
            ? byokTemplate.compactContext
            : {}),
          maxContextLength: limits.maxContextLength,
          maxOutputTokens: limits.reservedOutputTokens,
        }
      }
    }
  }

  let resolvePromise: (value: RunReturnType) => any = () => {}
  let _reject: (error: any) => any = () => {}
  const promise = new Promise<RunReturnType>((res, rej) => {
    resolvePromise = res
    _reject = rej
  })

  // Snapshot support: stop emitting the moment the run settles so a late
  // snapshot can never overwrite the final state persisted by the host.
  let settled = false
  let snapshotTimer: ReturnType<typeof setInterval> | null = null
  const resolve = (value: RunReturnType) => {
    settled = true
    if (snapshotTimer !== null) {
      clearInterval(snapshotTimer)
      snapshotTimer = null
    }
    resolvePromise(value)
  }

  async function onError(error: { message: string }) {
    if (handleEvent) {
      await handleEvent({ type: 'error', message: error.message })
    }
  }

  // The agent runtime mutates sessionState.mainAgentState as it progresses,
  // replacing messageHistory with a new array once it adds the user prompt.
  // Comparing array identity detects progress more robustly than length:
  // context pruning could shrink history below its starting length without
  // meaning the runtime never ran.
  let initialMessageHistory = sessionState.mainAgentState.messageHistory

  /** Calculates the current session state if cancelled.
   *
   * This is used when callMainPrompt throws an error. If the agent runtime made
   * any progress (replaced the shared messageHistory), those messages are
   * preserved. Otherwise the user's message is added so it isn't lost.
   */
  function getCancelledSessionState(message: string): SessionState {
    const runtimeMadeProgress =
      sessionState.mainAgentState.messageHistory !== initialMessageHistory

    return buildCancelledSessionState({
      sessionState,
      runtimeMadeProgress,
      // Only add the user's message if the runtime didn't get a chance to.
      promptMessage:
        prompt || content
          ? {
              role: 'user' as const,
              content: buildUserMessageContent(prompt, params, content),
              tags: ['USER_PROMPT'] as string[],
            }
          : undefined,
      message,
      logger,
    })
  }
  function getCancelledRunState(message?: string): RunState {
    message = message ?? 'Run cancelled by user.'
    return {
      sessionState: getCancelledSessionState(message),
      traceSessionId,
      output: {
        type: 'error',
        message,
      },
    }
  }

  const onResponseChunk = async (
    action: ServerAction<'response-chunk'>,
  ): Promise<void> => {
    if (signal?.aborted) {
      return
    }
    const { chunk } = action

    if (typeof chunk !== 'string') {
      if (chunk.type === 'reasoning_delta') {
        await handleStreamChunk?.({
          type: 'reasoning_chunk',
          chunk: chunk.text,
          // The agent's stable id (matches subagent_start/subagent_chunk), so
          // subagent reasoning attributes to the right agent. (Previously this
          // forwarded runId, which no consumer's agent map is keyed by.)
          agentId: chunk.agentId,
          ancestorRunIds: chunk.ancestorRunIds,
        })
      } else {
        await handleEvent?.(chunk)
      }
      return
    }

    if (handleStreamChunk) {
      await handleStreamChunk(chunk)
    }
  }
  const onSubagentResponseChunk = async (
    action: ServerAction<'subagent-response-chunk'>,
  ) => {
    if (signal?.aborted) {
      return
    }
    const { agentId, agentType, chunk } = action

    if (handleStreamChunk && chunk) {
      await handleStreamChunk({
        type: 'subagent_chunk',
        agentId,
        agentType,
        chunk,
      })
    }
  }

  // Opt-in (the project-understanding agent): every other run leaves both unset and
  // its tools behave exactly as before.
  const projectFence: ProjectFence | undefined =
    confineFileToolsToProject && cwd
      ? createProjectFence({ projectRoot: cwd, fs })
      : undefined
  const refuseNonText = refuseNonTextFileReads === true

  const agentRuntimeImpl = getAgentRuntimeImpl({
    logger,
    traceWriter,
    apiKey,
    byok,
    trustedAgentPublishers,
    disableAgentRegistry,
    requestHeaders,
    handleStepsLogChunk: () => {
      // Does nothing for now
    },
    requestToolCall: async ({ userInputId, toolName, input, mcpConfig, toolCallId }) => {
      return handleToolCall({
        action: {
          type: 'tool-call-request',
          toolCallId,
          requestId: crypto.randomUUID(),
          userInputId,
          toolName,
          input,
          timeout: undefined,
          mcpConfig,
        },
        overrides: overrideTools ?? {},
        readPage: async ({ url, max_chars, objective, live }) => {
          // A direct BYOK run has no hosted service allowance. Keep its local,
          // DNS-pinned reader and never send its provider key to the web API.
          if (byok) return readUrl({ url, max_chars, signal })
          const page = await callWebFetchAPI({
            url,
            maxChars: max_chars,
            objective,
            live,
            sessionId: promptId,
            signal,
            apiKey,
            fetch: agentRuntimeImpl.fetch,
            logger: agentRuntimeImpl.logger,
            env: { clientEnv: agentRuntimeImpl.clientEnv, ciEnv: agentRuntimeImpl.ciEnv },
          })
          return [{ type: 'json', value: page }]
        },
        customToolDefinitions: activeCustomToolDefinitions
          ? Object.fromEntries(
              activeCustomToolDefinitions.map((def) => [def.toolName, def]),
            )
          : {},
        cwd,
        fs,
        env: toolEnv,
        terminalCommandBroker,
        byok,
        signal,
        projectFence,
      })
    },
    requestMcpToolData: async ({ mcpConfig, toolNames }) => {
      const mcpClientId = await getMCPClient(mcpConfig)
      const listToolsResult = await listMCPTools(mcpClientId)
      const tools = listToolsResult.tools
      const filteredTools: typeof tools = []
      for (const tool of tools) {
        if (!toolNames) {
          filteredTools.push(tool)
          continue
        }
        if (toolNames.includes(tool.name)) {
          filteredTools.push(tool)
          continue
        }
      }

      return filteredTools
    },
    requestFiles: async ({ filePaths, fileWindows }) => {
      const read = (paths: string[]) =>
        readFiles({
          filePaths: paths,
          fileWindows,
          override: overrideTools?.read_files,
          fileFilter,
          cwd,
          fs,
          ...(refuseNonText ? { refuseNonText } : {}),
        })
      if (!projectFence || overrideTools?.read_files) return read(filePaths)
      // confineFileToolsToProject: refuse each outside path, read the rest.
      const refused = Object.create(null) as Record<string, string | null>
      const inside: string[] = []
      for (const filePath of filePaths) {
        if (!filePath) continue
        if (await projectFence.resolve(filePath)) {
          inside.push(filePath)
        } else {
          const { relativePath } = resolveFilePath(
            requireCwd(cwd, 'read_files'),
            filePath,
          )
          refused[relativePath] =
            `${FILE_READ_STATUS.OUTSIDE_PROJECT} ${PROJECT_UNDERSTANDING_OUTSIDE_PROJECT_MESSAGE}`
        }
      }
      const loaded = inside.length > 0 ? await read(inside) : {}
      return { ...refused, ...loaded }
    },
    // A remote host reads an image through its own read_image hook: its
    // read_files override is the text-only contract. Without a hook it keeps
    // images in that text read, and so does refuseNonTextFileReads, which
    // refuses them.
    requestImageFile: overrideTools?.read_image ?? (
      overrideTools?.read_files || refuseNonText
        ? undefined
        : async ({ filePath }) => {
            const imageCwd = requireCwd(cwd, 'read_files')
            if (projectFence && !(await projectFence.resolve(filePath))) {
              return {
                path: resolveFilePath(imageCwd, filePath).relativePath,
                error: `${FILE_READ_STATUS.OUTSIDE_PROJECT} ${PROJECT_UNDERSTANDING_OUTSIDE_PROJECT_MESSAGE}`,
              }
            }
            return getImageFile({
              filePath,
              cwd: imageCwd,
              fs,
              fileFilter,
            })
          }),
    requestOptionalFile: async ({ filePath }) => {
      const files = await readFiles({
        filePaths: [filePath],
        override: overrideTools?.read_files,
        fileFilter,
        cwd,
        fs,
        // str_replace/write_file use this path to compute edits. A truncated
        // read makes exact matches later in large files impossible.
        limitContent: false,
        enforceEnvPolicy: false,
      })
      const lookupKeys = cwd
        ? getProjectPathLookupKeys(cwd, filePath)
        : [filePath]
      const fileKey = findFileLookupKey(files, lookupKeys)
      return toOptionalFile(fileKey === undefined ? null : files[fileKey]!)
    },
    sendAction: async ({ action }) => {
      if (action.type === 'action-error') {
        onError({ message: action.message })
        return
      }
      if (action.type === 'response-chunk') {
        await onResponseChunk(action)
        return
      }
      if (action.type === 'subagent-response-chunk') {
        await onSubagentResponseChunk(action)
        return
      }
      if (action.type === 'prompt-response') {
        handlePromptResponse({
          action,
          resolve,
          onError,
          initialSessionState: sessionState,
          traceSessionId,
        })
        return
      }
      if (action.type === 'prompt-error') {
        handlePromptResponse({
          action,
          resolve,
          onError,
          initialSessionState: sessionState,
          traceSessionId,
        })
        return
      }
    },
    sendSubagentChunk: ({
      userInputId,
      agentId,
      agentType,
      chunk,
      prompt,
      forwardToPrompt = true,
    }) => {
      return onSubagentResponseChunk({
        type: 'subagent-response-chunk',
        userInputId,
        agentId,
        agentType,
        chunk,
        prompt,
        forwardToPrompt,
      })
    },
  })

  const promptId = Math.random().toString(36).substring(2, 15)

  // Send input. The lookup rides the run's signal: a socket that never answers here held the run
  // for the whole retry budget with the abort unable to reach it.
  let userId: string | undefined
  if (byok) {
    // BYOK is intentionally auth-free: no Codebuff identity, run ledger,
    // token-count, analytics, or helper request is necessary for inference.
    userId = requestedUserId ?? BYOK_LOCAL_USER_ID
  } else if (apiKeyUserId) {
    userId = requestedUserId ?? apiKeyUserId
  } else {
    let userInfo: { id: string } | null
    try {
      userInfo = await getUserInfoFromApiKey({
        ...agentRuntimeImpl,
        apiKey,
        fields: ['id'],
        signal,
      })
    } catch (error) {
      if (signal?.aborted) return getCancelledRunState('Run cancelled by user.')
      throw error
    }
    if (!userInfo) return getCancelledRunState('Invalid API key or user not found')
    userId = requestedUserId ?? userInfo.id
  }

  if (signal?.aborted) {
    return getCancelledRunState('Run cancelled by user.')
  }

  if (onStateSnapshot) {
    // The runtime replaces mainAgentState.messageHistory with a new array at
    // each step boundary, so reference identity is a cheap "has anything
    // durable changed" check. Skipping unchanged ticks avoids deep-cloning a
    // potentially multi-MB sessionState every interval while the run is just
    // waiting on a slow LLM call.
    let lastSnapshotHistory: unknown = null
    const emitStateSnapshot = () => {
      if (settled || signal?.aborted) {
        return
      }
      const history = sessionState.mainAgentState.messageHistory
      if (history === lastSnapshotHistory) {
        return
      }
      lastSnapshotHistory = history
      try {
        onStateSnapshot(
          getCancelledRunState(STATE_SNAPSHOT_INTERRUPTION_MESSAGE),
        )
      } catch (error) {
        logger?.debug?.(
          { error: error instanceof Error ? error.message : String(error) },
          'onStateSnapshot handler threw',
        )
      }
    }
    // Emit immediately so the user's prompt is checkpointed as soon as the
    // run starts, then keep checkpointing progress while it is in flight.
    emitStateSnapshot()
    snapshotTimer = setInterval(emitStateSnapshot, STATE_SNAPSHOT_INTERVAL_MS)
    // Don't let the checkpoint timer keep the host process alive.
    if (typeof (snapshotTimer as any).unref === 'function') {
      ;(snapshotTimer as any).unref()
    }
  }

  const report = <T>(callback: ((value: T) => void) | undefined) =>
    callback
      ? (value: T) => {
          try {
            callback(value)
          } catch (error) {
            agentRuntimeImpl.logger.debug?.(
              { error: error instanceof Error ? error.message : String(error) },
              'Run metrics handler threw',
            )
          }
        }
      : undefined
  const reportSignal = (callback: (() => void) | undefined) =>
    callback
      ? () => {
          try {
            callback()
          } catch (error) {
            agentRuntimeImpl.logger.debug?.(
              { error: error instanceof Error ? error.message : String(error) },
              'Run metrics handler threw',
            )
          }
        }
      : undefined

  const repoSnapshot = toRepoSnapshot(sessionState.fileContext?.gitChanges)

  const resolvedProjectKey = byok
    ? undefined
    : await resolveProjectKey({
        projectKey,
        cwd,
        readFile: (filePath) => fs.readFile(filePath, 'utf8'),
      })

  callMainPrompt({
    ...agentRuntimeImpl,
    ...(resolvedProjectKey
      ? createProjectProfileClient({
          apiKey,
          userId: requestedUserId,
          projectKey: resolvedProjectKey,
          surface: projectProfileSurface(extraCodebuffMetadata),
        })
      : {}),
    promptId,
    action: {
      type: 'prompt',
      promptId,
      prompt,
      promptParams: params,
      content,
      fingerprintId: fingerprintId,
      costMode: costMode ?? 'normal',
      sessionState,
      toolResults: extraToolResults ?? [],
      agentId,
    },
    drainSteeringMessages,
    promptIsContinuation,
    deadlineAt,
    repoUrl: undefined,
    repoId: undefined,
    clientSessionId: promptId,
    userId,
    extraCodebuffMetadata: {
      ...(extraCodebuffMetadata ?? {}),
      trace_session_id: traceSessionId,
      // Aggregate repository scale, collected once per thread by getGitChanges
      // (#2138) and reused on later turns. Sent as the allowlisted projection
      // so the legacy patch-content fields can never ride along.
      ...(repoSnapshot && { repo_snapshot: JSON.stringify(repoSnapshot) }),
    },
    signal: signal ?? new AbortController().signal,
    onAgentUsageReceived: report(onUsage),
    onAgentUsageIncomplete: reportSignal(onUsageIncomplete),
    onCompaction: report(onCompaction),
    onCompactionStart: report(onCompactionStart),
    onCompactionEnd: reportSignal(onCompactionEnd),
    onCompactionSummary: report(onCompactionSummary),
    onTurnReview: report(onTurnReview),
    turnReviewPrompt,
  }).catch((error) => {
    // A caller's own deadline (`AbortSignal.timeout`) rejects with a
    // `TimeoutError`, the same name Bun's 5-minute fetch idle timeout uses, so
    // the idle copy is only honest when the caller did NOT abort.
    let errorMessage = signal?.aborted
      ? createAbortError(signal).message
      : isFetchIdleTimeoutError(error)
        ? FETCH_IDLE_TIMEOUT_USER_MESSAGE
        : isTransientNetworkError(error)
          ? TRANSIENT_NETWORK_ERROR_USER_MESSAGE
          : error instanceof Error
            ? error.message
            : String(error ?? '')
    const apiErrorDetails = extractApiErrorDetails(error)
    const statusCode = apiErrorDetails.statusCode ?? getErrorStatusCode(error)
    const {
      countryBlockReason,
      countryCode,
      errorCode,
      ipPrivacySignals,
      message: parsedMessage,
    } = apiErrorDetails
    if (parsedMessage) {
      errorMessage = parsedMessage
    }

    resolve({
      sessionState: getCancelledSessionState(errorMessage),
      traceSessionId,
      output: {
        type: 'error',
        message: errorMessage,
        ...(statusCode !== undefined && { statusCode }),
        ...(errorCode !== undefined && { error: errorCode }),
        ...(countryCode !== undefined && { countryCode }),
        ...(countryBlockReason !== undefined && { countryBlockReason }),
        ...(ipPrivacySignals !== undefined && { ipPrivacySignals }),
      },
    })
  })

  return promise
}

function requireCwd(cwd: string | undefined, toolName: string): string {
  if (!cwd) {
    throw new Error(
      `cwd is required for the ${toolName} tool. Please provide cwd in CodebuffClientOptions or override the ${toolName} tool.`,
    )
  }
  return cwd
}

async function readFiles({
  filePaths,
  fileWindows,
  override,
  fileFilter,
  cwd,
  fs,
  limitContent,
  enforceEnvPolicy = true,
  refuseNonText,
}: {
  filePaths: string[]
  fileWindows?: Record<string, FileReadWindow[]>
  override?: NonNullable<
    Required<CodebuffClientOptions>['overrideTools']['read_files']
  >
  fileFilter?: FileFilter
  cwd?: string
  fs: CodebuffFileSystem
  limitContent?: boolean
  enforceEnvPolicy?: boolean
  /** RunOptions.refuseNonTextFileReads; local reads only. */
  refuseNonText?: boolean
}) {
  if (override) {
    // Windows are forwarded, not applied here: an override reads from a remote
    // workspace and applies its own output budget, so windowing after it has
    // already truncated a large file would answer an offset past the cut-off
    // with "beyond the end of the file". The override windows before its own
    // limiter, in the same order the local path does.
    if (!enforceEnvPolicy) {
      return await override({
        filePaths,
        ...(fileWindows ? { fileWindows } : {}),
        // `limitContent: false` is the edit path's read (requestOptionalFile).
        // Without forwarding it, the override applied the model-facing read
        // budget (~20k estimated tokens, roughly 65k characters) and every
        // str_replace past that offset was "not found".
        ...(limitContent === false ? { fullContent: true as const } : {}),
      })
    }

    const result = Object.create(null) as Record<string, string | null>
    const readablePaths: string[] = []
    for (const filePath of filePaths) {
      if (!filePath) continue
      if (isSensitiveEnvFilePath(filePath)) {
        result[filePath] = FILE_READ_STATUS.IGNORED
      } else {
        readablePaths.push(filePath)
      }
    }
    if (readablePaths.length > 0) {
      const loadedFiles = await override({
        filePaths: readablePaths,
        ...(fileWindows ? { fileWindows } : {}),
        ...(limitContent === false ? { fullContent: true as const } : {}),
      })
      if (
        !loadedFiles ||
        typeof loadedFiles !== 'object' ||
        Array.isArray(loadedFiles)
      ) {
        return { ...result }
      }
      for (const [filePath, content] of Object.entries(loadedFiles)) {
        result[filePath] = isSensitiveEnvFilePath(filePath)
          ? FILE_READ_STATUS.IGNORED
          : content
      }
    }
    return { ...result }
  }
  return getFiles({
    filePaths,
    cwd: requireCwd(cwd, 'read_files'),
    fs,
    fileWindows,
    fileFilter,
    limitContent,
    enforceEnvPolicy,
    ...(refuseNonText ? { refuseNonText } : {}),
  })
}

function confinedToolError(errorMessage: string): ToolResultOutput[] {
  return [{ type: 'json', value: { errorMessage } }]
}

/** glob under confineFileToolsToProject: a symlinked folder inside the project
 *  can point anywhere, and the project scan follows it, so a match is kept
 *  only when its real path is inside the project too. */
async function dropMatchesOutsideProject(
  output: ToolResultOutput[],
  fence: ProjectFence,
): Promise<ToolResultOutput[]> {
  return Promise.all(
    output.map(async (part) => {
      if (part.type !== 'json') return part
      const value = part.value as { files?: unknown }
      if (!value || !Array.isArray(value.files)) return part
      const kept: string[] = []
      for (const file of value.files) {
        if (typeof file === 'string' && (await fence.resolve(file))) {
          kept.push(file)
        }
      }
      return kept.length === value.files.length
        ? part
        : { ...part, value: { ...(part.value as object), files: kept } }
    }),
  ) as Promise<ToolResultOutput[]>
}

async function handleToolCall({
  action,
  overrides,
  readPage,
  customToolDefinitions,
  cwd,
  fs,
  env,
  terminalCommandBroker,
  byok,
  signal,
  projectFence,
}: {
  action: ServerAction<'tool-call-request'>
  overrides: NonNullable<CodebuffClientOptions['overrideTools']>
  readPage: (input: CodebuffToolCall<'read_url'>['input']) => Promise<ToolResultOutput[]>
  customToolDefinitions: Record<string, CustomToolDefinition>
  cwd?: string
  fs: CodebuffFileSystem
  env?: Record<string, string>
  terminalCommandBroker?: TerminalCommandBroker
  byok?: ResolvedByokConnection
  signal?: AbortSignal
  /** RunOptions.confineFileToolsToProject: unset for every other run. */
  projectFence?: ProjectFence
}): Promise<{ output: ToolResultOutput[] }> {
  const toolName = action.toolName
  const input = action.input

  if (
    byok &&
    ['web_search', 'read_docs', 'gravity_index', 'render_ui'].includes(toolName)
  ) {
    return {
      output: [{ type: 'json', value: { errorMessage: `${toolName} is unavailable in a direct BYOK run` } }],
    }
  }

  if (signal?.aborted) {
    return {
      output: [
        {
          type: 'json',
          value: {
            message: 'Tool call cancelled: the run was aborted by the user.',
          },
        },
      ],
    }
  }

  // Handle MCP tool calls when mcpConfig is present
  if (action.mcpConfig) {
    try {
      const mcpClientId = await getMCPClient(action.mcpConfig)
      const result = await callMCPTool(
        mcpClientId,
        {
          name: toolName,
          arguments: input,
        },
        undefined,
        signal ? { signal } : undefined,
      )
      return { output: result }
    } catch (error) {
      return {
        output: [
          {
            type: 'json',
            value: {
              errorMessage:
                error instanceof Error ? error.message : String(error),
            },
          },
        ],
      }
    }
  }

  let result: ToolResultOutput[]
  if (toolNames.includes(toolName as ToolName)) {
    clientToolCallSchema.parse(action)
  } else {
    const customToolHandler = customToolDefinitions[toolName]

    if (!customToolHandler) {
      throw new Error(
        `Custom tool handler not found for user input ID ${action.userInputId}`,
      )
    }
    return {
      output: await customToolHandler.execute(action.input, { toolCallId: action.toolCallId }),
    }
  }

  try {
    let override = overrides[toolName as PublishedClientToolName]
    if (
      !override &&
      (toolName === 'str_replace' || toolName === 'apply_patch')
    ) {
      // Reuse the write_file override for file editing tools.
      override = overrides['write_file']
    }
    if (override) {
      // Note: This type assertion is necessary because TypeScript cannot narrow
      // the union type of all possible tool inputs based on the dynamic toolName.
      // The input has been validated by clientToolCallSchema.parse above.
      // eslint-disable-next-line @typescript-eslint/no-explicit-any
      result = await override(input as any, { toolCallId: action.toolCallId })
    } else if (toolName === 'end_turn') {
      result = [{ type: 'json', value: { message: 'Turn ended.' } }]
    } else if (toolName === 'write_file' || toolName === 'str_replace') {
      result = await changeFile({
        parameters: input,
        cwd: requireCwd(cwd, toolName),
        fs,
      })
    } else if (toolName === 'apply_patch') {
      result = await applyPatchTool({
        parameters: input,
        cwd: requireCwd(cwd, toolName),
        fs,
      })
    } else if (toolName === 'run_terminal_command') {
      const resolvedCwd = requireCwd(cwd, 'run_terminal_command')
      result = await runTerminalCommand({
        ...input,
        cwd: path.resolve(resolvedCwd, input.cwd ?? '.'),
        env,
        scrubEnvironmentKeys: byok
          ? [
              'OPENROUTER_API_KEY', 'OPENAI_API_KEY', 'ANTHROPIC_API_KEY',
              ...(byok.credentialRef.startsWith('env:')
                ? [byok.credentialRef.slice(4)]
                : []),
            ]
          : undefined,
        scrubEnvironmentValues: byok ? [byok.apiKey] : undefined,
        signal,
        terminalCommandBroker,
      } as Parameters<typeof runTerminalCommand>[0])
    } else if (toolName === 'read_url') {
      result = await readPage(input as CodebuffToolCall<'read_url'>['input'])
    } else if (toolName === 'code_search' && projectFence) {
      result = confinedToolError(
        'code_search is not available in this run: it cannot be held to the project directory. Use glob and read_files instead.',
      )
    } else if (toolName === 'code_search') {
      result = await codeSearch({
        projectPath: requireCwd(cwd, 'code_search'),
        ...input,
        signal,
      } as Parameters<typeof codeSearch>[0])
    } else if (toolName === 'list_directory' && projectFence) {
      const directoryPath = (input as { path: string }).path
      const projectPath = requireCwd(cwd, 'list_directory')
      const fullPath = await projectFence.resolve(directoryPath, {
        allowRoot: true,
      })
      result = fullPath
        ? (
            await listDirectory({ directoryPath: fullPath, projectPath, fs })
          ).map((part) =>
            part.type === 'json' && 'files' in part.value
              ? { ...part, value: { ...part.value, path: directoryPath } }
              : part,
          )
        : confinedToolError(
            `Failed to list directory: ${directoryPath} is outside the project. Only folders inside the project directory can be listed.`,
          )
    } else if (toolName === 'list_directory') {
      result = await listDirectory({
        directoryPath: (input as { path: string }).path,
        projectPath: requireCwd(cwd, 'list_directory'),
        fs,
      })
    } else if (toolName === 'glob') {
      const globInput = input as {
        pattern: string
        cwd?: string
        max_results?: number
      }
      if (
        projectFence &&
        globInput.cwd !== undefined &&
        !(await projectFence.resolve(globInput.cwd, { allowRoot: true }))
      ) {
        result = confinedToolError(
          `Failed to search for files: cwd ${globInput.cwd} is outside the project. Only the project directory can be searched.`,
        )
      } else {
        result = await glob({
          pattern: globInput.pattern,
          projectPath: requireCwd(cwd, 'glob'),
          cwd: globInput.cwd,
          maxResults: globInput.max_results,
          fs,
        })
        if (projectFence) {
          result = await dropMatchesOutsideProject(result, projectFence)
        }
      }
    } else if (toolName === 'run_file_change_hooks') {
      // No-op: SDK doesn't run file change hooks
      result = [
        {
          type: 'json',
          value: {
            message: 'File change hooks are not supported in SDK mode',
          },
        },
      ]
    } else {
      throw new Error(
        `Tool not implemented in SDK. Please provide an override or modify your agent to not use this tool: ${toolName}`,
      )
    }
  } catch (error) {
    if (isRunPauseError(error)) {
      throw error
    }

    result = [
      {
        type: 'json',
        value: {
          errorMessage:
            error &&
            typeof error === 'object' &&
            'message' in error &&
            typeof error.message === 'string'
              ? error.message
              : typeof error === 'string'
                ? error
                : 'Unknown error',
        },
      },
    ]
  }
  return {
    output: result,
  }
}

/**
 * Extracts an HTTP status code from an error message string.
 * Parses common error patterns to identify the underlying status code.
 * Returns the status code if found, undefined otherwise.
 */
export const extractStatusCodeFromMessage = (
  errorMessage: string,
): number | undefined => {
  const lowerMessage = errorMessage.toLowerCase()

  // AI SDK's built-in retry error (e.g., "Failed after 4 attempts. Last error: Service Unavailable")
  // The AI SDK already retried 4 times, but we still want our SDK wrapper to retry 3 more times
  if (
    lowerMessage.includes('failed after') &&
    lowerMessage.includes('attempts')
  ) {
    // Extract the underlying error type from the message
    if (lowerMessage.includes('service unavailable')) {
      return 503
    }
    if (lowerMessage.includes('timeout')) {
      return 408
    }
    if (lowerMessage.includes('connection refused')) {
      return 503
    }
    // Default to 500 for other AI SDK retry failures
    return 500
  }

  if (
    errorMessage.includes('503') ||
    lowerMessage.includes('service unavailable')
  ) {
    return 503
  }
  if (errorMessage.includes('504')) {
    return 504
  }
  if (errorMessage.includes('502')) {
    return 502
  }
  if (lowerMessage.includes('timeout') || errorMessage.includes('408')) {
    return 408
  }
  if (
    lowerMessage.includes('econnrefused') ||
    lowerMessage.includes('connection refused')
  ) {
    return 503
  }
  if (lowerMessage.includes('dns') || lowerMessage.includes('enotfound')) {
    return 503
  }
  if (lowerMessage.includes('server error') || errorMessage.includes('500')) {
    return 500
  }
  if (errorMessage.includes('429') || lowerMessage.includes('rate limit')) {
    return 429
  }
  if (
    lowerMessage.includes('network error') ||
    lowerMessage.includes('fetch failed')
  ) {
    return 503
  }

  return undefined
}

async function handlePromptResponse({
  action,
  resolve,
  onError,
  initialSessionState,
  traceSessionId,
}: {
  action: ServerAction<'prompt-response'> | ServerAction<'prompt-error'>
  resolve: (value: RunReturnType) => any
  onError: (error: { message: string }) => void
  initialSessionState: SessionState
  traceSessionId: string
}) {
  if (action.type === 'prompt-error') {
    onError({ message: action.message })

    const statusCode = extractStatusCodeFromMessage(action.message)
    resolve({
      // A snapshot, not the object the runtime was handed: the host keeps and
      // resends this state, so it must not alias anything still in flight.
      sessionState: cloneSessionState(initialSessionState),
      traceSessionId,
      output: {
        type: 'error',
        message: action.message,
        ...(statusCode !== undefined && { statusCode }),
      },
    })
  } else if (action.type === 'prompt-response') {
    // Stop enforcing session state schema! It's a black box we will pass back to the server.
    // Only check the output schema.
    const parsedOutput = AgentOutputSchema.safeParse(action.output)
    if (!parsedOutput.success) {
      const message = [
        'Received invalid prompt response from server:',
        JSON.stringify(parsedOutput.error.issues),
        'If this issues persists, please contact support@freebuff.com',
      ].join('\n')
      onError({ message })
      resolve({
        sessionState: initialSessionState,
        traceSessionId,
        output: {
          type: 'error',
          message,
        },
      })
      return
    }
    const { sessionState, output } = action

    const state: RunState = {
      sessionState,
      traceSessionId,
      output: output ?? {
        type: 'error',
        message: 'No output from agent',
      },
    }
    resolve(state)
  } else {
    action satisfies never
    onError({
      message: 'Internal error: prompt response type not handled',
    })
    resolve({
      sessionState: initialSessionState,
      traceSessionId,
      output: {
        type: 'error',
        message: 'Internal error: prompt response type not handled',
      },
    })
  }
}
