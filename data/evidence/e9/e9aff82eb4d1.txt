```
import {
  BILLBOARD_SIDEBAR_PLACEMENT_ID,
  billboardDimensionsError,
  parseBillboardAssets,
} from './billboard'
import { desktopPlacementSource } from './web-desktop-placements'

/** Only these existing sidebar slots are opened to Gravity. */
export function isGravitySidebarPlacement(id: string): boolean {
  return desktopPlacementSource(id) === BILLBOARD_SIDEBAR_PLACEMENT_ID
}

export function isGravitySidebarRequest(ids: readonly string[]): boolean {
  return ids.length > 0 && ids.every(isGravitySidebarPlacement)
}

/** Gravity registration generates a hyphenated slug, unlike our :web IDs. */
export function gravitySidebarPlacementId(id: string): string {
  return id === `${BILLBOARD_SIDEBAR_PLACEMENT_ID}:web`
    ? `${BILLBOARD_SIDEBAR_PLACEMENT_ID}-web`
    : id
}

export function internalGravitySidebarPlacementId(id: string): string {
  return id === `${BILLBOARD_SIDEBAR_PLACEMENT_ID}-web`
    ? `${BILLBOARD_SIDEBAR_PLACEMENT_ID}:web`
    : id
}

/**
 * Proposed full-creative response contract for Gravity. brandImage is a
 * product image, NOT a finished ad. Neither it nor renderer_spec is inferred
 * as billboard artwork. No fetch, composition or generation occurs here.
 */
export function gravitySidebarAssets(value: unknown) {
  const asset = parseBillboardAssets(value)?.sidebar
  if (
    !asset ||
    !Number.isSafeInteger(asset.width) ||
    !Number.isSafeInteger(asset.height) ||
    billboardDimensionsError('sidebar', asset.width, asset.height)
  )
    return undefined
  try {
    const url = new URL(asset.url)
    if (
      url.protocol !== 'https:' ||
      url.username ||
      url.password ||
      (url.port && url.port !== '443')
    )
      return undefined
  } catch {
    return undefined
  }
  return { sidebar: asset }
}

/** Native ACK and browser proxies only forward Gravity's own tracking hosts. */
export function gravitySidebarTrackingUrl(value: unknown): boolean {
  if (typeof value !== 'string') return false
  try {
    const url = new URL(value)
    return (
      url.protocol === 'https:' &&
      !url.username &&
      !url.password &&
      (!url.port || url.port === '443') &&
      (url.hostname === 'trygravity.ai' ||
        url.hostname.endsWith('.trygravity.ai'))
    )
  } catch {
    return false
  }
}
```

```
export const BILLBOARD_SIDEBAR_PLACEMENT_ID = 'Desktop-Billboard-Sidebar'
export const BILLBOARD_PANEL_PLACEMENT_ID = 'Desktop-Billboard-Panel'

export const BILLBOARD_SHAPES = [
  'sidebar',
  'panel_tall',
  'panel_portrait',
  'panel_square',
  'panel_landscape',
] as const
export type BillboardShape = (typeof BILLBOARD_SHAPES)[number]

export function isBillboardShape(value: unknown): value is BillboardShape {
  return (BILLBOARD_SHAPES as readonly unknown[]).includes(value)
}

/**
 * Width:height and the minimum stored size of each shape. The minimums are
 * the 2x exports the Desktop renders at (a 320x240 CSS-pixel sidebar card),
 * so an upload is never upscaled on a high-density display.

  BillboardShape,
```

```
import { describe, expect, test } from 'bun:test'
import {
  gravitySidebarAssets,
  gravitySidebarPlacementId,
  internalGravitySidebarPlacementId,
  isGravitySidebarRequest,
} from './gravity-sidebar'
const SIDEBAR = 'Desktop-Billboard-Sidebar'
const asset = { url: 'https://cdn.example/ad.webp', width: 640, height: 480 }
describe('Gravity sidebar contract', () => {
  test('only sidebar-only requests open to Gravity', () => {
    expect(isGravitySidebarRequest([SIDEBAR])).toBe(true)
    expect(isGravitySidebarRequest([`${SIDEBAR}:web`])).toBe(true)
    for (const ids of [
      [],
      ['Desktop-Billboard-Panel'],
      [SIDEBAR, 'Desktop-Billboard-Panel'],
      [SIDEBAR, 'Desktop-Inline-Chat'],
    ])
      expect(isGravitySidebarRequest(ids)).toBe(false)
  })
  test('Web slug maps back to the existing internal placement', () => {
    expect(gravitySidebarPlacementId(`${SIDEBAR}:web`)).toBe(`${SIDEBAR}-web`)
    expect(internalGravitySidebarPlacementId(`${SIDEBAR}-web`)).toBe(
      `${SIDEBAR}:web`,
    )
    expect(gravitySidebarPlacementId(SIDEBAR)).toBe(SIDEBAR)
  })
  test('requires full, correctly sized HTTPS artwork; drops unknown shapes', () => {
    expect(
      gravitySidebarAssets({ sidebar: asset, panel_square: asset }),
    ).toEqual({ sidebar: asset })
    for (const patch of [
      { width: 320 },
      { height: 640 },
      { width: Infinity },
      { width: NaN },
      { width: 640.1 },
      { url: 'http://cdn.example/a.webp' },
      { url: 'https://user:pass@cdn.example/a.webp' },
      { url: 'javascript:alert(1)' },
      { url: 'https://cdn.example:8080/a.webp' },
    ])
      expect(
        gravitySidebarAssets({ sidebar: { ...asset, ...patch } }),
      ).toBeUndefined()
    for (const raw of [
      undefined,
      null,
      {},
      { brandImage: asset.url },
      { panel_square: asset },
    ])
      expect(gravitySidebarAssets(raw)).toBeUndefined()
  })
})
```
