- PR #1584: cli: skip checkpoint writes when the state is unchanged, and one fsync group per drain
  PR Description: ## What

Checkpoint drains used to unconditionally rewrite the whole persisted state: two multi-MB `JSON.stringify` passes + 2 file fsyncs + 2 dir fsyncs for `run-state.json` and `chat-messages.json`, plus a **third** atomic write (own file fsync + dir fsync) and an event-loop `statSync` for the `chat-meta` sidecar — every drain, ~720/hour. At a 1 MB transcript on the scaled cadence that's **~1.4 GB/hr of rewritten bytes**, much of it while the run sits waiting on a tool call with *nothing changed*.

## Changes (2 commits)

**1. Skip unchanged drains.** Per chat dir, keep the identity fingerprint of the last successful persist: `messages` array identity + length + tail-message identity (sound because `snapshotSessionState` shares message objects and producers mutate through new containers) + `runState` identity + a cheap digest of the mutation-relevant scalars (`contextTokenCount`, `creditsUsed`, `stepsRemaining`, `agentType`, `compactionSummary.text`). Unchanged → **0 serializes, 0 writes, 0 fsyncs**. Fingerprint failure errs toward writing. Authoritative saves (`saveChatState` — completion / interrupt / exit flush) always write.

**2. Coalesce durability; move the sidecar to turn end.** The two drain files now share one directory fsync (`prepareAtomicWrite`/`commitAtomicWrites` in `write-file-atomic.ts`): both tmp files fsync'd before any rename, one dir fsync after both renames. The `chat-meta` sidecar leaves the drain path and is refreshed at turn end — `readChatMeta`

// Choose the agent definition by explicit selection or mode-based fallback.
const resolveAgent = (
  agentMode: AgentMode,
  agentId: string | undefined,
  agentDefinitions: AgentDefinition[],
): AgentDefinition | string => {
  const selectedAgentDefinition =
    agentId && agentDefinitions.length > 0
      ? agentDefinitions.find((definition) => definition.id === agentId)
      : undefined

  return selectedAgentDefinition ?? agentId ?? getAgentIdForMode(agentMode)
}

// Respect bash context, but avoid sending empty prompts when only images are attached.
const buildPromptWithContext = (
  promptWithBashContext: string,
  messageContent: MessageContent[] | undefined,
) => {
  const trimmedPrompt = promptWithBashContext.trim()
  if (trimmedPrompt.length > 0) {
    return promptWithBashContext
  }

  if (messageContent && messageContent.length > 0) {
    return 'See attached image(s)'
  }

  return ''
}

      // This ensures the router can detect that we're busy and queue subsequent messages.

        if (sponsored.plan.signal.aborted) stop()

          updateChainInProgress,

        releaseRunOwnership()

                      id: 'validation_error',
                      message:

                }
                return {
                  ...msg,
                  validationErrors: errorsToAttach,

            createErrorChatMessage(

      // before any async work, so the router can correctly detect busy state.
      let actualCredits: number | undefined

      // (closed terminal, crash) can't lose the user's prompt, then keep the
      // checkpoint fresh from SDK run-state snapshots while the run streams.
      // The completion save below overwrites this with the final state.
      saveChatState(
        latestRunStateSnapshot,
        useChatStore.getState().messages,
        runChatDir,
      )

        // definition is repository-authored content a sponsored run has no
        // business loading, and its agent is the pinned sponsored one.

        const steeringLeftovers = deactivateSteering(runOwnerId)

          runChatIsCurrent() &&
          (!abortController.signal.aborted ||
            abortController.signal.reason === 'user-interrupt')
        ) {

            steeringLeftovers.map((entry) => entry.messageId),
          )

              content: entry.text,
              attachments: [],
            })
          }

        // checkpoint, on error) has been saved above. Owner-guarded so an
        // aborted run resolving late can't clear a newer run's provider.
        releaseRunOwnership()

        // interfering with any new run that may have started after the abort.

        if (!abortController.signal.aborted) {
          if (isChainInProgressRef.current) {
            logger.warn(
              {},
              '[send-message] Chain still in progress after try/catch, forcing reset',
            )
            updateChainInProgress(false)
            setStreamStatus('idle')
            setCanProcessQueue(!isQueuePausedRef?.current)
          }
          // Safety net: ensure lock is always released even if handleRunCompletion/handleRunError
          // didn't run (e.g., due to unexpected early return). Redundant releases are safe (idempotent).
          if (isProcessingQueueRef) {
            isProcessingQueueRef.current = false
          }
        }
        updater.dispose()
        // The verdict is the run service's, read off its own receipts; this
        // only says how the turn ended.
        settleSponsored(abortController.signal.aborted)
      }
    },
    [
      addActiveSubagent,
      addSessionCredits,
      agentId,
      inputRef,
      isChainInProgressRef,
      isProcessingQueueRef,
      isQueuePausedRef,
      mainAgentTimer,
      onBeforeMessageSend,
      onTimerEvent,
      prepareUserMessage,
      removeActiveSubagent,
      requeueMessageAtFront,
      resumeQueue,
      pauseQueue,
      scrollToLatest,
      setCanProcessQueue,
      setFocusedAgentId,
      setHasReceivedPlanResponse,
      setInputFocused,
      setIsRetrying,
      setMessages,
      getClient,
      setRunState,
      setStreamStatus,
      setStreamingAgents,
      streamRefs,
      updateChainInProgress,
    ],
  )

  return {
    sendMessage,
    clearMessages,
  }
}

// Chat-state saves fail in prod for reasons that are chronic, not transient:
// a transcript with a cyclic or over-string-limit payload fails identically on
// every 5s checkpoint, and a full disk stays full. Logging each attempt at
// error level made this a top-3 error by volume (~19k/day). Instead, each
// distinct issue per chat dir is logged at most once per interval, and
// environment/payload failures log as warnings — only genuinely unexpected
// errors keep the error level.
const SAVE_LOG_INTERVAL_MS = 5 * 60 * 1000
const saveIssueLastLoggedAt = new Map<string, number>()

function shouldLogSaveIssue(key: string): boolean {
  const now = Date.now()
  const last = saveIssueLastLoggedAt.get(key)
  if (last !== undefined && now - last < SAVE_LOG_INTERVAL_MS) {
    return false
  }
  saveIssueLastLoggedAt.set(key, now)
  return true
}

 * drains and exit flushes, where a throwing logger (e.g. analytics transport
 * not initialized) would otherwise lose the save or reject the drain.
 */
function bestEffortLog(
  level: 'warn' | 'error',
  payload: Record<string, unknown>,
  message: string,
): void {
  try {
    logger[level](payload, message)
  } catch {
    // Best-effort only.
  }
}

type SaveErrorClass = 'cyclic' | 'oom' | 'disk' | 'other'

function classifySaveError(error: unknown): SaveErrorClass {
  const fromStringify = classifyStringifyError(error)
  if (fromStringify) return fromStringify
  const msg = error instanceof Error ? error.message : String(error)
  // Environmental filesystem failures: retrying won't help and there is
  // nothing to fix server-side (full disk, AV holding a rename lock, etc.).
  if (/ENOSPC|EDQUOT|EROFS|EPERM|EACCES|EBUSY|EMFILE|ENFILE/.test(msg)) {
    return 'disk'
  }
  return 'other'
}

// Returns a plain boolean, not a `json is string` predicate: the caller's
// `json` is already typed `string`, so a predicate narrows the negative branch
// to `never` and the defensive `typeof` checks in the refusal log stop
// compiling. The runtime check stays, because SerializeResult's `string` comes
// from JSON.stringify's lib type, which answers `undefined` for undefined.
function isPersistableJson(
  json: unknown,
  part: 'runState' | 'messages',
): boolean {
  return (
    typeof json === 'string' && json.startsWith(CHAT_STATE_JSON_PREFIX[part])
  )
}

/**
 * Serialize the two chat-state files independently, so a poisoned run state
 * cannot block persisting the transcript (and vice versa). Cyclic and
 * over-limit payloads are rescued by serializeForPersistence's fallback pass;
 * the fallback diagnostics (cycle paths, truncation counts) are logged so the
 * source of a bad payload can be found from prod logs.
 */
function serializeChatState(
  runState: RunState,
  messages: ChatMessage[],
  chatDir: string,
): SerializedChatState {
  const result: SerializedChatState = {}
  for (const part of ['runState', 'messages'] as const) {
    const value = part === 'runState' ? runState : messages
    try {
      const { json, fallback } = serializeForPersistence(value)
      if (!isPersistableJson(json, part)) {
        // Rate-limited like every other save issue, but logged at error level:
        // this is a bug in whatever handed us the state, and until ticket #180
        // nothing reported it.
        if (shouldLogSaveIssue(`${chatDir}|nonpersistable|${part}`)) {
          bestEffortLog(
            'error',
            {
              part,
              // The first character and the length only. Anything more could
              // be the user's prompt text, and this is a diagnostic.
              jsonType: typeof json,
              jsonPrefix: typeof json === 'string' ? json.slice(0, 1) : null,
              jsonBytes: typeof json === 'string' ? json.length : null,
              ...chatShapeSummary(runState, messages),
            },
            'Refusing to persist non-document chat state (keeping the file on disk)',
          )
        }
        continue
      }
      if (part === 'runState') {
        result.runStateJson = json
      } else {
        result.messagesJson = json
      }
      if (fallback && shouldLogSaveIssue(`${chatDir}|fallback|${part}`)) {
        bestEffortLog(
          'warn',
          {
            part,
            reason: fallback.reason,
            cyclePaths: fallback.cyclePaths,
            truncatedStrings: fallback.truncatedStrings,
            jsonBytes: json.length,
            ...chatShapeSummary(runState, messages),
          },
          'Chat state serialized via fallback (broke cycles or truncated oversized strings)',
        )
      }
    } catch (error) {
      const errorClass = classifySaveError(error)
      if (shouldLogSaveIssue(`${chatDir}|serialize|${part}|${errorClass}`)) {
        bestEffortLog(
          errorClass === 'other' ? 'error' : 'warn',
          {
            part,
            errorClass,
            error: error instanceof Error ? error.message : String(error),
            ...chatShapeSummary(runState, messages),
          },
          'Failed to serialize chat state',
        )
      }
    }
  }
  return result
}

function logSaveWriteFailure(
  error: unknown,
  chatDir: string,
  message: string,
): void {
  const errorClass = classifySaveError(error)
  if (!shouldLogSaveIssue(`${chatDir}|write|${errorClass}`)) {
    return
  }
  bestEffortLog(
    errorClass === 'other' ? 'error' : 'warn',
    {
      errorClass,
      error: error instanceof Error ? error.message : String(error),
    },
    message,
  )
}

/**
 * Save both the RunState and ChatMessage[] to disk.
 *
 * `chatDir` should be captured when the state itself is captured (e.g. at run
 * start) and passed through. Defaulting to the current chat dir is only safe
 * when the caller knows no chat switch can have happened in between.
 */
export function saveChatState(
  runState: RunState,
  messages: ChatMessage[],
  chatDir: string = resolveCurrentChatDir(),
): void {
  // Compact JSON: these files are rewritten on every checkpoint and grow to
  // multiple MB; pretty-printing roughly doubles the write.
  const serialized = serializeChatState(runState, messages, chatDir)
  if (!serialized.runStateJson && !serialized.messagesJson) {
    return
  }
  try {
    // The dir existed when the save was captured, but may have been removed
    // since (e.g. the chat deleted from /history mid-run).
    fs.mkdirSync(chatDir, { recursive: true, mode: 0o700 })
    if (serialized.runStateJson) {
      writeFileAtomic(
        path.join(chatDir, RUN_STATE_FILENAME),
        serialized.runStateJson,
      )
    }
    if (serialized.messagesJson) {
      writeFileAtomic(
        path.join(chatDir, CHAT_MESSAGES_FILENAME),
        serialized.messagesJson,
      )
      // Sidecar summary so /history can list this chat without parsing the
      // (unbounded) chat-messages.json. Must be written after the messages
      // file: it records the file's size/mtime to detect staleness. Written
      // only by the authoritative saves (turn completion / interrupt / exit
      // flush); the periodic checkpoint below deliberately leaves the sidecar
      // stale, which readChatMeta detects and resolves with a full parse.
      writeChatMeta(chatDir, messages)
    }
    noteChatStatePersisted(chatDir, runState, messages)
  } catch (error) {
    logSaveWriteFailure(error, chatDir, 'Failed to save chat state')
  }
}

/** What makeCheckpointKey captures about one state. */
type CheckpointKey = {
  messages: unknown
  messagesLength: number
  lastMessage: unknown
  runState: unknown
  /** Identity marker for scalars that went into runStateScalarDigest. */
  runStateScalarDigest: string
}

/**
 * The state the SDK hands a checkpoint is a shallow snapshot whose message
 * objects are shared with the live run (snapshotSessionState): anything that
 * mutates them — streaming text appended to the tail message, the CLI's
 * setMessages replacing the array — always writes through a new container
 * (`blocks: [...]`, tail copy, or a fresh array). So container identity is a
 * sound "has anything changed" signal, and it is O(1) where a defensive
 * re-serialize of a multi-MB transcript on every drain is not.
 *
 * The run state gets no such discipline from its producers, so its
 * mutation-relevant scalars are digested cheaply instead — values the next
 * turn's continuation depends on, compared by content. The rest of the run
 * state changes only with the history and its containers carry identity.
 */
function makeCheckpointKey(
  runState: RunState,
  messages: ChatMessage[],
): CheckpointKey {
  const mainAgentState = (runState as { sessionState?: any } | null)
    ?.sessionState?.mainAgentState
  let runStateScalarDigest = ''
  if (mainAgentState && typeof mainAgentState === 'object') {
    try {
      runStateScalarDigest = JSON.stringify([
        mainAgentState.contextTokenCount ?? null,
        mainAgentState.creditsUsed ?? null,
        mainAgentState.stepsRemaining ?? null,
        mainAgentState.agentType ?? null,
        mainAgentState.compactionSummary?.text ?? null,
        mainAgentState.pendingCompactionTelemetry ?? null,
      ])
    } catch {
      // Never let a fingerprint break a save: unstringifyable scalars simply
      // stop comparing equal, which errs on the side of writing.
      runStateScalarDigest = randomUUID()
    }
  }
  return {
    messages,
    messagesLength: Array.isArray(messages) ? messages.length : -1,
    lastMessage: Array.isArray(messages) ? messages[messages.length - 1] : null,
    runState,
    runStateScalarDigest,
  }
}

function checkpointKeysEqual(a: CheckpointKey, b: CheckpointKey): boolean {
  return (
    a.messages === b.messages &&
    a.messagesLength === b.messagesLength &&
    a.lastMessage === b.lastMessage &&
    a.runState === b.runState &&
    a.runStateScalarDigest === b.runStateScalarDigest
  )
}

// What was last durably written, per chat dir. Checkpoint drains compare
// against this before serializing: an unchanged checkpoint — the SDK's
// snapshot timer fired while the run was waiting on a tool or a slow LLM
// call, or the CLI rescheduled the same settled state — would regenerate two
// multi-MB JSON documents and pay their fsyncs without changing a single byte
// on disk.
const persistedCheckpointKeys = new Map<string, CheckpointKey>()

function noteChatStatePersisted(
  chatDir: string,
  runState: RunState,
  messages: ChatMessage[],
): void {
  persistedCheckpointKeys.set(chatDir, makeCheckpointKey(runState, messages))
}

/**
 * True when this checkpoint would persist exactly what the last successful
 * save already wrote. Only drains consult it — the synchronous, authoritative
 [freebuff product prompt text omitted]
 * final state the user lost if a crash lands before the next checkpoint.
 */
function isCheckpointUnchanged(
  chatDir: string,
  runState: RunState,
  messages: ChatMessage[],

  return !!key && checkpointKeysEqual(key, makeCheckpointKey(runState, messages))

/**
 * Refresh only the /history sidecar for a chat directory, after the messages
 * file on disk already holds exactly `messages` (e.g. that state landed via
 * a skipped or regular checkpoint). Authoritative completion paths use this
 * when the components of the state were last persisted separately, so chat
 * completion always leaves a fresh sidecar behind.
 */
export function refreshChatMetaSync(
  chatDir: string,
  messages: ChatMessage[],
): void {
  try {
    writeChatMeta(chatDir, messages)
  } catch (error) {
    logSaveWriteFailure(error, chatDir, 'Failed to refresh chat meta')
  }
}

/**
 * Async counterpart to saveChatState. Serializes and writes off the caller's
 * tick so a multi-MB transcript doesn't block the CLI's render/input thread.
 */
async function saveChatStateAsync(
  runState: RunState,
  messages: ChatMessage[],
  chatDir: string,
): Promise<void> {
  if (isCheckpointUnchanged(chatDir, runState, messages)) {
    // Nothing to do at all: no serialize, no writes. An unchanged checkpoint
    // tick — the SDK fires at the interval regardless of progress — used to
    // cost two full stringifies of the transcript plus three atomic writes
    // (run-state, messages, meta) and their fsyncs, all producing the exact
    // bytes already on disk.
    return
  }
  const serialized = serializeChatState(runState, messages, chatDir)
  if (!serialized.runStateJson && !serialized.messagesJson) {
    return
  }
  try {
    await fs.promises.mkdir(chatDir, { recursive: true })
    // Both files went through tmp-file + fsync + rename + dir fsync each, and
    // the two dir fsyncs flush the same directory. Prepare both (contents
    // durable under their temp names), publish both renames, then pay that
    // directory fsync once for the pair.
    const pending = []
    if (serialized.runStateJson) {
      pending.push(
        await prepareAtomicWrite(
          path.join(chatDir, RUN_STATE_FILENAME),
          serialized.runStateJson,
        ),
      )
    }
    if (serialized.messagesJson) {
      pending.push(
        await prepareAtomicWrite(
          path.join(chatDir, CHAT_MESSAGES_FILENAME),
          serialized.messagesJson,
        ),
      )
    }
    await commitAtomicWrites(pending)
    // The chat-meta sidecar moved out of this path on purpose: it exists to
    // summarize a settled chat for /history, is validated against the messages
    // file's size/mtime with a full-parse fallback (readChatMeta), and cost a
    // third atomic write plus a statSync on every drain. It is refreshed by
    // the authoritative save at turn end instead; a mid-turn sidecar that
    // reads stale is exactly the case readChatMeta was written to detect.
    noteChatStatePersisted(chatDir, runState, messages)
  } catch (error) {
    logSaveWriteFailure(error, chatDir, 'Failed to save chat state (async)')
  }
}

// In-flight checkpoint writer. The SDK emits an onStateSnapshot roughly every
// 5s at step boundaries while a run streams; the CLI used to persist each one
// synchronously (cloneDeep in the SDK + two multi-MB JSON.stringify + two
// blocking disk writes on the CLI), all on the render/input thread. In long
// sessions that periodic stall is what users experience as "freezes". Instead,
// schedule the write asynchronously and collapse bursts to a single in-flight
// write, always persisting the latest state.
// Coalesced per chat directory: a burst of checkpoints for the same chat
// collapses to the latest, but a checkpoint for one chat can never displace a
// still-queued checkpoint for another (e.g. the final flush of a chat the
// user just switched away from).
const pendingCheckpoints = new Map<string, LiveChatState>()
let checkpointDrain: Promise<void> | null = null

async function drainCheckpoints(): Promise<void> {
  while (pendingCheckpoints.size > 0) {
    // Yield first so serialization never runs on the same tick that scheduled
    // us (that tick is often mid-render or handling a keystroke). Entries stay
    // in the map while we yield, so the synchronous exit flush can still write
    // them if the process quits before we resume.
    await new Promise<void>((resolve) => setImmediate(resolve))
    const entry = pendingCheckpoints.entries().next()
    if (entry.done) {
      break
    }
    const [chatDir, state] = entry.value
    pendingCheckpoints.delete(chatDir)
    // Write to the chat dir captured at schedule time: the current chat may
    // have rotated (/new, /history resume) while this write sat in the queue,
    // and resolving the dir here would dump this state into the wrong chat.
    await saveChatStateAsync(state.runState, state.messages, chatDir)
  }
}

/**
 * Schedule an asynchronous, coalescing checkpoint save. Safe to call at a high
 * rate: only one write runs at a time and intermediate states are dropped in
 * favor of the latest (per chat). Use this for periodic in-flight checkpoints;
 * use the synchronous saveChatState for one-shot authoritative saves (turn
 * completion, exit flush).
 */
export function scheduleCheckpointSave(
  runState: RunState,
  messages: ChatMessage[],
  chatDir: string = resolveCurrentChatDir(),
): void {
  pendingCheckpoints.set(chatDir, { runState, messages })
  if (!checkpointDrain) {
    checkpointDrain = drainCheckpoints().finally(() => {
      checkpointDrain = null
    })
  }
}

/**
 * Wait until all queued/in-flight checkpoint writes have flushed. Call this
 * before an authoritative synchronous save (turn completion / error) so that a
 * still-running async write can't land after — and clobber — the final state.
 * Relies on the run having stopped scheduling checkpoints (the SDK's snapshot
 * timer stops once the run settles), so the queue reaches idle.
 */
export async function settleCheckpointSave(): Promise<void> {
  await checkpointDrain
}

type PersistedChatFile = {
  /** The parsed document, or null when there was nothing usable to restore. */
  value: unknown
  outcome: 'ok' | 'missing' | 'damaged' | 'unpreserved' | 'unreadable'
}

/**
 * Read one chat-state file, and set its bytes aside before anything can
 * replace them.
 *
 * Substituting an empty transcript for an unreadable one used to be a one-way
 * door: the load returned `messages: []`, the next turn's save wrote that empty
 * array straight back over the file, and a transcript that a human (or a later
 * fix) could still have read was gone. A partial write, a half-flushed file or
 * a shape we simply do not recognise are all recoverable by hand — but only
 * while the bytes exist. So quarantine first, exactly as
 * freebuff-desktop's session journal and state file do, and only then let the
 * caller carry on with a fallback.
 *
 * Quarantining happens on the read, not the write: the writer would have to
 * re-read and re-parse a multi-MB transcript on every 5s checkpoint to notice,
 * and the read is already the moment we know the file is unusable.
 *
 * A file that parses but is not the right kind of document counts as damaged.
 * Four bytes of `null` parse perfectly well into `null`, and that is the exact
 * artifact support ticket #180 reported — it must be preserved too, not nodded
 * through as "no messages".
 */
function readPersistedChatFile(
  filePath: string,
  part: 'runState' | 'messages',
): PersistedChatFile {
  let raw: string
  try {
    raw = fs.readFileSync(filePath, 'utf8')
  } catch (error) {
    const code = (error as NodeJS.ErrnoException).code
    if (code === 'ENOENT') {
      return { value: null, outcome: 'missing' }
    }
    // EACCES, EISDIR, EIO: there may be perfectly good bytes behind this, and
    // we cannot even copy them. Do not touch the file.
    logger.warn({ filePath, part, code }, 'Could not open persisted chat state')
    return { value: null, outcome: 'unreadable' }
  }

  let parsed: unknown
  try {
    parsed = JSON.parse(raw)
  } catch (error) {
    return quarantinePersistedChatFile(filePath, part, {
      reason: 'unparseable',
      bytes: raw.length,
      error: error instanceof Error ? error.message : String(error),
    })
  }

  const shapeOk =
    part === 'messages'
      ? Array.isArray(parsed)
      : !!parsed && typeof parsed === 'object' && !Array.isArray(parsed)
  if (!shapeOk) {
    return quarantinePersistedChatFile(filePath, part, {
      reason: 'wrong-shape',
      bytes: raw.length,
      parsedType: parsed === null ? 'null' : typeof parsed,
    })
  }

  return { value: parsed, outcome: 'ok' }
}

function quarantinePersistedChatFile(
  filePath: string,
  part: 'runState' | 'messages',
  detail: Record<string, unknown>,
): PersistedChatFile {
  const quarantine = `${filePath}.corrupt-${randomUUID()}`
  try {
    fs.renameSync(filePath, quarantine)
  } catch (error) {
    logger.error(
      {
        filePath,
        part,
        ...detail,
        error: error instanceof Error ? error.message : String(error),
      },
      'Damaged chat state could not be set aside; leaving it untouched',
    )
    return { value: null, outcome: 'unpreserved' }
  }
  // Kept, not deleted: the user's transcript is in there, and a later fix or a
  // support request can still get it back.
  logger.warn(
    { filePath, part, quarantine, ...detail },
    'Damaged chat state moved aside; restoring what is left',
  )
  return { value: null, outcome: 'damaged' }
}

/**
 * Load both RunState and ChatMessage[] from a specific chat directory or the most recent one.
 * When chatId is provided, it is used to locate the chat directory; otherwise the most
 * recently modified chat directory is used.
 * Returns null if no previous chat exists or files can't be parsed.
 */
export function loadMostRecentChatState(
  chatId?: string,
): SavedChatState | null {
  try {
    let chatDir: string | null = chatDirOverride ?? null

    if (!chatDir && chatId && chatId.trim().length > 0) {
      const baseDir = path.join(getProjectDataDir(), 'chats')
      const candidateDir = path.join(baseDir, chatId.trim())
      if (
        fs.existsSync(candidateDir) &&
        fs.statSync(candidateDir).isDirectory()
      ) {
        chatDir = candidateDir
      } else {
        logger.debug(
          { candidateDir, chatId },
          'Requested chatId directory not found, falling back to most recent chat directory',
        )
      }
    }

    if (!chatDir) {
      chatDir = getMostRecentChatDir()
    }

    if (!chatDir) {
      logger.debug('No previous chat directory found')
      return null
    }

    const runStatePath = path.join(chatDir, RUN_STATE_FILENAME)
    const messagesPath = path.join(chatDir, CHAT_MESSAGES_FILENAME)

    // Parse the two files independently: a missing or torn run-state.json
    // must not lose the transcript, and vice versa. Restore whatever is
    // readable and fall back for the rest.
    const loadedRunState = readPersistedChatFile(runStatePath, 'runState')
    const loadedMessages = readPersistedChatFile(messagesPath, 'messages')

    // Either file holds bytes we could neither read nor move aside. We must
    // not hand back a state whose next save would land on top of them: a
    // rename replaces a file the process cannot read, so "unreadable" is not
    // "unwritable". Refusing the whole chat sends the session to a fresh chat
    // directory, which leaves this one exactly as the user found it.
    const unsafe = (outcome: PersistedChatFile['outcome']) =>
      outcome === 'unpreserved' || outcome === 'unreadable'
    if (unsafe(loadedRunState.outcome) || unsafe(loadedMessages.outcome)) {
      logger.error(
        {
          runStatePath,
          messagesPath,
          runStateOutcome: loadedRunState.outcome,
          messagesOutcome: loadedMessages.outcome,
        },
        'Refusing to resume a chat whose state could not be read or set aside',
      )
      return null
    }

    let runState: RunState | null = loadedRunState.value as RunState | null
    let messages: ChatMessage[] | null = loadedMessages.value as
      | ChatMessage[]
      | null

    if (!runState && !messages) {
      logger.debug(
        { runStatePath, messagesPath },
        'No readable state files in chat directory',
      )
      return null
    }

    runState ??= {
      output: {
        type: 'error',
        message: 'Previous run state could not be restored.',
      },
    } as RunState
    runState.traceSessionId ??= randomUUID()
    messages ??= []

    const resolvedChatId = path.basename(chatDir)

    logger.info(
      {
        runStatePath,
        messagesPath,
        messageCount: messages.length,
        chatId: resolvedChatId,
      },
      'Loaded chat state from chat directory',
    )

    return { runState, messages, chatId: resolvedChatId }
  } catch (error) {
    logger.error(
      {
        error: error instanceof Error ? error.message : String(error),
      },
      'Failed to load chat state',
    )
    return null
  }
}

import * as fs from 'fs'
import { randomUUID } from 'node:crypto'
import * as path from 'path'

// Unique per-write temp suffix. A plain `${pid}.tmp` collides when a sync
// exit-flush and an async checkpoint write target the same file concurrently
// (both share the pid): they'd write and rename the SAME temp path, tearing
// each other's output. A random component makes every write self-contained.
function tempPathFor(filePath: string): string {
  return `${filePath}.${process.pid}.${randomUUID()}.tmp`
}

/**
 * Owner read/write only: these files are chat transcripts and run state. The
 * temp file is created with this mode and the rename carries the inode, so
 * every rewrite also replaces an older world-readable file with a private one.
 */
const PRIVATE_FILE_MODE = 0o600

/**
 * Every filesystem call an atomic write makes, as one seam.
 *
 * A successful fsync leaves no trace in the resulting file: the only way to
 * tell a durable write from a non-durable one, short of cutting the machine's
 * power, is the sequence of syscalls it makes. Tests substitute a recording
 * delegate here to assert that sequence. Modelled on freebuff-desktop's
 * `JournalStore` (server/repo/session-journal.ts), which exists for the same
 * reason.
 */
export type AtomicWriteFs = {
  openSync: typeof fs.openSync
  writeFileSync: typeof fs.writeFileSync
  fsyncSync: typeof fs.fsyncSync
  closeSync: typeof fs.closeSync
  renameSync: typeof fs.renameSync
  unlinkSync: typeof fs.unlinkSync
  promises: {
    open: typeof fs.promises.open
    rename: typeof fs.promises.rename
    unlink: typeof fs.promises.unlink
  }
}

const nodeAtomicWriteFs: AtomicWriteFs = fs

let activeFs: AtomicWriteFs = nodeAtomicWriteFs

/**
 * Test-only seam, in the style of `setChatDirOverrideForTesting`: bun's
 * `mock.module` leaks across test files, so the durability tests inject a
 * recording delegate instead (see docs/testing.md: DI over module mocking).
 */
export function setAtomicWriteFsForTesting(
  override: AtomicWriteFs | undefined,
): void {
  activeFs = override ?? nodeAtomicWriteFs
}

/**
 * Flush the directory entry itself, so the *name* created by the rename is on
 * disk too. Best-effort: by the time this runs the rename has already
 * succeeded, so an fsync a filesystem refuses for a directory (win32 cannot
 * open one through node:fs at all; some network and fuse mounts answer EINVAL)
 * must not turn a completed write into a thrown error.
 */
function syncDirectory(directory: string): void {
  if (process.platform === 'win32') return
  try {
    const handle = activeFs.openSync(directory, 'r')
    try {
      activeFs.fsyncSync(handle)
    } finally {
      activeFs.closeSync(handle)
    }
  } catch {
    // See above: the data and the new name are already published.
  }
}

async function syncDirectoryAsync(directory: string): Promise<void> {
  if (process.platform === 'win32') return
  try {
    const handle = await activeFs.promises.open(directory, 'r')
    try {
      await handle.sync()
    } finally {
      await handle.close()
    }
  } catch {
    // See syncDirectory.
  }
}

/**
 * Write a file atomically *and durably*: write to a temp file in the same
 * directory, flush it, then rename over the target and flush the directory.
 * Chat files grow to multiple MB and are rewritten on every agent step, so a
 * plain writeFileSync interrupted by a crash/kill leaves truncated JSON that
 * hides the chat from /history.
 *
 * The fsync is not optional. Rename is atomic, not durable: flush the contents
 * before publishing the new name, or a power loss can leave a correctly-sized
 * file of NUL bytes — which is what a user whose machine was hard-crashed
 * mid-session got, in place of their whole transcript. Same reasoning, same
 * discipline, as freebuff-desktop's session journal and state file.
 *
 * The cost is deliberate. Every caller writes session state a user cannot
 * reproduce, at human rates — a few writes per turn, plus a checkpoint drain
 * that coalesces to one in-flight write per chat — so none of them pays an
 * fsync per keystroke or per token.
 */
export function writeFileAtomic(filePath: string, data: string): void {
  const tmpPath = tempPathFor(filePath)
  try {
    // 'wx' so a temp path that somehow already exists is a loud error rather
    // than a silent clobber of another write's bytes.
    const fd = activeFs.openSync(tmpPath, 'wx', PRIVATE_FILE_MODE)
    try {
      // Given an fd, writeFileSync loops until every byte is written; a bare
      // writeSync may not.
      activeFs.writeFileSync(fd, data)
      activeFs.fsyncSync(fd)
    } finally {
      activeFs.closeSync(fd)
    }
    activeFs.renameSync(tmpPath, filePath)
    syncDirectory(path.dirname(filePath))
  } catch (error) {
    try {
      activeFs.unlinkSync(tmpPath)
    } catch {
      // Ignore cleanup errors; the original error is what matters
    }
    throw error
  }
}

/**
 * Async counterpart to writeFileAtomic. Used by the in-flight checkpoint writer
 * so serializing + flushing a multi-MB transcript doesn't block the CLI's
 * render/input thread. Same tmp-flush-rename-flush guarantee; the flushes run
 * on the libuv threadpool, not the render thread.
 */

/**
 * The async pair of saves in a checkpoint drain both target files and both
 * flush the same directory after the rename (one fsync per write). The sibling
 * helpers below let a caller rename both files first and pay that directory
 * fsync once: flush each tmp file, publish both names, then flush the
 * directory that now holds both.
 *
 * Splitting the write at the rename changes nothing about ordering — the tmp
 * file is still flushed to the platter before its name exists — it only stops
 * flushing a directory whose entry is about to change again a microtask later.
 */
export type AtomicRename = {
  rename: () => Promise<void>
  tmpPath: string
  targetPath: string
}

export async function prepareAtomicWrite(
  filePath: string,
  data: string,
): Promise<AtomicRename> {
  const tmpPath = tempPathFor(filePath)
  try {
    const handle = await activeFs.promises.open(
      tmpPath,
      'wx',
      PRIVATE_FILE_MODE,
    )
    try {
      await handle.writeFile(data)
      await handle.sync()
    } finally {
      await handle.close()
    }
    return {
      tmpPath,
      targetPath: filePath,
      rename: () => activeFs.promises.rename(tmpPath, filePath),
    }
  } catch (error) {
    try {
      await activeFs.promises.unlink(tmpPath)
    } catch {
      // Ignore cleanup errors; the original error is what matters
    }
    throw error
  }
}

/** Commit prepared writes: publish every name, then flush the shared
 *  directory once. `prepareAtomicWrite` already dropped prepared tmp files it
 *  could not flush, so a failure here (a rename error) only needs the same
 *  unlink sweep for whatever has not been renamed yet. */
export async function commitAtomicWrites(
  writes: AtomicRename[],
): Promise<void> {
  try {
    for (const write of writes) {
      await write.rename()
    }
  } catch (error) {
    for (const write of writes) {
      try {
        await activeFs.promises.unlink(write.tmpPath)
      } catch {
        // Either renamed already or already gone.
      }
    }
    throw error
  }
  if (writes.length > 0) {
    await syncDirectoryAsync(path.dirname(writes[0].targetPath))
  }
}

+/**
+ * The state the SDK hands a checkpoint is a shallow snapshot whose message
+ * objects are shared with the live run (snapshotSessionState): anything that
+ * mutates them — streaming text appended to the tail message, the CLI's
+ * setMessages replacing the array — always writes through a new container

+ * mutation-relevant scalars are digested cheaply instead — values the next
