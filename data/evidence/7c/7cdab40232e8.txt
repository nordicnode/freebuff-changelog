/**
 * The two faces of the Freebuff workspace (web and hosted Desktop).
 *
 * `code` is the coding-agent product: local and cloud projects, terminals,
 * split panes, GitHub, templates, Sites controls and the full model catalog.
 *
 * `chat` is the cloud-only product for people who are not developers: plain
 * chats, simple cloud projects started from scratch (with the files the user
 * drops in), a one-row composer, a Files/Browser-only right panel and one
 * model name ("Freebuff 1.5") whose slider stops stand for the three presets.
 *
 * A chat belongs to a mode through its project: a cloud chat without a
 * project is a Chat-mode chat (Code mode has no standalone chats any more),
 * a project's chats take the project's mode, and a chat bound to one of the
 * user's computers (device execution) is always Code mode. A project's mode
 * is set once, at creation, and lives in the server-owned part of its
 * settings (`settings->>'mode'`); absent means `code`, which is what every
 * project created before modes existed is.
 *
 * `buffy` is the third face (buffy.ts): one endless thread per account with
 * Buffy, the assistant that works the other two modes on the user's behalf.
 * No project is ever in it and no chat joins it through a project: the
 * account's one Buffy thread is the only chat there, marked by the server
 * (`buffy` on the chat).
 */

File packages/agent-runtime/src/conversation-summary.ts:
/**
 * Compaction by structured summary, prepared in the background.
 *
 * The goals: a summary is ready before it is needed (low latency), it keeps
 * decisions, requirements and next steps (accurate), and a user who returns
 * after the prompt cache expired sends a small request (no cold start).
 *
 * 1. When a turn ends (or is cancelled or fails) with the context at
 *    FIRST_SUMMARY_TOKENS, or SUMMARY_GROWTH_TOKENS past the last summary,
 *    the runtime asks the agent's own model for a summary of the whole
 *    conversation. Mid-turn, after a step, it asks only once the context is
 *    in the band below the compaction threshold (`summaryBandStart`), the
 *    only summary the limit will use. The request is the agent's normal
 *    request plus one instruction, so the provider reads the history from its
 *    warm cache. It runs in the background. A new summary replaces the old
 *    one.
 * 2. The full history is still sent while the cache is warm.
 * 3. The summary is used only when the user returns after the model's idle
 *    threshold (`summaryIdleMs`), or when the context reaches its limit.
 *    Then the request is the summary, the messages after it, and the new
 *    message.
 *
 * The summary waits on the agent state (`compactionSummary`), outside the
 * history. A summary that finishes after its run ended reaches the next run
 * of the same conversation through an in-process store.
 */

/**
 * The instruction a root agent appends to its own history to ask its model
 * for a structured summary of the conversation (packages/agent-runtime/src/
 * conversation-summary.ts). It is the request's last message, after the same
 * system prompt, tools and history the agent's own requests send, so the
 * provider serves the history from its prompt cache.
 *
 * The section format follows Factory's anchored iterative summary
 * (https://factory.com/news/evaluating-compression) as OpenCode words it.
 */
export const COMPACTION_SUMMARY_PROMPT_OPENING =
  'Stop working on the task. Write a structured summary of this whole conversation.'

/** `codebuff_metadata` key that marks a summary request. */
export const COMPACTION_SUMMARY_METADATA_KEY = 'compaction_summary'

/**
 * Which summary a root asks for (`compactContext.summary`). The default is a
 * coding task's: one objective, its work state, its files. `personal` is for a
 * conversation that is not one task but an ongoing thread with one person
 * (a personal assistant's): many small subjects over days, where what must
 * survive is the person, what they asked of the agent, and what is still open.
 */
export type CompactionSummaryStyle = 'personal'

/** The first heading of each format: where a summary starts in a model's reply. */
export const COMPACTION_SUMMARY_FIRST_HEADING = '## Objective'
export const PERSONAL_SUMMARY_FIRST_HEADING = '## Standing Requests'

/**
 * The `personal` summary. It follows what the personal assistants that keep
 * one thread per person carry between turns: a small profile, the open items,
 * and a recap of what was said, with the newest turns kept word for word
 * beside it by the runtime (conversation-summary.ts, `personalTailTokens`).
 * Days are written as dates so "yesterday" still means something a week on.
 */
function buildPersonalSummaryPrompt(targetTokens: number): string {
  return `${COMPACTION_SUMMARY_PROMPT_OPENING}
[freebuff product prompt text omitted]

This conversation is one ongoing thread with one person, about many things over time. It is not one task. Summarize what the two of you said and what you did for them. Do not restate your system instructions. If the conversation starts with an earlier summary, merge it into this one, section by section: newer messages take precedence, and nothing still true or still open is dropped because it is old.

Use exactly this format. Write "(none)" for an empty section.

${PERSONAL_SUMMARY_FIRST_HEADING}
- [what they asked you to always or never do, and how they want to be addressed and answered; their exact words, with the date when known]

## About Them
- [lasting things they told you or that you learned for certain: their name, the people, places, work and tastes in their life, the accounts and apps they use; exact values]

## How They Text
- [the language they write in, how long, how formal, capitals, punctuation, emoji, slang: enough to keep matching them]

## Open Threads
- [each thing still in motion: what they want, where it stands, what happens next, and who is waiting on whom]

## Handed Off
-[freebuff product prompt text omitted]

## Scheduled
- [each reminder, schedule or watcher you set, with its id and when it fires; and anything you said you would do or tell them later]

## Timeline
- [YYYY-MM-DD: what they asked for that day and how it ended, a line per subject, oldest first]

## Details
- [exact things you could not easily find again: addresses, names, numbers, amounts, dates, links, ids, the paths of files and folders you made, the ids and titles of documents you made]

## Latest Exchange
- [their latest message, and what you last answered or did]

Rules:
- Use short single-line bullets, not paragraphs.
-[freebuff product prompt text omitted]
- Keep their exact wording for requests and preferences. Keep exact names, numbers, ids, links and paths.
- Say only what the conversation shows. Record something as sent, ordered, saved, started or finished only if it succeeded; otherwise record where it stopped.
- A thread that ended goes to the Timeline as one line and leaves Open Threads. When the Timeline grows long, merge its oldest days into one line per week, then per month.
- Leave out how you did things (tool calls, searches, retries) unless the next step depends on it.
- Name each image or file they sent or you showed, by name, with what it was. You cannot see it again without opening it.
- Keep every heading exactly as written here, in English. Write the text under the headings in the language they write in.

export function buildCompactionSummaryPrompt(
  targetTokens: number,
  style?: CompactionSummaryStyle,
): string {
  if (style === 'personal') return buildPersonalSummaryPrompt(targetTokens)
  return `${COMPACTION_SUMMARY_PROMPT_OPENING}
The summary replaces the conversation history. Another agent continues the work from it, with the same system prompt and tools, so it must be able to continue without the original messages.

Summarize what the user and the assistant said and did. Do not restate the system prompt, repository instructions or environment details: the next agent gets them separately. If the conversation starts with an earlier summary, merge it into this one. Newer messages take precedence over it.

Use exactly this format. Write "(none)" for an empty section.

## Objective
- [one or two sentences: what the user wants to accomplish]

## Requirements
- [constraints, preferences and scope limits the user stated; keep their exact wording when it matters]

## Decisions
- [decisions made and why]

## Work State
### Completed
- [finished items relevant to the objective, including from earlier summaries. Name each outcome and any check that confirmed it]
### Active
- [for each unfinished item, say what has already changed and what remains to do or verify]
### Blocked
- [what blocks progress, and why]

## Next Move
1. [the next actions, in order]

## Relevant Files
- \`path\`: [why it matters, what changed in it]

## Important Context
- [facts the next agent cannot easily find again: exact commands, error messages, values, identifiers, URLs, and test or build results]

## Latest Exchange
- [the user's latest message, and what the assistant last did or answered]

Rules:
- Use short single-line bullets, not paragraphs.
- Prefer short references over restating details. Leave out what the next agent can get again from the code or files.
- Keep exact file paths, symbols, commands, error text and identifiers.
- Keep user requests that are still open, with their exact wording. Drop the ones newer messages resolved.
- Keep the state of changes: uncommitted, committed, pushed, in review or merged.
- Name each image the user attached or a tool showed, by file name when it has one, with what it showed. The next agent cannot see any of them and must open the file again to look.
- Keep every heading exactly as written here, in English. Write the text under the headings in the language the user writes in.
- Keep the summary under about ${targetTokens} tokens.
-[freebuff product prompt text omitted]
}

/**
 * The two faces of the Freebuff workspace (web and hosted Desktop).
 *
 * `code` is the coding-agent product: local and cloud projects, terminals,
 * split panes, GitHub, templates, Sites controls and the full model catalog.
 *
 * `chat` is the cloud-only product for people who are not developers: plain
 * chats, simple cloud projects started from scratch (with the files the user
 * drops in), a one-row composer, a Files/Browser-only right panel and one
 * model name ("Freebuff 1.5") whose slider stops stand for the three presets.
 *
 * A chat belongs to a mode through its project: a cloud chat without a
 * project is a Chat-mode chat (Code mode has no standalone chats any more),
 * a project's chats take the project's mode, and a chat bound to one of the
 * user's computers (device execution) is always Code mode. A project's mode
 * is set once, at creation, and lives in the server-owned part of its
 * settings (`settings->>'mode'`); absent means `code`, which is what every
 * project created before modes existed is.
 *
 * `buffy` is the third face (buffy.ts): one endless thread per account with
 * Buffy, the assistant that works the other two modes on the user's behalf.
 * No project is ever in it and no chat joins it through a project: the
 * account's one Buffy thread is the only chat there, marked by the server
 * (`buffy` on the chat).
 */

/**
 * Compaction by structured summary, prepared in the background.
 *
 * The goals: a summary is ready before it is needed (low latency), it keeps
 * decisions, requirements and next steps (accurate), and a user who returns
 * after the prompt cache expired sends a small request (no cold start).
 *
 * 1. When a turn ends (or is cancelled or fails) with the context at
 *    FIRST_SUMMARY_TOKENS, or SUMMARY_GROWTH_TOKENS past the last summary,
 *    the runtime asks the agent's own model for a summary of the whole
 *    conversation. Mid-turn, after a step, it asks only once the context is
 *    in the band below the compaction threshold (`summaryBandStart`), the
 *    only summary the limit will use. The request is the agent's normal
 *    request plus one instruction, so the provider reads the history from its
 *    warm cache. It runs in the background. A new summary replaces the old
 *    one.
 * 2. The full history is still sent while the cache is warm.
 * 3. The summary is used only when the user returns after the model's idle
 *    threshold (`summaryIdleMs`), or when the context reaches its limit.
 *    Then the request is the summary, the messages after it, and the new
 *    message.

 * history. A summary that finishes after its run ended reaches the next run
 * of the same conversation through an in-process store.

/**
 * Whether the live prompt arrived after the conversation sat idle longer than
 * `idleMs`, measured from the later of the last turn end or checkpoint
 * (`lastActivityAt`) and the last assistant message. Never true after an
 * answer or for a continuation prompt.
 */
export function isIdleReturn(
  messages: Message[],
  idleMs = SUMMARY_IDLE_MS,
  now = Date.now(),
  lastActivityAt?: number,
): boolean {
  const promptIndex = messages.findLastIndex((m) =>
    m.tags?.includes('USER_PROMPT'),
  )
  if (promptIndex <= 0) return false
  if (messages.slice(promptIndex + 1).some((m) => m.role === 'assistant'))
    return false
  const prompt = messages[promptIndex]
  if (prompt.tags?.includes(CONTINUATION_PROMPT_TAG)) return false
  const sentAt = prompt.sentAt ?? now
  // The later of the two: a runtime that predates `lastActivityAt` keeps the
  // field but does not move it, so it can be older than the last reply.
  let last = lastActivityAt
  for (let i = promptIndex - 1; i >= 0; i--) {
    const message = messages[i]
    if (message.role !== 'assistant') continue
    if (message.sentAt !== undefined)
      last = Math.max(last ?? message.sentAt, message.sentAt)
    break
  }
  return last !== undefined && sentAt - last > idleMs
}

/** A short, stable id for a message's content (FNV-1a). */
export function messageFingerprint(message: Message): string {
  const text =
    message.role +
    '\u0000' +
    (message.role === 'tool' ? message.toolCallId : '') +
    '\u0000' +
    JSON.stringify(message.content)
  let hash = 0x811c9dc5
  for (let i = 0; i < text.length; i++) {
    hash ^= text.charCodeAt(i)
    hash = Math.imul(hash, 0x01000193)
  }
  return `${(hash >>> 0).toString(16)}:${text.length}`
}

/**

 * covered message is gone (an edit removed it). Searched from its recorded
 * position first, then from the end: dropped scaffolding can move it.

  messages: Message[],
  summary: ConversationSummary,
): number {

  if (
    messages[expected] &&
    messageFingerprint(messages[expected]) === summary.anchor
  )
    return expected
  for (let i = Math.min(expected, messages.length - 1); i >= 0; i--) {
    if (messageFingerprint(messages[i]) === summary.anchor) return i
  }
  return -1
}

/**

 * or its length when none fits. Never reaches back over an earlier summary.

function coveredTailStart(covered: Message[], tailTokens: number): number {

  for (let i = covered.length - 1; i >= 0; i--) {
    const message = covered[i]
    if (message.tags?.includes(CONVERSATION_SUMMARY_TAG)) break
    if (
      message.role !== 'user' ||
      message.tags?.includes('USER_PROMPT') !== true

    if (countTokensMessages(turn) > tailTokens) break

  }
  return start

/**
 * The summary in a model's reply, or '' when the reply is not one. It starts
 * at its first heading; a preamble before it is dropped. The format's headings
 * are in English, but a model can translate them with the rest (GLM 5.3 Flash
 * did on 12% of its summaries, prod 2026-10-30, and each one was thrown away),
 * so a reply with the format's shape counts without the name.
 */
export function summaryFromReply(
  reply: string,
  style?: CompactionSummaryStyle,
): string {
  const text = reply.replace(/<think>[\s\S]*?<\/think>/g, '')
  const english = text.indexOf(
    style === 'personal'
      ? PERSONAL_SUMMARY_FIRST_HEADING
      : COMPACTION_SUMMARY_FIRST_HEADING,
  )
  if (english >= 0) return text.slice(english).trim()
  const sections = [...text.matchAll(/^## /gm)]
  return sections.length >= MIN_TRANSLATED_SECTIONS
    ? text.slice(sections[0].index).trim()
    : ''
}

/**
 * Asks the model for the summary. `history` must be what the agent's next
 * request sends (`summaryRequestHistory`), or the cache misses.
 */
export async function generateConversationSummary(params: {
  summaryId?: string
  history: Message[]
  system: string
  /** The provider's count when the summary was requested, if known. */
  contextTokens?: number
  targetTokens: number
  /** Which format to ask for; absent is a coding task's. */
  style?: CompactionSummaryStyle
  maxOutputTokens?: number
  stream: SummaryStream
  signal: AbortSignal
  onUsage?: (usage: ModelUsageData) => void
}): Promise<{ summary: ConversationSummary; finishReason?: string }> {
  const { history } = params
  const summaryId = params.summaryId ?? crypto.randomUUID()
  const createdAt = Date.now()
  const anchor = history.findLastIndex((m) => !isEphemeral(m))
  if (anchor < 0)
    throw new SummaryError('no_history', 'No history to summarize')
  let finishReason: string | undefined
  const stream = params.stream({
    messages: [
      systemMessage(params.system),
      ...history,
      // Tagged so the server's prompt classifier skips it as product text.
      userMessage({
        content: withSystemInstructionTags(
          buildCompactionSummaryPrompt(params.targetTokens, params.style),
        ),
      }),
    ],
    maxOutputTokens:
      params.maxOutputTokens !== undefined && params.maxOutputTokens > 0
        ? Math.min(SUMMARY_MAX_OUTPUT_TOKENS, params.maxOutputTokens)
        : SUMMARY_MAX_OUTPUT_TOKENS,
    signal: params.signal,
    extraCodebuffMetadata: {
      [COMPACTION_SUMMARY_METADATA_KEY]: '1',
      compaction_summary_id: summaryId,
      compaction_request_id: summaryId,
      compaction_request_group: 'summary',
    },
    onFinishReason: (reason) => {
      finishReason = reason
    },
    onUsageReceived: (usage) => params.onUsage?.(usage),
  })
  let text = ''
  for (;;) {
    const next = await stream.next()
    if (next.done) {
      if (next.value.aborted) params.signal.throwIfAborted()
      break
    }
    const chunk = next.value
    if (chunk.type === 'error')
      throw new SummaryError('provider_error', chunk.message)
    if (chunk.type === 'text') text += chunk.text
  }
  params.signal.throwIfAborted()
  if (finishReason === 'length')
    throw new SummaryError('output_limit', 'The summary hit the output limit')
  const summaryText = summaryFromReply(text, params.style)
  if (summaryText.length < MIN_SUMMARY_CHARS)
    throw new SummaryError(
      text.trim().length < MIN_SUMMARY_CHARS ? 'empty' : 'unrecognized',
      'The model returned no summary',
    )
  return {
    finishReason,
    summary: {
      id: summaryId,
      text: summaryText,
      coveredMessages: anchor + 1,
      anchor: messageFingerprint(history[anchor]),
      ...(params.contextTokens === undefined
        ? {}
        : { contextTokens: params.contextTokens }),
      createdAt,
    },
  }
}

const conversations = new Map<string, Entry>()

function entry(key: string): Entry {
  let value = conversations.get(key)
  if (value) {
    conversations.delete(key)
  } else {
    value = {}
  }
  conversations.set(key, value)
  while (conversations.size > MAX_CONVERSATIONS) {
    const oldest = conversations.keys().next().value
    if (oldest === undefined) break
    conversations.delete(oldest)
  }
  return value
}

/** The newest of the stored summary and the one in `state`. */
function newer(
  a: ConversationSummary | undefined,
  b: ConversationSummary | undefined,
): ConversationSummary | undefined {
  if (!a) return b
  if (!b) return a
  return b.createdAt > a.createdAt ? b : a
}

/**
 * One conversation's summaries in this process: the newest finished one and
 * the one in progress. Keyed by conversation and agent, so a summary that
 * finishes after its run ended reaches the next turn.
 */
export function conversationSummaries(key: string) {
  return {
    /** The newest summary that still matches `messages`. */
    latest(
      fromState: ConversationSummary | undefined,
      messages: Message[],
    ): ConversationSummary | undefined {
      const summary = newer(fromState, entry(key).latest)
      return summary && summaryMatchesHistory(messages, summary)
        ? summary
        : undefined
    },
    pending() {
      return entry(key).pending
    },
    /** Records a summary the history now carries (after it was applied). */
    replace(summary: ConversationSummary | undefined) {
      entry(key).latest = summary
    },
    /** Starts a summary unless one is running. */
    start(
      contextTokens: number,
      run: () => Promise<ConversationSummary | null>,
    ): Promise<ConversationSummary | null> {
      const current = entry(key)
      if (current.pending) return current.pending.promise
      const promise = run()
        .then((summary) => {
          if (summary) {
            const target = entry(key)
            target.latest = newer(target.latest, summary)
          }
          return summary
        })
        .catch(() => null)
        .finally(() => {
          const target = conversations.get(key)
          if (target?.pending?.promise === promise) target.pending = undefined
        })
      current.pending = { promise, contextTokens }
      return promise
    },
  }
}

export function resetConversationSummariesForTest(): void {
  conversations.clear()
}

/** Runs `generateConversationSummary` and logs its outcome; never throws
 *  except on the caller's own abort, or a refused request when asked. */
export async function summarizeAndLog(
  params: Parameters<typeof generateConversationSummary>[0] & {
    logger: Logger
    runId: string
    model: string
    mode: 'background' | 'foreground'
    /** Started while the agent's next request may be running. */
    midTurn: boolean
    /** The run's own signal: its abort is logged and rethrown. */
    cancelSignal?: AbortSignal
    /**
     * Rethrow a request the API or provider refused (thrown by the stream: a
     * session that ended, a policy refusal, a BYOK 4xx), so the run ends on
     * that refusal as a turn's request would. An in-band failure (a 5xx or a
     * cut stream, an empty reply) still returns null: trying again can work.
     */
    rethrowRefusals?: boolean
    telemetry?: Record<string, string>
  },
): Promise<ConversationSummary | null> {
  const {
    logger,
    runId,
    model,
    mode,
    midTurn,
    cancelSignal,
    rethrowRefusals,
    telemetry,
    ...summaryParams
  } = params
  const started = Date.now()
  const summaryId = params.summaryId ?? crypto.randomUUID()
  let usage: ModelUsageData | undefined
  const log = (data: Record<string, unknown>, message: string) => {
    try {
      logger.info(
        {
          axiomEvent: COMPACTION_SUMMARY_EVENT,
          ...telemetry,
          summary_id: summaryId,
          agent_run_id: runId,
          model,
          mode,
          mid_turn: midTurn,
          context_token_count: params.contextTokens,
          duration_ms: Date.now() - started,
          ...(usage
            ? {
                input_tokens: usage.inputTokens,
                cached_input_tokens: usage.cachedInputTokens,
                output_tokens: usage.outputTokens,
              }
            : {}),
          ...data,
        },
        message,
      )
    } catch {
      // Logging must never fail a summary.
    }
  }
  try {
    const { summary, finishReason } = await generateConversationSummary({
      ...summaryParams,
      summaryId,
      onUsage: (received) => {
        usage = received
        summaryParams.onUsage?.(received)
      },
    })
    log(
      {
        outcome: 'completed',
        finish_reason: finishReason,
        summary_tokens: countTokens(summary.text),
      },
      'Conversation summary completed',
    )
    return summary
  } catch (error) {
    log(
      {
        outcome: cancelSignal?.aborted ? 'cancelled' : 'failed',
        error_kind:
          error instanceof SummaryError
            ? error.kind
            : isAbortError(error) || summaryParams.signal.aborted
              ? 'timeout'
              : 'provider_error',
        // Not allowlisted for Axiom; local/debug logs only.
        error: getErrorObject(error),
      },
      'Conversation summary failed',
    )
    if (cancelSignal?.aborted) throw error
    if (
      rethrowRefusals &&
      !(error instanceof SummaryError) &&
      !isAbortError(error) &&
      !summaryParams.signal.aborted
    )
      throw error
    return null
  }
}

/** Waits for `promise`, or rejects when `signal` aborts first. */
export function untilAborted<T>(
  promise: Promise<T>,
  signal: AbortSignal,
): Promise<T> {
  signal.throwIfAborted()
  return new Promise<T>((resolve, reject) => {
    const onAbort = () => reject(new AbortError())
    signal.addEventListener('abort', onAbort, { once: true })
    promise.then(resolve, reject).finally(() => {
      signal.removeEventListener('abort', onAbort)
    })
  })
}

/**
 * The reasoning settings for a summary request: the agent's own, at low
 * effort. On a high-effort model a summary at the limit thought for minutes
 * (a 150k-token summary took 304 s on DeepSeek V4.1 Flash) and one spent its
 * whole output budget on reasoning, leaving no summary. Anthropic models keep
 * their settings: changing thinking parameters there invalidates the cached
 * messages, the summary's whole saving.
 */
export function summaryReasoningOptions(
  model: string,
  reasoning: OpenRouterReasoningOptions | undefined,
): OpenRouterReasoningOptions | undefined {
  if (!reasoning || !('effort' in reasoning)) return reasoning
  if (model.startsWith('anthropic/')) return reasoning
  if (['none', 'minimal', 'low'].includes(reasoning.effort)) return reasoning
  return { ...reasoning, effort: 'low' }
}
