/** Strip SDK URL/person enrichment before either capture or the log mirror. */
export function sanitizeProductFunnelCapture<
  T extends {
    event: string
    properties: Record<string, unknown>
  },
>(event: T, projectToken: string): T | null {
  if (event.event !== PRODUCT_FUNNEL_EVENT) return event
  const fields: Record<string, unknown> = {}
  for (const key of productFunnelEventSchema.keyof().options) {
    if (event.properties[key] !== undefined) fields[key] = event.properties[key]
  }
  const parsed = productFunnelEventSchema.safeParse(fields)
  if (!parsed.success) return null
  // posthog-js adds this public ingest key before before_send; transport does
  // not restore it after sanitization. Never copy an untrusted event token.
  const properties: Record<string, unknown> = {
    ...parsed.data,
    token: projectToken,
  }
  for (const key of [
    'distinct_id',
    '$insert_id',
    '$session_id',
    '$is_identified',
  ]) {
    const value = event.properties[key]
    if (typeof value === 'string' || typeof value === 'boolean')
      properties[key] = value
  }
  const sanitized = { ...event, properties }
  Reflect.deleteProperty(sanitized, '$set')
  Reflect.deleteProperty(sanitized, '$set_once')
  return sanitized
}

- "prompt": what the user would send, in their voice. It must stand alone, because the agent that receives it never sees this review: name what to change or try and why, in one or two sentences, and leave the how to the agent.
When the turn changed code, include an action of kind "review" labeled "Code review" whose prompt asks the agent to review its own change with fresh eyes: switch to a better alternative where there is one, simplify the code, remove unnecessary cases, and cut anything not needed, for the cleanest, most maintainable result.
When the turn finished a feature or fix, such as merging a pull request or committing the change, include an action of kind "next" labeled "${BRAINSTORM_FOLLOW_UPS_LABEL}" whose prompt asks the agent to use the whole conversation as context and think up the most valuable follow-up ideas, including new ones no one has mentioned yet rather than only restating what was already discussed, each with why it is worth doing, ranked by value, without starting on any of them.
Every action, and "nextPrompt", must be something the user could sensibly send as their very next message, right now. Never offer a step that has to wait on something that has not happened yet, such as a merge, a deploy, a release, a review, or data arriving, like "Re-measure after deploy" while the pull request is still open: no one clicks it now. Offer only what the agent can start on as soon as it is sent.
Do not offer what was already done or declined, or two buttons that do the same thing.

Last, write "nextPrompt": the single message you would recommend the user send next, taking all of the above into account. It is offered in their empty message box, where one key fills it in for them to send. It may combine several instructions, such as the fixes the change needs and one valuable follow-up, but keep it concise: one to three short sentences in the user's voice that stand alone. Use null when nothing is worth sending.

/**
 * The review prompt for a turn's mode (TURN_REVIEW_PROMPT for Code,
 * TURN_REVIEW_CHAT_PROMPT for Chat) with the user's own recent messages from
 * across their chats, newest first, so the buttons and `nextPrompt` read the
 * way this user works: whether they open pull requests or keep changes local,
 * how much they ask for at once, their wording. Without any, the prompt alone.
 */
export function turnReviewPrompt(
  recentPrompts: readonly string[] = [],
  mode: WorkspaceMode = 'code',
): string {
  const prompt = mode === 'chat' ? TURN_REVIEW_CHAT_PROMPT : TURN_REVIEW_PROMPT
  const lines = recentPrompts
    .map((recent) => clip(recent, RECENT_PROMPT_CHARS))
    .filter(Boolean)
    .slice(0, TURN_REVIEW_RECENT_PROMPTS)

function clip(value: unknown, max: number): string {
  if (typeof value !== 'string') return ''
  const text = value.replace(/\s+/g, ' ').trim()
  return text.length > max ? `${text.slice(0, max - 1).trimEnd()}…` : text
}

/** The review in a model's reply, or null when the reply holds none (or an empty one). */
export function parseTurnReview(reply: string): TurnReview | null {
  const read = readTurnReview(reply)
  return read === 'empty' ? null : read
}

/**
 * The review in a model's reply: the first complete JSON object shaped like
 * one (it names `verdict`, `actions` or `nextPrompt`), ignoring prose and
 * other braces around it and anything after it, such as a stray `"}`.
 * `'empty'` is a well-formed review with nothing in it (the model judged there
 * was nothing to suggest); null is a reply that holds no review at all.
 */
export function readTurnReview(reply: string): TurnReview | 'empty' | null {
  const text = reply.replace(/<think>[\s\S]*?<\/think>/g, '')
  for (const candidate of jsonObjects(text)) {
    const review = reviewShaped(candidate)
    if (review !== undefined) return review
  }
  // A model sometimes drops the last string's closing quote ("nextPrompt":"…}).
  const start = text.indexOf('{')
  const end = text.lastIndexOf('}')
  if (start < 0 || end <= start) return null
  return reviewShaped(text.slice(start, end).trimEnd() + '"}') ?? null
}

/** The review in one JSON object's text, `'empty'`, or undefined when it is not one. */
function reviewShaped(json: string): TurnReview | 'empty' | undefined {
  let value: unknown
  try {
    value = JSON.parse(json)
  } catch {
    return undefined
  }
  if (!value || typeof value !== 'object' || Array.isArray(value)) return undefined
  if (!['verdict', 'actions', 'nextPrompt'].some((key) => key in value)) return undefined
  return turnReviewFrom(value) ?? 'empty'
}

/** At most this many `{` are tried, so a reply full of code costs little. */
const MAX_JSON_STARTS = 200

/**
 * Each balanced `{…}` in the text, outermost first, skipping braces inside
 * JSON strings. An object that fails to parse is followed by the ones inside
 * it, so a CSS block or a broken wrapper never hides the review after it.
 */
function* jsonObjects(text: string): Generator<string> {
  let tried = 0
  for (let start = text.indexOf('{'); start >= 0 && tried < MAX_JSON_STARTS; start = text.indexOf('{', start + 1)) {
    tried++
    let depth = 0
    let inString = false
    for (let i = start; i < text.length; i++) {
      const char = text[i]
      if (inString) {
        if (char === '\\') i++
        else if (char === '"') inString = false
      } else if (char === '"') inString = true
      else if (char === '{') depth++
      else if (char === '}' && --depth === 0) {
        yield text.slice(start, i + 1)
        break
      }
    }
  }
}

/** A review from untrusted JSON (a model's reply, a stored event), or null. */
export function turnReviewFrom(value: unknown): TurnReview | null {
  if (!value || typeof value !== 'object') return null
  const raw = value as Record<string, unknown>
  const verdict = TURN_REVIEW_VERDICTS.includes(raw.verdict as TurnReviewVerdict)
    ? (raw.verdict as TurnReviewVerdict)
    : null
  const actions: TurnReviewAction[] = []
  for (const item of Array.isArray(raw.actions) ? raw.actions : []) {
    const label = clip(item?.label, MAX_LABEL_CHARS)
    const prompt = clip(item?.prompt, MAX_PROMPT_CHARS)
    if (label && prompt)
      actions.push({
        label,
        prompt,
        kind: TURN_REVIEW_ACTION_KINDS.includes(item?.kind) ? item.kind : 'next',
      })
    if (actions.length === MAX_TURN_REVIEW_ACTIONS) break
  }
  const nextPrompt = clip(raw.nextPrompt, MAX_PROMPT_CHARS) || null
  if (!verdict && !actions.length && !nextPrompt) return null
  const reason = (verdict && clip(raw.reason, MAX_REASON_CHARS)) || null
  return { verdict, reason, actions, nextPrompt }
}

  'attempt',

/**
 * Fields a NEWER client may add to the exact set above (COD-365). Optional
 * on the check, not appended to the exact set, because a released CLI or
 * Desktop binary still sends the six-field shape and its events must stay
 * valid. Each is still validated when present -- a malformed value rejects
 * the whole event exactly as a malformed required field does.
 */
const FIRST_PARTY_VIEW_ACK_OPTIONAL_FIELDS = [
  'client_event_id',
  'sample_rate',
] as const
const FIRST_PARTY_VIEW_ACK_CLIENT_EVENT_ID_RE = /^[A-Za-z0-9._:-]{1,128}$/

const FIRST_PARTY_VIEW_ACK_PLACEMENTS = new Map<string, string>(
  PLACEMENT_SLOTS.map((slot) => [slot.id, slot.surface]),
)

/**
 * Validate the only telemetry payload a client may send for view
 * acknowledgement. Returning null rejects the entire event: dropping an
 * unsafe field but retaining the count would make malformed client input look
 * like a real rendering signal.
 */
export function createFirstPartyViewAckTelemetry(
  input: unknown,
): FirstPartyViewAckTelemetry | null {
  if (input == null || typeof input !== 'object' || Array.isArray(input)) {
    return null
  }
  const record = input as Record<string, unknown>
  const keys = Object.keys(record)
  const required = new Set<string>(FIRST_PARTY_VIEW_ACK_FIELDS)
  const optional = new Set<string>(FIRST_PARTY_VIEW_ACK_OPTIONAL_FIELDS)
  if (
    keys.some((key) => !required.has(key) && !optional.has(key)) ||
    FIRST_PARTY_VIEW_ACK_FIELDS.some((key) => !(key in record))
  ) {
    return null
  }
  const clientEventId = record.client_event_id
  const sampleRate = record.sample_rate
  if (
    (clientEventId !== undefined &&
      (typeof clientEventId !== 'string' ||
        !FIRST_PARTY_VIEW_ACK_CLIENT_EVENT_ID_RE.test(clientEventId))) ||
    (sampleRate !== undefined &&
      (typeof sampleRate !== 'number' ||
        !Number.isInteger(sampleRate) ||
        sampleRate < 1))
  ) {
    return null
  }

  const surface = record.surface
  const placementId = record.placement_id
  const outcome = record.outcome
  const attempt = record.attempt
  const durationMs = record.duration_ms
  const clientFamily = record.client_family
  if (
    typeof surface !== 'string' ||
    typeof placementId !== 'string' ||
    // Shipped clients still ack under legacy ids (COD-1043): look up by the
    // canonical id, keep the id as sent.
    FIRST_PARTY_VIEW_ACK_PLACEMENTS.get(canonicalPlacementId(placementId)) !==
      surface ||
    !FIRST_PARTY_VIEW_ACK_OUTCOMES.includes(
      outcome as FirstPartyViewAckOutcome,
    ) ||
    typeof attempt !== 'number' ||
    !Number.isInteger(attempt) ||
    attempt < 1 ||
    attempt > 3 ||
    typeof durationMs !== 'number' ||
    !Number.isFinite(durationMs) ||
    durationMs < 0 ||
    durationMs > FIRST_PARTY_VIEW_ACK_MAX_DURATION_MS ||
    !FIRST_PARTY_VIEW_ACK_CLIENT_FAMILIES.includes(
      clientFamily as FirstPartyViewAckClientFamily,
    )
  ) {
    return null
  }
  return {
    surface,
    placement_id: placementId,
    outcome: outcome as FirstPartyViewAckOutcome,
    attempt: attempt as 1 | 2 | 3,
    duration_ms: durationMs,
    client_family: clientFamily as FirstPartyViewAckClientFamily,
    ...(clientEventId !== undefined
      ? { client_event_id: clientEventId as string }
      : {}),
    ...(sampleRate !== undefined ? { sample_rate: sampleRate as number } : {}),
  }
}

/** Keep only the allowlisted keys whose value matches the declared type
 *  (strings truncated); everything else is dropped. */
function sanitizeAllowlistedFields(
  record: Record<string, unknown>,
  fields: AxiomOnlyFieldSchema,
): Record<string, string | number | boolean> {
  const sanitized: Record<string, string | number | boolean> = {}
  for (const [key, expectedType] of Object.entries(fields)) {
    const value = record[key]
    if (typeof value !== expectedType) continue
    if (typeof value === 'string') {
      sanitized[key] = value.slice(0, 200)
    } else if (typeof value === 'number' && Number.isFinite(value)) {
      sanitized[key] = value
    } else if (typeof value === 'boolean') {
      sanitized[key] = value
    }
  }
  return sanitized
}

/**
 * Return a sanitized Axiom-only event, or null for ordinary logger payloads.
 * The event name comes from `data.axiomEvent` (the in-process marker set at
 * the log call site) or the `event` param (the wire-format field a caller
 * already extracted, e.g. the server-side sink re-checking a persisted
 * `LogRow`). Unknown keys and unexpected value types are deliberately
 * discarded.
 *
 * Matched by exact equality (not a lookup keyed on the caller-supplied name)
 * so a value like 'constructor' can't resolve through an object's prototype.
 */
export function getAxiomOnlyLogEvent(
  data: unknown,
  event?: string | null,
): AxiomOnlyLogEvent | null {
  const record =
    data != null && typeof data === 'object' && !Array.isArray(data)
      ? (data as Record<string, unknown>)
      : {}
  const eventName =
    typeof record.axiomEvent === 'string' ? record.axiomEvent : event

  if (eventName === CONTEXT_PRUNING_COMPLETED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, CONTEXT_PRUNING_FIELDS),
    }
  }
  if (eventName === STREAM_RECOVERY_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, STREAM_RECOVERY_FIELDS),
    }
  }
  if (eventName === FILE_WRITE_GUARD_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, FILE_WRITE_GUARD_FIELDS),
    }
  }
  if (eventName === TOOL_CALL_REPAIR_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, TOOL_CALL_REPAIR_FIELDS),
    }
  }
  if (eventName === MECHANICAL_COMPACTION_SKIPPED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(
        record,
        MECHANICAL_COMPACTION_SKIPPED_FIELDS,
      ),
    }
  }
  if (eventName === CONTEXT_COMPACTION_COMPLETED_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(
        record,
        CONTEXT_COMPACTION_COMPLETED_FIELDS,
      ),
    }
  }
  if (eventName === COMPACTION_FOLLOWUP_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, COMPACTION_FOLLOWUP_FIELDS),
    }
  }
  if (eventName === COMPACTION_SUMMARY_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, COMPACTION_SUMMARY_FIELDS),
    }
  }
  if (eventName === TURN_REVIEW_EVENT) {
    return {
      event: eventName,
      data: sanitizeAllowlistedFields(record, TURN_REVIEW_FIELDS),

/** Asks for the review. `history` must be what the agent's next request sends. */
export async function reviewTurn(params: {
  history: Message[]
  system: string
  /** TURN_REVIEW_PROMPT, or `turnReviewPrompt` with the user's recent messages. */
  prompt?: string
  stream: SummaryStream
  logger: Logger
  runId: string
  model: string
  telemetry?: Record<string, string>
}): Promise<TurnReview | null> {
  const started = Date.now()
  let usage: ModelUsageData | undefined
  let finishReason: string | undefined
  let outcome = 'failed'
  let review: TurnReview | null = null
  try {
    const stream = params.stream({
      messages: [
        systemMessage(params.system),
        ...params.history,
        // Tagged so the server's prompt classifier skips it as product text.
        userMessage({
          content: withSystemInstructionTags(params.prompt ?? TURN_REVIEW_PROMPT),
        }),
      ],
      signal: AbortSignal.timeout(TURN_REVIEW_TIMEOUT_MS),
      extraCodebuffMetadata: { [TURN_REVIEW_METADATA_KEY]: '1' },
      onFinishReason: (reason) => {
        finishReason = reason
      },
      onUsageReceived: (received) => {
        usage = received
      },
    })
    let text = ''
    for (;;) {
      const next = await stream.next()
      if (next.done) break
      if (next.value.type === 'error') throw new Error(next.value.message)
      if (next.value.type === 'text') text += next.value.text
    }
    const read = readTurnReview(text)
    // `empty`: a well-formed review with nothing to suggest, not a broken reply.
    outcome = read === 'empty' ? 'empty' : read ? 'completed' : 'unrecognized'
    review = read === 'empty' ? null : read
    return review
  } catch (error) {
    params.logger.debug?.({ error: getErrorObject(error) }, 'Turn review failed')
    return null
  } finally {
    try {
      params.logger.info(
        {
          axiomEvent: TURN_REVIEW_EVENT,
          ...params.telemetry,
          agent_run_id: params.runId,
          model: params.model,
          outcome,
          finish_reason: finishReason,
          verdict: review?.verdict ?? undefined,
          action_count: review?.actions.length ?? 0,
          offers_brainstorm: review?.actions.some(isBrainstormAction) ?? false,
          duration_ms: Date.now() - started,
          ...(usage
            ? {
                input_tokens: usage.inputTokens,
                cached_input_tokens: usage.cachedInputTokens,
                output_tokens: usage.outputTokens,
              }
            : {}),
        },
        'Turn review finished',
      )
    } catch {
      // Logging must never fail a review.
    }
  }
}
