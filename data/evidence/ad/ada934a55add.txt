export function createBase3(
  model: SecretAgentDefinition['model'] = OPUS_MODEL,
  options: {
    /** Overrides the per-model policy. A server-catalog row supplies it,
     *  because its `model` is an opaque handle the lookup cannot key on; its
     *  `maxContextLength` replaces the runtime's per-model budget lookup for
     *  the same reason. `lean` (with a `maxContextLength`) adds the
     *  context-budget prompt section. */
    compaction?: RootCompactionPolicy
    /** Web/Cloud owns persistent processes through managed sessions. */
    supportsBackgroundCommands?: boolean
    /** Append the "Work in batches" section, on a model in BATCHING_MODELS.
     *  For the roots that run the user's coding agent in build mode: the CLI
     *  (`createBase3CliRoot`), Desktop and Desktop Cloud. Plan mode, fast mode,
     *  sponsored runs and Web/Cloud (whose platform runs the checks) leave it
     *  off, and a lean row never gets it: its context budget says to read
     *  narrowly. */
    batching?: boolean
    /** The compiled model a catalog handle stands for, which decides the
     *  per-model parts below; defaults to `model`. */
    compiledModelId?: string
    /** Desktop and the current Cloud workspace omit todos and ask only when blocked. */
    workspace?: boolean
  } = {},
): Omit<SecretAgentDefinition, 'id'> {
  const turnEndingNote = TURN_ENDING_NOTE_MODELS.some((id) =>
    freebuffModelIdMatches(options.compiledModelId ?? model, id),
  )
  const batchingModel = BATCHING_MODELS.some((id) =>
    freebuffModelIdMatches(options.compiledModelId ?? model, id),
  )
  const { lean, ...compactContext }: RootCompactionPolicy =
    options.compaction ?? compactionPolicyForModel(model)
  const leanGuidance =
    lean && compactContext.maxContextLength
      ? `\n${leanContextGuidance(compactContext.maxContextLength)}\n`
      : ''
  const batching = options.batching === true && batchingModel && !lean
  return {
    publisher,
    model,
    providerOptions: model.startsWith('anthropic/')
      ? { only: ['amazon-bedrock'], data_collection: 'deny' }
      : { data_collection: 'deny' },
    displayName: 'Buffy',
    spawnerPrompt:
      'Single-loop coding agent that explores, edits, and verifies directly with its own tools',
    inputSchema: {
      prompt: {
        type: 'string',
        description: 'A coding task to complete',
      },
    },
    outputMode: 'last_message',
    includeMessageHistory: true,
    windowedFileReads: true,
    // A root built on this that can spawn lists those agents once, at the end
    // of its system prompt, not in a message after every user prompt that
    // stays in history.
    spawnableAgentsInSystemPrompt: true,
    // Per-model idle gap and token floor (common/src/constants/
    // compaction-policy.ts). Explicit, not `true`: a bare `true` would take
    // the runtime default instead of the model's.
    compactContext,
    ...(turnEndingNote ? { instructionsPrompt: TURN_ENDING_NOTE } : {}),
    // The section's reminder at the end of every step, dropped from history
    // after it. Read once, Flash drifts back to one call per step; repeated,
    // it keeps batching. A step prompt is deleted each step, so no request
    // extends the previous one: a provider that caches only exact prefix
    // extensions then misses most of every request (MiMo and Luna did,
    // #1158), while Flash's lanes cache partial prefixes and kept their hit
    // rate. The server skips it where it reads the user's own words
    // (`isRootStepReminderMessage`).
    ...(batching ? { stepPrompt: BASE3_BATCHING_STEP_PROMPT } : {}),
    toolNames: [
      'read_files',
      'str_replace',
      'write_file',
      'run_terminal_command',
      'code_search',
      'glob',
      'list_directory',
      ...(options.workspace ? [] : ['write_todos'] as const),
    ],

    // No deadline, time-budget or "minimal result early" wording: no product
    // host gives the model a deadline, and that wording made models invent
    // one and stop short. A host with a real deadline passes `deadlineAt`; the
    // runtime then states it.
    systemPrompt: `[freebuff product prompt text omitted]

Current date: ${PLACEHOLDER.CURRENT_DATE}.

- Size the work to the request. Do everything the user asked for, completely, and nothing they did not: no unrequested features, refactors, new files, dependencies or polish. A question gets an answer, not a change. A small or specific request gets a direct fix, without a plan, a to-do list or a survey of the codebase.
- ${options.workspace ? WORKSPACE_QUESTION_GUIDANCE : '[freebuff product prompt text omitted]'}
- Match the project's existing conventions. Verify a library is already used in the project before employing it.
- Prefer editing existing files over creating new ones. Make the fewest changes that address the request.
- Note the request's explicit requirements (exact output paths, file names, interfaces, schemas, formats, performance or resource constraints) and check the result against them before finishing.
- For performance work, do not claim a speedup without a measurement.
- Verify the final edited files: after a repair, rerun the affected checks; earlier passing results do not cover later changes. When filtering test or build output through a pipeline, preserve the tested command's exit status (for example, Bash pipefail or explicit status capture), and inspect the failure output.
- Take screenshots or drive a browser for visual inspection only when the user explicitly asks for a screenshot or a visual check.
-[freebuff product prompt text omitted]
- ${options.supportsBackgroundCommands === false ? 'Terminal commands support only SYNC on this host. Use the host-managed preview controls for persistent services; do not request BACKGROUND or use shell backgrounding. /**
 * The CLI's own base3 roots — Codebuff's DEFAULT and LITE modes, and every
 * Freebuff model the picker offers.
 *
 * Kept separate from `createBase3` rather than folded into it, because Freebuff
 * Desktop derives its toolset from `createBase3().toolNames` and would silently
 * inherit whatever is added here (THREAD_AGENT_TOOLS in
 * freebuff-desktop/.../thread-agent.ts unions that array with its own extras).
 * The bare eight are the harness; what follows is CLI product surface.
 *
 * One thing is load-bearing, for the same reason it is on the Web roots
 * (docs/freebuff-base3-harness.md): the appendix is APPENDED. The root prompt
 * must keep its canonical opening first (`hasFreebuffRootSystemPromptOpening`).
 */
export function createBase3CliRoot(
  options: {
    model?: SecretAgentDefinition['model']
    /** Freebuff branding and meta-information instead of Codebuff's. */
    isFreebuff?: boolean
    /** Drop the tools that address a human. For the eval harness, where an
     *  ask_user call would stall the run rather than gather anything. */
    noAskUser?: boolean
    /** Drop web_search and read_url. For the eval harness: the evaluated
     *  repositories are public, and a search result that quotes their
     *  current code is the answer key (36% of base3's search results in the
     *  2026-10-04 runs cited the repository under evaluation). */
    noWeb?: boolean
    /** Harness guidance that belongs BETWEEN base3's prompt and the CLI
     *  appendix (base3-fast's fan-out section). Appended, never prepended,
     *  for the reason the appendix is: the canonical opening stays at byte 0. */
    extraSystemPrompt?: string
    /** The model's human name, for the meta-information line. A server
     *  catalog root's `model` is an opaque handle, which must not be what the
     *  prompt tells the model it is running on. */
    modelLabel?: string
    /** See createBase3. */
    compaction?: RootCompactionPolicy
    /** See createBase3. */
    compiledModelId?: string
    /** See createBase3. On for CLI roots; the sponsored and fast-mode roots
     *  turn it off. */
    batching?: boolean
  } = {},
): Omit<SecretAgentDefinition, 'id'> {
  const {
    model = OPUS_MODEL,
    isFreebuff = false,
    noAskUser = false,
    noWeb = false,
    extraSystemPrompt,
  } = options
  const base3 = createBase3(model, {
    compaction: options.compaction,
    compiledModelId: options.compiledModelId,
    batching: options.batching ?? true,
  })

  const root: Omit<SecretAgentDefinition, 'id'> = {
    ...base3,
    // base3's own eight, then the CLI surface: `web_search`/`read_url`
    // replace the researcher subagents base2 spawned; the last five drive the
    // ask-user panel, the followup cards, service discovery, rendered UI and
    // skills. (The shipped-agents CI guard checks this toolset by importing
    // the definitions, so it may be assembled rather than restated.)
    toolNames: [
      ...base3.toolNames!,
      'web_search',
      'read_url',
      'ask_user',
      'suggest_followups',
      'gravity_index',
      'render_ui',
      'skill',
    ],
    systemPrompt: `${base3.systemPrompt}
${extraSystemPrompt ? `${extraSystemPrompt}\n` : ''}${buildCliAppendix({
      isFreebuff,
      model: options.modelLabel ?? model,
      noAskUser,
    })}`,
  }

  const withheld = new Set<string>([
    ...(noAskUser ? HUMAN_TOOL_NAMES : []),
    ...(noWeb ? WEB_TOOL_NAMES : []),
  ])
  if (withheld.size === 0) return root
  return {
    ...root,
    toolNames: root.toolNames?.filter((name) => !withheld.has(name)),
  }
}

/** The web, which the eval harness withholds from both arms (see `noWeb`). */
const WEB_TOOL_NAMES: ReadonlySet<string> = new Set(['web_search', 'read_url'])

/** Offered only when there is a human on the other end. */
const HUMAN_TOOL_NAMES: ReadonlySet<string> = new Set([
  'ask_user',
  'suggest_followups',
])

function buildCliAppendix({
  isFreebuff,
  model,
  noAskUser = false,
}: {
  isFreebuff: boolean
  /** The model id, or its human name when the id is an opaque handle. */
  model: string
  noAskUser?: boolean
}): string {
  return `
# Working with the user
${
  noAskUser
    ? ''
    : `
- **Suggest next steps:** At the end of your turn, use the suggest_followups tool to suggest ~3 next steps the user might want to take. ${FOLLOWUP_STYLE_GUIDANCE}`
}
${gravityIndexGuidance()}
${SKILL_DISCOVERY_GUIDANCE}

# ${isFreebuff ? 'Freebuff' : 'Codebuff'} Meta-information

[freebuff product prompt text omitted]

${
  isFreebuff
    ? '[freebuff product prompt text omitted]'
    : [
        'Users send prompts to you in one of a few user-selected modes, like DEFAULT, LITE, MAX, or PLAN.',
        "Every prompt sent consumes the user's credits, which is calculated based on the API cost of the models used.",
        '[freebuff product prompt text omitted]',
        '[freebuff product prompt text omitted]',
      ].join('\n')
}

${PLACEHOLDER.SYSTEM_INFO_PROMPT}
`
}

/**
 * The chat's own user messages, oldest first, bounded the way Desktop's
 * `sponsoredTaskEvidence` bounds them: at most eight, whole messages up to
 * 8,192 characters, and only the newest cut (head and tail) when it alone
 * exceeds the budget.
 */
export function labTaskContext(messages: readonly string[]): string[] {
  const recent = messages
    .filter((message) => message.trim().length > 0)
    .slice(-LAB_TASK_CONTEXT_MAX_MESSAGES)
  const context: string[] = []
  let chars = 0
  for (let index = recent.length - 1; index >= 0; index -= 1) {
    const message = recent[index]!
    if (chars + message.length > LAB_TASK_CONTEXT_MAX_CHARS) {
      if (context.length === 0) context.unshift(excerpt(message))
      break
    }
    context.unshift(message)
    chars += message.length
  }
  return context
}

/** Desktop's `excerptTaskMessage`: tail-weighted, the cut marked in the text. */
function excerpt(text: string, max = LAB_TASK_CONTEXT_MAX_CHARS): string {
  if (text.length <= max) return text
  const marker = (omitted: number) =>
    ` [… ${omitted.toLocaleString('en-US')} characters omitted] `
  const keep = Math.max(0, max - marker(text.length).length)
  let head = Math.floor(keep * 0.4)
  let tail = keep - head
  if (/[\ud800-\udbff]/.test(text.charAt(head - 1))) head -= 1
  if (/[\udc00-\udfff]/.test(text.charAt(text.length - tail))) tail -= 1
  return `${text.slice(0, head)}${marker(text.length - head - tail)}${text.slice(text.length - tail)}`
}

 * whose subject is LAUNCHING what was built.

/**

 * costs a line of colour above the composer while a false negative costs the
 * impression entirely.

 * their own: they are in a large share of all coding requests, and including
 * them would make the trigger "the user typed something", which is not a
 * trigger. `pull` IS included even though it also matches `git pull`, which
 * is the loose end we accepted knowingly -- it is the only way "pull request"
 * spelled in full matches, since the set is checked word by word.

/**
 * Whether `text` mentions pull requests, merging or code review.
 *
 * WHOLE WORDS, not substrings. A substring test is the one way this shape of
 * rule goes obviously wrong: `pr` appears in "print", "improve" and
 * "reprocess", and an ad that appeared while somebody typed "print the logs"
 * would read as an ad that appears always.
 *
 * Splitting on non-alphanumerics is what makes the punctuation cases work
 * without a list of them: `PR#12`, `pr-123`, `(review)` and `merge?` all
 * yield the bare keyword.
 */
export function mentionsPrKeyword(text: string): boolean {
  return mentionsAnyWord(text, PR_INTENT_KEYWORDS)
}

/** The whole-word test both triggers share, so they tokenize identically. */
function mentionsAnyWord(text: string, words: ReadonlySet<string>): boolean {
  return text
    .toLowerCase()
    .split(/[^a-z0-9]+/)
    .some((word) => words.has(word))
}

  T extends {
    event: string
    properties: Record<string, unknown>
  },
>(event: T, projectToken: string): T | null {
  if (event.event !== PRODUCT_FUNNEL_EVENT) return event
  const fields: Record<string, unknown> = {}
  for (const key of productFunnelEventSchema.keyof().options) {
    if (event.properties[key] !== undefined) fields[key] = event.properties[key]
  }
  const parsed = productFunnelEventSchema.safeParse(fields)
  if (!parsed.success) return null
  // posthog-js adds this public ingest key before before_send; transport does
  // not restore it after sanitization. Never copy an untrusted event token.
  const properties: Record<string, unknown> = {
    ...parsed.data,
    token: projectToken,
  }
  for (const key of [
    'distinct_id',
    '$insert_id',
    '$session_id',
    '$is_identified',
  ]) {
    const value = event.properties[key]
    if (typeof value === 'string' || typeof value === 'boolean')
      properties[key] = value
  }
  const sanitized = { ...event, properties }
  Reflect.deleteProperty(sanitized, '$set')
  Reflect.deleteProperty(sanitized, '$set_once')
  return sanitized

+/** Wire contract shared by the server, native Desktop and hybrid Web UI. */
+export type CloudStreamEvent = { id: string; kind: string; payload: unknown }
+export type CloudStreamPage = {
+  status: string
+  error: string | null
+  events: CloudStreamEvent[]
+  questions?: unknown[]
+  preview?: CloudPreview
+}
+export const CLOUD_RUN_STREAM_VERSION = 1
+export const CLOUD_STREAM_MAX_FRAME_BYTES = 8 * 1024 * 1024
+
+/** Incremental SSE decoder: caps a frame, handles split UTF-8 and cancels on detach. */
+export async function* readCloudRunStream(
+  response: Response,
+): AsyncGenerator<CloudStreamPage> {
+  if (
+    !response.ok ||
+    !response.body ||
+    !response.headers.get('content-type')?.includes('text/event-stream')
+  )
+    throw new Error('Cloud stream unavailable')
+  const reader = response.body.getReader()
+  const decoder = new TextDecoder()
+  let buffer = ''
+  try {
+    while (true) {
+      const { done, value } = await reader.read()
+      if (done) break
+      buffer = (buffer + decoder.decode(value, { stream: true })).replace(
+        /\r\n/g,
+        '\n',
+      )
+      let boundary: number
+      while ((boundary = buffer.indexOf('\n\n')) !== -1) {
+        const frame = buffer.slice(0, boundary)
+        buffer = buffer.slice(boundary + 2)
+        if (frame.length > CLOUD_STREAM_MAX_FRAME_BYTES)
+          throw new Error('Cloud stream frame too large')
+        const data = frame
+          .split('\n')
+          .filter((line) => line.startsWith('data:'))
+          .map((line) => line.slice(5).trimStart())
+          .join('\n')
+        if (!data) continue
+        const item = JSON.parse(data)
+        if (
+          item.v !== CLOUD_RUN_STREAM_VERSION ||
+          !Array.isArray(item.events) ||
+          typeof item.status !== 'string'
+        )
+          throw new Error('Invalid Cloud stream frame')
+        if (item.preview !== undefined && !validCloudPreview(item.preview))
+          throw new Error('Invalid Cloud preview')
+        for (const event of item.events) {
+          if (
+            !event ||
+            typeof event.id !== 'string' ||
+            !/^(0|[1-9][0-9]{0,18})$/.test(event.id) ||
+            typeof event.kind !== 'string'
+          )
+            throw new Error('Invalid Cloud event')
+        }
+        yield item as CloudStreamPage
+      }
+      if (buffer.length > CLOUD_STREAM_MAX_FRAME_BYTES)
+        throw new Error('Cloud stream frame too large')
+    }
+  } finally {
+    await reader.cancel().catch(() => {})
+    reader.releaseLock()
+  }

+  constructor(readonly status: number) {
+    super(`Cloud stream request failed (${status})`)
+  }

+/** A single transport per viewer: SSE first, bounded fallback, then a jittered probe. */
+export class CloudRunReader {
+  private iterator?: AsyncGenerator<CloudStreamPage>
+  private connection?: AbortController
+  private retryAt = 0
+  private emptyPolls = 0
+  streaming = false
+  constructor(
+    private options: {
+      open: (after: string, signal: AbortSignal) => Promise<Response>
+      poll: (after: string) => Promise<CloudStreamPage>
+      signal: AbortSignal
+      onDisconnect?: () => void
+    },
+  ) {}
+
+  async next(after: string): Promise<CloudStreamPage> {
+    this.options.signal.throwIfAborted()
+    if (!this.iterator && Date.now() >= this.retryAt) {
+      this.connection = new AbortController()
+      try {
+        const response = await this.options.open(
+          after,
+          AbortSignal.any([
+            this.options.signal,
+            this.connection.signal,
+            AbortSignal.timeout(150_000),
+          ]),
+        )
+        if ([401, 403].includes(response.status)) {
+          await response.body?.cancel()
+          throw new CloudStreamHttpError(response.status)
+        }
+        if (
+          response.ok &&
+          response.headers.get('content-type')?.includes('text/event-stream')
+        ) {
+          this.iterator = readCloudRunStream(response)
+          this.streaming = true
+        } else {
+          await response.body?.cancel()
+          this.cooldown()
+        }
+      } catch (error) {
+        if (error instanceof CloudStreamHttpError) throw error
+        this.options.signal.throwIfAborted()
+        this.cooldown()
+      }
+    }
+    if (this.iterator) {
+      try {
+        const result = await this.iterator.next()
+        if (!result.done) return result.value
+      } catch {
+        this.options.signal.throwIfAborted()
+      }
+      await this.close()
+      this.cooldown()
+    }
+    const page = await this.options.poll(after)
+    this.emptyPolls = page.events.length ? 0 : Math.min(3, this.emptyPolls + 1)
+    return page
+  }
+  get delayMs() {
+    if (this.streaming) return 0
+    return (
+      Math.min(5000, 1000 * 2 ** this.emptyPolls) * (1 + Math.random() * 0.2)
+    )
+  }
+  private cooldown() {
+    this.options.onDisconnect?.()
+    this.streaming = false
+    this.retryAt = Date.now() + 30_000 * (1 + Math.random() * 0.2)
+  }
+  async close() {
+    this.connection?.abort()
+    await this.iterator?.return(undefined).catch(() => {})
+    this.iterator = undefined
+    this.streaming = false
+  }

+
+export class DictationUploadError extends Error {
+  constructor(readonly status: number, message: string) {
+    super(message)
+  }
+}

+/** Read only after authentication. Bounds actual bytes, including chunked uploads. */
+export async function readDictationUpload(request: Request): Promise<ArrayBuffer> {
+  if (request.headers.get('content-type')?.split(';')[0].trim() !== DICTATION_CONTENT_TYPE)
+    throw new DictationUploadError(415, 'Record audio to use dictation.')
+  const length = request.headers.get('content-length')
+  if (length !== null && (!/^\d+$/.test(length) || Number(length) > DICTATION_MAX_AUDIO_BYTES))
+    throw new DictationUploadError(413, 'Recordings must be two minutes or shorter.')
+  const reader = request.body?.getReader()
+  if (!reader) throw new DictationUploadError(400, 'The recording is empty.')
+  const signal = AbortSignal.any([request.signal, AbortSignal.timeout(15_000)])
+  const cancel = () => { void reader.cancel().catch(() => {}) }
+  signal.addEventListener('abort', cancel, { once: true })
+  let size = 0
+  const chunks: Uint8Array[] = []
+  try {
+    for (;;) {
+      signal.throwIfAborted()
+      const { done, value } = await reader.read()
+      signal.throwIfAborted()
+      if (done) break
+      size += value.byteLength
+      if (size > DICTATION_MAX_AUDIO_BYTES)
+        throw new DictationUploadError(413, 'Recordings must be two minutes or shorter.')
+      chunks.push(value)
+    }
+    if (!size) throw new DictationUploadError(400, 'The recording is empty.')
+    const buffer = new ArrayBuffer(size)
+    const bytes = new Uint8Array(buffer)
+    let offset = 0
+    for (const chunk of chunks) {
+      bytes.set(chunk, offset)
+      offset += chunk.byteLength
+    }
+    return buffer
+  } catch (error) {
+    if (error instanceof DictationUploadError) throw error
+    throw new DictationUploadError(signal.aborted ? 408 : 400,
+      signal.aborted ? 'The recording upload was interrupted. Try again.' : 'Could not read the recording.')
+  } finally {
+    signal.removeEventListener('abort', cancel)
+    // A source may never settle its cancel promise. Cancellation must not
+    // prevent the request deadline or byte-limit refusal from being returned.
+    void reader.cancel().catch(() => {})
+  }

+const MAX_WALLPAPER_SOURCE_CHARS = 4_000_000
+
+/** Validate the bounded, static raster produced by the renderer without a native image decoder. */
+export function decodeWallpaper(source: unknown): Buffer | null {
+  if (
+    typeof source !== 'string' ||
+    source.length > MAX_WALLPAPER_SOURCE_CHARS ||
+    !/^data:image\/webp;base64,[a-z\d+/]+={0,2}$/i.test(source)
+  )
+    return null
+  const image = Buffer.from(source.slice(source.indexOf(',') + 1), 'base64')
+  if (
+    image.length < 30 ||
+    image.toString('ascii', 0, 4) !== 'RIFF' ||
+    image.toString('ascii', 8, 12) !== 'WEBP' ||
+    image.readUInt32LE(4) + 8 !== image.length
+  )
+    return null
+  let hasPixels = false
+  for (let offset = 12; offset + 8 <= image.length; ) {
+    const kind = image.toString('ascii', offset, offset + 4)
+    const length = image.readUInt32LE(offset + 4)
+    const start = offset + 8
+    if (start + length > image.length) return null
+    let width = 0,
+      height = 0
+    if (kind === 'VP8X' && length >= 10) {
+      if (image[start] & 0x02) return null // Animated wallpapers never enter the cache.
+      width = image.readUIntLE(start + 4, 3) + 1
+      height = image.readUIntLE(start + 7, 3) + 1
+    } else if (kind === 'VP8 ' && length >= 10) {
+      if (image.toString('hex', start + 3, start + 6) !== '9d012a') return null
+      width = image.readUInt16LE(start + 6) & 0x3fff
+      height = image.readUInt16LE(start + 8) & 0x3fff
+      hasPixels = true
+    } else if (kind === 'VP8L' && length >= 5 && image[start] === 0x2f) {
+      const dimensions = image.readUInt32LE(start + 1)
+      width = (dimensions & 0x3fff) + 1
+      height = ((dimensions >>> 14) & 0x3fff) + 1
+      hasPixels = true
+    } else if (kind === 'ICCP') {
+      // Chromium's canvas encoder adds its own sRGB profile. Keep this bounded
+      // color information; EXIF/XMP and animation chunks remain unsupported.
+      if (length > 65_536) return null
+    } else if (kind !== 'ALPH') return null
+    if (
+      kind !== 'ALPH' &&
+      kind !== 'ICCP' &&
+      (!width || !height || width > 2560 || height > 2560)

 /** `app-3.4.10` before `app-3.4.9`: numeric per segment, newest first. */
 function compareAppVersionsDescending(a: string, b: string): number {
   const parts = (name: string) => name.slice('app-'.length).split('.').map((n) => Number(n) || 0)
   const [left, right] = [parts(a), parts(b)]
   for (let i = 0; i < Math.max(left.length, right.length); i++) {
     const diff = (right[i] ?? 0) - (left[i] ?? 0)
     if (diff !== 0) return diff
   }
   return 0
 }
