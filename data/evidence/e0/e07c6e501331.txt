import type { ChangesScope, CloudChanges, FileDiffResult } from './changes'
import type { HostedBrowserScope } from './hosted-workspace'
import type { PullRequestSnapshot, PullRequestStatus } from './pull-request'

/**
 * Where a chat's changes live and where they go: its branch, what "branch changes" compare
 * against, and its pull request. `live` asks Desktop to refresh the PR (one cached `gh` read);
 * without it the answer is what Desktop already saved, which the sidebar reads for every row.
 * Desktop builds before 2026-11 reject the operation; callers treat that as "nothing known".
 */
export interface GitSummary {
  branch: string | null
  /** The branch "Branch changes" diff against; null when the chat has none (a shared checkout). */
  base: string | null
  worktree: boolean
  pullRequest: PullRequestStatus | null
  /** The open PR no longer merges cleanly. */
  conflict: boolean
  /**
   * Where this chat's PR would be on github.com: its `origin` and the branch it owns. Lets the
   * browser read the PR with the account's GitHub connection when `gh` cannot answer (not installed,
   * signed out, another account). Null for a shared checkout; absent from Desktop before 2026-11.
   */
  github?: { repository: string; branch: string } | null
}

/** Git reads are scoped to the native chat, with no renderer-supplied repository path. */
export type HostedGitRequest = HostedBrowserScope & (
  | { operation: 'changes'; scope: ChangesScope }
  | { operation: 'diff'; scope: ChangesScope; path: string; untracked: boolean }
  | { operation: 'summary'; live: boolean }
  /** A PR the chat touched, read with the computer's `gh`; `pullRequest` is null when `gh` cannot see it.
   *  An object, never a bare null, which the device relay refuses. Desktop builds before 2026-11-16 reject it. */
  | { operation: 'pullRequest'; repository: string; number: number }
)
export type HostedGitResponses = {
  changes: CloudChanges
  diff: FileDiffResult
  summary: GitSummary
  pullRequest: PullRequestSnapshot
}
export type HostedGitHost = {
  request<T extends HostedGitRequest>(input: T): Promise<HostedGitResponses[T['operation']]>
}

export function countTokensJson(value: unknown): number {
  // Inspect lengths without serializing (and duplicating) multi-MB tool output.
  // Bound traversal too: a deeply nested/wide payload is oversized, not zero.
  let remaining = 100_000
  const ancestors = new Set<object>()
  const length = (value: unknown, depth: number): number => {
    if (--remaining < 0 || depth > 100) return Number.MAX_SAFE_INTEGER
    if (typeof value === 'string') return estimatedTextLength(value) + 2
    if (value === null) return 4
    if (typeof value === 'number' || typeof value === 'boolean')
      return String(value).length
    if (typeof value !== 'object') return 0
    if (ancestors.has(value)) return Number.MAX_SAFE_INTEGER
    ancestors.add(value)
    let total = 2
    for (const key in value) {
      if (!Object.hasOwn(value, key)) continue
      total +=
        (Array.isArray(value) ? 1 : estimatedTextLength(key) + 4) +
        length((value as Record<string, unknown>)[key], depth + 1)
      if (total >= Number.MAX_SAFE_INTEGER) {
        total = Number.MAX_SAFE_INTEGER
        break
      }
    }
    ancestors.delete(value)
    return total
  }
  return Math.ceil(length(value, 0) / CHARS_PER_TOKEN)
}

/** Number blank projects after the highest existing Untitled name. */
export function nextUntitledProjectName(
  existingNames: Iterable<string | null | undefined>,
): string {
  let highest = 0
  for (const name of existingNames) {
    const match = name?.trim().match(/^untitled(?: ([1-9]\d*))?$/i)
    if (!match) continue
    const number = match[1] ? Number(match[1]) : 1
    if (Number.isSafeInteger(number) && number < Number.MAX_SAFE_INTEGER)
      highest = Math.max(highest, number)
  }
  return highest === 0 ? 'Untitled' : `Untitled ${highest + 1}`
}
