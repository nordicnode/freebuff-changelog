/**
 * THE AGENTIC OFFER ROUTE's wire contract (COD-665), shared by Freebuff
 * Desktop and `POST /api/v1/ads/agentic/offer` on freebuff-web.
 *
 * The agentic pool (COD-662, `agentic-pool.ts`) gave agentic cards a
 * placement of their own, but still decided them INSIDE the display request:
 * the display chain waited on the roster, the identity check and generic
 * selection, and a card could only be discovered when a display-ad threshold
 * fired mid-stream. This route asks the agentic question on its own request,
 * once per user turn, and the display request stops asking it.
 *
 * Deliberately thin: the response names WHAT was offered, never how it is
 * drawn. The card UX is being reworked, and the rework must not need a
 * contract change -- an invitation carries the existing invitation payload,
 * and a proposal is read from the proposal routes as every proposal already
 * is. Since revamp S10 a proposal may also carry its `preview`: the two
 * answers those routes would give, so the card draws and Accept is one POST
 * without a second read. The response objects are not `.strict()`, so the
 * server may add fields without a version bump and an older client simply
 * drops them.
 *
 * CUTOVER. There is no server knob: the Desktop release that sends
 * `agenticOfferRoute: 1` is the cutover. For such a client this endpoint
 * serves and `/api/ads` skips agentic discovery, wherever the channel's kill
 * switch (`FREEBUFF_AGENTIC_ADS`) admits the caller. Where it does not, this
 * endpoint answers `none/disabled` and `/api/ads` ignores the flag -- one
 * predicate on the server, so exactly one path serves. `FREEBUFF_AGENTIC_ADS=off`
 * therefore stops agentic on BOTH paths; there is no position that sends an
 * updated Desktop back to the old one.
 *
 * THE CLI (`Freebuff-CLI/` UA) is served here too, under the same
 * `FREEBUFF_AGENTIC_ADS` audience as Desktop; it has no switch of its own. A
 * CLI request must also carry `inPlaceExecutionVersion: 1` and report macOS
 * or Linux; a CLI request without the in-place flag, or from Windows, is
 * `ineligible_client`, so the CLI's old worktree flow is never served. The CLI reads offers from the `CLI-Chat-Inline` book,
 * mints rows with `surface: 'cli'`, and is never answered an invitation (it
 * cannot draw one). There is no `/api/ads` fallback for the CLI: it has never
 * been asked the agentic question there.
 */

/**
 * Whether a client build can run a skill's pinned installs.
 *
 * A sponsored run may install only the packages its skill declares
 * (`evaluateSponsoredInstallCommand`), and the CLIENT supplies that list: the
 * skill kept with the run, else its own bundled registry, else (Desktop only)
 * the server's copy. A build that cannot resolve the skill runs with an empty
 * list and refuses even the skill's own pinned install, so the run writes code
 * against a package it never installed (2026-10-13/14: 9 of 30 Exa runs, on
 * Desktop builds older than declared installs or the skill fetch, and every
 * CLI build, none of which bundles exa-web-search-feature@2.0.0).
 *
 * So a skill that declares packages is offered, and accepted, only on a build
 * that can resolve it:
 *
 *  - Desktop from {@link SPONSORED_DESKTOP_SKILL_FETCH_MIN}: the first build
 *    with declared installs AND the server skill fetch, which resolves any
 *    skill version whether or not the build bundles it.
 *  - The CLI from {@link SPONSORED_CLI_SKILL_DELIVERY_MIN}, which takes the
 *    skill the Accept delivers (#6709). An older CLI only runs a skill it
 *    BUNDLES: the first such release is listed in
 *    {@link SPONSORED_SKILL_MIN_CLIENT}.
 *
 * A skill that declares no packages, a free-text procedure, and the web client
 * (server-run, always current) are never refused here.
 *
 * A client that runs the RUN SPEC (`runSpecVersion`, `./sponsored-run-spec.ts`)
 * is never refused for its version: the Accept hands it the skill and its
 * packages, and it resolves nothing itself. It is refused only when the
 * SERVER cannot resolve the skill the procedure names
 * (`skill_unresolved`), because then there is no spec to hand it.
 *
 * A skill that carries a CONVERSION DEFINITION (`conversion`, either
 * version) is the exception to all of the above, run-spec clients, the web
 * client and free-text procedures included: only a Desktop build that
 * advertises the conversion tools (`conversionToolsVersion`,
 * `SPONSORED_CONVERSION_TOOLS_VERSION` in `./sponsored-conversion.ts`) can
 * run it, so it is never offered to, nor accepted from, any other build. It
 * is a capability rather than a version floor because the tools ship on a
 * branch: a Desktop release cut without them must not pass on its number.
 */

/**
 * THE per-advertiser skill format for sponsored (agentic-ad) runs.
 *
 * The advertiser no longer writes a procedure. They state a goal, and we write
 * and test a skill for it: numbered steps, each marked as something the agent
 * does, something only the user can do, or a pure check, with a code check

 * runs a command or sees a secret. The step driver evaluates checks; this
 * module only defines and validates them.
 *
 * A skill is DATA, never code: it is reviewed as text, shown at consent,
 * locked by {@link sponsoredSkillSha256} at Accept, and run by our own agent.
 */

/**
 * The step driver: the harness, not the model, decides when a sponsored
 * skill's steps are done (piece 3 of COD-822; the format is
 * `docs/ads/agentic/sponsored-skills.md`).
 *
 * After the run's turn has done its work, the driver walks the skill's steps
 * in order and runs each step's checks through THE skill evaluator
 * (`evaluateSponsoredSkillChecks`), the same one the sponsored eval scores
 * with. A step with no failing check is done, whatever the model said. A
 * failing `agent` or `check` step goes back to the agent with the failed
 * checks (`sponsoredSkillCheckFeedback`), to fix and stop; the checks run
 * again after each fix. The step's `maxAttempts` counts the first try (the
 * turn's own work) plus fixes, and the run as a whole gets at most
 * {@link SPONSORED_STEP_MAX_FIXES_PER_RUN}. Then the step is BLOCKED, and the
 * walk ends there: later steps usually build on it, and prompting them would
 * spend the run's compute on work that cannot pass. They are still checked
 * once, so the result says what is true on disk.
 *
 * - A `user` step is never prompted or retried. It is done only once its
 *   checks pass (the user handed over the credential), else it needs the user.
 * - A step that passed before a later fix is checked again at the end: a fix
 *   for step 4 can break step 2.
 *
 * DELIVERED means every `agent` and `check` step passed. That is the run's
 * completion, not the model writing "Done:".
 *
 * Pure: the caller hands in the check context (its worktree, its sandbox) and
 * how to prompt the agent, so the whole loop runs in tests without a model,
 * a sandbox or a disk.
 */

 * Bench receipts (bench-v2, B5): what one bench run of one skill version
 * found, committed next to the version as
 * `common/src/ads/sponsored-skills/<id>/<version>.bench.json` and indexed in
 * `sponsored-skills/receipts.ts`. CI checks the newest version of every skill

const rateOf = (cells: readonly BenchReceiptCell[]) => {
  const run = cells.filter((c) => c.expected === 'should_run')
  const hits = run.reduce((n, c) => n + Math.min(c.passes, 1), 0)
  return { hits, n: run.length, rate: run.length ? hits / run.length : null }
}

/** One cell's attempts folded: a pass stops the retries, a mismatch is never forgiven. */
export function benchReceiptCell(
  cell: Pick<
    BenchReceiptCell,
    'id' | 'entry' | 'os' | 'surface' | 'env' | 'expected'
  >,
  outcomes: readonly BenchRunOutcome[],
  traces?: string[],
): BenchReceiptCell {
  const results = outcomes.map((o) => benchCellPass(cell.expected, o))
  return {
    ...cell,
    attempts: results.filter((r) => r !== 'infra').length,
    passes: results.filter((r) => r === 'pass').length,
    results,
    causeKeys: outcomes.map((o) => o.causeKey ?? 'none'),
    ...(traces?.length ? { traces } : {}),
  }
}

/**
 * A receipt's own verdict, with no previous version: no label mismatch, at
 * least one pass in every language family the skill runs on, and an overall
 * `should_run` rate of at least BENCH_MIN_PASS_RATE.
 */
export function benchOwnVerdict(
  cells: readonly BenchReceiptCell[],
  families: ReadonlyMap<string, string>,
): string[] {
  const reasons: string[] = []
  for (const cell of cells)
    if (cell.results.includes('mismatch'))
      reasons.push(
        `${cell.id}: expected ${cell.expected}, the run did the opposite`,
      )
  const run = cells.filter((c) => c.expected === 'should_run')
  const perFamily = new Map<string, number>()
  for (const cell of run) {
    const family = families.get(cell.entry) ?? 'other'
    perFamily.set(
      family,
      (perFamily.get(family) ?? 0) + Math.min(cell.passes, 1),
    )
  }
  for (const [family, passes] of perFamily)
    if (!passes) reasons.push(`No should_run cell passed for ${family}.`)
  const { rate } = rateOf(cells)
  if (rate !== null && rate < BENCH_MIN_PASS_RATE)
    reasons.push(
      `should_run pass rate ${(rate * 100).toFixed(0)}% is below ${BENCH_MIN_PASS_RATE * 100}%.`,
    )
  for (const cell of cells)
    if (cell.attempts === 0 && cell.results.length)
      reasons.push(`${cell.id}: every attempt was infra; not measured.`)
  return reasons
}

/**
 * THE regression rule between two receipts of the same skill (B5): a cell
 * regresses if it had at least one pass before and passes 0 of
 * BENCH_CELL_REGRESSION_ATTEMPTS now; the overall `should_run` rate
 * regresses when it drops BENCH_MAX_RATE_DROP or more at z ≥ BENCH_MIN_DROP_Z.
 * Only cells both receipts ran are compared.
 */
export function compareBenchReceipts(
  previous: Pick<BenchReceipt, 'skill' | 'cells'>,
  next: Pick<BenchReceipt, 'cells'>,
): { regressed: boolean; reasons: string[]; shared: number } {
  const reasons: string[] = []
  const before = new Map(previous.cells.map((c) => [c.id, c]))
  const shared = next.cells.filter((c) => before.has(c.id))
  for (const cell of shared) {
    const old = before.get(cell.id)!
    if (
      old.passes >= 1 &&
      cell.passes === 0 &&
      cell.attempts >= BENCH_CELL_REGRESSION_ATTEMPTS
    )
      reasons.push(
        `${cell.id}: passed ${old.passes}/${old.attempts} in ${previous.skill}, 0/${cell.attempts} now`,
      )
  }
  const a = rateOf(shared.map((c) => before.get(c.id)!))
  const b = rateOf(shared)
  if (a.rate !== null && b.rate !== null) {
    const drop = a.rate - b.rate
    const z = dropZ(a, b)
    if (drop >= BENCH_MAX_RATE_DROP && z >= BENCH_MIN_DROP_Z)
      reasons.push(
        `should_run rate fell ${(drop * 100).toFixed(0)} points (${(a.rate * 100).toFixed(0)}% → ${(b.rate * 100).toFixed(0)}%, z ${z.toFixed(1)})`,
      )
  }
  return { regressed: reasons.length > 0, reasons, shared: shared.length }
}

/** A receipt from a plan and its live outcomes, judged against the previous version's receipt. */
export function buildBenchReceipt(input: {
  plan: BenchPlan
  cells: BenchReceiptCell[]
  labelsSha256: string
  families: ReadonlyMap<string, string>
  previous: BenchReceipt | null
  labBatchIds: string[]
  actionsRunUrl: string | null
  costUsd: number | null
  createdAt: string
  surfaces?: string[]
}): BenchReceipt {
  const { plan } = input
  const byExpected = {
    should_run: 0,
    should_refuse: 0,
    should_fail_gracefully: 0,
  }
  for (const cell of plan.static) byExpected[cell.expected]++
  const fidelity = plan.static.flatMap((c) =>
    c.fidelity ? [`${c.id}: ${c.fidelity}`] : [],
  )
  const own = benchOwnVerdict(input.cells, input.families)
  if (fidelity.length)
    own.push(
      `${fidelity.length} static cell(s) where the fit oracle and the preflight disagree.`,
    )
  const compared = input.previous
    ? compareBenchReceipts(input.previous, { cells: input.cells })
    : null
  return {
    v: 1,
    skill: plan.skill,
    skillSha256: plan.skillSha256,
    benchVersion: plan.benchVersion,
    mode: plan.mode,
    bump: plan.bump,
    surfaces:
      input.surfaces ??
      [
        ...new Set(
          input.cells.map((c) =>
            c.surface === 'lab' ? 'lab' : `desktop-${c.os}`,
          ),
        ),
      ].sort(),
    createdAt: input.createdAt,
    labBatchIds: input.labBatchIds,
    actionsRunUrl: input.actionsRunUrl,
    costUsd: input.costUsd,
    static: {
      cells: plan.static.length,
      byExpected,
      labelsSha256: input.labelsSha256,
      fidelity,
    },
    cells: input.cells,
    verdict: {
      result: own.length
        ? 'failed'
        : compared?.regressed
          ? 'regressed'
          : 'pass',
      against: input.previous?.skill ?? null,
      reasons: [...own, ...(compared?.reasons ?? [])],
    },
  }
}

    `**Bench (${receipt.benchVersion}, ${receipt.mode}): ${receipt.verdict.result.toUpperCase()}**${receipt.verdict.against ? ` against ${receipt.verdict.against}` : ''}`,
    '',
    `Static oracle: ${receipt.static.cells} cells (${receipt.static.byExpected.should_run} should_run, ${receipt.static.byExpected.should_refuse} should_refuse, ${receipt.static.byExpected.should_fail_gracefully} should_fail_gracefully).`,
    '',
    '| Cell | Expected | Passes | Causes | Trace |',
    '| --- | --- | ---: | --- | --- |',
    ...receipt.cells.map(
      (c) =>
        `| ${c.id} | ${c.expected} | ${c.passes}/${c.attempts}${c.results.includes('mismatch') ? ' (mismatch)' : ''} | ${[...new Set(c.causeKeys)].join(', ')} | ${c.passes < c.attempts && c.traces?.length ? c.traces.map((t, i) => `[${i + 1}](${t.replace(/^gs:\/\//, 'https://storage.cloud.google.com/')})`).join(' ') : ''} |`,
    ),
    '',
    ...receipt.verdict.reasons.map((r) => `- ${r}`),
    ...(receipt.actionsRunUrl ? ['', `Run: ${receipt.actionsRunUrl}`] : []),
    '<!-- /bench -->',
  ]

 * THE AGENTIC OFFER ROUTE's wire contract (COD-665), shared by Freebuff
 * Desktop and `POST /api/v1/ads/agentic/offer` on freebuff-web.
 *
 * The agentic pool (COD-662, `agentic-pool.ts`) gave agentic cards a
 * placement of their own, but still decided them INSIDE the display request:
 * the display chain waited on the roster, the identity check and generic
 * selection, and a card could only be discovered when a display-ad threshold
 * fired mid-stream. This route asks the agentic question on its own request,
 * once per user turn, and the display request stops asking it.
 *

 * drawn. The card UX is being reworked, and the rework must not need a
 * contract change -- an invitation carries the existing invitation payload,
 * and a proposal is read from the proposal routes as every proposal already

/** One run/definition identity. Retries and later weaker reports cannot erase observed evidence. */
export function mergeConversionReport(
  current: ConversionAssessment,
  report: ConversionReport,
  now: number,
): ConversionAssessment {
  const evidence = new Map(current.evidence.map((e) => [e.criterionId, e]))
  for (const item of report.evidence) {
    const criterion = current.definition.criteria.find(
      (c) => c.id === item.criterionId,
    )
    if (
      !criterion ||
      criterion.origin !== item.origin ||
      item.observedAt < current.acceptedAt - CONVERSION_CLOCK_SKEW_MS ||
      item.observedAt > now + CONVERSION_CLOCK_SKEW_MS ||
      item.observedAt >
        current.acceptedAt +
          CONVERSION_EVIDENCE_WINDOW_MS +
          CONVERSION_CLOCK_SKEW_MS
    )
      continue
    if (item.source === 'browser_metadata') {
      if (
        current.definition.version !== 1 ||
        item.createdAt < current.acceptedAt ||
        item.createdAt > item.observedAt + CONVERSION_CLOCK_SKEW_MS
      )
        continue
    } else {
      if (current.definition.version === 1) continue
      if (
        (criterion.claim === 'product_download') !==
        (item.source === 'agent_install')
      )
        continue
      if (
        item.verdict === 'satisfied' &&
        ((criterion.claim !== 'product_download' &&
          (current.userAnswer === 'unknown'
            ? report.userAnswer
            : current.userAnswer) !== 'confirmed') ||
          !validConversionBasis(criterion.claim, item.basis) ||
          (criterion.claim === 'product_download'
            ? item.source !== 'agent_install' ||
              item.installationVerified !== true
            : item.source !== 'agent_browser') ||
          (criterion.claim === 'new_api_key' &&
            item.projectCredentialAvailable !== true))
      )
        continue
    }
    const prior = evidence.get(item.criterionId)
    if (
      !prior ||
      (prior.source !== 'browser_metadata' && prior.verdict !== 'satisfied')
    )
      evidence.set(item.criterionId, item)
  }
  const satisfied = current.definition.criteria.every((c) => {
    const item = evidence.get(c.id)
    return (
      item &&
      (item.source === 'browser_metadata' || item.verdict === 'satisfied')
    )
  })
  const userAnswer =
    current.userAnswer === 'unknown' ? report.userAnswer : current.userAnswer
  return {
    ...current,
    userAnswer,
    updatedAt: now,
    evidence: [...evidence.values()],
    lifecycle: satisfied
      ? 'completed'
      : current.lifecycle !== 'pending' && report.lifecycle === 'pending'
        ? current.lifecycle
        : report.lifecycle,
    reason: satisfied
      ? 'criteria_satisfied'
      : report.reason === 'criteria_satisfied'
        ? current.definition.version !== 1
          ? 'insufficient_evidence'
          : 'metadata_missing'
        : current.assessment === 'inconclusive' &&
            WEAK_REPORT_REASONS.has(report.reason)
          ? // A later "not assessed"/interruption adds nothing, so it must not
            // replace the evidence-backed reason the assessment rests on.
            current.reason
          : report.reason,
    assessment: satisfied
      ? 'satisfied'
      : [
            'browser_unavailable',
            'tab_closed',
            'origin_mismatch',
            'metadata_missing',
            'insufficient_evidence',
            'project_key_missing',
            'existing_resource',
            'criteria_satisfied',
          ].includes(report.reason)
        ? 'inconclusive'
        : current.assessment,
  }
}

/**
 * The criterion ids whose reported evidence `merged` did not keep: outside the
 * time window, the wrong origin, unknown, or a satisfied verdict without the
 * confirmation, basis or saved key it needs. Evidence replaced only because a
 * stronger item was already stored is not rejected.
 */
export function rejectedConversionEvidence(
  merged: ConversionAssessment,
  report: ConversionReport,
): string[] {
  const rejected = report.evidence.flatMap((item) => {
    const kept = merged.evidence.find((e) => e.criterionId === item.criterionId)
    if (!kept) return [item.criterionId]
    if (kept.source === item.source && kept.observedAt === item.observedAt)
      return []
    const stronger =
      kept.source === 'browser_metadata' || kept.verdict === 'satisfied'
    return stronger ? [] : [item.criterionId]
  })
  return [...new Set(rejected)]
}

    })

export const SPONSORED_EVIDENCE_CHECK_LIMIT = 20

/** Also used by Convex: keep this module free of runtime/zod dependencies. */
export function parseSponsoredRunEvidence(
  value: unknown,
): SponsoredRunEvidence | undefined {
  if (!value || typeof value !== 'object' || Array.isArray(value))
    return undefined
  const v = value as Record<string, unknown>
  if (
    Object.keys(v).some(
      (k) =>
        ![
          'version',
          'agentSummary',
          'ending',
          'checks',
          'checksRecorded',
          'verificationPending',
          'truncated',
        ].includes(k),
    ) ||
    (v.agentSummary !== undefined &&
      (typeof v.agentSummary !== 'string' ||
        !v.agentSummary.trim() ||
        v.agentSummary.length > SPONSORED_RUN_SUMMARY_MAX)) ||
    v.version !== 1 ||
    !(SPONSORED_RUN_ENDINGS as readonly unknown[]).includes(v.ending) ||
    (v.verificationPending !== undefined &&
      (v.verificationPending !== true ||
        !['completed', 'closed', 'stopped', 'interrupted'].includes(
          v.ending as string,
        ))) ||
    (v.checksRecorded !== undefined && v.checksRecorded !== true) ||
    (v.truncated !== undefined && v.truncated !== true) ||
    !Array.isArray(v.checks) ||
    v.checks.length > SPONSORED_EVIDENCE_CHECK_LIMIT
  )
    return undefined
  const checks: SponsoredRunEvidence['checks'] = []
  for (const raw of v.checks) {
    if (!raw || typeof raw !== 'object' || Array.isArray(raw)) return undefined
    const c = raw as Record<string, unknown>
    if (
      Object.keys(c).some(
        (k) => !['step', 'checkId', 'category'].includes(k),
      ) ||
      !Number.isInteger(c.step) ||
      (c.step as number) < 0 ||
      (c.step as number) > 100 ||
      typeof c.checkId !== 'string' ||
      !/^[a-zA-Z0-9_-]{1,64}$/.test(c.checkId) ||
      !(SPONSORED_BLOCKER_CATEGORIES as readonly unknown[]).includes(c.category)
    )
      return undefined
    checks.push({
      step: c.step as number,
      checkId: c.checkId,
      category: c.category as SponsoredBlockerCategory,
    })
  }
  return {
    version: 1,
    ...(scrubSponsoredRunSummary(v.agentSummary)
      ? { agentSummary: scrubSponsoredRunSummary(v.agentSummary)! }
      : {}),
    ending: v.ending as SponsoredRunEvidence['ending'],
    checks,
    ...(v.verificationPending ? { verificationPending: true as const } : {}),
    ...(v.checksRecorded ? { checksRecorded: true } : {}),
    ...(v.truncated ? { truncated: true } : {}),
  }
}

      })

  return {
    version: 1,
    ending: /cloud runner restarted/i.test(errorText ?? '')
      ? 'runner_restarted'
      : ending,
    ...(steps &&
    sponsoredUserVerificationPending(steps) &&
    ['completed', 'closed', 'stopped', 'interrupted'].includes(ending) &&
    !/cloud runner restarted/i.test(errorText ?? '')
      ? { verificationPending: true as const }
      : {}),
    ...(steps?.preflight?.length ||
    steps?.steps.some((step) => step.checks.length)
      ? { checksRecorded: true as const }
      : {}),
    checks: checks.slice(0, SPONSORED_EVIDENCE_CHECK_LIMIT),
    ...(checks.length > SPONSORED_EVIDENCE_CHECK_LIMIT
      ? { truncated: true }
      : {}),
  }

/**
 * Why `clientVersion` (`desktop/<v>` or `cli/<v>`, from
 * `sponsoredClientVersionFromUserAgent`) cannot run the named skill, or null
 * when it can.
 *
 * `skill` is the resolved skill, or null when the procedure names one that
 * could not be read: that is treated as declaring packages, since an unknown
 * verdict must not serve. `dev` builds always pass.
 *
 * `procedure` is the text the client would run. When `skill` is null it is
 * the only way to see a conversion definition (the `conversion-definition:`
 * line the skill renders), so pass it wherever it is at hand. A free-text
 * procedure (`skillRef` null) with that line is gated too.
 *
 * `conversionToolsVersion` is the request's field, as sent: a Desktop runs a
 * conversion skill only when it advertises the tools, `dev` builds included.
 */
export function sponsoredSkillClientRefusal(input: {
  client: 'desktop' | 'cli' | 'web' | null
  clientVersion: string | null
  /** The skill the procedure names (`sponsoredSkillDirective`), or null for free text. */
  skillRef: { id: string; version: string } | null
  skill: {
    packages: readonly unknown[]
    format?: number
    conversion?: unknown
  } | null
  /** The procedure text the skill renders to, when the caller has it. */
  procedure?: string | null
  /** The request's `runSpecVersion`, as sent. */
  runSpecVersion?: unknown
  /** The request's `conversionToolsVersion`, as sent. */
  conversionToolsVersion?: unknown
}): SponsoredSkillClientRefusal | null {
  // CONVERSION FIRST, for every client, run-spec, web and free text
  // included: only a Desktop build with the conversion tools can run it.
  const conversion = sponsoredConversionClientRefusal(input)
  if (conversion) return conversion
  if (!input.skillRef) return null
  // FORMAT 2 FIRST, for every Desktop and CLI build, run-spec ones included:
  // a build that cannot parse the skill cannot drive it.
  if (
    input.skill?.format === 2 &&
    (input.client === 'desktop' || input.client === 'cli')
  ) {
    const version = clientVersionOf(input.client, input.clientVersion)
    if (version === null) return 'client_version_unknown'
    if (
      version !== 'dev' &&
      compareSponsoredSkillVersions(
        version,
        SPONSORED_SKILL_FORMAT_2_MIN_CLIENT[input.client],
      ) < 0
    )
      return input.client === 'desktop' ? 'desktop_too_old' : 'cli_too_old'
  }
  if (clientRunsSponsoredRunSpec(input.runSpecVersion)) {
    return input.skill ? null : 'skill_unresolved'
  }
  if (input.client !== 'desktop' && input.client !== 'cli') return null
  if (input.skill && input.skill.packages.length === 0) return null

  const prefix = `${input.client}/`
  if (!input.clientVersion?.startsWith(prefix)) return 'client_version_unknown'
  const version = input.clientVersion.slice(prefix.length)
  if (version === 'dev') return null
  if (!/^\d+\.\d+\.\d+(?:-[0-9A-Za-z.-]+)?$/.test(version)) {
    return 'client_version_unknown'
  }

  const listed =
    SPONSORED_SKILL_MIN_CLIENT[`${input.skillRef.id}@${input.skillRef.version}`]
  if (input.client === 'desktop') {
    const min = maxVersion(SPONSORED_DESKTOP_SKILL_FETCH_MIN, listed?.desktop)
    return compareSponsoredSkillVersions(version, min) < 0
      ? 'desktop_too_old'
      : null
  }
  // The Accept delivers the skill only when the server could read it: with
  // no skill here the CLI would get none and cannot fetch one later.
  if (
    input.skill &&
    compareSponsoredSkillVersions(version, SPONSORED_CLI_SKILL_DELIVERY_MIN) >=
      0
  ) {
    return null
  }
  return listed?.cli && compareSponsoredSkillVersions(version, listed.cli) >= 0
    ? null
    : 'cli_too_old'
}

/**
 * The conversion part of {@link sponsoredSkillClientRefusal} alone: why this
 * client cannot run a skill (or free-text procedure) that carries a
 * conversion definition, or null when it carries none or the client can.
 * Needs no skill lookup when `procedure` is the rendered text, which always
 * carries the `conversion-definition:` line of a skill that has one.
 */
export function sponsoredConversionClientRefusal(input: {
  client: 'desktop' | 'cli' | 'web' | null
  skill?: { conversion?: unknown } | null
  procedure?: string | null
  conversionToolsVersion?: unknown
}): 'conversion_tools_missing' | 'desktop_too_old' | 'client_version_unknown' | null {
  if (
    input.skill?.conversion == null &&
    !procedureCarriesConversion(input.procedure)
  )
    return null
  if (input.client === 'cli' || input.client === 'web')
    return 'conversion_tools_missing'
  if (input.client !== 'desktop') return 'client_version_unknown'
  const definition =
    input.skill?.conversion ??
    conversionDefinitionFromProcedure(input.procedure)
  if (
    definition &&
    typeof definition === 'object' &&
    'version' in definition &&
    definition.version === 3 &&
    !(
      typeof input.conversionToolsVersion === 'number' &&
      Number.isInteger(input.conversionToolsVersion) &&
      input.conversionToolsVersion >= 2
    )
  )
    return 'desktop_too_old'
  return clientHasSponsoredConversionTools(input.conversionToolsVersion)
    ? null
    : 'desktop_too_old'
}

/** `desktop/<v>` → `<v>` (or `dev`); null when it is not this client's or not a version. */
function clientVersionOf(
  client: 'desktop' | 'cli',
  clientVersion: string | null,
): string | null {
  const prefix = `${client}/`
  if (!clientVersion?.startsWith(prefix)) return null
  const version = clientVersion.slice(prefix.length)
  if (version === 'dev') return version
  return /^\d+\.\d+\.\d+(?:-[0-9A-Za-z.-]+)?$/.test(version) ? version : null
}

/**
 * Whether a procedure text carries a conversion definition line, well-formed
 * or not: a malformed one still asks the client for conversion tools.
 */
function procedureCarriesConversion(
  procedure: string | null | undefined,
): boolean {
  return (
    procedure
      ?.replace(/\r\n?/g, '\n')
      .split('\n')
      .some((line) => line.startsWith(CONVERSION_DIRECTIVE)) === true
  )
}

function maxVersion(a: string, b: string | undefined): string {
  return b && compareSponsoredSkillVersions(b, a) > 0 ? b : a
}
